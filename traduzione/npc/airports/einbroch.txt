airport,143,43,5	script	Airport Staff#airport1a::airport1	90,{
	mes "Personale aeroportuale";
	mes "Benvenuto/a al";
	mes "Airport di Einbroch,";
	mes "dove offriamo senza sosta";
	mes "voli per le città di";
	mes "Juno, Lighthalzen e Hugel.";
	next;
	if (select("Salire sull'aereo-nave:Annulla") == 1) {
		mes "Personale aeroportuale";
		mes "La tariffa di imbarco dell'diramidione";
		mes "sono 1.200 zeny, ma se hai";
		mes "ho preso un biglietto gratuito per Airship,";
		mes "la tariffa sarà annullata. Sarà";
		mes "Prendi l'aereo?";
		next;
		if (select("Sì:No") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "airport",148,51;
				end;
			}
			mes "Personale aeroportuale";
			mes "Mi dispiace, ma tu non";
			mes "ho un biglietto gratuito per";
			mes "Zecchino e tu non hai";
			mes "abbastanza zeny per salire a bordo";
			mes "l'Aerei. Ricorda, il";
			mes "la tassa di imbarco è di 1.200 zeny.";
			close;
		}
	}
	mes "Personale aeroportuale";
	mes "Grazie e";
	mes "ti auguro una buona giornata.";
	close;
}
