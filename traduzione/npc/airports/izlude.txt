-	script	::Airship_Staff_izlude	-1,{
	mes "[Personale del Dirigibile]";
	mes "Benvenuto/a! Vorresti";
	mes "salire sull'aeronave che";
	mes "parte sul volo che ferma";
	mes "in Juno e Rachel?";
	next;
	if (select("Imbarcati nell'aereo-nave per Juno/Rachel.:Annulla.") == 1) {
		mes "[Personale del Dirigibile]";
		mes "La tassa di imbarco è";
		mes "1.200 zeni. Tuttavia, questo";
		mes "il costo è annullato se usi";
		mes "Un biglietto gratuito per l'Airship. Ora,";
		mes "vuoi ancora salire a bordo?";
		next;
		if (select("Sì:No") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "airplane_01",244,58;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "airplane_01",244,58;
				end;
			}
			mes "[Personale del Dirigibile]";
			mes "Mi dispiace, ma tu non";
			mes "ho 1.200 zeny da pagare";
			mes "per la tassa di imbarco.";
			close;
		}
	}
	mes "[Personale del Dirigibile]";
	mes "Grazie e";
	mes "per favore torna.";
	mes "Buona giornata~";
	close;
}
