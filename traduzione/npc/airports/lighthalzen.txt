lhz_airport,143,43,5	script	Airport Staff#lhz_air1a::lhz_airport1	90,{
	mes "Personale aeroportuale";
	mes "Benvenuto/a al";
	mes "Aeroporto di Lighthalzen,";
	mes "dove offriamo senza sosta";
	mes "voli per Einbroch, Juno e Hugel.";
	next;
	if (select("Imbarcati nell'aereo-nave.:Annulla.") == 1) {
		mes "Personale aeroportuale";
		mes "La tassa di imbarco è";
		mes "1.200 zeni, ma puoi";
		mes "rinunciare alla tariffa se riscatti";
		mes "Un biglietto gratuito per l'aerostato.";
		next;
		if (select("Sì:No") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "lhz_airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "lhz_airport",148,51;
				end;
			}
			mes "[Personale del Dirigibile]";
			mes "Mi dispiace, ma tu non";
			mes "ho 1.200 zeny da pagare";
			mes "per la tassa di imbarco.";
			close;
		}
	}
	mes "Personale aeroportuale";
	mes "Grazie e";
	mes "per favore torna.";
	mes "Buona giornata~";
	close;
}

lhz_airport,143,49,3	script	Arrival Staff#lhz_air2a::lhz_airport2	90,{
	mes "Personale di arrivo";
	mes "Benvenuti all'aeroporto di Lighthalzen.";
	mes "Lasciatemi guidarvi a";
	mes "terminal principale se arrivi dal tuo volo. Altrimenti, per favore";
	mes "salpiate l'Aerei-nave di partenza per raggiungere la vostra destinazione prevista.";
	next;
	if (select("Uscita al terminale principale.:Annulla.") == 1) {
		mes "Personale di arrivo";
		mes "Una volta che sarai nel terminal principale, dovrai pagare di nuovo la tariffa";
		mes "per salire su un dirigibile. Dovresti";
		mes "uscire solo se Lighthalzen è la tua destinazione prevista. Andiamo?";
		mes "procedere al terminale principale?";
		next;
		if (select("Sì:No") == 1) {
			warp "lhz_airport",142,40;
			end;
		}
	}
	mes "Personale di arrivo";
	mes "Va bene, grazie";
	mes "per il vostro patrocinio";
	mes "e spero che tu abbia";
	mes "un piacevole volo~";
	close;
}
