ra_fild12,295,208,0	script	toairplane#rachel	45,1,1,{
	end;
OnTouch:
	mes "Per usare l'aerei, è richiesto di pagare 1.200 zeni o un Biglietto Aereo Gratuito.";
	mes "Vuoi usare il servizio?";
	next;
	if (select("Sì:No") == 1) {
		if (countitem(7311) > 0) {
			delitem 7311,1; //Free_Flying_Ship_Ticket
			warp "airplane_01",245,60;
			end;
		}
		if (Zeny >= 1200) {
			set Zeny, Zeny - 1200;
			warp "airplane_01",245,60;
			end;
		}
		mes "Mi dispiace, ma non hai abbastanza soldi.";
		mes "Si prega di ricordare che è richiesto di pagare 1.200 zeny per utilizzare il servizio.";
		close;
	}
	mes "Grazie, per favore torna.";
	close;
}
