y_airport,143,43,5	script	Airport Staff#y_air1a::y_airport1	90,{
	mes "Personale aeroportuale";
	mes "Benvenuti all'aeroporto di Juno dove offriamo voli nazionali per Einbroch, Lighthalzen e Hugel,";
	mes "e voli internazionali per Izlude e Rachel.";
	mes "Come posso esserle utile?";
	next;
	if (select("Imbarcati nell'aereo-nave.:Annulla.") == 1) {
		mes "Personale aeroportuale";
		mes "La tassa di imbarco per tutti";
		mes "voli costano 1.200 zeni. Se tu";
		mes "usa un biglietto gratuito per Airship,";
		mes "La tassa di imbarco sarà annullata. Quindi vorresti partire?";
		next;
		if (select("Sì:No") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "y_airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "y_airport",148,51;
				end;
			}
			mes "Personale aeroportuale";
			mes "Mi dispiace, ma tu non";
			mes "ho 1.200 zeny da pagare";
			mes "per la tassa di imbarco.";
			close;
		}
	}
	mes "Personale aeroportuale";
	mes "Grazie e";
	mes "ti auguro una buona giornata.";
	close;
}
