	case 2:
		mes "[Erundek]";
		mes "Abbiamo molti articoli, quindi per favore dai un'occhiata e acquista con deliberazione.";
		close2;
		readbook 11010,1;
		end;
	}
	set .@menu$,"";
	if (.@menu1$ != "") set .@menu$, .@menu1$;
	else for(set .@i,0; .@i<getarraysize(.@Items); set .@i,.@i+2)
		set .@menu$, .@menu$+getitemname(.@Items[.@i])+":";
	set .@i, (select(.@menu$)-1)*2;
	mes "[Erundek]";
	mes "Hai scelto ^3131FF"+getitemname(.@Items[.@i])+"^000000.";
	switch(.@Items[.@i]) {
		case 2720: mes "Questo articolo è solo per la classe di Swordman e Taekwon Master."; break;
		case 2721: mes "Questo oggetto è solo per la Classe Ladro."; break;
		case 2722: mes "Questo oggetto è solo per la Classe Acolita."; break;
		case 2723: mes "Questo oggetto è solo per la Classe Mago."; break;
		case 2724: mes "Questo articolo è solo per la Classe Arciera."; break;
		case 2725: mes "Questo articolo è solo per la Classe Commerciante."; break;
		case 2733: mes "Questo oggetto è solo per Gunslinger."; break;
		default: break;
	}
	mes "Puoi scambiare per questo articolo con ^FF0000"+.@Items[.@i+1]+" "+getitemname(7828)+" o "+.@Items[.@i+1]+" "+getitemname(7829)+"^000000.";
	mes "Vorresti scambiare?";
	next;
	switch(select("Non scambiare:Scambio")) {
	case 1:
		mes "[Erundek]";
		mes "Hai bisogno di più tempo per controllare gli articoli?";
		break;
	case 2:
		mes "[Erundek]";
		mes "Quale distintivo vuoi scambiare?";
		mes "Hai bisogno di ^3131FF"+.@Items[.@i+1]+" Medaglie^000000 da scambiare.";
		next;
		if (.@item[0] < 12269 || .@item[0] > 12273 ) {
			mes "[Erundek]";
			mes "Ricorda che gli oggetti ricompensa di Battleground sono vincolati al personaggio ^FF0000Character Bound^000000. Sei sicuro di voler questo oggetto?";
			next;
		}
		set .@j, select("Medaglia del coraggio:Valore Badge:Annulla");
		mes "[Erundek]";
		if (.@j == 3) {
			mes "Hai annullato lo scambio.";
			break;
		}
		set .@cost, ((.@j==1)?7828:7829);
		if (countitem(.@cost) >= .@Items[.@i+1]) {
			mes "Grazie per lo scambio.";
			delitem .@cost, .@Items[.@i+1];
			getitem .@Items[.@i],1;
		}
		else mes "Non hai abbastanza "+getitemname(.@cost)+"s.";
		break;
	}
	close;
}

// Badge Reward Function
//============================================================
function	script	F_BG_Badge	{
	.@team$ = getarg(1);
	.@bg_type$ = getarg(2);

	if (.@bg_type$ == "Tierra") {
		set .@badge,7828; //BF_Badge1
		set .@amount_win,3;
		set .@amount_lose,1;
	} else if (.@bg_type$ == "Flavius") {
		set .@badge,7829; //BF_Badge2
		set .@amount_win,9;
		set .@amount_lose,3;
	} else
		return;

	if (getarg(0) == 1) {
		set .@amount, .@amount_win;
		if (.@team$ == "Guillaume") {
			mes "[Axl Rose]";
			mes "Beato Guglielmo!";
		} else if (.@team$ == "Croix") {
			mes "[Swandery]";
			mes "Benedetta Croce!";
		}
		mes "Godiamoci la nostra gloriosa vittoria!";
		mes strcharinfo(0)+", it's a sign reflecting victory.";
	} else {
		set .@amount, .@amount_lose;
		if (.@team$ == "Guillaume") {
			mes "[Axl Rose]";
			mes "Hai perso, ma sei dedicato a questa battaglia.";
			mes "Questo è un premio per la tua grande dedizione di Guillaume Marollo!";
			mes "Consider questa sconfitta come una lezione e la prossima volta vincerai sicuramente.";
		} else if (.@team$ == "Croix") {
			mes "[Swandery]";
			mes "Oh, "+strcharinfo(0)+" Non essere triste.";
			mes "Anche se non abbiamo vinto, abbiamo fatto del nostro meglio.";
			mes "Questo è un regalo reale da Croix, e per favore non dimenticare questa battaglia. Vinceremo la prossima.";
		}
	}
	close2;

	// If the VIP system is enabled, VIP players receive an extra 2 badges.
	if (VIP_SCRIPT && vip_status(VIP_STATUS_ACTIVE))
		set .@amount, .@amount+2;

	set .@medal_gap, 500 - countitem(.@badge);
	if (.@medal_gap >= .@amount)
		getitem .@badge,.@amount;
	else
		getitem .@badge,.@medal_gap;
	return;
}
