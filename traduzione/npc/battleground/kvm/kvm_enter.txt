// KvM Officer - Guillaume
//============================================================
bat_room,164,178,5	script	KVM Mercenary Officer#1	418,{
	if (getbattleflag("feature.bgqueue"))
		end;
	/*
	set .@kvm,checkquest(6026,PLAYTIME);
	if ((.@kvm == 0) || (.@kvm == 1)) {
		mes "Ufficiale mercenario Croce";
		mes "So che hai già firmato con la Croce. Torna indietro e unisciti al loro team. Non accetteremo traditori per combattere per noi.";
		close;
	}
	else if (.@kvm == 2) erasequest 6026;
	*/
	set .@permit,checkquest(6025,PLAYTIME);
	if ((.@permit == 0) || (.@permit == 1)) {
		mes "Ufficiale mercenario Guillaume";
		mes "Sembra che tu abbia partecipato di recente a una battaglia KVM. Devi aspettare 5 minuti prima di registrarti nuovamente.";
		close;
	}
	else if (.@permit == 2) erasequest 6025;
	mes "Ufficiale mercenario Guillaume";
	mes "Fagli sapere il vero potere di Guillaume!";
	next;
	switch(select("Ti prenderò in giro.:Fine Conversazione.")) {
	case 1:
		mes "Ufficiale mercenario Guillaume";
		mes "Mostra loro quanto siamo forti.";
		mes "Oggi tutti sentiranno il grido di trionfo di Guillaume!";
		close2;
		//setquest 6025;
		if (BaseLevel > 79)  warp "bat_room",169,223;
		else if (BaseLevel < 60) warp "bat_room",197,223;
		else warp "bat_room",225,223;
		end;
	case 2:
		mes "Ufficiale mercenario Guillaume";
		mes "Vinceremo!";
		close;
	}
}

// Guillaume Knight - KvM
//============================================================
bat_room,167,178,5	script	Guillaume Knight#kvm	417,{
	mes "[Guillaume Cavaliere]";
	mes "Ciao.";
	mes "Cosa vuoi sapere?";
	next;
	switch(select("Candidarsi per KVM.:Cos'è KVM?:Come posso partecipare a KVM?:Voglio sapere i miei Punti Kreiger.")) {
	case 1:
		mes "[Guillaume Cavaliere]";
		if (getbattleflag("feature.bgqueue"))
			mes "Se sei pronto per la battaglia, clicca il pulsante ^0000FFbattle dall'interfaccia^000000 per iscriverti alla battaglia KVM!";
		else {
			mes "Le applicazioni non sono ancora disponibili.";
			mes "Per candidarti, devi andare da un Ufficiale Mercenario KVM.";
		}
		close;
	case 2:
		mes "[Guillaume Cavaliere]";
		mes "KVM è l'abbreviazione di Kreiger Von Midgard.";
		mes "Avventuriero, sei consapevole che la via al nuovo mondo è stata aperta?";
		next;
		mes "[Guillaume Cavaliere]";
		mes "Noi, l'Amministrazione Guillaume, abbiamo diversi piani per avanzare nel nuovo mondo.";
		mes "E KVM è uno di questi.";
		next;
		mes "[Guillaume Cavaliere]";
		mes "Significa che noi, uno dei molti paesi nel continente di Midgard,";
		mes "ha deciso di impiegare molti avventurieri per l'avanzata immediata nel nuovo mondo.";
		next;
		mes "[Guillaume Cavaliere]";
		mes "Pertanto, per selezionare i migliori avventurieri, stiamo tenendo il KVM.";
		mes "Ci aspettiamo risposte da molti avventurieri.";
		next;
		mes "[Guillaume Cavaliere]";
		mes "E daremo loro ricompense per la loro partecipazione!";
		mes "Per quanto riguarda i premi, si prega di contattare un Ufficiale Logistico KVM.";
		close;
	case 3:
		mes "[Guillaume Cavaliere]";
		if (getbattleflag("feature.bgqueue")) {
			mes "In pratica, KVM è una battaglia 3 contro 3.";
			mes "Clicca sul pulsante ^0000FFbattle dall'interfaccia^000000, come gruppo o individualmente.";
		} else {
			mes "In pratica, KVM è una battaglia 5 contro 5.";
			mes "Innanzitutto, ti candidi con un receptionist KVM, in gruppo o individualmente.";
		}
		next;
		mes "[Guillaume Cavaliere]";
		if (getbattleflag("feature.bgqueue"))
			mes "Le applicazioni di gruppo sono per quando hai intenzione di entrare nel KVM con i membri del tuo gruppo o gilda,";
		else
			mes "Le applicazioni di gruppo sono per quando hai intenzione di entrare nella KVM con i membri del tuo gruppo,";
		mes "e un'applicazione personale è per quando si ha intenzione di entrare nel KVM individualmente.";
		next;
		mes "[Guillaume Cavaliere]";
		if (getbattleflag("feature.bgqueue"))
			mes "Please be noted, once you cancel you ^0000FFcannot apply again for 1 minute^000000.";
		else {
			mes "Si prega di candidarsi con un ufficiale KVM, e lui la contatterà più tardi quando sarà a Prontera.";
			next;
			mes "[Guillaume Cavaliere]";
			mes "Poi entri e segui le istruzioni sul campo di battaglia.";
			next;
			mes "[Guillaume Cavaliere]";
			mes "Tuttavia, si prega di notare che a meno che non vi troviate nell'ufficio KVM, non potrà contattarvi.";
		}
		close;
	case 4:
		mes "[Guillaume Cavaliere]";
		mes "I tuoi Punti Kreiger sono:";
		mes ""+kvm_point+".";
		close;
	}
}
