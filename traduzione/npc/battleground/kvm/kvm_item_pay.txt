PurchaseItem:
	// Pricing for Weapon, Suit1, Muffler1, Shoes1, Ring1m Suit2, Shoes2, Ring2, Suit3, Shoes3, Ring3
	setarray .@prices[0],2000,840,630,580,1200,120,70,200,15,10,30;

	.@item_id = getarg(0);
	.@req_setting = getarg(1);
	.@item_type = getarg(2);

	mes "[Logistica]";

	if (!.@item_type)
		mes "Hai selezionato ^ff0000"+ getitemname(.@item_id) +"^000000.";
	else if (.@item_type == 1)
		mes "C'è solo uno "+ callfunc("F_OttieniTipoArma",.@item_id) +": "+ getitemname(.@item_id) +".";
	else if (.@item_type == 2) {
		mes "Hai selezionato '"+ callfunc("F_OttieniTipoArmatura",.@item_id) +"'.";
		mes "L'armatura per chi ha un Livello superiore a 80 è: ^ff0000"+ getitemname(.@item_id) +"^000000.";
	}
	else if (.@item_type == 3) {
		mes "Hai selezionato 'Produzione di massa "+ callfunc("F_OttieniTipoArmatura",.@item_id) +"'.";
		mes "Esiste un solo Mass-Production "+ callfunc("F_OttieniTipoArmatura",.@item_id) +": ^ff0000"+ getitemname(.@item_id) +"^000000.";
	}
	else if (.@item_type == 4) {
		mes "Hai selezionato 'Popularizzato "+ callfunc("F_OttieniTipoArmatura",.@item_id) +"'.";
		mes "Esiste solo un Popolarizzato "+ callfunc("F_OttieniTipoArmatura",.@item_id) +": ^ff0000"+ getitemname(.@item_id) +"^000000.";
	}

	if (!.@req_setting)
		mes "Richiede ^0000ff2,000^000000 Punti KVM, e "+ .@name$ +", i tuoi punti sono: ^580080"+ kvm_point +"^000000.";
	else if (.@req_setting == 4)
		mes "Richiede ^0000ff1,200^000000 Punti KVM, e "+ .@name$ +", i tuoi punti sono: ^580080"+ kvm_point +"^000000.";
	else
		mes "Richiede ^0000ff"+ .@prices[.@req_setting] +"^000000 Punti KVM, e "+ .@name$ +", i tuoi punti sono: ^580080"+ kvm_point +"^000000.";

	mes "Sei sicuro di voler questo articolo?";
	next;
	switch(select("No, non lo comprerò.:Sì, lo comprerò.")) {
	case 1:
		mes "[Logistica]";
		mes "Hai selezionato 'Non lo comprerò'.";
		mes "Quando si acquista un articolo, si prega di fare attenzione che non ci sono rimborsi.";
		break;
	case 2:
		if (kvm_point >= .@prices[.@req_setting]) {
			kvm_point = kvm_point -.@prices[.@req_setting];
			getitem .@item_id,1;
			mes "[Logistica]";
			mes "Hai acquistato un "+ getitemname(.@item_id) +".";
			if (!.@req_setting)
				mes "I tuoi Punti KVM sono ridotti di ^0000ff"+ .@req_setting +"^000000 points, your KVM Points are now ^580080"+kvm_point+"^000000.";
			if (.@req_setting == 4)
				mes "I tuoi Punti KVM sono ridotti di ^0000ff1,200^000000 punti, i tuoi Punti KVM sono ora ^580080"+kvm_point+"^000000.";
			else
				mes "I tuoi Punti KVM sono ridotti di ^0000ff"+ .@prices[.@req_setting] +"^000000 points, your KVM Points are now ^580080"+kvm_point+"^000000.";
		}
		else {
			mes "[Logistica]";
			if (!.@req_setting)
				mes "Ti servono ^0000ff2,000^000000 Punti KVM per acquistare questo articolo.";
			else if (.@req_setting == 4)
				mes "Ti servono ^0000ff1,200^000000 Punti KVM per acquistare questo articolo.";
			else
				mes "Hai bisogno di ^0000ff"+ .@prices[.@req_setting] +"^000000 Punti per acquistare questo articolo.";
			mes "Tuttavia, i tuoi Punti KVM sono ora ^580080"+ kvm_point +"^000000.";
			mes "Non sono abbastanza per comprarlo.";
			mes "Quando avrai abbastanza punti, per favore torna di nuovo.";
		}
	}
	close;
