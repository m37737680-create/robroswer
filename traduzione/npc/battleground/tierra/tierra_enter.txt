bat_room,124,178,5	script	Tierra Gorge Officer#01a	418,{
	if (getbattleflag("feature.bgqueue"))
		end;
	if (checkweight(1201,1) == 0) {
		mes "Aspetta un minuto!!";
		mes "Attualmente stai portando";
		mes "troppi oggetti con te.";
		mes "Per favore riprova";
		mes "dopo che perderai un po' di peso.";
		close;
	}
	mes "Ufficiale dell'Esercito Guillaume";
	mes "Mostriamo il potere dell'Esercito di Guillaume a quei puzzolenti Creoli!";
	next;
	switch(select("Voglio unirmi al vostro esercito!:Fine Conversazione")) {
	case 1:
		if ((BaseJob == Job_Novice) || (BaseJob == Job_Super_Novice)) {
			mes "Ufficiale dell'Esercito Guillaume";
			mes "Sono molto lieto che vogliate unirvi a noi e combattere l'Esercito Croce, ma mi dispiace: non posso mandare un neofita come te a morire sul crudele campo di battaglia.";
		}
		else {
			if (BaseLevel < 80) {
				mes "Ufficiale dell'Esercito Guillaume";
				mes "Sono molto lieto che vogliate unirvi a noi e combattere l'Esercito Croce, ma mi dispiace: non posso mandare un neofita come te a morire sul crudele campo di battaglia.";
			}
			else {
				set .@chk_urtime,checkquest(2069,PLAYTIME);
				if ((.@chk_urtime == 0)|| (.@chk_urtime == 1)) {
					mes "Ufficiale dell'Esercito Guillaume";
					mes "Sembra di essere appena tornato dal campo di battaglia.";
					mes "È troppo presto per te tornare. Vai a riposare e lascia i nemici a noi!";
				}
				else {
					if (.@chk_urtime == 2)
						erasequest 2069;
					set .@mapcount,getmapusers("bat_a01");
					if (.@mapcount > 0) {
						mes "Ufficiale dell'Esercito Guillaume";
						mes "Ho ricevuto un rapporto che mi informa che un corpo d'élite è stato inviato sul campo di battaglia per combattere già i Croixs.";
						mes "Perché non vai ad aspettare un po'? Ti suggerisco di affilare le tue armi e preparare i tuoi rifornimenti fino ad allora.";
					}
					else {
						if ($@TierraBG1) {
							mes "Ufficiale dell'Esercito Guillaume";
							mes "Un corpo d'élite è già pronto per essere inviato sul campo di battaglia.";
							mes "Perché non vai ad aspettare un po'? Ti suggerisco di affilare le tue armi e preparare i tuoi rifornimenti fino ad allora.";
						}
						else {
							mes "Ufficiale dell'Esercito Guillaume";
							mes "Sei decisamente pronto per la battaglia!";
							mes "Vai e mostra ai Croixs cosa significa veramente la paura!";
							mes "Oggi, il nostro grido di vittoria riecheggerà su tutto il campo di battaglia!";
							close2;
							warp "bat_room",57,223;
							end;
						}
					}
				}
			}
		}
		break;
	case 2:
		mes "Ufficiale dell'Esercito Guillaume";
		mes "Oggi saremo vittoriosi!";
		break;
	}
	close;
}

bat_room,127,178,5	script	Guillaume Knight#1	417,{
	mes "[Guillaume Cavaliere]";
	mes "Tierra Gorge è composto da due versanti ripidi disposti verticalmente, e dispone di depositi di razioni per gli eserciti Guillaume e Croix nelle direzioni delle 11 e delle 7.";
	next;
	mes "[Guillaume Cavaliere]";
	mes "La battaglia inizia dalla nave del tuo esercito e l'obiettivo è avanzare e distruggere il deposito razioni del nemico più velocemente di quanto riescano a distruggere il vostro.";
	next;
	mes "[Guillaume Cavaliere]";
	mes "Il guaritore del campo di battaglia risusciterà i soldati una volta ogni 25 secondi affinché possano rientrare nella battaglia.";
	mes "L'esercito che cattura la bandiera neutrale al centro del campo di battaglia riceverà punti di rigenerazione extra, il che significa che i suoi soldati risusciteranno più del lato opposto, dandogli un vantaggio.";
	next;
	mes "[Guillaume Cavaliere]";
	mes "Quei due depositi di razioni sono collegati tra loro attraverso sentieri brevi e stretti, ma i cancelli sono bloccati con barricate.";
	mes "Pensa e muoviti il più velocemente possibile. La vittoria del tuo esercito dipende dal tuo contributo.";
	next;
	mes "[Guillaume Cavaliere]";
	if (getbattleflag("feature.bgqueue"))
		mes "Se sei pronto, inizia cliccando il pulsante ^0000FFbattle dall'interfaccia^000000.";
	else {
		mes "Ricorda: quando ti unisci a una battaglia, riceverai un token che indica la durata per cui non potrai partecipare allo stesso tipo di battaglia.";
		mes "Puoi controllare la durata premendo i tasti Alt+U.";
	}
	next;
	mes "[Guillaume Cavaliere]";
	if (getbattleflag("feature.bgqueue"))
		mes "Tieni presente che una volta annullato, ^0000FFpotresti non poter ripresentare per 1 minuto^000000!";
	else
		mes "Sei pronto per la battaglia? Allora fai domanda con il reclutatore accanto a me!";
	close;
}
