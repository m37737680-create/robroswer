		mes "[Aibakthing]";

		next;
		mes "[Aibakthing]";
		mes "La nostra cucina tradizionale è famosa in tutto il mondo. Non sorprende quando serviamo piatti come il Tom Yum Goong, fatto di Gamberi, Limone e Peperoncini, che ha un aroma e un sapore irresistibili oltre l'immaginazione.";
		next;
		mes "[Aibakthing]";
		mes "Per coloro che bramano l'avventura, esiste una storia antica sul Tigre Sa-mhing, una creatura malvagia che tormenta le rovine del nostro tempio.";
		next;
		mes "[Aibakthing]";
		mes "Hoo! Hahh! Hoo! Hahh!";
		mes "Nessun uomo vivo può resistere al";
		mes "chiamata di questa bella terra.";
		mes "I Midgardiani sono";
		mes "sempre benvenuto!";
		close;
	case 2:
		mes "[Aibakthing]";
		mes "Devi pagare 10.000 Zeny";
		mes "se desideri visitare Ayutthaya.";
		mes "Se sei pronto, possiamo partire";
		mes "adesso~";
		next;
		if (select("Sono pronto, partiamo!:No.") == 1) {
			if (Zeny > 9999) {
				mes "[Aibakthing]";
				mes "Hoo! Hah! Hoo! Hah!";
				mes "Andiamo via! Tornato al mio";
				mes "bellissima Ayutthaya!";
				close2;
				set Zeny, Zeny-10000;
				warp "ayothaya",149,71;
				end;
			}
			mes "[Aibakthing]";
			mes "Mi dispiace, ma non hai i 10.000 Zeny per viaggiare ad Ayutthaya. Un prezzo del genere è nulla rispetto alle esperienze che ti aspettano!";
			close;
		}
	case 3:
		mes "[Aibakthing]";
		mes "Hai mai sognato";
		mes "di un bellissimo posto pieno";
		mes "con serena spiritualità? Tu";
		mes "deve aver sognato";
		mes "di Ayutthaya, amico mio.";
		close;
	}
}
