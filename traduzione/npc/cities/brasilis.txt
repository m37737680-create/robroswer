//============================================================
-	script	::Crewman_bra2	-1,{
	mes "[Marinaro]";
	mes "Ehi, hai sentito parlare di un posto chiamato Brasilis?";
	mes "È una città tropicale che è calda come il deserto ma anche piovosa. È un posto molto misterioso.";
	next;
	mes "[Marinaro]";
	mes "Abbiamo recentemente trovato una nuova rotta oceanica per arrivarci facilmente.";
	if (VIP_SCRIPT) {
		set .@cost, vip_status(VIP_STATUS_ACTIVE)?1000:10000;
		mes "Sono solo 10.000 zeny per un viaggio di andata e ritorno, e 1.000 per il VIP! Quindi vuoi andare?"; //custom translation (VIP)
	} else {
		set .@cost,10000;
		mes "Sono solo 10.000 zeny per un viaggio di andata e ritorno! Quindi vuoi andare?";
	}
	next;
	switch(select("Portami a Brasilia!:Resto qui.")) {
	case 1:
		if (Zeny < .@cost) {
			mes "[Marinaro]";
			mes "Ho detto "+callfunc("F_InserisciVirgola",.@cost)+" zeny.";
			close;
		}
		else {
			mes "[Marinaro]";
			mes "Cool~!! Andiamo~!";
			set Zeny, Zeny-.@cost;
			close2;
			warp "brasilis",314,60;
			end;
		}
	case 2:
		mes "[Marinaro]";
		mes "Se ti interessa, fammi sapere e posso portarti lì.";
		close;
	}
}

brasilis,137,77,5	script	Ice-Cream Maker	85,{
	mes "Gelatiera";
	mes "Come~come~";
	mes "Il gelato è lo spuntino perfetto per una giornata calda~";
	mes "È solo ^3355FF100 Zeny^000000~";
	mes "Gelato~";
	mes "Prendi il tuo gelato!";
	next;
	switch(select("Dammi uno!:Gelato?:Annulla.")) {
	case 1:
		mes "Gelatiera";
		mes "Dato che ci sono così tante persone che vogliono un gelato fantastico, puoi ordinarne solo 5 alla volta.";
		mes "Quanti ne vuoi?";
		next;
		while(1) {
			input .@input; //,1,5;
			if (.@input == 0) {
				mes "Gelatiera";
				mes "Niente?";
				mes "Via, schivati, ho clienti da servire.";
				close;
			}
			else if ((.@input < 0) || (.@input > 5)) {
				mes "Gelatiera";
				mes "Wow.";
				mes "Ne hai ordinato troppo.";
				mes "Se ne mangi più di 5 potresti dover combattere con un mostro nello stomaco. Calmati amico.";
				next;
			}
			else
				break;
		}
		set .@icecream_hap,.@input*100;
		if (Zeny < .@icecream_hap) {
			mes "Gelatiera";
			mes "Dood~! Non hai abbastanza soldi.";
			mes "È solo ^3355FF100 Zeny^000000~ Seriamente!";
			close;
		}
		if (!checkweight(536,.@input)) {
			mes "Gelatiera";
			mes "Sembra che tu abbia troppe cose.";
			mes "Alleggerisci il tuo zaino prima di comprare questo.";
			close;
		}
		set Zeny, Zeny-.@icecream_hap;
		getitem 536,.@input; //Ice_Cream
		close;
	case 2:
		mes "Gelatiera";
		mes "Il gelato è...";
		mes "Aspetta, non lo sai";
		mes "Cos'è il gelato?";
		mes "Quale roccia hai";
		mes "vissuto sotto?";
		next;
		mes "Gelatiera";
		mes "Non voglio nemmeno iniziare con quanto suona strano.";
		mes "Comunque, prendi il tuo gelato proprio qui mentre è bello fresco.";
		close;
	case 3:
		mes "Gelatiera";
		mes "Non perdere l'occasione di mangiare il miglior gelato di tutto il paese~!";
		close;
	}
}
