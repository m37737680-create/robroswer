-	script	::Sailor_izlude	-1,{
	mes "[Marinaro]";
	mes "Ciao a tutti!";
	mes "Attenzione, attenzione!";
	mes "Vieni e cavalca il vento";
	mes "su una affascinante nave!";
	mes "Sbrigati, sbrigati!";
	next;
	switch(select("Isola Byalan -> 150 Zeny.:Alberta Marina -> 500 Zeny.:Annulla.")) {
	case 1:
		if (Zeny < 150) {
			mes "[Marinaro]";
			mes "150 Zeny!";
			mes "Solo 150 Zeny per cavalcare!";
			close;
		}
		set Zeny, Zeny - 150;
		warp "izlu2dun",107,50;
		end;
	case 2:
		if (Zeny < 500) {
			mes "[Marinaro]";
			mes "500 Zeny!";
			mes "Solo 500 Zeny per cavalcare!";
			close;
		}
		set Zeny, Zeny - 500;
		warp "alberta",188,169;
		end;
	case 3:
		close;
	}
}
