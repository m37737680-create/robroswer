-	script	::PR_Officer_moscovia	-1,{
	mes "Ufficiale di Moscovia P.R.";
	mes "Mosca! Conosci Mosca?";
	mes "il paradiso che si espande sui mari infiniti...";
	mes "Benvenuti a Moscovia";
	mes "È avventuroso e mistico.";
	next;
	switch(select("Su Moscovia...:Vai a Moscovia.:Annulla")) {
		case 1:
			mes "Ufficiale di Moscovia P.R.";
			mes "Mosca è un bellissimo regno";
			mes "su un'isola situata a nord di Rune-";
			mes "Midgardi";
			next;
			mes "Ufficiale di Moscovia P.R.";
			mes "Sono sicuro che lo sarai";
			mes "assolutamente affascinato";
			mes "della bellissima scenografia di Mosca";
			mes "e splendido palazzo.";
			next;
			mes "Ufficiale di Moscovia P.R.";
			mes "Ora che il nostro lungo inverno ha";
			mes "superato,";
			mes "Sono felice di poterti mostrare ora";
			mes "la nostra splendida città natale.";
			next;
			mes "Ufficiale di Moscovia P.R.";
			mes "Se hai voglia di visitare la Moscovia";
			mes "cogli l'occasione ora!";
			mes "Ti aiuterò ad avere un bel viaggio";
			mes "a Mosca!";
			close;

		case 2:
			mes "Ufficiale di Moscovia P.R.";
			mes "Allora, cominciamo adesso.";
			mes "Dovresti pagarmi 10.000 zeny";
			mes "andare a Mosca";
			mes "Ma quando tornerai,";
			mes "Non devi pagare.";
			next;
			mes "Ufficiale di Moscovia P.R.";
			mes "Possiamo andare via adesso?";
			next;
			if (select("Andiamo!:Annulla") == 2) {
				mes "Ufficiale di Moscovia P.R.";
				mes "Se sei troppo impegnato ora,";
				mes "dimmi di nuovo quando vuoi.";
				mes "Sono sempre pronto a guidare chiunque a Moscovia.";
				close;
			}
			if (Zeny < 10000) {
				mes "Ufficiale di Moscovia P.R.";
				mes "Mi dispiace ma non hai";
				mes "abbastanza zeny ora";
				mes "Hai bisogno di 10.000 zeny";
				mes "andare a Mosca";
				mes "Grazie.";
				close;
			} else {
				mes "Ufficiale di Moscovia P.R.";
				mes "Allora, partiamo adesso.";
				close2;
				set Zeny,Zeny - 10000;
				warp "moscovia",163,55;
				end;
			}
		case 3:
			mes "Ufficiale di Moscovia P.R.";
			mes "Se sei troppo impegnato ora,";
			mes "dimmi di nuovo quando vuoi.";
			mes "Sono sempre pronto a guidare chiunque a Moscovia.";
			close;
	}
}
