//= rAthena Project
//===== Description: =========================================
//= Removes cards from equipped items.
//===== Additional Comments: =================================
//= 1.0 First version. [TyrNemesis^]
//= 1.2 Optimized and fixed getequipname menu. [Kisuka]
//= 1.2a Added 'disable_items' command. [Euphy]
//============================================================

prt_in,28,73,4	script	Wise Old Woman#eAcustom	78,{

	set .zenycost,200000;    // base cost of the card remover services (in Zeny)
	set .percardcost,25000;  // cost per card of the card remover services (in Zeny)
	set .faildestroy,1;      // should the card remover have a chance of failure that destroys items? (1=yes, 0=no)

	disable_items;
	mes "[Donna saggia]";
	mes "Buona giornata, giovane. Ho il potere di rimuovere le carte che hai compattato sul tuo equipaggiamento. Questa idea ti piace?";
	next;
	switch(select("Sì, lo fa.:Quanto costi?:No grazie.")) {
	case 1:
		mes "[Donna saggia]";
		mes "Molto bene. Quale articolo dovrei esaminare per lei?";
		next;

		setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
		for( set .@i,1; .@i <= 10; set .@i,.@i+1 ) {
			if( getequipisequiped(.@indices[.@i]) )
				set .@menu$, .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			set .@menu$, .@menu$ + ":";
		}
		set .@part, .@indices[ select(.@menu$) ];
		if(!getequipisequiped(.@part)) {
			mes "[Donna saggia]";
			mes "Giovane... Non indossi nulla lì da cui potrei rimuovere carte.";
			close;
		}
		if(getequipcardcnt(.@part) == 0) {
			mes "[Donna saggia]";
			mes "Giovane... Non ci sono carte composte su questo articolo. Non posso fare nulla con esso, temo.";
			close;
		}
		set .@cardcount,getequipcardcnt(.@part);
		
		if (!checkweight(1202,(.@cardcount+1))) { 
			mes "^3355FFUn momento!";
			mes "Non posso offrire nessuno dei miei";
			mes "servizi a te perché";
			mes "porti troppo peso";
			mes "cose. Metti i tuoi articoli extra in";
			mes "Stoccaggio Kafra e ancora~";
			close;
		}
		mes "[Donna saggia]";
		mes "Questo articolo ha " + .@cardcount + " carte su cui sono composte. Per eseguire la mia magia, avrò bisogno " + (.zenycost+(.@cardcount * .percardcost)) + " zeny, a ^0000FFStar Crumb^000000, and a ^0000FFYellow Gemstone^000000.";
		next;
		if(select("Molto bene. Fallo.:Non importa.") == 2) {
			mes "[Donna saggia]";
			mes "Molto bene. Ritorna subito se cerchi i miei servizi.";
			close;
		}
		if((zeny < (.zenycost+(.@cardcount * .percardcost))) || (countitem(1000) < 1) || (countitem(715) < 1)) {
			mes "[Donna saggia]";
			mes "Non hai tutti gli oggetti di cui ho bisogno per compire la mia magia, bambina. Torna quando li avrai.";
			close;
		}
		mes "[Donna saggia]";
		mes "Prima di iniziare, devo avvertirti—potrei fallire. Se lo faccio, potrei distruggere le carte, l'oggetto o entrambi. Non offro rimborsi. Detto questo, cosa è più importante per te: Le carte, o l'oggetto?";
		next;
		switch(select("Ho cambiato idea su questo.:L'articolo.:Le carte.")) {
		case 1:
			mes "[Donna saggia]";
			mes "Molto bene. Ritorna subito se cerchi i miei servizi.";
			close;
		case 2:
			set .@failtype,1;
			break;
		case 3:
			set .@failtype,2;
		}
		mes "[Donna saggia]";
		mes "Molto bene. Inizierò.";
		set Zeny, Zeny - (.zenycost+(.@cardcount * .percardcost));
		delitem 1000,1; //Star_Crumb
		delitem 715,1; //Yellow_Gemstone
		
		// Replace the constants in the next 3 lines with failure chance values defined in refine_db.txt
		// First value = Total failure chance (item and cards destroyed)
		// Second value = Partial failure chance (one or the other is destroyed, player decides which one is safe)
		// Third value = Harmless failure chance (all that's lost is your investment)

		set .@failchance,rand(100);
		if (.faildestroy==1) {
			if(.@failchance < 2) {
				next;
				failedremovecards .@part,0;
				mes "[Donna saggia]";
				mes "Il processo è stato un fallimento totale. Temo che l'articolo e le carte siano stati distrutti.";
				close;
			}

			if(.@failchance < 8) {
				if (.@failtype == 1) {
					next;
					failedremovecards .@part,1;
					mes "[Donna saggia]";
					mes "Sebbene io sia riuscito a rimuovere le carte dall'oggetto, sono state distrutte nel processo. L'oggetto, tuttavia, va bene.";
					close;
				}

				if (.@failtype == 2) {
					next;
					failedremovecards .@part,2;
					mes "[Donna saggia]";
					mes "Più sfortunato. Sono riuscito a rimuovere le carte, ma l'oggetto stesso è stato distrutto nel processo.";
					close;
				}
			}
		}

		if(.@failchance < 10) {
			next;
			failedremovecards .@part,3;
			mes "[Donna saggia]";
			mes "Non sono riuscito a rimuovere le carte. Fortunatamente, sia l'oggetto che le carte stanno ancora bene.";
			close;
		}
		next;
		successremovecards .@part;
		mes "[Donna saggia]";
		mes "Il processo è stato un successo. Ecco le vostre carte e il vostro articolo. Addio.";
		close;
	case 2:
		mes "[Donna saggia]";
		mes "addebito una tariffa fissa di "+callfunc("F_InserisciVirgola",.zenycost)+" zeny, più "+callfunc("F_InserisciVirgola",.percardcost)+" zeny per ogni carta che rimuovo dall'oggetto. Inoltre, ho bisogno di una briciola di stella e un gemme giallo per fare la mia magia.";
		close;
	case 3:
		mes "[Donna saggia]";
		mes "Molto bene. Ritorna subito se cerchi i miei servizi.";
		close;
	}
}
