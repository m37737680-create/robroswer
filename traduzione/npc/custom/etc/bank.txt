//= An account wide Banker to store Zeny
//===== Additional Comments: ================================= 
//= Syrus22 - There's an optional transaction fee at the top of
//= the script. To use it simply change the first set command
//= to set the cost variable to whatever you want the fee to be.
//= Version 2.0: Optimized and brought the script up to standard. [Jguy]
//============================================================ 

prontera,132,217,5	script	Banker	109,{
	set @cost,500;
	mes "Banchiere";
	mes "Benvenuti alla Prima Banca di Prontera. Come posso aiutarvi oggi?";
	next;
	switch(select("Vorrei fare un deposito.:Vorrei fare un prelievo.:Qual è il mio saldo attuale?:Annulla")) {
	case 1:
		mes "Banchiere";
		mes "Molto bene... Quanto vorresti depositare? Il massimo che puoi depositare in una volta sola è 999.999 Zeny.";
		next;
		if (@cost > 0) {
			mes "Banchiere";
			mes "Oh, e renditi conto che c'è un " +@cost + " Tassa Zeny su tutte le transazioni!";
			next;
		}
		input @deposit;
		if (@deposit < 1) {
			mes "Assicurati di chiedermi di depositare un importo reale.";
			close;
		}
		else if (@deposit > Zeny) {
			mes "Non sembra che tu abbia la quantità di zeny che stai cercando di depositare!";
			close;
		}
		else if (@deposit > (Zeny - @cost)) {
			mes "Hai bisogno di " + @cost + " Zeny per coprire la commissione di transazione!";
			close;
		}
		else {
			set Zeny,Zeny - @deposit;
			set Zeny,Zeny - @cost;
			set #bankstorage,#bankstorage + @deposit;
			mes "Banchiere";
			mes "Grazie mille... Il tuo zeny è in buone mani.";
			close;
		}
	case 2:
		mes "Banchiere";
		mes "Molto bene... Quanto vorresti prelevare? Il massimo che puoi prelevare in una volta è 999.999 Zeny";
		next;
		if (@cost > 0) {
			mes "Banchiere";
			mes "Oh, e renditi conto che c'è un " +@cost + " Tassa Zeny su tutte le transazioni!";
			next;
		}
		input @withdrawl;
		if (@withdrawl < 1) {
			mes "Per favore non fare giochi. Ho bisogno di una vera somma da prelevare.";
			close;
		}
		else if (@withdrawl > #bankstorage) {
			mes "Hai solo ^00FF00" + callfunc("F_InserisciVirgola",#bankstorage) +"^000000 zeny nel tuo conto!";
			close;
		}
		else if ((@cost > Zeny) && ((Zeny + @withdrawl) > @cost)) {
			mes "Banchiere";
			mes "Non hai i Zeny per la commissione di transazione al momento. Vorresti che prenda io la commissione direttamente dal tuo prelievo?";
			next;
			switch(select("Sì per favore.:No, Grazie")){
			case 1:
				mes "Banchiere";
				mes "Rimuovere " + @cost + " dal tuo prelievo per coprire la commissione di deposito...";
				set @withdrawl,@withdrawl - @cost;
				set #bankstorage,#bankstorage - @cost;
				set @cost,0;
				next;
				set Zeny,Zeny - @cost;
				set Zeny,Zeny + @withdrawl;
				set #bankstorage,#bankstorage - @withdrawl;
				mes "Banchiere";
				mes "Ecco i tuoi Zeny. Buona giornata.";
				close;
			case 2:
				mes "Banchiere";
				mes "Molto bene... torna presto.";
				close;
			}
		}
		else {
			set Zeny,Zeny - @cost;
			set Zeny,Zeny + @withdrawl;
			set #bankstorage,#bankstorage - @withdrawl;
			mes "Banchiere";
			mes "Ecco i tuoi Zeny. Buona giornata.";
			close;
		}
	case 3:
		mes "Banchiere";
		mes "Hmmmm... devo controllare qualche documento.";
		next;
		mes "Sussurro, Sussurro";
		next;
		mes "Banchiere";
		mes "Hai attualmente " + callfunc("F_InserisciVirgola",#bankstorage) + " Zenny nel tuo conto.";
		close;
	case 4:
		mes "Banchiere";
		mes "Molto bene... torna presto.";
		close;
	}
}
