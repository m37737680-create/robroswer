//===== Description: =========================================
//= Black Jack card game.  Gameplay based off standard casino
//= black jack rules.  Dealer must have at least 17 to stay and will
//= automatically stay at 17 and up.  Player must have at least
//= 13 to stay.  Aces counted as 11 or 1.  Option to "Double Down".
//= Currently does not allow for "insurance", or "splitting"
//= of pairs.
//===== Additional Comments: =================================
//= 1.0 First version.
//============================================================

cmd_in02,188,89,1	script	Black Jack Dealer	57,{
	mes "[Mazziere]";
	mes "Ciao! Ti piacerebbe giocare a Blackjack?";
M_Menu:
	next;
	menu "Quali sono le regole?",M_0, "Voglio giocare.",L_Play, "No grazie.",M_End;

M_0:
	mes "[Mazziere]";
	mes "Cosa vorresti sapere?";
sM_Menu0:
	next;
	menu "Le basi.",sM_0a, "Vincere e perdere.",sM_0b, "Raddoppiare la posta",sM_0c,
	"Valore Asso.",sM_0d, "Nulla.",M_Menu;

sM_0a:
	mes "[Mazziere]";
	mes "Black Jack è un gioco di carte in cui l'obiettivo è raggiungere 21, o come";
	mes "il più vicino possibile a 21 punti, senza superare i 21.";
	next;
	mes "[Mazziere]";
	mes "Prima che inizi la partita, il giocatore deve fare una puntata. Una volta fatta la puntata";
	mes "è stato distribuito, sia al concessionario che al giocatore vengono distribuite 2 carte";
	mes "Ogni. A seconda delle carte che hai, puoi scegliere di ^5533FF'stay'^000000,";
	mes "o puoi scegliere di ^5533FF'tirare'^000000.";
	next;
	mes "[Mazziere]";
	mes "Quando scegli di 'restare', stai dicendo al croupier che non";
	mes "ho bisogno di altre carte. Questo permette al croupier di tirare se lui/lei";
	mes "wants to.  In order to stay, you must have ^FF3355at least 13 points^000000. The";
	mes "il concessionario può rimanere solo quando ha^FF3355 17 punti o più^000000.";
	next;
	mes "[Mazziere]";
	mes "Quando scegli di 'tirare', stai dicendo al croupier che vuoi";
	mes "un'altra carta. Tirando più carte puoi aumentare il tuo punto";
	mes "totale. Finché hai ^FF3355minori di 21^000000 punti puoi tirare un";
	mes "carta dal mazzo.";
	goto sM_Menu0;
sM_0b:
	mes "[Mazziere]";
	mes "Ci sono 3 modi per vincere e perdere al Blackjack.";
	next;
	mes "[Mazziere]";
	mes "Alla fine di un turno di Blackjack, se hai più punti";
	mes "che il concessionario vinci il turno. Viceversa se hai";
	mes "meno punti del dealer perderai.";
	next;
	mes "[Mazziere]";
	mes "Se tiri una carta che fa superare il tuo punteggio totale i 21 tu";
	mes "perdere automaticamente il turno. Questo è chiamato un 'bust'^5533FF'bust'^000000. Se";
	mes "il casinò busta e allora vincerai il round.";
	next;
	mes "[Mazziere]";
	mes "Se hai un totale di punti di 21 con le prime 2 carte, tu";
	mes "will automatically win the round.  This is called a ^5533FF'Black Jack'^000000";
	mes "e succede quando prendi un 'As' e una carta con valore '10'. Se il";
	mes "il croupier fa un Blackjack e vincerà automaticamente il turno.";
	next;
	mes "[Mazziere]";
	mes "4.) Oltre a vincere e perdere, puoi pareggiare con il dealer. Se";
	mes "sia tu che il croupier avete lo stesso totale di punti alla fine di un";
	mes "arrotondato, questo risulterà in un pareggio senza vincitore e senza perdita o guadagno";
	mes "in denaro.";
	next;
	mes "[Mazziere]";
	mes "This is called a ^5533FF'push'^000000 with the dealer.  This also";
	mes "si applica sia a te che al concessionario che hanno Blackjack allo stesso";
	mes "tempo.";
	goto sM_Menu0;
sM_0c:
	mes "[Mazziere]";
	mes "L'opzione 'Raddoppiare' ti permette di raddoppiare la tua puntata attuale,";
	mes "ma con lo svantaggio che potrai tirarne solo uno";
	mes "carta aggiuntiva. Questa opzione è disponibile solo all'inizio di";
	mes "ogni turno.";
	next;
	mes "[Mazziere]";
	mes "Un esempio di quando puntare tutto è utile è quando";
	mes "i tuoi primi 2 carte ti danno un totale di punti di 11. Hai un buon";
	mes "probabilità di ottenere 21 o 20 con la prossima carta che peschi. Questo";
	mes "sarebbe un buon asso per puntare tutto.";
	goto sM_Menu0;
sM_0d:
	mes "[Mazziere]";
	mes "La carta 'Asso' è una carta unica nel gioco del Blackjack perché";
	mes "può avere 2 valori. Un Asso può essere contato come 11 punti,";
	mes "o solo 1 punto. Per esempio se avessi un As e un 4, sarebbe";
	mes "ti do o 15 o 5 punti.";
	next;
	mes "[Mazziere]";
	mes "Se avessi deciso di restare, l'As verrebbe automaticamente contato come";
	mes "11 punti per darti un totale di 15 punti.";
	next;
	mes "[Mazziere]";
	mes "Se avessi deciso di tirare e avessi ricevuto un 9, l'asso avrebbe";
	mes "automaticamente contati come 1 punto per darti un totale di 14";
	mes "punti. Se l'As fosse contato come 11 punti, avresti un";
	mes "punteggio totale sopra i 21 e avrei perso.";
	next;
	mes "[Mazziere]";
	mes "È a causa della flessibilità che hai con l''Ace' che fa";
	mes "è la carta più potente del gioco.";
	goto sM_Menu0;
M_End:
	mes "[Mazziere]";
	mes "Sentiti libero di tornare quando vuoi";
	close;

//================
L_Play:
	mes "[Mazziere]";
	mes "Scommetti pure...";
	next;
	menu "2z",M_1a, "10z",M_1b, "20z",M_1c, "100z",M_1d, "Troppo ricco per il mio sangue...",M_End;

M_1a:
	if(Zeny < 2) goto sL_NotEnuf;
	set @bet, 2;
	goto L_Cont0;
M_1b:
	if(Zeny < 10) goto sL_NotEnuf;
	set @bet, 10;
	goto L_Cont0;
M_1c:
	if(Zeny < 20) goto sL_NotEnuf;
	set @bet, 20;
	goto L_Cont0;
M_1d:
	if(Zeny < 100) goto sL_NotEnuf;
	set @bet, 100;
	goto L_Cont0;

sL_NotEnuf:
	mes "[Mazziere]";
	mes "Mi dispiace ma non hai abbastanza zeny per fare quella puntata.";
	close;

L_Cont0:
	mes "(le carte vengono distribuite)";
	next;
	deletearray $@card[0],13;
	set @dealerTurn, 0;
	set @numP, 0;
	set @numD, 0;
	set @pAce, 0;
	set @dAce, 0;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	callsub sF_GetCards, @numD, @dealCard[@numD], @dealCard$[@numD], @dAce;
	callsub sF_GetCards, @numD, @dealCard[@numD], @dealCard$[@numD], @dAce;

//==============
L_Start:
	callsub sF_GetTot;

	mes "Ecco le carte ^FF5533DEALER'S^000000:";
	if (@numD==2) callsub sF_D2cards;
	if (@numD==3) callsub sF_D3cards;
	if (@numD==4) callsub sF_D4cards;
	if (@numD==5) callsub sF_D5cards;
	mes "    Il DEALER ha: ^FF5533"+@dealTot+"^000000";
	mes " ";
	mes "Ecco le carte ^5533FFYOUR^000000:";
	if (@numP==2) callsub sF_P2cards;
	if (@numP==3) callsub sF_P3cards;
	if (@numP==4) callsub sF_P4cards;
	if (@numP==5) callsub sF_P5cards;
	if(@pAce != 1 || @playTot == 21) mes "    Hai: ^5533FF" +@playTot+ "^000000";
	if(@pAce == 1 && @playTot != 21) mes "    Hai: ^5533FF" +@playTot+ "^000000, o ^5533FF" +(@playTot-10)+ "^000000";
	next;
	if(@playTot==21 && @dealTot==21) goto sL_Push;
	if(@numP==2 && @playTot == 21) goto sL_Win;
	if(@numD==2 && @dealTot == 21) goto sL_Lose;
	if(@playTot > 21) goto sL_Lose;
	if(@dealTot > 21) goto sL_Win;
	if(@numP==2 && @dealerTurn==0) menu "Colpami(tirare)",M_Hit, "Rimani",M_Stay, "Raddoppiare la posta",M_Double;
	if(@dealerTurn == 0) menu "Colpami(tirare)",M_Hit, "Rimani",M_Stay;

M_Stay:
	mes "[Mazziere]";
	if(@playTot < 13) goto sL_PlayToLow;
	if(@dealTot > 16) mes "Il concessionario resta.";
	if(@dealTot > 16 || @numD == 5) goto L_Check;
	mes "Il concessionario sta per tirare";
	next;
	callsub sF_GetCards, @numD, @dealCard[@numD], @dealCard$[@numD], @dAce;
	set @dealerTurn, 1;
	goto L_Start;

sL_PlayToLow:
	mes "Mi dispiace ma non hai un totale abbastanza alto per restare. Devi ritirarti.";
	next;
	goto M_Hit;

M_Hit:
	if(@numP == 5) goto M_Stay;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	goto L_Start;
M_Double:
	mes "[Mazziere]";
	mes "Il giocatore ha scelto di raddoppiare. La tua puntata attuale sarà";
	mes "raddoppiato e potrai pescare solo 1 carta extra.";
	next;
	set @dealerTurn, 1;
	set @bet, @bet*2;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	goto L_Start;

//=============
L_Check:
	next;
	if(@playTot < @dealTot) goto sL_Lose;
	if(@playTot == @dealTot) goto sL_Push;

sL_Win:
	mes "[Mazziere]";
	mes "Congratulazioni, hai vinto!";
	next;
	set Zeny, Zeny + @bet;
	goto L_Play;
sL_Lose:
	mes "[Mazziere]";
	mes "Mi dispiace ma hai perso.";
	set Zeny, Zeny - @bet;
	next;
	goto L_Play;
sL_Push:
	mes "[Mazziere]";
	mes "È una spinta. Hai pareggiato con il Dealer.";
	next;
	goto L_Play;


//==================================
	// Sub function for dealing/pulling the cards
sF_GetCards:
	set @rnd, rand(1,13);
	if($@card[@rnd] == 4) goto sF_GetCards;
	set $@card[@rnd], $@card[@rnd] + 1;
	set getarg(1), @rnd;
	if(getarg(1) > 10) set getarg(1), 10;
	if(getarg(1) == 1 && getarg(3) < 1) set getarg(1), 11;
	if(getarg(1) == 11) set getarg(3), 1;
	set getarg(2), " " + getarg(1) + " ";
	if(@rnd == 10) set getarg(2), getarg(1);
	if(@rnd == 1) set getarg(2), " A ";
	if(@rnd == 11) set getarg(2), " J ";
	if(@rnd == 12) set getarg(2), " Q ";
	if(@rnd == 13) set getarg(2), " K ";
	set getarg(0), getarg(0) + 1;
	return;

//==================================
	// Sub function for getting the total score for each hand
sF_GetTot:
	set @i, 0;
	set @dealTot, 0;
	set @playTot, 0;

GetDealTot:
	set @dealTot, @dealTot + @dealCard[@i];
	set @i, @i + 1;
	if(@i < @numD) goto GetDealTot;
	if(@dAce == 1 && @dealTot > 21) set @dealTot, @dealTot - 10;
	set @i, 0;
GetPlayTot:
	set @playTot, @playTot + @playCard[@i];
	set @i, @i + 1;
	if(@i < @numP) goto GetPlayTot;
	if(@pAce == 1 && @playTot > 21) set @pAce, 2;
	if(@pAce > 0 && @playTot > 21) set @playTot, @playTot - 10;
	return;

//=======================================
	// Sub function for displaying the Cards
sF_D2cards:
	mes "    .-----.   .-----. ";
	mes "    | "+@dealCard$[0]+" |   | "+@dealCard$[1]+" | ";
	mes "    '-----'   '-----' ";
	return;
sF_P2cards:
	mes "    .-----.   .-----. ";
	mes "    | "+@playCard$[0]+" |   | "+@playCard$[1]+" | ";
	mes "    '-----'   '-----' ";
	return;
sF_D3cards:
	mes "    .-----.   .-----.   .-----. ";
	mes "    | "+@dealCard$[0]+" |   | "+@dealCard$[1]+" |   | "+@dealCard$[2]+" | ";
	mes "    '-----'   '-----'    '-----' ";
	return;
sF_P3cards:
	mes "    .-----.   .-----.   .-----. ";
	mes "    | "+@playCard$[0]+" |   | "+@playCard$[1]+" |   | "+@playCard$[2]+" | ";
	mes "    '-----'   '-----'    '-----' ";
	return;
sF_D4cards:
	mes " .-----.  .-----.  .-----. ";
	mes " | "+@dealCard$[0]+" |  | "+@dealCard$[1]+" |  | "+@dealCard$[2]+" | ";
	mes " '-----'  '-----'   '-----' ";
	mes "    .-----.  ";
	mes "    | "+@dealCard$[3]+" |";
	mes "    '-----'   ";
	return;
sF_P4cards:
	mes " .-----.  .-----.  .-----. ";
	mes " | "+@playCard$[0]+" |  | "+@playCard$[1]+" |  | "+@playCard$[2]+" |";
	mes " '-----'  '-----'   '-----' ";
	mes "    .-----.  ";
	mes "    | "+@playCard$[3]+" |";
	mes "    '-----'   ";
	return;
sF_D5cards:
	mes " .-----.  .-----.  .-----. ";
	mes " | "+@dealCard$[0]+" |  | "+@dealCard$[1]+" |  | "+@dealCard$[2]+" | ";
	mes " '-----'  '-----'   '-----' ";
	mes "    .-----.   .-----. ";
	mes "    | "+@dealCard$[3]+" |   | "+@dealCard$[4]+" | ";
	mes "    '-----'   '-----' ";
	return;
sF_P5cards:
	mes " .-----.  .-----.  .-----. ";
	mes " | "+@playCard$[0]+" |  | "+@playCard$[1]+" |  | "+@playCard$[2]+" | ";
	mes " '-----'  '-----'   '-----' ";
	mes "    .-----.   .-----. ";
	mes "    | "+@playCard$[3]+" |   | "+@playCard$[4]+" | ";
	mes "    '-----'   '-----' ";
	return;
}
