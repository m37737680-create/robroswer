function SF_Groom {
	if ($wed_bride_progress == 0) {
		mes "["+@name$+"]";
		mes "Sembra che la tua sposa non sia ancora arrivata e si sia registrata.";
		close;
	}
	if (SF_equip_check() == 0)
		close;

	switch($wed_groom_progress) {
	case 1:
		SF_AcceptBride();
		break;
	case 2:
		mes "["+@name$+"]";
		mes "Aspetto che il tuo partner ti accetti per iniziare la cerimonia.";
		close;
	case 3:
		SF_RetrieveRingM();
		break;
	case 4:
		mes "["+@name$+"]";
		mes "L'anello di matrimonio del tuo partner non è ancora stato ritirato. La cerimonia inizierà non appena avrete entrambi reclamato i vostri anelli.";
		close;
	case 5:
		mes "["+@name$+"]";
		SF_StartCeremony();
		break;
	default:
		mes "["+@name$+"]";
		mes "Non interrompermi ora.";
		close;
	}
}

function SF_AcceptGroom {
	mes "["+@name$+"]";
	mes $wed_bride$+", "+$wed_groom$+" has requested to be your husband for the rest of your life. Do you accept?";
	next;
	switch(select("Ho bisogno di tempo per pensarci.","No, non lo faccio!","Sì, lo faccio!")) {
	case 1:
		mes "["+@name$+"]";
		mes "Che cosa!?";
		mes "err.. *tosse* *tosse* molto bene... torna quando ti sarai deciso.";
		emotion ET_ANGER;
		close;
	case 2:
		mes "["+@name$+"]";
		mes "!!";
		mes "Ah... ehm... ok. Sembra che abbiate delle differenze da risolvere tra voi due prima.";
		close2;
		emotion ET_HUK;
		npctalk @name$ +" : Ladies and gentlemen, "+$wed_bride$+" has rejected to marry "+$wed_groom$+"!";
		SF_wed_end();
		break;
	case 3:
		set $wed_bride_progress,2;
		if ($wed_groom_progress == 2) {
			SF_RingsAccepted();
			break;
		}
		emotion ET_OK;
		mes "["+@name$+"]";
		mes "Dopo che il tuo sposo avrà dato l'approvazione, ti verranno consegnati gli anelli, inizierà la cerimonia e sarai ufficialmente sposata.";
		close;
	}
}

function SF_RingsAccepted {

function SF_RetrieveRingM {
	mes "["+@name$+"]";
	if ($@wed_ring && countitem($@wed_ring) < 1) {
		mes "Cosa è successo al tuo "+getitemname($@wed_ring)+"Non l'hai perso... vero? Ne abbiamo bisogno per continuare con la cerimonia!";
		close;
	}
	if ($wed_bride_sex)
		set @item, 2634;	//Groom's wedding ring
	else
		set @item, 2635;	//Bride's wedding ring
	if (getnameditem(@item,$wed_groom$) == 0) {
		mes "Non sembri avere abbastanza spazio per portare l'anello... libera un po' di spazio e torna a reclamare l'anello del tuo partner.";
		close;
	}
	mes "Ecco l'anello da sposa per la tua sposa.";
	if ($@wed_ring) delitem $@wed_ring,1;
	set $wed_groom_progress,4;
	
	if ($wed_bride_progress == 4)
		SF_StartCeremony();
	else {
		mes "Quando la sposa ritroverà l'anello, la cerimonia inizierà.";
		close;
	}
}

function SF_RetrieveRingF {

	else {
		mes "Una volta che lo sposo ritirerà l'anello, la cerimonia inizierà.";
		close;
	}
}

function SF_StartCeremony {
	mes "Inizio ora la cerimonia di matrimonio e voi sarete dichiarati marito e moglie.";
	set $wed_bride_progress,5;
	set $wed_groom_progress,5;
	set $@msg$,$wed_groom$;
	if (strcharinfo(0) == $wed_groom$)
		set $@msg$,$wed_bride$;
	if (marriage($@msg$) == 0) {
		next;
		mes "["+@name$+"]";
		mes "Dove è "+$@msg$+"Non posso sposarvi entrambi se ne manca uno...";
		close;
	}
	set $wed_bride_progress,6;
	set $wed_groom_progress,6;
	initnpctimer;
	close;
}

//Subfunction: Checks that the groom/bride is still wearing their stuff.
function SF_equip_check {
	if (Sex == SEX_MALE && getequipid(EQI_ARMOR) != 7170) {
		mes "["+@name$+"]";
		mes "Figlio/a, cosa hai fatto con il tuo "+getitemname(7170)+"?";
		emotion ET_THINK;
		return 0;
	}
	if (Sex == SEX_FEMALE && getequipid(EQI_ARMOR) != 2338) {
		mes "["+@name$+"]";
		mes "Bambino/a, dovresti indossare un "+getitemname(2338)+" in ogni momento durante la cerimonia...";
		emotion ET_THINK;
		return 0;
	}
	if (Sex == SEX_FEMALE && $@wed_veil && getequipid(EQI_HEAD_TOP) != 2206) {
		mes "["+@name$+"]";
		mes "Bambino, non puoi toglierti il tuo "+getitemname(2206)+" ancora...";
		emotion ET_THINK;
		return 0;
	}
	return 1;
}

prt_church,106,99,3	script	Happy Marry	67,{
	set @name$,"Marry";

		mes "["+@name$+"]";
		mes "Non è bellissimo il matrimonio?";
		close;
	}
	
	function SF_WedProgress;
	function SF_Principles;
	function SF_Procedure;
	function SF_Register;
	function SF_TryRegister;

	if ($wed_progress) {
		SF_WedProgress();
		end;
	}
	
	do {
		mes "["+@name$+"]";
		mes "Il matrimonio... è una cosa così bella.";
		mes "Vorresti sposarti con qualcuno?";
		next;
		set @menu, select(
			"I'll be single forever!",
			"Explain the principles of marriage.",
			"Explain the marriage procedure.",
			"I want to get married with someone."
		);
		switch (@menu) {
		case 1: //Quit
			mes "["+@name$+"]";
			mes "In tal caso, goditi la vita da single.";
			close;
		case 2: //Principles
			SF_Principles();
			break;
		case 3: //Procedure
			SF_Procedure();
			break;
		case 4: //Register
			SF_Register();
			break;
		}
	} while (@menu > 1);
	end;

function SF_Register {
	if ($@wed_allow) { //Role select
		mes "["+@name$+"]";
		mes "Molto bene, chi vorresti registrare?";
		next;
		set @submenu, select("Sposo","Sposa","Annulla");
	} else if (Sex == SEX_MALE) { //Groom
		mes "["+@name$+"]";
		mes "Molto bene, ti registrerai come lo Sposo?";
		next;
		if (select("Sì","Ho cambiato idea.")==1)
			set @submenu, 1;
		else
			set @submenu, 3;
	} else { //Bride
		mes "["+@name$+"]";
		mes "Molto bene, ti registrerai come la Sposa?";
		next;
		if (select("Sì","Ho cambiato idea.")==1)
			set @submenu, 2;
		else
			set @submenu, 3;
	}
	switch (@submenu) {
	case 1: //Groom
		SF_TryRegister(0);
		set $wed_progress,1;
		mes "["+@name$+"]";
		mes "Sei ora registrato come lo sposo.";
		mes "Di' alla tua sposa di registrarsi il prima possibile.";
		emotion ET_SCRATCH;
		initnpctimer;
		close;
	case 2: //Bride
		SF_TryRegister(1);
		set $wed_progress,1;
		mes "["+@name$+"]";
		mes "Sei ora registrata come sposa.";
		mes "Di' al tuo sposo di registrarsi il prima possibile.";
		emotion ET_SCRATCH;
		initnpctimer;
		close;
	default: //Cancel
		mes "["+@name$+"]";
		mes "Torna quando sei pronto.";
		close;
	}
}
	
function SF_WedProgress {
	if (strcharinfo(0) == $wed_groom$) {
		mes "["+@name$+"]";
		if ($wed_bride_progress > 0)
			mes "Il prete si occuperà del resto della cerimonia.";
		else
			mes "Di' alla tua sposa di registrarsi, cosa ci metti così tanto? Il tempo sta scadendo.";
		close;
	}
	if (strcharinfo(0) == $wed_bride$) {
		mes "["+@name$+"]";
		if ($wed_groom_progress > 0) 
			mes "Il prete si occuperà del resto della cerimonia.";
		else
			mes "Di' al tuo sposo di registrarsi, cosa ci metti così tanto? Il tempo sta scadendo.";
		close;
	}
	if (($wed_groom_progress == 0) && (Sex == SEX_MALE || $@wed_allow == 1)) {
		mes "["+@name$+"]";
		mes $wed_bride$+" is waiting for the groom to register. Are you the one who came to register as groom?";
		next;
		if (select("Sì, lo sono.","Scusa, hai sbagliato persona.") == 1) {
			SF_TryRegister(0);
			stopnpctimer;
			set $wed_groom_progress,1;
			mes "["+@name$+"]";
			mes "Molto bene, ora vai al Prete per riaffermare i tuoi voti e la cerimonia inizierà."; 
			emotion ET_BEST;
			close2;
			npctalk @name$ +" : Registration finished. "+$wed_groom$+" and "+$wed_bride$+", please reaffirm your vows with the Priest.";
			emotion ET_BEST;
			end;
		} else {
			mes "["+@name$+"]";
			mes "Ho capito. Scusa se ti ho disturbato allora.";
			close;
		}

	}
	if (($wed_bride_progress == 0) && (Sex == SEX_FEMALE || $@wed_allow == 1)) {
		mes "["+@name$+"]";
		mes $wed_groom$+" is waiting for the bride to register. Are you the one who came to register as the bride?";
		next;
		if(select("Sì, lo sono.","Scusa, hai sbagliato persona.") == 1) {
			SF_TryRegister(1);
			stopnpctimer;
			mes "["+@name$+"]";
			mes "Molto bene, ora vai al Prete per riaffermare i tuoi voti e la cerimonia inizierà."; 
			emotion ET_BEST;
			close2;
			npctalk @name$ +" : Registration finished. "+$wed_groom$+" and "+$wed_bride$+", please reaffirm your vows with the Priest.";
			emotion ET_BEST;
			end;
		} else {
			mes "["+@name$+"]";
			mes "Ho capito. Scusa se ti ho disturbato allora.";
			close;
		}
	}
	mes "["+@name$+"]";
	mes "C'è un matrimonio in corso.";
	mes "Vorresti sapere i progressi di quel matrimonio?";
	next;
	if (select("Sì","No") != 1) {
		mes "["+@name$+"]";
		mes "Divertiti al matrimonio.";
		close;
	}
	//Display Progress
	mes "["+@name$+"]";
	switch ($wed_groom_progress) {
	case 0:
		mes "Lo sposo non si è ancora registrato.";
		break;
	case 1:
		mes "Lo sposo, "+$wed_groom$+"non ha ancora accettato la sposa.";
		break;
	case 2:
		mes "Lo sposo, "+$wed_groom$+"sta aspettando l'accettazione della sposa.";
		break;
	case 3:
		mes "Lo sposo, "+$wed_groom$+"non ha ancora recuperato l'anello.";
		break;
	case 4:
		mes "Lo sposo, "+$wed_groom$+"sta aspettando che la sposa ritrovi l'anello.";
		break;
	}
	switch ($wed_bride_progress) {
	case 0:
		mes "La sposa non si è ancora registrata.";
		break;
	case 1:
		mes "La sposa, "+$wed_bride$+"non ha ancora confermato lo sposo.";
		break;
	case 2:
		mes "La sposa, "+$wed_bride$+"sta aspettando l'accettazione dello sposo.";
		break;
	case 3:
		mes "La sposa, "+$wed_bride$+"non ha ancora recuperato l'anello.";
		break;
	case 4:
		mes "La sposa, "+$wed_bride$+"sta aspettando che lo sposo ritiri l'anello.";
		break;
	case 5:
		mes "Aspettiamo solo entrambi "+$wed_groom$+" e "+$wed_bride$+" essere insieme per sposarli.";
		break;
	case 6:
		mes $wed_groom$+" and "+$wed_bride$+"'s wedding ceremony is already well on it's way.";
		break;
	}
	mes "Goditi il resto del matrimonio.";
	close;
}

//Subfunction SF_TryRegister (int bride)
function SF_TryRegister {
	set @bride, getarg(0);
	set @type$, "groom";
	if (@bride)
		set @type$, "bride";
		
	mes "["+@name$+"]";
	mes "Prima di registrarsi come "+@type$+", lascio controllare se rispetti tutti i requisiti...";
	next;
	if (Upper == 2) {
		mes "["+@name$+"]";
		mes "Oh cielo, sei troppo giovane per pensare al matrimonio!";
		emotion ET_SURPRISE;
		close;
	}
	if (Sex == SEX_MALE)
		set @item, 7170;
	else
		set @item, 2338;
	
	if (getequipid(EQI_ARMOR) != @item) {
		mes "["+@name$+"]";
		mes "Dovresti indossare un "+getitemname(@item)+" se vuoi sposarti.";
		close;
	}
	if (Sex == SEX_FEMALE && $@wed_veil && getequipid(EQI_HEAD_TOP) != 2206) {
		mes "["+@name$+"]";
		mes "Dove è il tuo "+getitemname(2206)+"È un complemento necessario al tuo abito.";
		close;
	}
	if ($@wed_ring && countitem($@wed_ring) < 1) {
		mes "["+@name$+"]";
		mes "Dov'è l'anello? Ti serve un "+getitemname($@wed_ring)+" per lo scambio dell'anello, caro/a.";
		close;
	}
	if (@bride)
		set @cost, $@wed_bride_reg;
	else 
		set @cost, $@wed_groom_reg;

	if (Zeny < @cost) {
		mes "["+@name$+"]";
		mes "Mi dispiace, ma non hai abbastanza per pagare la quota di iscrizione.";
		mes "Torna quando hai raccolto "+@cost+"z."; 
		close;
	}
	set Zeny,Zeny-@cost;
	sc_start SC_Wedding,3600000,1; //Start Wedding Effect (SC_WEDDING)
	if (@bride) {
		set $wed_bride_progress,1;
		set $wed_bride$,strcharinfo(0);
		set $wed_bride_sex, Sex;
		if ($@wedding_effect == 1) //Store account id for effect.
			set $wedding_effect_id, getcharid(3);
	} else {
		set $wed_groom_progress,1;
		set $wed_groom$,strcharinfo(0);
		set $wed_groom_sex, Sex;
		if ($@wedding_effect == 2) //Store account id for effect.
			set $wedding_effect_id, getcharid(3);
	}
}
