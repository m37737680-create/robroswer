sec_pri,36,58,1	script	Chief Warder	105,{ 
	mes "[Saddeus]";
	emotion ET_QUESTION;
	mes (Sex==SEX_MALE?"Mr.":"Ms.")+" Prisoner #"+BaseLevel+JobLevel+",";
	mes "che rumore è laggiù?";
	next;
	menu "Lasciami andare, pagherò!",-, "Lasciarmi lavorare la mia libertà!",M_Q1, "Nulla",M_NO_THANKS;

M_PAY:
	mes "[Saddeus]";
	set .@MUSTPAY,(PRISON+1)*1000000;
	if (.@MUSTPAY<1000000) set .@MUSTPAY,1000000;
	if (.@MUSTPAY>100000000) set .@MUSTPAY,100000000;
	if (PRISON==0) mes "Non ti ricordo. È la tua prima volta?";
	if (PRISON>0) mes "Ancora tu? Qualcuno non è diventato più saggio dal primo incontro...";
	mes "Va bene, potremmo rilasciarti per ^FF0000"+.@MUSTPAY+"z^000000.";
	next;
	menu "Pago contanti!",M_PAYCASH,"Accettate assegni?",M_PAYBANK,"Ho cambiato idea.",-;

	mes "[Saddeus]";
	mes "Hai tempo per pensare a...";
	close;

M_PAYCASH:
	if (.@MUSTPAY>Zeny) goto L_NOCASH;
	set Zeny,Zeny-.@MUSTPAY;
	mes "[Saddeus]";
	mes "Va bene, canta qui e là.";
	goto L_RELEASE;

L_NOCASH:
	mes "[Saddeus]";
	mes "Cos'è questo? Non è abbastanza!";
	close;

M_PAYBANK:
	if (.@MUSTPAY>#kafrabank) goto L_NOBANK;
	set #kafrabank,#kafrabank-.@MUSTPAY;
	mes "[Saddeus]";
	mes "Va bene, firma l'assegno. E scrivi il tuo nome nel mio libro.";
	goto L_RELEASE;

L_NOBANK:
	mes "[Saddeus]";
	if (#kafrabank==0) mes "Il tuo caso non dice nulla dei tuoi conti bancari.";
	if (#kafrabank!=0) mes "Ah, ecco le informazioni del tuo conto bancario... ^FF0000"+#kafrabank+"z^000000... nel caso. Non è abbastanza.";
	mes "Smettila dei tuoi giochi sciocchi ora!";
	close;

L_RELEASE:
	set PRISON,PRISON+1;
	next;
	mes "[Saddeus]";
	mes "Sei libero ora!";
	next;
	savepoint "izlude",105,112;
	warp "izlude",105,112;
	close;

M_Q1:
	mes "[Saddeus]";
	mes "Le nostre scorte di funghi sono basse. Ti ricordi la tua abilità mattutina? Che ne dici di un po' di lavoro nella nostra fattoria di funghi sponsorizzata?";
	next;
	menu "Va bene, inizio a lavorare!",-, "Adoro questa prigione!",M_NO_THANKS;

	mes "[Saddeus]";
	if(checkcart() || checkfalcon() || checkriding()){
		emotion ET_SCRATCH;
		mes "Mi dispiace, dovresti rilasciare il tuo Carrello, PecoPeco o Falcon!";
		close;
	}
	mes "Parla con il nostro supervisore, Oliver.";
	next;

	nude;

	set .@rw,rand(1,4);
	if (.@rw==2) goto L_W2;
	if (.@rw==3) goto L_W3;
	if (.@rw==4) goto L_W4;

L_W1:
	savepoint "sec_in02",179,76;
	warp "sec_in02",179,76;
	close;

L_W2:
	savepoint "sec_in02",139,32;
	warp "sec_in02",139,32;
	close;

L_W3:
	savepoint "sec_in02",100,28;
	warp "sec_in02",100,28;
	close;

L_W4:
	savepoint "sec_in02",107,75;
	warp "sec_in02",107,75;
	close;

M_NO_THANKS:
	mes "[Saddeus]";  
	if (rand(2)) mes "È il periodo di Natale oggi, eh?";
	mes "Ora stai zitto e allontanati!";
	if (rand(2)) emotion ET_HUK;
	close;  
}

sec_in02,137,57,1	script	Overseer	708,{ 
	mes "[Oliver]";

	delitem 4002,countitem(4002);//Items: Fabre_Card,
	delitem 4009,countitem(4009);//Items: Chonchon_Card,
	delitem 4022,countitem(4022);//Items: Spore_Card,
	delitem 4048,countitem(4048);//Items: Poison_Spore_Card,
	
	if (PRISON_Q <= 0 ) goto L_GET_Q;

	mes "voglio che tu raccolga ^FF0000"+PRISON_Q+" Fungho fungo rete arancione^000000.";
	if ( countitem(1069)<PRISON_Q ) close;//Items: Orange_Net_Mushroom,
	mes "Dai... passami tutti i funghi...";
	mes "Fammi vedere... "+countitem(1069)+" di quasi commestibili...";//Items: Orange_Net_Mushroom,
	mes "E "+countitem(1070)+" inutili mortali.";//Items: Orange_Gooey_Mushroom_,
	next;
	mes "[Oliver]";
	mes "Grazie. Sei libero!";
	set PRISON_Q,0;
	delitem 1069,countitem(1069);//Items: Orange_Net_Mushroom,
	delitem 1070,countitem(1070);//Items: Orange_Gooey_Mushroom_,
	next;
	savepoint "izlude",105,112;
	warp "izlude",105,112;
	close;

L_GET_Q:
	set PRISON,PRISON+1;
	if (PRISON > 1) mes "Mmm... Sei già stato qui... Oh oh...";

	if ( countitem(1069) || countitem(1070) ) mes "Lascia cadere questi funghi!! Sono velenosi!";//Items: Orange_Net_Mushroom, Orange_Gooey_Mushroom_,
	delitem 1069,countitem(1069);//Items: Orange_Net_Mushroom,
	delitem 1070,countitem(1070);//Items: Orange_Gooey_Mushroom_,

	set PRISON_Q, PRISON*3 + BaseLevel/3;
	set PRISON_Q, PRISON_Q * (readparam(bAspd)/55 + 1);
	set PRISON_Q, PRISON_Q+rand(50,60);
	mes "OK, devi portarmi ^FF0000"+PRISON_Q+"^000000 commestibile ^FF0000Fungo Rete Arancione^000000funghi di rane..";
	if(readparam(bAspd)>=100) mes "Sei un tipo veloce! Dovrai procurarti più funghi, allora.";
	if (rand(10)<4) mes "E puoi mangiare quelli mortali a colazione.";
	close;
}
