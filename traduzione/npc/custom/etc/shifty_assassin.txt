//= Players buy ninjas to assassinate other players
//===== Additional Comments: =================================
//= 1.1.1 Changed all gmcommand to atcommand as Poki#3 suggested. [Vicious]
//= 1.1.2 Updated WoE Check. [Paradox924X]

morocc,148,86,5	script	Shifty Assassin	725,{
	set .@ninja_price,250000;

	// STARTS THE MENU //
M_Start:
	mes "Assassino furtivo";
	mes "Cosa vuoi?";
	next;
	menu "Compra Ninja",M_Buy,"Assassinare qualcuno",M_Kill,"Controlla i tuoi Ninja",M_Check, ( getgmlevel() > 90 ? "Aggiungi Ninja" : "" ),-,"Annulla",M_Exit;

	// GM MENU TO ADD NINJAS //
	mes "Assassino furtivo";
	mes "Quanti ninja vuoi rendere disponibili?";
	next;
	input .@add;
	set $ninja_avail, $ninja_avail+.@add;
	mes .@add +" ninjas added.";
	close;

	// BUY NINJAS //
M_Buy:
	mes "Assassino furtivo";
	mes "Quanti ninja vuoi comprare?";
	mes "Ci sono ^0000FF" + $ninja_avail + "^000000 ninja disponibili.";
	mes "Costano ^0000FF" + .@ninja_price + " zeny ^000000ogni.";

	input .@buy;
	next;
	if ($ninja_avail < 1) goto NoNinjas;
	if ($ninja_avail < .@buy) goto NotEnoughNinjas;
	set .@price, .@buy*.@ninja_price;
	if (Zeny < .@price) goto NoZeny;

	mes "Assassino furtivo";
	mes "Ti costerà ^0000FF"+ .@price +" zeny^000000.";
	next;
	menu "Continua",-,"Annulla",M_Exit;

	set Zeny, Zeny - .@price;
	set #ninjas, #ninjas + .@buy;
	set $ninja_avail, $ninja_avail - .@buy;

	mes "Assassino furtivo";
	mes "Grazie.";
	close;

	// ASSASSINATE SOMEBODY //
M_Kill:
	if (agitcheck()) goto M_Busy;
	mes "Assassino furtivo";
	mes "Inserisci il nome del bersaglio.";
	mes "^FF0000Type the name exactly, otherwise I won't be able to find the victim.^000000";
	next;
	menu "Continua",-,"Annulla",M_Exit;
	input .@name$;
	if ( !getcharid( 3,.@name$ ) ) {
		mes "Assassino furtivo";
		mes .@name$ +" is not online.";
		close;
	}
	next;
	mes "Assassino furtivo";
	mes "Ninjas Attivi: "+#ninjas;
	mes "Ninja a riposo: "+#ninjasr;
	mes "Quanti ne vuoi inviare?";
	input .@number;
	if (.@number < 1) goto NoNinjasSent;
	if (.@number > #ninjas) goto NotEnoughNinjas1;
	if (.@number > 10) goto TooManyNinjas;
	set .@chance, rand(1,12);
	set #ninjas,#ninjas-.@number;
	set #ninjas,#ninjas+#ninjasr;
	set #ninjasr,0;
	if (.@number < .@chance) goto M_Failure;

	// SUCCESSFUL ATTACK //
	mes "Invio i ninja ora.";
	next;
	mes "Assassino furtivo";
	set .@ninjasurvived, rand(1,.@number);
	set #ninjasr,.@number-.@ninjasurvived;
	mes "Il tuo attacco è riuscito ma solo ^FF0000" + #ninjasr + "^000000 I ninja sono sopravvissuti.";
	set .@acc_id, getcharid( 3,.@name$ );
	if ( .@acc_id )
		unitkill .@acc_id;
	announce .@name$ +" has been assassinated by " + strcharinfo(0) +"'s Ninjas.",bc_npc;
	close;

	// FAILED ATTACK //
M_Failure:
	mes "Invio i ninja ora.";
	next;
	mes "Assassino furtivo";
	set .@ninjasurvived, rand(1,.@number);
	set #ninjasr, .@number-.@ninjasurvived;
	mes "Il tuo attacco è fallito e solo ^FF0000" + #ninjasr + "^000000 I ninja sono sopravvissuti.";

	announce .@name$+" has survived " + strcharinfo(0) +"'s Ninja attack.",8;
	close;

	// NINJAS BUSY FOR WOE //
M_Busy:
	mes "Assassino furtivo";
	mes "Scusa, tutti i miei ninja sono impegnati a fare la Guerra di Emperium.";
	close;

	// CHECK YOUR NINJAS //
M_Check:
	mes "Assassino furtivo";
	mes "Hai:";
	mes "^FF0000" + #ninjas + "^000000 Ninja Attivi.";
	mes "^0000FF" + #ninjasr + "^000000 Ninja a riposo.";
	next;
	goto M_Start;


	// LIMIT //
NoNinjasSent:
	mes "Assassino furtivo";
	mes "Non puoi uccidere nessuno senza ninja.";
	next;
	goto M_Start;

TooManyNinjas:
	mes "Assassino furtivo";
	mes "Puoi inviare al massimo 10 ninja.";
	next;
	goto M_Start;

NoZeny:
	mes "Assassino furtivo";
	mes "Non hai abbastanza zeny.";
	close;

NotEnoughNinjas:
	mes "Assassino furtivo";
	mes "Non ci sono molti ninja da comprare.";
	next;
	goto M_Start;

NoNinjas:
	mes "Assassino furtivo";
	mes "Non ci sono ninja rimasti da comprare.";
	close;

NotEnoughNinjas1:
	mes "Assassino furtivo";
	mes "Non hai tanti ninja.";
	next;
	goto M_Start;

M_Exit:
	mes "Assassino furtivo";
	mes "Arrivederci.";
	close;

// TIMER DELAY NINJA ADDER //
OnClock0600:
OnClock1200:
OnClock1500:
OnClock1900:
OnClock2000:
OnClock0000:
	set $ninja_avail,$ninja_avail+2;
	end;
OnClock1800:
	set $ninja_avail,$ninja_avail+3;
	end;
OnInit:
	set $ninja_avail,$ninja_avail+1;
	end;
}
