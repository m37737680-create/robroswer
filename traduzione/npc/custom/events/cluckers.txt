//===== Description: =========================================
//= Click the chicken and try retrieve the item at a low 
//= chance. If you fail he will nuke, freeze, stone, 
//= stun, or make you fall asleep.
//= The prize is configurable and triggered by the NPC.
//===== Additional Comments: =================================
//= 1.0 First release.
//= 1.1 Using 'switch rand' instead.
//= 1.2 Cleaned and standardized. [Euphy]
//= 1.2a Switched 'atcommand' to 'unitskilluseid'.
//= 1.3 Switched 'unitskilluseid' to 'unitkill'.
//============================================================

prontera,156,219,4	script	Cluckers	800,{

	set .@GMAccess,80;	// GM level required to access NPC

	if (.startcluck) {
		specialeffect2 EF_HIT3;
		switch(rand(15)) {
			case 0: npctalk "Cluckers : CLUUUUUUCK!!!"; unitkill getcharid(3); skilleffect "NPC_SELFDESTRUCTION",1; emotion ET_HUK; break;
			case 1: npctalk "Gallinacci: Cluuuuuck!~"; break;
			case 2: unitkill getcharid(3); skilleffect "NPC_SELFDESTRUCTION",1; break;
			case 3: sc_start SC_Freeze,10000,0; break;
			case 4: npctalk "Cluckers : CLUUUUUUUUUCK!!!"; unitkill getcharid(3); skilleffect "NPC_SELFDESTRUCTION",1; emotion ET_HUK; break;
			case 5: sc_start SC_Sleep,10000,0; break;
			case 6: sc_start SC_Stone,10000,0; emotion ET_KIK; break;
			case 7: npctalk "Cluckers : CLUUUUUUCK!!!"; unitkill getcharid(3); skilleffect "NPC_SELFDESTRUCTION",1; emotion ET_HUK; break;
			case 8: npctalk "Cluckers: Coccì! COCCÌ!!"; unitkill getcharid(3); skilleffect "NPC_SELFDESTRUCTION",1; emotion ET_HUK; break;
			case 9: sc_start SC_Stun,10000,0; break;
			case 10: sc_start SC_Sleep,10000,0; emotion ET_KIK; break;
			case 11: npctalk "Gallinacci: Coccù! Coccù!"; break;
			case 12: sc_start SC_Stun,10000,0; break;
			case 13: unitkill getcharid(3); skilleffect "NPC_SELFDESTRUCTION",1; break;
			default:
				if (rand(50) < 4) {
					npctalk "Cluckers : WOOF!...........";
					specialeffect2 EF_SPHERE;
					announce ""[Cluck! Cluck! Boom!] " + strcharinfo(0) + " squeezed out the prize! Well done!",0;
					getitem $cluck_item_id,$cluck_item_amount;
					set .startcluck,0;
				} else {
					npctalk "Cluckers: Coccì! COCCÌ!!";
					unitkill getcharid(3); skilleffect "NPC_SELFDESTRUCTION",1;
				}
				break;
		}
		end;
	}
	mes "[Ciacchiolatori]";
	if (getgmlevel() >= .@GMAccess) {
		mes "Ciacciorío! Ciuuuucc? ^FF0000~Ciao GM " + strcharinfo(0) + "Vuoi giocare oggi?~^000000";
		mes "Ciacciorriacchi... CACCIO! ^FF0000~Dimmi solo cosa fare!~^000000";
		next;
		while(1) switch(select("Evento di inizio:Controlla Premio:Imposta Premio:Non oggi Cluckers")) {
			case 1:
				mes "[Ciacchiolatori]";
				mes "CUCU! ^FF0000~Certamente!~^000000";
				emotion ET_OK;
				close2;
				announce ""[Cluck! Cluck! Boom!] sta per iniziare a Prontera!",bc_blue;
				initnpctimer;
				end;
			case 2:
				mes "[Ciacchiolatori]";
				mes "Ciac, ciac ciac... Ciac! ^FF0000~Il premio attuale è^000000 ^008000"+ $cluck_item_amount +" "+ getitemname($cluck_item_id) +".^000000";
				next;
				break;
			case 3:
				mes "[Ciacchiolatori]";
				mes "Coccia coccia? Coccia??? ^FF0000~Quale dovrebbe essere il premio per vincere? Si prega di inserire l'ID.~^000000";
				input $cluck_item_id;
				next;
				mes "[Ciacchiolatori]";
				mes "Coccollo? cocuuccco? ^FF0000~Quanti di questo articolo dovrei regalare?~^000000";
				input $cluck_item_amount;
				next;
				mes "[Ciacchiolatori]";
				mes "Ciacciorellare...? Ciacco. ^FF0000~Quindi, il premio è^000000 ^008000"+ $cluck_item_amount +" "+ getitemname($cluck_item_id) +"^000000? ^FF0000Great.~^000000";
				emotion ET_OK;
				next;
				break;
			case 4:
				mes "[Ciacchiolatori]";
				mes "Ciacciorriacchio ciacciorriacchio ciacciorriacchio...";
				close;
		}
	}
	mes "Coccollo coccollo! Coccollo?";
	mes "Ciacciò...";
	close;

OnTimer10000:
	announce ""Per favore, corre dietro la fontana se vuoi giocare con il pollo pazzo!",bc_blue;
	end;
OnTimer20000:
	announce ""Cluckers ha mangiato uno dei miei oggetti! Ho troppo paura per recuperarlo!",bc_blue;
	end;
OnTimer30000:
	announce ""Clicca il pollo pazzo e prova a spremere fuori l'oggetto, se sei fortunato vincerai! Sei pronto?",bc_blue;
	end;
OnTimer40000:
	announce ""Vai! Clicca sul pollo per ottenere il premio!",bc_blue;
	stopnpctimer;
	if (!$cluck_item_id) set $cluck_item_id,512;
	if (!$cluck_item_amount) set $cluck_item_amount,1;
	set .startcluck,1;
	end;
}
