prontera,151,190,5	script	Hallows' Eve Event	807,{
	mes "[Evento della Vigilia]";
	if(Class == Job_Novice){
		mes "Mi dispiace ma devi trovare un lavoro per completare l'evento.";
		emotion ET_SWEAT, playerattached();
		close;
	}
	if((EVENT_HALWN06&1) && (EVENT_HALWN06&2)){
		mes "Mi dispiace ma hai già completato l'evento.";
		emotion ET_SORRY;
		close;
	}
	mes "Saluti caro umano vivace.";
	next;
	mes "[Evento della Vigilia]";
	mes "Halloween è alle porte ed è quasi ora di avventurarsi in luoghi che pochi osano raggiungere.";
	next;
	mes "[Evento della Vigilia]";
	mes "Zucca o dolciumi, quale preferisci?";
	next;
	menu "Trucco",-, "Trattare",L_Treat;

	mes "[Evento della Vigilia]";
	if(EVENT_HALWN06&2){
		mes "Mi dispiace ma hai già avuto un trucco. Perché non vai a prendere uno snack adesso.";
		emotion ET_SORRY;
		close;
	}
	mes "Dammi un min...";
	next;
	set BaseExp,BaseExp+BaseLevel*1000;
	set EVENT_HALWN06,EVENT_HALWN06|2;

	mes "[Evento della Vigilia]";
	mes "Ecco il tuo regalo. Il tuo EXP Base è stato leggermente aumentato.";
	next;
	mes "[Evento della Vigilia]";
	mes "Spero ti diverta il tuo trucco e Buon Halloween!";
	close;

L_Treat:
	mes "[Evento della Vigilia]";
	if(EVENT_HALWN06&1){
		mes "Mi dispiace ma hai già avuto un premio. Perché non vai a prendere un trucco adesso.";
		emotion ET_SORRY;
		close;
	}
	if(!(EVENT_HALWN06&4)){
		mes "Ottimo! Sarà così divertente.";
		next;
		mes "[Evento della Vigilia]";
		mes "Ti invierò per una caccia al tesoro di oggetti.";
		next;
		mes "[Evento della Vigilia]";
		mes "Riporta tutti gli oggetti necessari e ti creerò una prelibatezza.";
		next;
		mes "[Evento della Vigilia]";
		mes "Non un semplice dolcetto normale che puoi mangiare, ma è qualcosa che puoi indossare sulla testa.";
		next;
		mes "[Evento della Vigilia]";
		mes "Ma con tutte le cose fatte, alcuni elementi sono necessari prima.";
		mes "La lista:";
		mes "Cappello,";
		mes "30 Zucca, e";
		mes "20 Pezzi di Stoffa Neri";
		next;
		mes "[Evento della Vigilia]";
		mes "Sbrigati ora.";
		set EVENT_HALWN06,EVENT_HALWN06|4;
		close;
	}
	mes "Hai ricevuto gli articoli che ho richiesto o hai bisogno di vedere la lista di nuovo?";
	next;
	menu "Ci ho",-, "Non avere",L_Dont_Have, "Vedi Lista",L_List;

	mes "[Evento della Vigilia]";
	if((countitem(2220) < 1) || (countitem(535) < 30) || (countitem(7205) < 20)){
		mes "Non hai abbastanza articoli. Torna quando ne avrai.";
		emotion ET_SORRY;
		close;
	}
	delitem 2220,1;
	delitem 535,30;
	delitem 7205,20;
	getitem 5134,1;
	set EVENT_HALWN06,EVENT_HALWN06|1;
	mes "Ottimo lavoro! Spero che tu sia felice del tuo articolo.";
	emotion ET_CONGRATULATION;
	next;
	mes "[Evento della Vigilia]";
	mes "Buon resto di Halloween.";
	emotion ET_SCISSOR, playerattached();
	close;

L_Dont_Have:
	mes "[Evento della Vigilia]";
	mes "Torna fuori e cercali. Vorresti visualizzare l'elenco?";
	next;
	menu "No",-, "Sì",L_List;
	close;

L_List:
	mes "[Evento della Vigilia]";
	mes "Richiedo i seguenti articoli:";
	mes "Cappello,";
	mes "30 Zucca, e";
	mes "20 Pezzi di Stoffa Neri.";
	close;
}
