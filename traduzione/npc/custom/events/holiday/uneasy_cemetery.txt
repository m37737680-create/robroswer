//===== Description: ========================================= 
//= A periodical quest of the Uneasy Cemetery (Kill undead / Prevent their appearance)
//= Every day, at the midnight Prontera receive a wave of Undeads.
//= They come from Uneasy Cemetery of Prontera. To protect the players
//= from the undeads terror you may either kill the enemy. Or supply Mother Mathana
//= with needed amount of Holy Water. Every citizen can take his part in the 
//= saving of Prontera city. After some days of quiet life... the Cemetery strikes back.
//===== Additional Comments: ================================= 
//= 1.1 More advanced ver. Added some bonus for the one who'd kill the last walking undead
//= 1.2 Added coords to the script to make label OmMobDead working
//= 1.2a Changed item names to item IDs. [Samuray22]

prontera,3,3,3	script	Uneasy_Check	-1,{
	end;

OnHour00:
	set $UNEASY_DL,$UNEASY_DL-1;
	set $UNEASY_BL,$UNEASY_BL+30; //add need of HW for 30 bottles per day
	if ($UNEASY_BL>666) set $UNEASY_BL,666; //keep needed bottles not <=666
	if ($UNEASY_DL < 0) goto L_Start_Undead;
//The Cemetery is OK yet.
	disablenpc "Mother Mathana";
	end;
OnInit:
	if ($UNEASY_DL >= 0) disablenpc "Mother Mathana";
	end;

OnHour06:
	killmonsterall "prontera";	//The Sun kills undead in the morning
	end;

OnHour01:
	if ($@UNEASY_MOB > 0) mapannounce "prontera",""[Mother Mathana]: Nel nome di Odino, per favore finite questi resti di non morti vaganti!",0;
	end;

OnZombieDead:
	set $@UNEASY_MOB,$@UNEASY_MOB-1;
	if ($@UNEASY_MOB>0) end;
	set $UNEASY_DL,0;
	set $UNEASY_H$,strcharinfo(0);
	if (Sex==SEX_MALE) mapannounce "prontera",""[Mother Mathana]: Coraggiosa "+$UNEASY_H$+" has just killed the last undead in Prontera!",0;
	else mapannounce "prontera",""[Mother Mathana]: Signora "+$UNEASY_H$+" has just killed the last undead in Prontera!",0;
	set JobExp,JobExp+100;
	set BaseExp,BaseExp+50;
	end;

L_Start_Undead:
	killmonsterall "prontera";	//kills any left monsters
	enablenpc "Mother Mathana";
//call some monsters in the city
	set $@UNEASY_MOB, 65;
	areamonster "prontera",0,0,0,0,"Zombie",1015,30,"Uneasy_Check::OnZombieDead";
//in the Cemetery
	monster "prontera",268,349,"Zombie",1015,30,"Uneasy_Check::OnZombieDead";
	monster "prontera",269,350,"Ghoul",1036,5,"Uneasy_Check::OnZombieDead";
//announce
	mapannounce "prontera",""[Madre Mathana]: Il cimitero è diventato irrequieto! Nel nome di Odino, affrettatevi al Santuario! Salvate la città di Prontera!",0;
}

prontera,257,313,5	script	Mother Mathana	79,{
	mes "Madre Mathana";
	if ($UNEASY_DL <= 0) goto L_Undead_Walk;
	mes "Temo che c'sia qualcosa di sbagliato nel nostro vecchio cimitero...";
	if ($UNEASY_H$==strcharinfo(0)) mes "Ma grazie a te, "+$UNEASY_H$+"potremo dormire " + $UNEASY_DL + " notte!";
	if ($UNEASY_H$!=strcharinfo(0)) mes "Ma grazie a "+$UNEASY_H$+"il supporto di', abbiamo " + $UNEASY_DL + " serate facili!";
	emotion ET_SURPRISE;
	close;

L_Undead_Walk:
	if ($UNEASY_DL == 0) mes "Potrebbero tornare stasera di nuovo!";
	if ($UNEASY_DL == 0 && $UNEASY_H$==strcharinfo(0)) mes "Grazie, "+$UNEASY_H$+"Ora riusciremo a riposare fino alla prossima mezzanotte!";
	if ($UNEASY_DL == 0 && $UNEASY_H$!=strcharinfo(0)) mes "Ma a causa di "+$UNEASY_H$+"con il tuo aiuto riusciremo a riposare fino alla prossima mezzanotte!";
	mes "Per placare il quieto cimitero, dovremmo versare tutte queste tombe con l'Acqua Santa. Ma le nostre sorelle e i nostri fratelli se ne sono scaricati.";
	mes "Potrebbe fornirci Acqua Santa?";
	next;
	menu "Sì, ho tutta la mia Acqua Santa!",-, "No, ne ho bisogno.",M_NO, "Non ne ho nessuno.",M_DONT_HAVE;

	if ( countitem(523)<1 ) goto M_DONT_HAVE;
	set $UNEASY_BL,$UNEASY_BL-countitem(523);
	delitem 523,countitem(523);

	if ( $UNEASY_BL > 0 ) goto L_NEED_MORE;
//set quiet days!!! no more undead for this period!
	set $UNEASY_DL,5+((0-$UNEASY_BL)/30);
	set $UNEASY_H$,strcharinfo(0);
	mes "Madre Mathana";
	mes "Grazie, "+$UNEASY_H$+"Ora abbiamo abbastanza Acqua Santa!";
	next;
	mes "Madre Mathana";
	mes "Versando il cimitero con quell'acqua otterremo " + $UNEASY_DL + " buoyanti notti!";
	next;
	killmonsterall "prontera";	//kills any left monsters
	mes "Madre Mathana";
	mes "Guarda, "+ $UNEASY_H$ +"Sono tutti spariti ora!";
	next;
	mes "Madre Mathana";
	mes "La nostra Chiesa vi ringrazierà personalmente...";
	next;
	if (Sex==SEX_MALE) mapannounce "prontera",""[Mother Mathana]: Nel nome di Odino dichiariamo bello "+$UNEASY_H$+" as a Prontera savior!",0;
	else mapannounce "prontera",""[Mother Mathana]: Nel nome di Odino dichiariamo bello "+$UNEASY_H$+" as a Prontera savior!",0;
	mes "Madre Mathana";
	mes "Nel nome di Odino ti benediciamo e ti presentiamo con decoro un modesto dono direttamente dalla cantina di Mareusis.";
	getitem 505,1; //Blue_Potion
	set JobExp,JobExp+100;
	set BaseExp,BaseExp+50;
	close;

L_NEED_MORE:
	mes "Madre Mathana";
	mes "Grazie, bene "+strcharinfo(0)+"ma abbiamo ancora bisogno di " + $UNEASY_BL + " più bottiglie d'acqua santa.";
	close;

M_NO:
	mes "Madre Mathana";
	mes "Temo che il vecchio cimitero stia per sfuggirci di mano... Per favore, portaci tutto l'Acqua Santa che puoi trovare.";
	close;

M_DONT_HAVE:
	mes "Madre Mathana";
	mes "Ah! Abbiamo ancora bisogno di " + $UNEASY_BL + " altre bottiglie di Acqua Santa... Perché non vai a chiedere ad altre persone un po' di Acqua Santa in più?";
	mes "Il vecchio cimitero sta per sfuggire al controllo...";
	mes "Per favore, nel nome di Odino, aiuta la città di Prontera.";
	close;
}
