prontera,142,228,6	script	Find the Mushroom	1084,{
	mes "Trova il fungo";
	if (.status == 1) {
		.@count = mobcount( .event_map$, strnpcinfo(0) + "::OnMobKilled" );
		if (.@count > 0) {
			mes "Ci sono " + .@count + " Funghi rimasti in " + .event_map$ + "!";
			mes "Trova e uccidi i funghi per guadagnare " + getitemname(.prize) + "!";
			close;
		}
		.event_map$ = "";
		.status = 0;
		mes "Tutti i funghi sono morti?";
	}
	mes "Non c'è nessun evento al momento!";
	if (.status || getgmlevel() < .GM) close;
	mes "Avviare l'evento?";
	next;
	if (select("No:Sì") == 1) close;
	donpcevent strnpcinfo(0) + "::OnMinute10";
	mes "Trova il fungo";
	mes "Evento iniziato!";
	close;

OnInit:
	.prize = 512;	// Reward item ID
	.amount = 10;	// Reward item amount
	.GM = 60;		// GM level required to access NPC
	setarray .maps_list$[0],"izlude","geffen","morocc","prontera"; // Possible maps
	end;

OnMinute10:	// Start time (every hour)
	if (.status) end;
	.status = 1;
	.@spawn = rand(1,10);	// How many Mushrooms should spawn?
	.event_map$ = .maps_list$[ rand(getarraysize(.maps_list$)) ];
	.@label$ = strnpcinfo(0) + "::OnMobKilled";
	killmonster .event_map$, .@label$;
	monster .event_map$,0,0,"Please don't kill me!",1084,.@spawn, .@label$;
	announce ""Trova il fungo: Totale di " + .@spawn + " Mushrooms have been spawned in " + .event_map$ + "!",0;
	sleep 2500;
	announce ""Trova il fungo: Ogni fungo che uccidi ti darà " + getitemname(.prize) + "!",0;
	end;

OnMobKilled:
	if (playerattached() == 0)
		end;
	getitem .prize, .amount;
	.@spawn = mobcount( .event_map$, strnpcinfo(0) + "::OnMobKilled" );
	if (.@spawn > 0)
		announce ""[ " + strcharinfo(0) + " ] has killed a Mushroom. There are now " + .@spawn + " Mushroom(s) left.",bc_map;
	else {
		announce ""L'evento 'Trova il Fungo' è terminato. Tutti i funghi sono stati uccisi.",0;
		.status = 0;
		.event_map$ = "";
	}
	end;
}
