umbala,126,129,4	script	Bearded Man#bandit	120,{
	if(BEARD_QUEST == 4 ) goto L_MAKE;
	if(BEARD_QUEST >= 5 ) goto L_DONE;
	if(BEARD_QUEST >= 1 ) goto L_THREAD;

	mes "Uomo barbuto";
	mes "Barbe! Prendete le vostre meravigliose barbe!";
	mes "Ti piacerebbe una barba?";
	next;
	menu "Sì, vorrei un po' di barba!",-,"No grazie, è disgustoso!",M_NO;

	mes "Uomo barbuto";
	mes "HoHo~ Allora vuoi dei baffi?";
	emotion ET_SMILE;
	mes "Beh, ti darei le mie barbe ma non sono economiche.";
	mes "Di fatto ne ho solo uno rimasto...";
	mes "ma non lo sto vendendo.";
	mes "Ma non arrabbiarti! Posso fartene uno.";
	next;
	set BEARD_QUEST,1;
L_THREAD:
	mes "Uomo barbuto";
	mes "Avrei bisogno di del filo ^8080FFThread^000000 per la barba.";
	mes "Parla con il mio amico a Izlude per qualche ^8080FFThread^000000.";
	next;

	mes "Uomo barbuto";
	mes "Anche avrei bisogno che mi prendessi alcune cose per la barba.";
	mes "Sono:";
L_LIST:
	mes "^8080FFElastic Band^000000";
	mes "^8080FF5 Pelli di Animali^000000";
	mes "^8080FFMateriale Colorante Nero^000000";
	mes "^8080FFGranpa Beard^000000";
	mes "^8080FF100 muco appiccicoso^000000";
	mes "^8080FFContra-agente^000000";
	mes " e un Diamante da ^8080FF2 Carati^000000";
	close;

M_NO:
	mes "Uomo barbuto";
	mes "Va bene, che sia così!";
	emotion ET_ANGER;
	close;

L_DONE:
	mes "Uomo barbuto";
	mes "Scusa, posso farne solo uno a persona.";
	mes "Non vogliamo troppi baffi ora. Giusto?";
	emotion ET_KIK;
	close;

L_MAKE:
	mes "Uomo barbuto";
	mes "Facciamoti la barba!";
	next;
	if(countitem(7200) < 1 || countitem(919) < 1 ||	countitem(983) < 1 || countitem(2241) < 1 ||
	countitem(938) < 100 || countitem(973) < 1 || countitem(731) < 1) goto L_NOITEMS;
	delitem 7200, 1;
	delitem 919, 1;
	delitem 983, 1;
	delitem 2241, 1;
	delitem 938, 100;
	delitem 973, 1;
	delitem 731, 1;
	mes "^8080FF~Lo vedi tagliare e cucire la barba insieme~^000000";
	next;
	mes "^8080FF~He hands you the finished beard~^000000";
	next;
	getitem 2237, 1;
	set BEARD_QUEST, 5;
	mes "Uomo barbuto";
	mes "Buona giornata!";
	emotion ET_KIK;
	close;

L_NOITEMS:
	mes "Uomo barbuto";
	mes "Che diavolo! Non riesco a fare una barba senza gli oggetti!";
	mes "Ecco la lista di nuovo:";
	emotion ET_HUK;
	goto L_LIST;
}

alberta,65,123,6	script	Man#bandit	85,{
	if(BEARD_QUEST == 3 && $@beardMobD) goto L_KILLED;
	if(BEARD_QUEST > 3) goto L_DONE;

	mes "[Merchant Filo]";
	mes "ARGGG!! Non riesco mai a portare le mie cose ad Al de Baran!";
	mes "Vengo assillato da alcuni bug.";
	mes "Rubano tutte le mie scorte e ^8080FFThread^000000.";
	next;
	mes "[Merchant Filo]";

	if(BEARD_QUEST != 3) mes "Cosa devo fare?";
	if(BEARD_QUEST != 3) close;

	if($@beardmob > 0) mes "Uccidi! Uccidili già!!!";
	if($@beardmob > 0) close;
	mes "Sei qui per aiutarmi?";
	emotion ET_QUESTION;
	next;
	menu "Sì",-,"No",M_NO;

	mes "[Merchant Filo]";
	mes "Oh grazie a Dio!";
	mes "Se potessi solo uccidere i bug riuscirei a fare la mia consegna.";
	next;
	mes "[Merchant Filo]";
	mes "Oh NO! Arrivano!!";
	emotion ET_PROFUSELY_SWEAT;
	donpcevent "BRDQ_MOBS::OnStart";
	close;

L_KILLED:
	set $@beardMobD,0;
	mes "[Merchant Filo]";
	mes "Grazie per aver ucciso i Bug Ladri.";
	mes "Come segno di gratitudine ecco una scatola piena di ^8080FFThread^000000.";
	set BEARD_QUEST,4;
	close;

L_DONE:
	mes "[Merchant Filo]";
	mes "Grazie ancora!";
	emotion ET_THANKS;
	close;

M_NO:
	mes "[Merchant Filo]";
	mes "Cosa fare...";
	close;
}
