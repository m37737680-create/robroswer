prontera,165,178,8	script	Old Woman	103,{

	setarray .@Items[0],714,25,1061,100,1095,200,618,10,1551,2,1614,1,727,20;
	set .@n$,"[Old Woman]";

	mes .@n$;
	if(BerzQuest == 1) {
		mes "Ehi, mi ricordo di te! Ti avevo già detto che potresti completare questa missione solo una volta.";
		close; }
	mes "Ciao, ragazzo. Ho trovato un oggetto davvero carino mentre raccoglievo funghi vicino a Glast Heim.";
	next;
	if(select("Continua a parlare:Lasciare")==2) goto M_Leave;

	mes .@n$;
	mes "Non ho mai visto un articolo come questo prima. Mi dicono che può essere inserito all'interno di qualsiasi accessorio con uno slot disponibile.";
	next;
	mes .@n$;
	mes "Se sei interessato a questo oggetto, ti dirò cosa ti serve per fartene uno tuo. Posso darne solo uno a ogni giocatore, quindi una volta che avrai finito questa missione, non potrai farlo più.";
	next;
	if(select("Dimmi di più a riguardo:Non me ne importa")==2) goto M_Leave;

	mes .@n$;
	mes "Credo che l'oggetto si chiami una Carta ^FF0000Berzebub Card^000000. Ti fa lanciare incantesimi molto velocemente!";
	next;
	mes .@n$;
	mes "Ho bisogno di tutti gli elementi seguenti:";
	for(set .@i,0; .@i<getarraysize(.@Items); set .@i,.@i+2) {
		mes " ~ "+.@Items[.@i+1]+"x "+getitemname(.@Items[.@i]);
		if (countitem(.@Items[.@i]) < .@Items[.@i+1]) set .@nr,1; }
	next;
	if(select("Ce l'ho tutto!:Te li darò il prima possibile")==2) goto M_Leave;

	mes .@n$;
	if (.@nr) {
		mes "Menti! Vai via e prendi quegli oggetti, sei un dispiacevole scusa per uno rockstar.";
		close; }
	for(set .@i,0; .@i<getarraysize(.@Items); set .@i,.@i+2)
		delitem .@Items[.@i], .@Items[.@i+1];
	getitem 4145,1;
	set BerzQuest, 1;
	mes "Congratulazioni! Avete tutti gli oggetti. Ecco la vostra ^FF0000Carta Berzebub^000000, come promesso.";
	emotion ET_CONGRATULATION;
	close;

M_Leave:
	mes .@n$;
	mes "Un articolo così bello che ho qui...";
	close;
}
