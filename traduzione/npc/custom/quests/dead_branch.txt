//===== Description: ========================================= 
// Simple item trade-in quest, a person can get a Dead Branch
// by simply providing the items: Log x3 (7201), Wooden Heart x1 (7189), 
// Trunk x5 (1019), and Wooden Gnarl x1 (7222).
//===== Additional Comments: ================================= 
//= 1.1 Optimized, added Bloody Branch with 0.01% chance [Lupus]
//============================================================

niflheim,204,179,3	script	Mister Mobry	121,{

	mes "[Signor Mobry]";
	mes "Gli alberi possiedono spiriti sai...";
	next;

	mes "[Signor Mobry]";
	mes "Infatti conosco un segreto su questi spiriti che può in qualche modo... forse aiutarti...";
	next;

	mes "[Signor Mobry]";
	mes "Posso creare un ramo vivace... Cioè, se mi fornisci i materiali corretti...";
	next;
	goto L_MENU;

L_MENU:
	menu "Di cosa ho bisogno?",-,"Dove posso trovare questa merda?",M_FIND,"Fammi un ramo morto!",M_CREATE,"Dimenticalo...",M_END;

	mes "[Signor Mobry]";
	mes "Mwehehe... Vedo che vuoi controllare gli spiriti degli alberi, giusto? Ok... Ho bisogno di:";
	mes "^1354453 Log^000000";
	mes "^1354451 Wooden Heart^000000";
	mes "^1354455 Trunk^000000";
	mes "^1354451 Nodume di legno^000000";
	next;
	goto L_MENU;

M_FIND:
	mes "[Signor Mobry]";
	mes "Puoi trovare i seguenti oggetti da certi mostri heheh...";
	mes "^135445Logs can be found off of Tree Golems.^000000";
	mes "^135445Wooden Hearts the essence of Tree Golems.^000000";
	mes "^135445Tronchi? Seriamente, pensa al legno...^000000";
	mes "^135445Nodulo di legno... Patibolo...^000000";
	next;
	goto L_MENU;

L_NOLOG:
	mes "[Signor Mobry]";
	mes "^135445Logs^000000 sono il corpo... Ne ho bisogno di più.";
	mes "Dammi tre tronchi e ti farò lo spirito...";
	close;

L_NOHEART:
	mes "[Signor Mobry]";
	mes "Il Cuore di Legno ^135445Wooden Heart^000000 è l'essenza dello spirito.";
	mes "Dammi un cuore di legno e ti farò lo spirito...";
	close;

L_NOTRUNK:
	mes "[Signor Mobry]";
	mes "Cosa? Non sei riuscito nemmeno a trovare ^1354455 Trunks^000000?";
	mes "Uccidere l'anziano Salice o qualcosa del genere... Accidenti... nessun tronco eh... patetico.";
	close;

L_NOGNARL:
	mes "[Signor Mobry]";
	mes "Come posso farne uno senza un ^135445Nocio di legno^000000?";
	mes "Ho detto Gibbet... sono tutti intorno a questo posto...";
	close;

M_CREATE:
	mes "[Signor Mobry]";
	mes "Beh... vediamo cosa mi hai portato...";
	next;
	if(countitem(7201)<3) goto L_NOLOG;
	if(countitem(7189)<1) goto L_NOHEART;
	if(countitem(1019)<5) goto L_NOTRUNK;
	if(countitem(7222)<1) goto L_NOGNARL;
	delitem 7201,3;
	delitem 7189,1;
	delitem 1019,5;
	delitem 7222,1;
	mes "[Signor Mobry]";
	mes "Ehi... Spero ti divertirai con gli spiriti degli alberi... Attento ora...";
	mes "Ehi... stai attenta ora? Cosa mi importa se tu fai... Non importa.";
	if(rand(1000)==0) goto L_GIVE2;
	getitem 604,1; //Dead Branch
	close;
L_GIVE2:
	getitem 12103,1; //Bloody Branch
	close;

M_END:
	mes "[Signor Mobry]";
	mes "Nessuno spirito per te... allora vai!";
	close;
}
