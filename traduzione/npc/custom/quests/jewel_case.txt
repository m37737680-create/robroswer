lighthalzen,108,229,2	script	La Fenn	766,{
	mes "[La Fenn]";
	if(BaseLevel<99) {
		if(Q_99LVL) {
			mes "Tu... Sei più giovane!!! Ma sei sicuramente TU!";
			mes "Non pensavo che fossi in grado di soddisfare la mia condizione...";
			emotion ET_SCRATCH;
			next;
			mes "[La Fenn]";
			mes "Ecco il tuo premio. È tuo per sempre.";
			set Q_99LVL,0;
			getitem 12106,1; //Jewel Case
			emotion ET_CONGRATULATION;
		} else {
			mes callfunc("F_Hi");
			mes "...";
			emotion ET_SLEEPY;
			next;
			mes "[La Fenn]";
			mes "Ho un indovinello per un uomo forte di Seconda Classe e Livello Base 99.";
			mes "*Yawn* Se ne incontri alcuni/qualcuno, per favore parlane del mio indovinello.";
		}
		close;
	}
	if(BaseLevel==99 && Upper==0) {
		if(Q_99LVL) {
			mes "Ora vai e diventa più giovane!";
			mes "Se risolvi il mio indovinello, allora ti darò un incantato Contenitore Gioielli con un raro accessorio.";
		} else {
			mes callfunc("F_Hi")+" The Hero!";
			if(BaseExp<1000000) {
				mes "Ora sei forte.";
				mes "Ma... non abbastanza forte per il mio indovinello.";
				if(rand(2)) {
					mes "Uccidere "+((1000000-BaseExp)/3765+1)+" più sigilli e restituisci.";
				} else {
					mes "Uccidere "+((1000000-BaseExp)/81+1)+" più Poporings e torna indietro.";
				}
			} else {
				mes "Ora sei VERAMENTE forte.";
				mes "E se diventi più giovane di adesso ti darò il mio tesoro.";
				set Q_99LVL,1;
				emotion ET_BEST;
			}
		}
		close;
	}
	emotion ET_SCRATCH;
	mes "...";
	close;
}
