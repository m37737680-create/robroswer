prt_castle,80,170,4	script	King of Prontera	108,{
	mes "Re di Prontera";
	mes "Ciao, puoi aiutarmi?";
	next;
	if(select("Certo, cosa posso fare?:Scusa, sono occupato/a.")==2) {
		mes "Re di Prontera";
		mes "Va bene, sono sicuro che qualcun altro aiuterà.";
		close; }
	mes "Re di Prontera";
	mes "Grazie. Beh, mia figlia accanto a me è un po' birichina...";
	mes "Se le dai questi oggetti sono sicuro che si calmerà.";
	mes "Ti darò la mia corona o la mia spada come ricompensa.";
	next;
	setarray .@Items1[0],5007,742,30,741,50,753,20,732,1,710,20,629,20;
	setarray .@Items2[0],1137,2254,1,2282,1,714,1;
	switch(select("Di cosa hai bisogno?:Ho gli oggetti!:Dimenticalo.")) {
		case 1:
			for(set .@i,1; .@i<3; set .@i,.@i+1) {
				mes "Re di Prontera";
				mes "Se potessi prendermi:";
				for(set .@j,1; .@j<getarraysize(getd(".@Items"+.@i)); set .@j,.@j+2)
					mes " ~ ^0055FF"+getd("@.Items"+.@i+"["+(.@j+1)+"]")+"^000000x "+getitemname(getd("@.Items"+.@i+"["+.@j+"]"));
				mes "Darò il mio "+((.@i == 2 )?"spada":"corona.");
				if (.@i==2) close;
				next; }
		case 2:
			mes "Re di Prontera";
			mes "Quale articolo vuoi?";
			next;
			set .@i, select("Corona:Spada");
			for(set .@j,1; .@j<getarraysize(getd(".@Items"+.@i)); set .@j,.@j+2)
				if (countitem(getd(".@Items"+.@i+"["+.@j+"]")) < getd(".@Items"+.@i+"["+(.@j+1)+"]")) set .@nr,1;
			mes "Re di Prontera";
			if (.@nr) { 
				mes "Ti manca qualcosa.";
				mes "Torna dopo aver preso TUTTI gli oggetti.";
				close; }
			for(set .@j,1; .@j<getarraysize(getd(".@Items"+.@i)); set .@j,.@j+2)
				delitem getd(".@Items"+.@i+"["+.@j+"]"), getd(".@Items"+.@i+"["+(.@j+1)+"]");
			getitem getd(".@Items"+.@i+"[0]"),1;
			mes "GRAZIE MOLS!";
			mes "Questo la calmerà sicuramente.";
			mes "Come promesso, ecco il mio "+((.@i==1)?"corona.":"spada");
			close;
		case 3:
			mes "Re di Prontera";
			mes "Va bene, sono sicuro che qualcun altro aiuterà.";
			close;
	}
}
