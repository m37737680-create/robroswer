morocc,208,90,6	script	Turban Thief	58,{
	mes "Ladro di turbanti";
	if(event_umbala < 2) {
		mes "Cosa vuoi con me? Via'!";
		close; }
	if(sphmask_q) {
		mes "Non hai più affari con me, via'!";
		close; }
	mes "Ciao mio amico, saresti interessato a questa maschera rara? Il suo valore te lo assicuro mio amico. È meravigliosa!";
	next;
	callsub L_Menu,1,"1,000,000",1000000;
	callsub L_Menu,2,"750,000",750000;
	callsub L_Menu,3,"500,000",500000;
	next;
	mes "Ladro di turbanti";
	mes "Accà! Forgiatelo! Posso fare meglio altrove!";
	set sphmask_q,1;
	close;

L_Menu:
	mes "Ladro di turbanti";
	switch(getarg(0)) {
		case 1: mes "Wah? Oh ho ho ho, sai che ti rubo questo da quei selvaggi, vero? Beh non te lo ridarò. Ma lo farò per un prezzo... cosa dici?"; break;
		case 2: mes "Ah un uomo d'affari sei tu, vero? Bene, che ne dici di..."; break;
		case 3: mes "Mmm... sei un negoziatore difficile, ok... la mia ultima offerta..."; break; }
	next;
	if(select("Paga "+getarg(1)+"z:Nessun accordo")==2) return;
	mes "Ladro di turbanti";
	if (Zeny < getarg(2)) {
		mes "Stai giocando con me? Non hai i soldi!";
		close; }
	set Zeny, Zeny-getarg(2);
	mes "Oh oh oh, è un affare, allora!";
	getitem 7114,1;
	set sphmask_q,1;
	close;
}
