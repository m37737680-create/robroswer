yuno_in01,124,164,6	script	Guild Leader	108,{
	mes "[Keegan]";
	mes "Cosa ti porta qui? Hai qualcosa da dire?";
	next;
	menu "Voglio essere un cacciatore di tesori",-,"Accetta una nuova missione.",N_NQuest,"Portami al negozio ufficiale.",N_OShop,"Non importa.",N_NVM;
	if(Event_THQS==1) goto N_AlreadyReg;
	mes "[Keegan]";
	mes "Beh "+strcharinfo(0)+" se vuoi essere un cacciatore di tesori devi prima.";
	mes " ";
	mes "^FF00001.^000000 Ti verrà addebitato ^FF000025,000z^000000 per l'addestramento di base e la prova della tua appartenenza.";
	mes " ";
	mes "Ehm... Beh, è tutto quello che devi fare. Hahaha.";
	next;
	menu "Paga ^FF000025.000z^000000.",-,"Troppo alto!!",N_HighPrice;
	if (Zeny < 25000) goto N_NoZeny;
	set Zeny,Zeny-25000;
	//getitem 7950, 1;
	//getitem 7951, 1;
	set Event_THQS,1;
	mes "[Keegan]";
	mes "Congratulazioni!";
	emotion ET_CONGRATULATION;
	next;
	mes "[Keegan]";
	mes "Benvenuti nella Gilda dei Cercatori di Tesori.";
	mes "Puoi ora intraprendere una missione di caccia al tesoro come preferisci.";
	close;
N_NoZeny:
	mes "[Keegan]";
	mes "Mmmh, non sembri avere ^FF000025,000z^000000 "+strcharinfo(0)+"Ti prego di tornare quando sarai pronto/a.";
	close;
N_HighPrice:
	mes "[Keegan]";
	mes "Beh se non puoi permetterti un piccolo ^FF000025,000z^000000 non dovresti essere ancora un Cacciatore di Tesori.";
	close;
N_AlreadyReg:
	mes "[Keegan]";
	mes "Ah ah ah...sei già un membro.";
	close;
N_NQuest:
	if (Event_THQS!=1) goto A_NeedReg;
	mes "[Keegan]";
	mes "Beh "+strcharinfo(0)+" se desideri accettare una missione parla con il compagno Cercatore di Tesori nell'altra stanza lì.";
	close;
A_NeedReg:
	mes "[Keegan]";
	mes "Scusa, devi essere un membro della Gilda dei Cacciatori di Tesori se desideri accettare una missione.";
	mes " ";
	mes "Anche ora che sei un membro sentiti libero di usare i nostri negozi quando vuoi. Sono tutti situati su questo piano.";
	mes " ";
	mes "Abbiamo anche un altro negozio solo per i membri, in cui usiamo piccoli diamanti metallici chiamati Gettoni Cacciatore di Tesori per scambiare beni rari.";
	mes " ";
	mes "Avrai anche accesso al nostro esclusivo Dungeon personale.";
	close;
N_OShop:
	if (Event_THQS!=1) goto N_NeedMem;
	mes "[Keegan]";
	mes "Sì "+strcharinfo(0)+" Ti porterò subito al nostro negozio!";
	next;
	warp "prt_in",166,171;
	close;
	end;
N_NeedMem:
	mes "[Keegan]";
	mes "Mi dispiace, solo i membri possono visitare il nostro negozio privato.";
	close;
N_NVM:
	mes "[Keegan]";
	mes "Tornaci quando hai tempo libero da dedicare.";
	close;
}
