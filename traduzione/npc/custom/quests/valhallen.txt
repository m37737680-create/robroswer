prontera,147,171,5	script	Lenneth	811,{
	mes "[Lenneth]";
	mes "Vengo da Valhalla per aiutarti.";
	mes "Posso trasmutare per te molti degli oggetti di Valhalla,";
	mes "sebbene richiedano molti elementi rari per essere realizzati.";
	next;
	switch(select("Mjolnir:Sleipnir:Brisingamen:Gleipnir:Megingjard:Asprika:Brynhild:Fuori:Jormungand:Muspellium:Essenza di Runa:Chiudi")) {
		case 1:
			mes "[Lenneth]";
			mes "^3355FFMjolnir^000000 è il potente Martello di Thor.";
			setarray .@Items[0],1530,1522,2,984,20,985,5,969,40,7074,2,7075,4,7078,5,7091,5,7089,5;
			break;
		case 2:
			mes "[Lenneth]";
			mes "^3355FFSleipnir^000000 sono stivali ispirati al Cavallo di Guerra di Odino.";
			setarray .@Items[0],2410,2406,2,969,20,985,10,7076,3,7079,5,7083,3,7086,4,7087,3;
			break;
		case 3:
			mes "[Lenneth]";
			mes "^3355FFBrisingammen^000000 è la magica collana di Freyja, dea della Bellezza.";
			setarray .@Items[0],2630,2603,1,726,2,722,3,727,10,724,5,969,20,7073,4,7077,4,7088,3,7090,3,7092,3;
			break;
		case 4:
			mes "[Lenneth]";
			mes "Il ^3355FFGleipnir^000000 è una corda leggera ma forte necessaria per creare ^3355FFMegingjard^000000";
			setarray .@Items[0],7058,7080,4,7081,5,7082,4,7084,3,7085,3;
			break;
		case 5:
			mes "[Lenneth]";
			mes "Il ^3355FFMegingjard^000000 è la potente Cintura di Thor.";
			setarray .@Items[0],2629,7058,1,2627,1,969,20,726,10,984,10;
			break;
		case 6:
			mes "[Lenneth]";
			mes "^3355FFAsprika^000000 è la luce degli dei.";
			setarray .@Items[0],2541,7835,1,7836,1,7837,1,7838,1,2513,1,7063,100,7291,10,7293,10,985,20;
			break;
		case 7:
			mes "[Lenneth]";
			mes "La ^3355FFBrynhild^000000 è l'armatura del guerriero.";
			setarray .@Items[0],2383,7830,1,7831,1,7832,1,7833,1,7834,1,7510,100,2357,1,969,10,985,20;
			break;
		case 8:
			mes "[Lenneth]";
			mes "Il ^3355FFHervor^000000 prende il nome dalla valchiria che assicura la vittoria ai suoi guerrieri.";
			setarray .@Items[0],2181,6595,2,6596,1,6594,4,6597,3,6602,1,6604,1,6605,1,2115,1;
			break;
		case 9:
			mes "[Lenneth]";
			mes "Il ^3355FFJormungand^000000 è famoso per la sua capacità di avvelenare il nemico.";
			setarray .@Items[0],2020,6603,4,6599,1,6598,1,6601,4,6600,1,6604,1,6605,1,1473,1;
			break;
		case 10:
			mes "[Lenneth]";
			mes "Il ^3355FFMuspellium^000000 è necessario per creare ^3355FFJormungand^000000 e ^3355FFHervor^000000.";
			setarray .@Items[0],6605,984,50,985,50;
			break;
		case 11:
			mes "[Lenneth]";
			mes "The ^3355FFEssence Of Rune^000000 is required to make ^3355FFJormungand^000000 and ^3355FFHervor^000000.";
			setarray .@Items[0],6604,12735,200,12736,100;
			break;
		default:
			close;
	}
	
	mes "^FF0000The items I need are as follows:^000000";
	set .@itemmissing,0;
	for(set .@i,1; .@i<getarraysize(.@Items); set .@i,.@i+2) {
		set .@amount,.@Items[.@i+1];
		set .@id,.@Items[.@i];
		mes " ~ "+.@amount+"x "+((countitem(.@id)<.@amount)?("^dd3355"):("^33dd55"))+getitemname(.@id)+((getitemslots(.@id)>0)?" ["+getitemslots(.@id)+"]":"")+"^000000";
		if (countitem(.@id) < .@amount) 
			set .@itemmissing,1;
	}
	if (.@itemmissing) {
		next;
		mes "[Lenneth]";
		mes "Vai e trova tutti questi articoli.";
		mes "Aspetterò qui per assisterti.";
		close;
	}
	next;
	mes "[Lenneth]";
	mes "Desideri ^3355FF"+getitemname(.@Items[0])+"^000000?";
	next;
	if(select("No","Sì")==1) {
		mes "[Lenneth]";
		mes "Per favore, fai sapere se cambi idea.";
		close;
	}
	mes "[Lenneth]";
	mes "Ecco il tuo ^3355FF"+getitemname(.@Items[0])+"^000000.";
	mes "Che ti sia d'aiuto.";
	//second check if the player tries to bypass the first
	for(set .@i,1; .@i<getarraysize(.@Items); set .@i,.@i+2) {
		set .@amount,.@Items[.@i+1];
		set .@id,.@Items[.@i];
		if (countitem(.@id) < .@amount) 
			close;
	}
	// Delete the items _after_ the second check
	for(set .@i,1; .@i<getarraysize(.@Items); set .@i,.@i+2) {
		set .@amount,.@Items[.@i+1];
		set .@id,.@Items[.@i];
		delitem .@id, .@amount;
	}
	getitem .@Items[0],1;
	close;
}
