//===== rAthena Script =======================================
//= Reset NPC
//===== Description: =========================================
//= Resets skills, stats, or both.
//===== Additional Comments: =================================
//= 1.0 First Version
//= 1.1 Optimized for the greater good. [Kisuka]
//= 1.2 Cleaning [Euphy]
//= 1.3 All statuses removed upon skill reset. [Euphy]
//= 1.4 Compressed Script, Added limit use option [Stolao]
//=	Changed set -> setarray, Improved text with F_InsertPlural
//= 1.5 Added sc_end_class to reset related status changes [sader1992]
//============================================================
prontera,150,193,4	script	Reset Girl	124,{
	//	 		Skills,	Stats,	Both,	Limit
	setarray .@Reset,	5000,	5000,	9000,	0;
	mes "[Ragazza Reset]";
	if(.@Reset[3] && reset_limit > .@Reset[3]) {
		mes "Scusa puoi solo resettare "+callfunc("F_InserisciPlurale",.@Reset[3],"tempo")+" nella tua vita.";
		close;
	}
	mes "Sono la Ragazza del Reset.";
	mes "Resetta Statistiche: "+ callfunc("F_InserisciVirgola",.@Reset[1]) +"z";
	mes "Resetta Abilità: "+ callfunc("F_InserisciVirgola",.@Reset[0]) +"z";
	mes "Resetta Entrambi: "+ callfunc("F_InserisciVirgola",.@Reset[2]) +"z";
	if(.@Reset[3]) mes "Puoi resettare solo "+callfunc("F_InserisciPlurale",.@Reset[3],"tempo")+", quindi usa "+((.@Reset[3]>1)?"loro":"esso")+" saggiamente.";
	mes "Seleziona il servizio che desideri:";
	next;
	set .@i,(select("^FF3355Ripristina Abilità:Resetta Statistiche:Resetta Entrambi^000000:Annulla"));
	if(.@i > 3) close;
	mes "[Ragazza Reset]";
	if (Zeny < .@Reset[.@i-1]) {
		mes "Scusa, non hai abbastanza Zeny.";
		close;
	}
	if(.@Reset[3]){
		mes "Puoi solo resettare "+callfunc("F_InserisciPlurale",.@Reset[3],"tempo")+" nella tua vita, sei sicuro?";
		if(select("Fammi pensare:Va bene") == 1) close;
	}
	set Zeny, Zeny-.@Reset[.@i-1];
	if(.@i&1){
		sc_end_class;
		ResetSkill;
	}
	if(.@i&2) ResetStatus;
	mes "Eccolo/a!";
	if(.@Reset[3]) set reset_limit,reset_limit + 1;
	close;
}
