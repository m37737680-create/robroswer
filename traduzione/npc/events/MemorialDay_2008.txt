			getexp 93750,43750;
		}

		mes "C'è un altro messaggio.";
		next;
		mes "Forse non sono andato dove intendevo andare,-";
		mes "ma penso di essere arrivato dove dovevo stare.";
		mes "- Douglas Adams.-";

		}
		next;
