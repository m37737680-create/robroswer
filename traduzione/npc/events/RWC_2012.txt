prontera,147,61,3	script	Driller#pron	87,{
	disable_items;
	if (checkweight(1201,1) == 0) {
		mes "Stai portando troppi oggetti nel tuo inventario. Visita il deposito di Kafra e riprova.";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "Non puoi procedere perché sei in sovrappeso.";
		close;
	}
	set .@part, EQI_ACC_L;
	if (!getequipisequiped(.@part)) {
		mes "Perforatore";
		mes "Il mio lavoro è forare uno slot per schede negli accessori RWC Memorial.";
		next;
		mes "Perforatore";
		mes "Mi dispiace ma non hai nessun oggetto equipaggiato nella tua posizione accessorio destra.";
		close;
	}
	mes "Perforatore";
	mes "Il mio lavoro è forare uno slot per schede negli accessori RWC Memorial. Inoltre, tratto solo gli articoli ^ff0000puri^000000, o quelli che non sono stati incantati.";
	next;
	set .@equip_id, getequipid(.@part);
	if (.@equip_id != 2966 && .@equip_id != 2968) {
		mes "Perforatore";
		mes "Tuttavia, posso vedere che l'accessorio che indossi sul lato destro non può essere trattato. Per favore, equipaggia un accessorio RWC Memorial.";
		close;
	}
	mes "Perforatore";
	mes "Dovresti anche sapere che creare uno slot per schede è estremamente pericoloso. ^ff0000Le possibilità di successo sono circa il 50%.^000000 Desideri procedere?";
	next;
	if(select("Annulla:Andiamo!") == 1) {
		mes "Perforatore";
		mes "A presto.";
		close;
	}
	if (.@equip_id == 2966) {
		set .@slotted, 2967; //RWC_2012_Ring_
		set .@name$,"RWC 2012 Memorial Ring";
		set .@str$,"ring";
	} else if (.@equip_id == 2968) {
		set .@slotted, 2969; //RWC_2012_Pendant_
		set .@name$,"RWC 2012 Memorial Pendant";
		set .@str$,"pendant";
	} else {
		mes "Perforatore";
		mes "Non riesco a identificare l'accessorio che indossi sulla mano destra. Non posso lavorarci sopra.";
		close;
	}
	if (getequipcardid(.@part,3) > 0) {
		mes "Perforatore";
		mes "Questo oggetto è già stato incantato. Non posso lavorarci perché va contro le regole.";
		close;
	}

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@equip_id))
		close;

	delequip .@part;
	if (rand(1,10) > 5) {
		getitem .@slotted,1;
		specialeffect2 EF_REPAIRWEAPON;
		mes "Perforatore";
		mes "Yay! Successo! Tuo "+.@name$+" ora ha uno slot per la scheda. Guardatelo!";
		close;
	} else {
		specialeffect2 EF_LORD;
		mes "Perforatore";
		mes "Aww... Dann debole "+.@str$+"È rotto durante la procedura. Mi dispiace.";
		close;
	}
}

prontera,147,59,3	script	Goldberg#pron	878,{
	disable_items;
	if (checkweight(1201,1) == 0) {
		mes "Stai portando troppi oggetti. Torna indietro dopo aver organizzato il tuo inventario.";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "Non puoi procedere perché sei in sovrappeso.";
		close;
	}
	mes "[Goldberg]";
	mes "Ciao! Sono responsabile di incantare gli accessori commemorativi RWC con qualche potere mistico.";
	next;
	set .@part, EQI_ACC_L;
	if (!getequipisequiped(.@part)) {
		mes "[Goldberg]";
		mes "Mi dispiace ma non hai nessun oggetto equipaggiato nella tua posizione accessorio destra.";
		close;
	}
	set .@equip_id, getequipid(.@part);
	if (.@equip_id < 2966 || .@equip_id > 2969) {
		mes "[Goldberg]";
		mes "Tuttavia, posso vedere che l'accessorio che indossi non è qualcosa su cui posso lavorare. Per favore, equipaggia un accessorio RWC Memorial.";
		close;
	}
	set .@select, select("Scusa, non sono interessato/a.:Per favore, potenzia il mio accessorio.:Rimuovi l'Incantesimo.")-1;
	if (.@select == 0) {
		mes "[Goldberg]";
		mes "Allora, ci vediamo la prossima volta...";
		close;
	}
	setarray .@equip_card[0], getequipcardid(.@part,0),getequipcardid(.@part,1),getequipcardid(.@part,2),getequipcardid(.@part,3);
	if (.@select == 1) {
		switch(getequipid(.@part)) {
		case 2966:
			setarray .@option[0],2,2,4,4;
			break;
		case 2967:
			setarray .@option[0],0,2,4,4;
			break;
		case 2968:
			setarray .@option[0],1,1,3,3;
			break;
		case 2969:
			setarray .@option[0],0,1,3,3;
			break;
		default:
			mes "[Goldberg]";
			mes "Mi dispiace, ma non posso lavorare sull'accessorio che indossi attualmente.";
			mes "Se hai equipaggiato l'accessorio RWC Memorial sul lato sinistro, prova a spostarlo sul lato destro.";
			close;
		}
		for(set .@i,3; .@i>=0; set .@i,.@i-1) {
			if (.@equip_card[.@i] == 0) {
				set .@slot, .@i;
				set .@op_type, .@option[.@i];
				break;
			}
		}
		switch(.@op_type) {
		case 4:
			mes "[Goldberg]";
			mes "Quale incantesimo vorresti infondere?";
			next;
			setarray .@enchant_select[0],1,2,3,4;
			set .@i, select("Annulla:Spirito combattivo:ATK (%):Max PV:HP")-2;
			break;
		case 3:
			setarray .@enchant_select[0],5,6,7;
			set .@i, select("Annulla:Spellere:MATK (%):SP")-2;
			break;
		case 2:
			mes "[Goldberg]";
			mes "^ff0000Attenzione! C'è circa una possibilità del 25% che l'incantesimo fallisca. Se succede questo, l'oggetto verrà distrutto.^000000 Quale incantesimo vorresti infondere?";
			next;
			setarray .@enchant_select[0],8,9,10,11,12,13,14;
			set .@i, select("Annulla:STR:Intelligenza Artificiale Generale:VIT:Interno:DEX:LUK:SP")-2;
			break;
		case 1:
			mes "[Goldberg]";
			mes "^ff0000C'è circa una possibilità del 25% che l'incantesimo fallisca. Se ciò accade, l'oggetto verrà distrutto.^000000Quale incantesimo vorresti infondere?";
			next;
			setarray .@enchant_select[0],8,9,10,11,12,13,15,16;
			set .@i, select("Annulla:STR:Intelligenza Artificiale Generale:VIT:Interno:DEX:LUK:MHP:HP")-2;
			break;
		case 0:
			mes "[Goldberg]";
			mes "Il tuo accessorio ha ricevuto così tanti incantesimi che faccio fatica a lavorarci ancora.";
			close;
		}
		if (.@i == -1) {
			mes "[Goldberg]";
			mes "Allora, ci vediamo la prossima volta.";
			close;
		}
		mes "[Goldberg]";
		mes "Il potere dell'incantesimo sarà scelto casualmente. ^ff0000Una volta infuso, l'incantesimo non può essere rimosso.^000000 Continuiamo?";
		next;
		if(select("No, per favore smetti.:Sì, per favore procedi.") == 1) {
			mes "[Goldberg]";
			mes "Allora, ci vediamo la prossima volta...";
			close;
		}
		set .@enchant_type, .@enchant_select[.@i];
		if (!getequipisequiped(.@part)) {
			mes "[Goldberg]";
			mes "Non togliere l'attrezzatura mentre sto lavorando, ok?";
			close;
		}
		switch(.@enchant_type) {
		case 1:
			setarray .@enc[0],4811,4810,4809; //Fighting_Spirit1,Fighting_Spirit2,Fighting_Spirit3
			break;
		case 2:
			setarray .@enc[0],4819,4766,4767; //Atk1,Atk2,Atk3
			break;
		case 3:
			setarray .@enc[0],4861,4862,4867; //MHP1,MHP2,MHP3
			break;
		case 4:
			setarray .@enc[0],4795,4796,4797; //HP100,HP200,HP300
			break;
		case 5:
			setarray .@enc[0],4760,4761,4806; //Matk1,Matk2,Matk3
			break;
		case 6:
			setarray .@enc[0],4815,4814,4813; //Spell1,Spell2,Spell3
			break;
		case 7:
			setarray .@enc[0],4870,4800,4871; //SP25,SP50,SP75
			break;
		case 8:
			setarray .@enc[0],4700,4701,4702; //Strength1,Strength2,Strength3
			break;
		case 9:
			setarray .@enc[0],4730,4731,4732; //Agility1,Agility2,Agility3
			break;
		case 10:
			setarray .@enc[0],4740,4741,4742; //Vitality1,Vitality2,Vitality3
			break;
		case 11:
			setarray .@enc[0],4710,4711,4712; //Inteligence1,Inteligence2,Inteligence3
			break;
		case 12:
			setarray .@enc[0],4720,4721,4722; //Dexterity1,Dexterity2,Dexterity3
			break;
		case 13:
			setarray .@enc[0],4750,4751,4752; //Luck1,Luck2,Luck3
			break;
		case 14:
			setarray .@enc[0],4870,4800,4871; //SP25,SP50,SP75
			break;
		case 15:
			setarray .@enc[0],4861,4862,4867; //MHP1,MHP2,MHP3
			break;
		case 16:
			setarray .@enc[0],4795,4796,4797; //HP100,HP200,HP300
			break;
		default:
			mes "[Goldberg]";
			mes "Abbiamo un problema, lascio controllare.";
			close;
		}

		if (.@enchant_type < 8)
			set .@i, rand(1,300); // 0% break chance.
		else
			set .@i, rand(1,400); // 25% break chance.

		     if (.@i < 151) set .@enchant, .@enc[0];
		else if (.@i < 251) set .@enchant, .@enc[1];
		else if (.@i < 301) set .@enchant, .@enc[2];
		else set .@enchant,9;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) ||
		    callfunc("F_IsEquipCardHack", .@part, .@equip_card[0], .@equip_card[1], .@equip_card[2], .@equip_card[3]))
			close;

		set .@equip_card[.@slot], .@enchant;
		if (.@slot == 2 && .@enchant == 0) {
			set .@equip_card[3],0;
		} else if (.@slot == 1 && .@enchant == 0) {
			set .@equip_card[2],0;
			set .@equip_card[3],0;
		} else if (.@slot == 0 && .@enchant == 0) {
			set .@equip_card[1],0;
			set .@equip_card[2],0;
			set .@equip_card[3],0;
		}

		set .@equip_refine, getequiprefinerycnt(.@part);
		delequip .@part;
		if (.@enchant == 9) {
			mes "[Goldberg]";
			mes "Oh cavolo!";
 			mes "L'oggetto non era abbastanza resistente da sopportare l'incantesimo e quindi è stato distrutto. Mi dispiace.";
			specialeffect2 EF_LORD;
			close;
		}
		if (.@enchant == 0) { // Should never happen.
			mes "[Goldberg]";
			mes "Oh... Sembra che ci sia stata un'instabilità di qualche tipo tra tutti i poteri infusi. Questo ha fatto svanire tutti gli incantesimi. È un peccato, ma per favore riprova!";
		} else {
			mes "[Goldberg]";
			mes "Ottimo!";
			mes "L'incantesimo è riuscito! Verrà applicato nel socket N.º^990000"+(.@slot+1)+"^000000.";
			specialeffect2 EF_REPAIRWEAPON;
		}

//		GetNonSlotItemSock2 .@equip_refine .@equip_id .@equip_card[0] .@equip_card[1] .@equip_card[2] .@equip_card[3]
		getitem2 .@equip_id,1,1,.@equip_refine,0,.@equip_card[0],.@equip_card[1],.@equip_card[2],.@equip_card[3];

		close;
	} else if (.@select == 2) {
		mes "[Goldberg]";
		mes "Inizializzerò solo l'opzione incantesimo senza fare nulla alla carta inserita. Vuoi continuare?";
		next;
		if(select("Mi fermerò.:Sì, certo, continua pure.") == 1) {
			mes "[Goldberg]";
			mes "Torna indietro se cambi idea.";
			close;
		}
		if (countitem(6665) == 0) {
			mes "[Goldberg]";
			mes "Mi dispiace. Ma non hai il coupon di inizializzazione RWC. Puoi controllare il tuo inventario?";
			close;
		}
		if (getiteminfo(.@equip_card[3], ITEMINFO_SUBTYPE) != CARD_ENCHANT) {
			mes "[Goldberg]";
			mes "Hmm... questa attrezzatura è pulita. Non posso inizializzarla se non c'è nulla! Ricontrollala.";
			close;
		}
		specialeffect2 EF_REPAIRWEAPON;
		mes "[Goldberg]";
		mes "L'opzione incantesimo del tuo oggetto verrà inizializzata.";
		delitem 6665,1; //RWC_Inicializer
		delequip .@part;
	
//		GetNonSlotItemSock2 .@equip_refine .@equip_id .@equip_card[0] .@equip_card[1] .@equip_card[2] .@equip_card[3]
		for ( .@i = getiteminfo(.@equip_id, ITEMINFO_SLOT); .@i < MAX_SLOTS; .@i++ ) {
			if (getiteminfo(.@equip_card[.@i], ITEMINFO_SUBTYPE) == CARD_ENCHANT)
				.@equip_card[.@i] = 0;// Armor Enchant System
		}
		getitem2 .@equip_id,1,1,.@equip_refine,0,.@equip_card[0],.@equip_card[1],.@equip_card[2],.@equip_card[3];
		
		close;
	}
}
