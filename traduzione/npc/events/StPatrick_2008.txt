	end;

	if (StPatrick2008 < 1) {
		mes "Hai trovato un mucchio di rocce coperto di terra.";
		close;
	}
	if (StPatrick2008 == 1) {
		mes "Hai trovato un mucchio di rocce coperto di terra.";
		next;
		if (countitem(12018) < 200) {
			mes "Le rocce non si muoveranno.";
			mes "Forse O'Riley sa un modo per spostare le rocce.";
			close;
		}
		else {
			if (select("Usare i petardi.:Ignora") == 2) {
				close;
			}
			mes "Hai seppellito 200 petardi sotto l'accumulo di rocce.";
			mes "Accendi la miccia.";
			next;
			mes "*BOOM!*";
			specialeffect2 EF_LORD;
			next;
			mes "Dopo che una nube di polvere e fumo si è dissipata,";
			mes "Hai trovato una scatola tra le rocce e il terreno.";
			mes "Questa scatola deve contenere i beni di valore di O'Riley.";
			mes "Portiamo la scatola a O'Riley.";
			close2;
			set StPatrick2008,2;
			delitem 12018,200;
			getitem 7721,1;
		}
	}
	end;
}

		end;

		end;
	case 6:
		getitem 7915,2;
		end;
	}
}
