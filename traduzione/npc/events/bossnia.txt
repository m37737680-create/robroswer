	mes "[Riss]";

	mes "[Riss]";
	mes "Um... comunque voglio dire... qualcosa..";
	mes "Mentre stavo viaggiando attraverso alcuni posti,";
	mes "Ho trovato un posto davvero spaventoso.";
	next;
	mes "[Riss]";
	mes "La maggior parte delle volte quando arrivi in un posto,";
	mes "c'è un mostro forte e spaventoso.";
	mes "Non è vero?";
	next;
	mes "[Riss]";
	mes "Ma... ma... lì dentro...";
	mes "Ci sono molti mostri spaventosi e forti lì dentro...";
	mes "È stato davvero spaventoso.";
	next;
	mes "[Riss]";
	mes "Se avessi reagito un po' più tardi... qualche secondo...";
	mes "Potrei essere stato ucciso.";
	next;
	mes "[Riss]";
	mes "Cosa?";
	mes "Vuoi entrare lì?";
	mes "Oh~ Ragazzo~ non mi hai capito.";
	mes "Lì dentro.......";
	next;
	mes "[Riss]";
	mes "Uh... lo sai già?";
	mes "Anche se conosci il posto, vuoi andare in...";
	mes "Bene, te lo invierò lì.";
	mes "Ma dopo che ci sei andato, non trattenere";

	// If the VIP system is enabled, players pay a Reset Stone for 5 entrances (with a choice of warps).
	// If it's disabled, players pay 5,000z per entrance.
	if (VIP_SCRIPT) {
		mes "un rancore contro di me. Ti costerà 1 Pietra di Resetta per 5 accessi.";
		next;
		mes "[Riss]";
		if (bossnia_event > 0) {
			set .@type,1;
			mes "Accesso rimanente: "+bossnia_event;
		} else if (countitem(6320)) {
			set .@type,2;
			mes "Vuoi andare?";
		} else {
			mes "Non hai un biglietto ora.....";
			mes "Quindi torna da me con una Pietra di Resetta più tardi.";
			close;
		}
		next;
		if(select("Entra:Lasciare") == 2)
			close;
		set .@i, select("Primo:Secondo:Terzo:Quarto");
		switch(select("Distorsione 1:Warp 2:Warp 3:Warp 4")) {
			case 1: set .@x,31; set .@y,208; break;
			case 2: set .@x,31; set .@y,31; break;
			case 3: set .@x,208; set .@y,31; break;
			case 4: set .@x,208; set .@y,208; break;
		}
		if (.@type == 1)
			set bossnia_event, bossnia_event-1;
		else {
			delitem 6320,1; //Premium_Reset_Stone
			set bossnia_event,4;
		}
		specialeffect2 EF_MAXPOWER;
		warp "bossnia_0"+.@i,.@x,.@y;
		close;
	} else {
		mes "un rancore contro di me. Inoltre costa 5.000 zeny.";
		next;
		if (Zeny > 4999) {
			mes "[Riss]";
			mes "Ti piacerebbe davvero accettare la sfida?";
			mes "Va bene, scegli solo il corso.";
			next;
			set .@i, select("Primo:Secondo:Terzo:Quarto");
			mes "[Riss]";
			mes "Stai attento, ragazzo~";
			mes "Non tenermi rancore.";
			close2;
			set Zeny, Zeny - 5000;
			warp "bossnia_0"+.@i,rand(202,204),rand(202,204);
			end;
		} else {
			mes "[Riss]";
			mes "Non hai abbastanza soldi...";
			mes "Torna quando avrai almeno 5.000 zeny.";
			close;
		}
	}
}
