		mes "[Nerlen]";

		mes "portarmi...";
		next;
		mes "[Nerlen]";
		mes "^4D4DFF1 Erba bianca^000000,";
		mes "^4D4DFF1 Erba Verde^000000,";
		mes "^4D4DFF1 Erba Gialla^000000,";
		mes "^4D4DFF1 Acqua Santa^000000, e";
		mes "^4D4DFF1 Uovo di pollo^000000.";
		close;
	case 2:
		mes "[Nerlen]";
		mes "Allora, quanti";
		mes "Sacri uova vorresti";
		mes "farmi come te?";
		mes "Posso fare fino a";
		mes "10 Uova di Pasqua alla volta.";
		mes "Inserisci '0' per annullare.";
		next;
		while(1) {
			input .@input;
			if (.@input == 0) {
				mes "[Nerlen]";
				mes "Oh, hai cambiato il tuo";
				mes "mente? Sentiti libero di visitarmi";
				mes "quando vuoi che io";
				mes "ti faccio delle uova di Pasqua~";
				close;
			}
			else if (.@input < 0 || .@input > 10) {
				mes "[Nerlen]";
				mes "Mi dispiace, ma posso solo";
				mes "fare fino a 10 uova di Pasqua sacre";
				mes "alla volta. Se volessi";
				mes "cancellare, poi inserire solo '0.'";
				next;
			}
			else {
				if ((countitem(509) < .@input) || (countitem(511) < .@input) || (countitem(508) < .@input) || (countitem(523) < .@input) || (countitem(7605) < .@input)) {
					mes "[Nerlen]";
					mes "Mi dispiace, ma tu non";
					mes "avere abbastanza articoli per fare";
					mes "Uova sacre. Hai bisogno del";
					mes "seguimi per creare";
					mes "ogni Uovo Santo per te.";
					next;
					mes "[Nerlen]";
					mes "^4D4DFF1 Erba bianca^000000,";
					mes "^4D4DFF1 Erba Verde^000000,";
					mes "^4D4DFF1 Erba Gialla^000000,";
					mes "^4D4DFF1 Acqua Santa^000000, e";
					mes "^4D4DFF1 Uovo di pollo^000000.";
					close;
				}
				else {
					break;
				}
			}
		}
		if (checkweight(12019,.@input) == 0) {
			mes "[Nerlen]";
			mes "Mi dispiace, ma tu";
			mes "non sarà in grado di portare";
			mes "tanti uova di Pasqua ora...";
			mes "Meglio liberare altro spazio";
			mes "spazio nel tuo Inventario prima.";
			close;
		}
		else {
			delitem 509,.@input; //White_Herb
			delitem 511,.@input; //Green_Herb
			delitem 508,.@input; //Yellow_Herb
			delitem 523,.@input; //Holy_Water
			delitem 7605,.@input; //Egg
			getitem 12019,.@input; //Holy_Egg
			mes "[Nerlen]";
			mes "Grazie!";
			mes "Spero ti piaccia";
			mes "queste uova sacre~";
			mes "Spero che tu sia con l'amore di Dio.";
			close;
		}
	case 3:
		mes "[Nerlen]";
		mes "Va bene, capisco.";
		mes "Che Dio ti benedica.";
		close;
	}
}
