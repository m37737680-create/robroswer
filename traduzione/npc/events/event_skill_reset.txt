//= Skill Reset with payment, minimum BaseLv.60 to reset skill.
//= Each 1 BaseLv need 20.000 Zeny, so be careful to use skill.
//============================================================
yuno,138,187,4	script	Hypnotist Teacher	124,{
	set @npcname$,"[^D5A500Hypnotist^000000]";
	mes @npcname$;

	if(MISC_QUEST & 1024){
		mes "Ti ho già detto che puoi completare questo evento solo una volta.";
		close;
	}
	if(BaseLevel < 60) {
		mes "Per favore torna quando raggiungerai BaseLv 60 o superiore.";
		close;
	}
	if(SkillPoint != 0) {
		mes "Avrai bisogno di usare tutti i tuoi punti abilità se vuoi che io continui.";
		mes "Torna presto!";
		close;
	}
	if(Weight || checkfalcon() || checkcart() || checkriding()) {
		mes "Vuoi resettare le competenze?";
		mes "Mi dispiace, ma...";
		next;
		mes @npcname$;
		if(Weight) {
			mes "Non puoi resettare le abilità";
			mes "quando tieni";
			mes "nessun articolo.";
		} else if(checkcart())
			mes "Per favore, lascia il carrello e continueremo.";
		else if(checkfalcon())
			mes "Per favore, libera il tuo Falco e continueremo.";
		else if(checkriding())
			mes "Per favore, libera il tuo PecoPeco e continueremo.";
		next;
		mes @npcname$;
		mes "Torna presto!";
		close;
	}
	mes "Ciao, sono il Resetatore di Abilità.";
	mes "Il tuo nome è ^0080FF"+strcharinfo(0)+"^000000.";
	mes "Come posso aiutarti?";
	next;
	menu "^009500Information about Reset skills.^000000",L_Info,"^00B6FFResettare abilità.^000000",L_Reset,"^000088Non importa^000000",-;

	mes @npcname$;
	mes "Sai dove trovarmi,";
	mes "se vorrai un reset!!"; 
	close;

L_Info:
	mes @npcname$;
	mes "Questo reset di abilità non è GRATUITO!!";
	mes "Spesa per il reset di abilità è ^D5A50020000 Zeny x BaseLv^000000.";
	mes "Sì...ogni BaseLv costa 20000 Zeny per resettare la abilità.";
	next;
	mes @npcname$;
	mes "Oh sì, un'altra cosa!";
	mes "Quali carri, falconi o pecos hai equipaggiato";
	mes "verrà rimosso se reimposti le tue abilità.";
	next;
	mes @npcname$;
	mes "Solo una volta fa di nuovo per scuotere il punto abilità";
	mes "Attento con le tue abilità d'ora in poi.";
	close;
	
L_Reset:
	mes @npcname$;
	mes "Prima del reset delle abilità all'inizio.";
	mes "Dovrai prima dirmi il tuo Livello Base."; 
	next;
	mes "^D5A500[" + strcharinfo(0) + "]^000000";
	mes "Il mio livello base è ^AA00AALevel " + BaseLevel + "BaseLv.^000000"; 
	next;
	set @zeny,BaseLevel*20000;
	mes @npcname$;
	mes "Totale zeny per il ripristino dell'importo delle abilità ^529DFF" + @zeny + "Zeny^000000 per il servizio di reset abilità.";
	next;
	if(Zeny < @zeny) {
		mes @npcname$;
		mes "Sembra che non abbia abbastanza soldi.";
		mes "Inoltre aspettiamo l'opportunità.";
		emotion ET_SCRATCH;
		close;
	}
	set Zeny, Zeny-@zeny;
	set MISC_QUEST,MISC_QUEST | 1024;
	resetskill;
	mes @npcname$;
	mes "Grazie.";
	emotion ET_THANKS;
//	logmes "SKILL RESET EVENT";
	close;
}
