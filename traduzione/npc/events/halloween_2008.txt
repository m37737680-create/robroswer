payon,162,176,4	script	Halloween Magician#iRO08	704,{
	if (Hallow08 < 1) {
		mes "[Mago di Halloween]";
		mes "Kkkkkkkkk!";
		mes "Ho un evento speciale questo Halloween che mette alla prova la tua fortuna e agilità.";
		mes "Sei interessato?";
		next;
		mes "[Mago di Halloween]";
		mes "Dai! Non fare il codardo!";
		mes "Se raccogli abbastanza biglietti puoi ottenere bei premi!";
		mes "Allora cosa dici?";
		next;
		goto MainMenu;
	}
	if (Hallow08Kill == 1) {
		mes "[Mago di Halloween]";
		mes "Sei un gatto spaventato o qualcosa del genere?!";
		mes "Sai che vuoi riprovare...";
		mes "Conosci le regole?";
		next;
		set Hallow08Kill,0;
		if(select("Sì, lo so.:No, non lo so.") == 2) {
			goto Rules;
		}
		goto Participate;
	}
	if (Hallow08Kill == 2) {
		mes "[Mago di Halloween]";
		mes "Oh, ben fatto! Stai bene!";
		mes "Non è divertente con i zombie?";
		mes "Sai, i zombie erano persone anche loro!";
		mes "Ah!";
		mes "Kkkkkkk.";
		next;
		getitem 7941,1;
		set Hallow08Kill,0;
		mes "[Mago di Halloween]";
		mes "Come promesso";
		mes "Puoi ottenere biglietti per Halloween per oggetti fantastici.";
		close;
	}
	if (Hallow08 > 0) {
		mes "[Mago di Halloween]";
		mes "Volete che vi rileggia le regole o torniamo a quello...";
		next;
		goto MainMenu2;
	}

		Rules:
			mes "[Mago di Halloween]";
			mes "Questo villaggio è come un Payon virtuale.";
			mes "Ci sono zombie e ghoul che vagano intorno e tre uscite meridionali, ma solo una funziona.";
			mes "Dipende da te scoprirlo.";
			next;
			mes "[Mago di Halloween]";
			mes "Non puoi usare nessuna abilità per uccidere i ghoul o gli zombie.";
			mes "E un'altra cosa...";
			mes "non dovresti dimenticare...";
			next;
			mes "[Mago di Halloween]";
			mes "Tutti i partecipanti dovrebbero essere vestiti con niente.";
			mes "Metti tutti i tuoi effetti nel tuo deposito e torna qui quando il tuo peso è '0'.";
			next;
			mes "[Mago di Halloween]";
			mes "E un'altra cosa!";
			mes "Non puoi cavalcare un PecoPeco o avere un Carrello.";
			mes "Se lo sei, allora li rimuoverò prima che tu entri.";
			mes "Capito?";
			next;
			mes "[Mago di Halloween]";
			mes "Ricorda, ci sono tre uscite ma solo una funziona casualmente, i zombie e gli scheletri che vagano intorno non possono essere uccisi e non puoi indossare nulla.";
			next;
			if (Hallow08 > 0) {
				mes "[Mago di Halloween]";
				mes "Ehi...";
				mes "Torna quando sarai pronto.";
				close;
			} else {
				goto MainMenu;
			}

		Participate:
			mes "[Mago di Halloween]";
			mes "Ok, sei pronto.";
			mes "Fammi controllare il tuo peso.";
			next;
			if (Weight > 0) {
				mes "[Mago di Halloween]";
				mes "Oh mio Dio!";
				mes "C'è sempre una pecora nera ovunque.";
				next;
				mes "[Mago di Halloween]";
				mes "Pensi che non noterò che il tuo peso è superiore a '0'?";
				mes "Sei in sovrappeso...";
				close;
			} else {
				mes "[Mago di Halloween]";
				mes "Sembra che tu sia pronto/a e il tuo peso è giusto.";
				next;
				mes "[Mago di Halloween]";
				mes "Ero piuttosto sommerso dal mio lavoro, quindi sono esausto.";
				next;
				mes "[Mago di Halloween]";
				mes "A volte mi dimentico di mandartelo lì...";
				next;
				mes "[Mago di Halloween]";
				mes "Spero che tu torni bene.";
				close2;
				if (Hallow08 < 1) {
					set Hallow08,1;
				}
				set Hallow08Kill,1;
				set @Hallow08Warp, rand(1,3);
				percentheal -98,0;
				setriding 0;
				setcart 0;
				warp "evt_zombie",155,246;
				end;
			}

		TicketExchange:
			mes "[Mago di Halloween]";
			mes "Vuoi scambiare biglietti con premi?";
			mes "Ottimo lavoro! Kkkkkk!";
			next;
			mes "[Mago di Halloween]";
			mes "Lasciarmi dire quali oggetti puoi scambiare per.";
			next;
			mes "[Mago di Halloween]";
			mes "5 biglietti per Torta di Zucca.";
			mes "20 biglietti per Pumpkin-Head.";
			mes "50 biglietti per Old Blue Box.";
			mes "70 biglietti per la Scatola Viola Vecchia.";
			mes "200 biglietti per l'album di vecchie carte.";
			next;
			mes "[Mago di Halloween]";
			mes "Cosa vorresti scambiare?";
			next;
			switch(select("Torta di zucca:Testa di zucca:Scatola Blu Vecchia:Scatola Viola Vecchia:Album di vecine cartoline")) {
			case 1:
				if (countitem(7941) < 5) {
					goto NotEnough;
				} else {
					delitem 7941,5;
					getitem 12192,1;
					goto Enough;
				}
				break;
			case 2:
				if (countitem(7941) < 20) {
					goto NotEnough;
				} else {
					delitem 7941,20;
					getitem 5134,1;
					goto Enough;
				}
				break;
			case 3:
				if (countitem(7941) < 50) {
					goto NotEnough;
				} else {
					delitem 7941,50;
					getitem 603,1;
					goto Enough;
				}
				break;
			case 4:
				if (countitem(7941) < 70) {
					goto NotEnough;
				} else {
					delitem 7941,70;
					getitem 617,1;
					goto Enough;
				}
				break;
			case 5:
				if (countitem(7941) < 200) {
					goto NotEnough;
				} else {
					delitem 7941,200;
					getitem 616,1;
					goto Enough;
				}
			}

		NextTime:
			mes "[Mago di Halloween]";
			mes "A dopo.";
			mes "Kkkkkkkk.";
			close;

	MainMenu:
		menu "Spiegamelo.",Rules,"Voglio partecipare.",Participate,"Voglio scambiare biglietti con premi.",TicketExchange,"Tornerò la prossima volta.",NextTime;

	MainMenu2:
		menu "Tornami lì adesso!",Participate,"Per favore, dimmi le regole",Rules,"Voglio scambiare biglietti con premi.",TicketExchange,"Tornerò la prossima volta.",NextTime;

	NotEnough:
		mes "[Mago di Halloween]";
		mes "Non hai abbastanza biglietti!";
		mes "Non riesci nemmeno a contare?";
		mes "Per favore vieni qui con il numero giusto di biglietti.";
		close;

	Enough:
		mes "[Mago di Halloween]";
		mes "Ecco qui.";
		mes "Hai bisogno di...";
		mes "qualcos'altro?";
		next;
		if (Hallow08 == 1) {
			goto MainMenu2;
		} else {
			goto MainMenu;
		}
}
