-	script	Halloween Wizard#iRO09::09HallowWiz	737,{
	mes "[Mago di Halloween]";
	mes "...";
	mes "Vuoi fare uno scherzo a qualcuno?";
	next;
	while(1) {
		switch(select("Quale trucco?:Certo:No.")) {
			case 1:
				mes "[Mago di Halloween]";
				mes "Posso evocare mostri in altre parti del mondo con solo pochi materiali.";
				next;
				mes "[Mago di Halloween]";
				mes "Sembra interessante, vero?";
				next;
				mes "[Mago di Halloween]";
				mes "Se mi porti Tessuto, Jack o' Pumpkin, Tessuto Usurato, o Zucca Frantumata posso evocare i mostri.";
				next;
				break;
			case 2:
				mes "[Mago di Halloween]";
				mes "Quale città vuoi ingannare?";
				next;
				getmapxy(.@mapname$,.@mapx,.@mapy,BL_PC,""+strcharinfo(0)+"");
				if (.@mapname$ == "prontera") {
					switch(select("Geffen:Payon:Alberta:Aldebaran")) {
						case 1:
							set .@HallowTown,3;
							break;
						case 2:
							set .@HallowTown,2;
							break;
						case 3:
							set .@HallowTown,4;
							break;
						case 4:
							set .@HallowTown,5;
							break;
					}
				}
				else if (.@mapname$ == "payon") {
					switch(select("Prontera:Geffen:Alberta:Aldebaran")) {
						case 1:
							set .@HallowTown,1;
							break;
						case 2:
							set .@HallowTown,3;
							break;
						case 3:
							set .@HallowTown,4;
							break;
						case 4:
							set .@HallowTown,5;
							break;
					}
				}
				else if (.@mapname$ == "geffen") {
					switch(select("Prontera:Payon:Alberta:Aldebaran")) {
						case 1:
							set .@HallowTown,1;
							break;
						case 2:
							set .@HallowTown,2;
							break;
						case 3:
							set .@HallowTown,4;
							break;
						case 4:
							set .@HallowTown,5;
							break;
					}
				}
				else if (.@mapname$ == "alberta") {
					switch(select("Prontera:Geffen:Payon:Aldebaran")) {
						case 1:
							set .@HallowTown,1;
							break;
						case 2:
							set .@HallowTown,3;
							break;
						case 3:
							set .@HallowTown,2;
							break;
						case 4:
							set .@HallowTown,5;
							break;
					}
				}
				else if (.@mapname$ == "aldebaran") {
					switch(select("Prontera:Geffen:Payon:Alberta")) {
						case 1:
							set .@HallowTown,1;
							break;
						case 2:
							set .@HallowTown,3;
							break;
						case 3:
							set .@HallowTown,2;
							break;
						case 4:
							set .@HallowTown,4;
							break;
					}
				}
				setarray .@HallowTowns$[1],"prontera","payon","geffen","alberta","aldebaran";
				mes "[Mago di Halloween]";
				mes "Allora passiamo al prossimo passo.";
				next;
				mes "[Mago di Halloween]";
				mes "Quanti Tessuti o Zucca da Halloween vuoi usare? Non superare i 100 perché è la quantità massima che posso usare.";
				next;
				input .@input;
				if (.@input == 0) {
					mes "[Mago di Halloween]";
					mes "Non hai idea definito.";
					mes "Non è un grosso problema.";
					mes "Fammi sapere.";
					next;
					break;
				}
				else if (.@input > 100) {
					mes "[Mago di Halloween]";
					mes "Ti ho detto che deve essere tra 1 e 100!";
					mes "Non hai prestato attenzione!";
					next;
					break;
				}
				else {
					set .@fabric,countitem(1059);
					set .@jack,countitem(1062);
					set .@worn,countitem(6299);
					set .@crushed,countitem(6298);
					set .@whispers,0;
					set .@darklords,0;

					set .@total,.@fabric + .@jack + .@worn + .@crushed;

					if(.@total < .@input) {
						mes "[Mago di Halloween]";
						mes "Riconta gli articoli che hai e dimmi il totale.";
						mes "Huhuhuhuhuhu...";
						next;
						break;
					}

					if(.@fabric > 0) {
						if(.@fabric >= .@input) {
							delitem 1059,.@input;
							set .@whispers,.@whispers+.@input;
							set .@input,0;
						}
						else{
							delitem 1059,.@fabric;
							set .@input,.@input - .@fabric;
							set .@whispers,.@whispers+.@fabric;
						}
					}
					if(.@worn > 0 && .@input != 0) {
						if(.@worn >= .@input) {
							delitem 6299,.@input;
							set .@whispers,.@whispers+.@input;
							set .@input,0;
						}
						else{
							delitem 6299,.@worn;
							set .@input,.@input - .@worn;
							set .@whispers,.@whispers+.@worn;
						}
					}
					if(.@jack > 0 && .@input != 0) {
						if(.@jack >= .@input) {
							delitem 1062,.@input;
							set .@darklords,.@darklords+.@input;
							set .@input,0;
						}
						else{
							delitem 1062,.@jack;
							set .@input,.@input - .@jack;
							set .@darklords,.@darklords+.@jack;
						}
					}
					if(.@crushed > 0 && .@input != 0) {
						if(.@crushed >= .@input) {
							delitem 6298,.@input;
							set .@darklords,.@darklords+.@input;
							set .@input,0;
						}
						else{
							delitem 6298,.@crushed;
							set .@input,.@input - .@crushed;
							set .@darklords,.@darklords+.@crushed;
						}
					}
					if (.@input > 0) {
						mes "C'è un problema.";
						close;
					}
					monster ""+.@HallowTowns$[.@HallowTown]+"",0,0,"Halloween Whisper",3014,.@whispers;
					monster ""+.@HallowTowns$[.@HallowTown]+"",0,0,"Halloween Dark Lord",3015,.@darklords;
					mes "[Mago di Halloween]";
					mes "Ecco quello che volevi.";
					mes "Immagina cosa devono pensare le persone nei villaggi vicini?";
					mes "Muahahaha";
					close;
				}
			case 3:
				mes "[Mago di Halloween]";
				mes "Se cambi idea, torna qui...";
				mes "Rimarrò qui per un po'...";
				mes "Kkkk...";
				close;
		}
	}
}
