payon_in01,193,30,4	script	Rice Mill Grandma#rat	78,{
	mes "Nonna del Mulino di Riso";
	if (!lunar_rat) {
		mes "Quei sporchi piccoli";
		mes "creature! Scappare";
		mes "attorno, strappando cose";
		mes "da me! Se solo loro";
		mes "non erano così colpevoli velocemente...";
		next;
		select("Cosa c'è che non va?");
		mes "Nonna del Mulino di Riso";
		mes "Oh, stavo solo facendo";
		mes "alcuni cracker di riso e";
		mes "dolci per festeggiare";
		mes "il nuovo anno, ma questi";
		mes "gli animali hanno rubato";
		mes "i sacchetti di riso che ho preparato.";
		next;
		select("Sacchetti di riso?");
		mes "Nonna del Mulino di Riso";
		mes "Sì, mio figlio ne ha bisogno";
		mes "Sacchetti per riso da sbattere il";
		mes "riso in un mortaio, ma non posso";
		mes "faccio dei dolci se non ne ho";
		mes "anche il riso. Tu hai?";
		mes "pensi che tu possa aiutarmi?";
		next;
		if (select("Certo.:Non ti conosco nemmeno.") == 1) {
			set lunar_rat,1;
			mes "[" + strcharinfo(0)+"]";
			mes "Certo.";
			mes "Cosa posso fare?";
			next;
			mes "Nonna del Mulino di Riso";
			mes "Beh, pensi che tu";
			mes "possono catturare gli animali che";
			mes "mi hai rubato i miei sacchetti di riso?";
			mes "Sono questi blu e";
			mes "ratti bianchi e questi";
			mes "piccoli e fastidiosi tassi.";
			next;
			select("Nessun problema.");
			mes "[" + strcharinfo(0)+"]";
			mes "Nessun problema.";
			mes "Catturerò quegli animali,";
			mes "e riportare qualsiasi Riso";
			mes "Sacchetti che posso trovare.";
			next;
			mes "Nonna del Mulino di Riso";
			mes "Oh, grazie!";
			mes "Buona fortuna nel prendere";
			mes "que parassiti per me~";
			close;
		}
		mes "Nonna del Mulino di Riso";
		mes "Io... Beh...";
		mes "È vero, ma io ero";
		mes "sto pianificando di darti";
		mes "qualcosa di carino in cambio";
		mes "per il tuo aiuto. So che sei";
		mes "non una persona cattiva, quindi...";
		close;
	}
	else if ((lunar_rat >= 1 && lunar_rat <= 2) || lunar_rat == 4) {
		if (countitem(7770) >= 1 && lunar_rat == 2) {
			mes "Oh, è il";
			mes "Riso dolce mio figlio ha fatto?";
			mes "Mi lo lasci avere?";
			mes "Aspetta qui un momento,";
			mes "e ti farò dei dolciumi~";
			delitem 7770,1; //Sweet_rice
			set lunar_rat,3;
			close;
		}
		else if (lunar_rat == 4) {
			if (countitem(7770) >= 1) {
				mes "Oh, è il";
				mes "Riso dolce mio figlio ha fatto?";
				mes "Mi lo lasci avere?";
				mes "Aspetta qui un momento,";
				mes "e ti farò dei dolciumi~";
				delitem 7770,1; //Sweet_rice
				set lunar_rat,5;
				close;
			}
			mes "Oh, dove sono tutti i";
			mes "Sacchetti per riso? Abbiamo bisogno";
			mes "farli fare più gallette di riso...";
			close;
		}
		else if (countitem(7869) < 1 && lunar_rat <= 2) {
			mes "Oh, quei bianchi e";
			mes "i topi blu potrebbero avere";
			mes "mi sono strappati i sacchetti di riso";
			mes "Quei lombrini probabilmente hanno preso";
			mes "anche loro. Così cattivi piccoli";
			mes "creature, non è vero?";
			close;
		}
		mes "Oh, è uno dei miei";
		mes "Sacchetti di riso? Mi dispiace,";
		mes "ma il dolore alle mie anche..";
		mes "Ti dispiacerebbe essere";
		mes "caro, e consegnando";
		mes "per mio figlio per me?";
		set lunar_rat,2;
		close;
	}
	else if (lunar_rat == 3 || lunar_rat == 5) {
		set .@reward,rand(1,100);
		if (.@reward <= 5) {
			if (lunar_rat == 3) {
				getitem 9038,1; // New_Year_Doll_Egg
				set lunar_rat,4;
			}
			else if (lunar_rat == 5) {
				getitem 668,1; // Red_Envelope
			}
		}
		else if (.@reward >= 6 && .@reward <= 10) getitem 12198,2; // Lucky_Rice_Cake
		else if (.@reward >= 11 && .@reward <= 30) getitem 12195,3; // Plain_Rice_Cake
		else if (.@reward >= 31 && .@reward <= 40) getitem 12196,2; // Hearty_Rice_Cake
		else if (.@reward >= 41 && .@reward <= 55) getitem 12123,2; // Honey_Pastry
		else if (.@reward >= 56 && .@reward <= 70) getitem 12122,2; // Sesame_Pastry
		else if (.@reward >= 71 && .@reward <= 80) getitem 12124,2; // Rainbow_Cake
		else if (.@reward >= 81 && .@reward <= 90) getitem 12198,2; // Korean_Rice_Cake
		else if (.@reward >= 91 && .@reward <= 100) getitem 12197,3; // Salty_Rice_Cake
		// After getting the egg there is a chance to get an envelope with each turn in
		if (rand(1,20) <= 5 && lunar_rat == 4) getitem 668,1; // Red_Envelope
		// Set the quest accordingly for repeat turn-ins.
		if (lunar_rat == 3) set lunar_rat,2;
		else if (lunar_rat == 5) set lunar_rat,4;
		mes "Non è quasi abbastanza";
		mes "per ripagarti di quello che hai";
		mes "fatto per me, ma mi piacerebbe";
		mes "avere questo dolce che";
		mes "L'ho appena fatto. Goditelo~";
		next;
		mes "Nonna del Mulino di Riso";
		mes "Grazie per il";
		mes "Riso dolce! Io sarò";
		mes "Certezza di creare qualcosa";
		mes "delizioso per te se tu";
		mes "dammi altro, ok?";
		close;
	}
}
