function	script	F_GldManager	{
	.@npc_name$ = getarg(0);
	.@map_name$ = getarg(1);
	.@gid = getcastledata(.@map_name$,1);
	.@gm_name$ = getguildmaster(.@gid);

	mes "[ "+ .@npc_name$ +" ]";
	if (.@gid == 0){
		mes "Aspetto il mio maestro. Coraggioso avventuriero, segui il tuo destino!";
		return;
	}
	if (getcharid(2) != .@gid){
		mes "Sono qui per seguire ^5533FF" + getguildmaster(.@gid) + "Il comando di '^000000' ! Ehi! Tu non fai nemmeno parte della gilda!!";
		mes "Dove sono i guardiani? Distruggete questi intrusi!";
		return;
	}
	if (is_guild_leader() == false){
		mes "Non sei ^5533FF" + getguildmaster(.@gid) + "^000000! Sono qui per seguire ^5533FF" + getguildmaster(.@gid) + "Il comando di '^000000' solo";
		return 0;
	}

	mes "Benvenuto Maestro ^5533FF" + getguildmaster(.@gid) + "^000000 ! Ti aiuterò in ogni modo che posso!";
	next;

	menu "Lavoro/Licenziamento del Personale di Kafra",M_Kaf, "Entra nella Stanza del Tesoro",M_Treas, "Annulla",M_End;

	//===========================
	M_Kaf:
	//======
		mes "[ "+ .@npc_name$ +" ]";
		if (GetCastleData(.@map_name$,9) == 1) goto L_Dismiss;
		if (getgdskilllv(.@gid,10001) == 0){
			mes "Signore, lei non ha un contratto con la Kafra Staff Company.";
			mes "Per assumere un Kafra, devi prima imparare la abilità del Gilda ^5533FFContratto con il Kafra^000000.";
			return;
		}

		L_Hire:
			mes "Vorresti utilizzare i servizi di un Kafra? Avrai bisogno di ^5533FF10,000 Zeny^000000 per farlo... ";
			next;
			menu "Impiega Kafra.",-,"Annulla",sM_KafEnd;

				mes "[ "+ .@npc_name$ +" ]";
				if (Zeny < 10000){
					mes "Padrone, non ha abbastanza soldi per impiegare un Kafra. L'impiego è stato annullato.";
					return;
				}
				set Zeny,Zeny-10000;
				enablenpc "Kafra Staff#"+getarg(4);
				SetCastleData .@map_name$,9,1;
				mes "Hai creato un contratto con la Kafra Staff Company.";
				next;
				cutin "kafra_01",2;
				mes "[ Kafra Personale ]";
				mes "Come stai? Sono qui per fornirti un servizio utile! Farò del mio meglio per servirti.";
				next;
				cutin "kafra_01",255;
				mes "[ "+ .@npc_name$ +" ]";
				//mes "Your employment contract lasts ^5533FF1 month^000000.  After this term is over you will have to create a new contract.";
				mes "Penso che lo staff di Kafra beneficerà i membri della nostra gilda.";
				return;

			sM_KafEnd:
				mes "[ "+ .@npc_name$ +" ]";
				mes "Come desideri Maestro. Ma suggerisco che prendiamo un Kafra il prima possibile!";
				return;

		L_Dismiss:
			mes "Vorresti scartare la corrente Kafra?";
			next;
			menu "Licenziamento",-,"Annulla",sM_KafEnd2;

				cutin "kafra_01",2;
				mes "[ Kafra Personale ]";
				mes "Ho fatto qualcosa di sbagliato? Se sì, potresti perdonarmi?";
				next;
				menu "Scartare",-,"Annulla",ssM_KafEnd2;

					mes "[ Kafra Personale ]";
					mes "È un peccato che non potrò più servire la vostra gilda....";
					next;
					disablenpc "Kafra Staff#"+getarg(4);
					SetCastleData .@map_name$,9,0;
					cutin "kafra_01",255;
					mes "[ "+ .@npc_name$ +" ]";
					mes "La Kafra è stata annullata. Ma... dovremmo davvero prendere una Kafra il prima possibile!";
					return;
				ssM_KafEnd2:
					mes "[ Kafra Personale ]";
					mes "Grazie maestro, farò del mio meglio! ^^.";
					cutin "kafra_01",255;
					return;
			sM_KafEnd2:
				mes "[ "+ .@npc_name$ +" ]";
				mes "Maestro, credo che lei dovrebbe mantenere la attuale Kafra Staff perché sta già facendo del suo meglio per servirci";
				return;

	//=========================
	M_Treas:
	//========
		mes "[ "+ .@npc_name$ +" ]";
		mes "Vorresti andare nella nostra Stanza del Tesoro? Solo tu, il Maestro di Gilda, hai il permesso di entrare in questa stanza.";
		next;
		menu "Entra nella stanza dei tesori.",-,"Annulla",sM_TresEnd;

			mes "[ "+ .@npc_name$ +" ]";
			mes "Seguimi attraverso il passaggio segreto.";
			mes "Devi tirare sulla levetta segreta per uscire.";
			next;
			warp .@map_name$,getarg(2),getarg(3);
			return;
		sM_TresEnd:
			mes "[ "+ .@npc_name$ +" ]";
			mes "I beni vengono prodotti ogni giorno.";
			mes "Dovresti prenderli quando puoi perché potrebbero scomparire se li prendi al momento sbagliato.";
			return;


	//==========================
	M_End:
	//=======
		mes "[ "+ .@npc_name$ +" ]";
		mes "Come desideri, maestro.";
		return;
}
