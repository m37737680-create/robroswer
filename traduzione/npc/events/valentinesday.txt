// Stephen - Valentine Event Chocolate seller
//============================================================
alberta,26,243,4	script	Stephen	58,{
	mes "[Stephen]";
	mes "Indovina cosa ho?";
	mes "Un bocconcino gustoso non facilmente trovato a Midgard....";
	next;
	mes "[Stephen]";
	mes "Cioccolato!";
	mes "Hai ragione, non ami il cioccolato... Io sì.";
	mes "E siete fortunati, perché li sto vendendo solo a 5.000 zeny l'uno!";
	next;
	if(select("Voglio del cioccolato!:No grazie.")==2) {
		mes "[Stephen]";
		mes "Non vuoi del cioccolato?";
		mes "Te lo dico io! Te ne pentirai!";
		mes "Meglio che ne prenda un po' adesso... non troverà mai il Cioccolato così di nuovo!";
		mes "Ci pensaci e rivisitamici un'altra volta.";
		close; }
	mes "[Stephen]";
	mes "Ah!";
	mes "Lo sapevo!";
	mes "Ma non posso venderti più di 5 alla volta... ma, se ne hai davvero bisogno di più....";
	mes "puoi tornare di nuovo.";
	mes "Quanti ne vuoi?";
	next;
	input .@i;
	if (.@i <= 0) close;
	mes "[Stephen]";
	if (.@i > 5)  {
		mes "Mi dispiace, ma non posso darti così tanti.";
		close; }
	if (Zeny < .@i*5000) {
		mes "Mi dispiace, ma sembra che non ti possa permettere di comprarli da me.";
		close; }
	set Zeny, Zeny - .@i*5000;
	getitem 558,.@i;
	mes "Eccolo/a!";
	mes "Puoi darlo come regalo a qualcuno, o godertelo tu stesso!";
	mes "Mmh... cioccolato dolce...";
	mes "Venimi a trovare quando vuoi...!";
	close;
}
