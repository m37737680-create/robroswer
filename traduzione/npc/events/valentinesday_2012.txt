mosk_in,21,246,5	script	Pinkamenia	71,{
	set .@n$,"^0000FF[Pinkamenia]^000000";
	if (#V_QUE12==3) {
		set .@LFlowerID,7864;
		setarray .@SweetsID[0],538,558,539,573,559,560,12062,596,597,12414,12319;
		setarray .@SweetsCost[0],1,2,5,10,10,10,15,15,15,20,20;
		mes .@n$;
		mes "Ciao "+strcharinfo(0)+", fa";
		mes "vuoi comprare degli articoli?";
		mes "Dovrai darmi Amore";
		mes "Fiori per loro, ovviamente!";
		next;
		if(select("Sì, certo!","No, non importa.") == 2) close;
		next;
		mes .@n$;
		mes "Seleziona un elemento.";
		mes "La quantità di Fiori d'Amore di cui avrai bisogno è tra parentesi.";
		for(set .@i,0; .@i < getarraysize(.@SweetsID); set .@i,.@i+1)
			set .@menu$, .@menu$+"^00AA00["+.@SweetsCost[.@i]+"]^000000 "+getitemname(.@SweetsID[.@i])+":";
		next;
		set .@choice, select(.@menu$)-1;
		mes .@n$;
		if (countitem(.@LFlowerID) < .@SweetsCost[.@choice]) {
			mes "Avrai bisogno di più Fiori d'Amore se vuoi quello!";
			close;
		}
		mes "Sei sicuro di voler scambiare ^00aa00"+.@SweetsCost[.@choice]+"x Fiore d'amore ^000000 per ^0055FF"+getitemname(.@SweetsID[.@choice])+"^000000?";
		if(select("No, ho cambiato idea.:Sì, commercio!") == 1) close;
		set .@new, .@LFlowerID - .@SweetsCost[.@choice];
		if (.@new < 0) set .@new, 0;
		delitem .@LFlowerID, .@SweetsCost[.@choice];
		getitem .@SweetsID[.@choice],1;
		mes "Divertiti con il tuo articolo!";
		close;
	} else if (#V_QUE12==2) {
		mes .@n$;
		if (countitem(7864) < 15) {
			mes "Devi portarmi 15 Fiori d'Amore!";
			close;
		}
		mes "Grazie mille per";
		mes "riprendere i nostri fiori!";
		delitem 7864,15;
		next;
		mes .@n$;
		mes "Considera questo un piccolo";
		mes "grazie";
		getexp 500000,400000;
		getitem 617,1;
		getitem 12319,2;
		next;
		mes .@n$;
		mes "Se ottieni più Amore";
		mes "Fiori, puoi scambiare";
		mes "loro per dei dolci qui.";
		mes "A presto!";
		set #V_QUE12,3;
		close;
	} else if (#V_QUE12==1) {
		mes .@n$;
		mes "Per favore porta un Cappello Torta +8";
		mes "al Fornaio Straordinario";
		mes "accanto a me!";
		close;
	} else if (BaseLevel >= 45) {
		mes .@n$;
		mes "Ciao "+strcharinfo(0)+", è";
		mes "San Valentino e noi";
		mes "adorare consegnare dolci";
		mes "cioccolatini.";
		next;
		mes .@n$;
		mes "Il nostro problema è che noi";
		mes "ho bisogno di una persona forte";
		mes "come te che potresti aiutare";
		mes "noi, ma prima avrai";
		mes "portare un cappello da torta +8";
		mes "al Fornaio Straordinario, che";
		mes "è proprio accanto a me!";
		set #V_QUE12,1;
		close;
	}
	mes .@n$;
	mes "Ciao "+strcharinfo(0)+"!";
	close;

OnInit:
	// Unknown item drop rate (currently set at 5%)
	setitemscript 5105,"{ bonus bDex,1; bonus bMaxSP,80; bonus3 bAddMonsterDropItem,7864,RC_DemiHuman,500; }",0;
	end;
}

mosk_in,21,244,3	script	Baker Extraordinaire	410,{
	set .@n$,"^0000FF[Baker Extraordinaire]^000000";
	if(#V_QUE12==3) {
		mes .@n$;
		mes "Scambia i tuoi Fiori d'Amore con Pinkamenia!";
		close;
	} else if(#V_QUE12==2) {
		mes .@n$;
		mes "Porta 15 Fiori d'Amore a Pinkamenia!";
		close;
	} else if(#V_QUE12==1) { 
		mes .@n$;
		if (getequipid(EQI_HEAD_TOP) == 5024 && getequiprefinerycnt(EQI_HEAD_TOP) >= 8) {
			mes "Ah, quindi Pinkamenia te l'ha detto";
			mes "portarmi la torta +8";
			mes "Cappello. Adesso vado";
			mes "scambiare il tuo +8 Torta";
			mes "Cappello per un altro Cappello Torta.";
			next;
			mes .@n$;
			delitem 5024,1;
			getitem 5105,1;
			equip 5105;
			mes "Ora, se indossi il";
			mes "nuovo cappello da cuoco, lì";
			mes "è una possibilità che";
			mes "Mostri semi-umani";
			mes "lancia un Fiore d'Amore!";
			next;
			mes .@n$;
			mes "I mobili hanno rubato i nostri Flowers,";
			mes "che dobbiamo creare ";
			mes "i nostri cioccolatini. Ora prendi";
			mes "15 Fiori d'amore e porta";
			mes "a Pinkamenia!";
			set #V_QUE12,2;
			close;
		} else {
			mes "Scusa, ma dov'è il tuo";
			mes "Cappello da torta +8? Porta";
			mes "detto a me, e ricorda";
			mes "averlo equipaggiato!";
			close;
		}
	} else {
		mes .@n$;
		mes "Ciao, "+strcharinfo(0)+"!";
		close;
	}
}
