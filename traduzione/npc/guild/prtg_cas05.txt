// Guild Flags (Outside Castle)
//============================================================
-	script	Gondul#1::OutsideFlagsPR5	722,{

	set .@GID, GetCastleData("prtg_cas05",1);

	if (.@GID == 0) {
		mes "[ Editto del Regno della Divina Runo-Midgard ]";
		mes " ";
		mes "Seguire l'ordinanza del Regno di The Divine Rune-Midgarts, ";
		mes "Dichiariamo che";
		mes "non c'è un maestro formale di questo castello.";
		mes " ";
		mes "A chi può ";
		mes "superare tutte le prove";
		mes "e distruggere l'Imperium,";
		mes "il re donerà a colui con";
		mes "proprietà di questo castello.";
		close;
	}
	if (getcharid(2) == .@GID) {
		mes "[ Voce che echeggia ]";
		mes "Coraggiosi...";
		mes "Desideri tornare al tuo onorevole posto?";
		next;
		if(select("Tornare al castello della gilda.:Basta.") == 1) {
			close2;
			if (getcharid(2) == GetCastleData("prtg_cas05",1)) {
				warp "prtg_cas05",52,41;
				end;
			}
			end;
		}
		close;
	}
	mes "[ Editto del Regno della Divina Runo-Midgard ]";
	mes " ";
	mes "Seguire l'ordinanza del Regno di The Divine Rune-Midgarts, ";
	mes "approviamo che questo posto sia in";
	mes "la proprietà privata di ^ff0000"+GetGuildName(.@GID)+"^000000 Gilda.";
	mes " ";
	mes "Il Maestro della gilda di ^ff0000"+GetGuildName(.@GID)+"La Gilda ^000000 è";
	mes "^ff0000"+GetGuildMaster(.@GID)+"^000000";
	mes "Se c'è qualcuno che si oppone a questo,";
	mes "prova la tua forza e onore con una lama d'acciaio nella mano.";
	close;

OnRecvCastlePt05:
	FlagEmblem GetCastleData("prtg_cas05",1);
	end;
}
