	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Assassin Cross]";
			mes "Congratulazioni...";
			next;
			mes "[Assassin Cross]";
			mes "...";
			next;
			mes "[Assassin Cross]";
			mes "...";
			mes "......";
			next;
			mes "[Assassin Cross]";
			mes "...";
			mes "......";
			mes "Onore a";
			mes "i guerrieri.";
			close;
		}
		mes "[Assassin Cross]";
		mes "Siamo i guerrieri";
		mes "del deserto. Nessuno";
		mes "ci guarda dall'alto.";
		mes "Nessuno...";
		close;
	}
	else if (ADVJOB == Job_Assassin_Cross && Class == Job_Thief_High && JobLevel > 39) {
		mes "[Assassin Cross]";
		mes "È arrivato il momento.";
		mes "Il mondo ha bisogno di te...";
		mes "Più che mai.";
		next;
		mes "[Assassin Cross]";
		mes "Chiedo che tu continui a vivere nelle ombre, ma come un Assassino ancora più grande con un nuovo aspetto.";
		next;
		mes "[Assassin Cross]";
		mes "Diventerai";
		mes "un incrocio di Assassino?";
		next;
		if (select("No.:Sì.") == 1) {
			mes "[Assassin Cross]";
			mes "Quando sei";
			mes "pronto, torna indietro.";
			next;
			mes "[Assassin Cross]";
			mes "Onore a";
			mes "i guerrieri.";
			close;
		}
		if (SkillPoint) {
			mes "[Assassin Cross]";
			mes "Non hai ancora";
			mes "imparato tutto";
			mes "che puoi.";
			next;
			mes "[Assassin Cross]";
			mes "Usa tutto il tuo";
			mes "Punti Abilità";
			mes "e poi torna indietro.";
			close;
		}
		jobchange Job_Assassin_Cross;
		set ADVJOB,0;
		mes "[Assassin Cross]";
		mes "Congratulazioni.";
		mes "Come un Assassino Incrociato,";
		mes "Spero che tu lotti per un futuro più luminoso nell'oscurità.";
		close;
	}
	mes "[Assassin Cross]";
	mes "Benvenuto";
	mes "a Valhalla,";
	mes "la Sala d'Onore.";
	next;
	mes "[Assassin Cross]";
	mes "Per favore fai";
	mes "ti senti a tuo agio";
	mes "mentre sei qui.";
	mes "Onore ai guerrieri.";
	close;
}
