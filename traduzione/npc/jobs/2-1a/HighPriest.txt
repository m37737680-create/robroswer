valkyrie,44,42,5	script	High Priest#Valkyrie	60,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "Sacerdote capo";
			mes "Congratulazioni.";
			mes "Onore ai guerrieri!";
			close;
		}
		mes "Sacerdote capo";
		mes "Attraverso il potere";
		mes "di santità, possiamo";
		mes "trova pace, forza";
		mes "e protezione. Consegnare";
		mes "da dalle forze del male...";
		close;
	}
	else if (ADVJOB == Job_High_Priest && Class == Job_Acolyte_High && JobLevel > 39) {
		mes "Sacerdote capo";
		mes "Il nostro mondo è in";
		mes "bisogno di persone di";
		mes "talento e convinzione.";
		mes "Continua il tuo";
		mes "buone opere come un evento";
		mes "più grande eroe di santità...";
		next;
		mes "Sacerdote capo";
		mes "Vorresti";
		mes "diventare un Sommo Sacerdote?";
		next;
		if (select("No.:Sì.") == 1) {
			mes "Sacerdote capo";
			mes "Quando sei pronto,";
			mes "sentiti libero di tornare.";
			mes "Onore ai guerrieri!";
			close;
		}
		if (SkillPoint) {
			mes "Sacerdote capo";
			mes "È ancora possibile che tu impari più abilità. Per favore usa";
			mes "tutti i tuoi Punti Abilità rimanenti prima di tornare da me.";
			close;
		}
		jobchange Job_High_Priest;
		set ADVJOB,0;
		mes "Sacerdote capo";
		mes "Congratulazioni.";
		mes "Come Sommo Sacerdote,";
		mes "Spero che mi guiderai";
		mes "altri sul sentiero";
		mes "alla santità...";
		close;
	}
	mes "Sacerdote capo";
	mes "Benvenuto";
	mes "a Valhalla,";
	mes "la Sala d'Onore.";
	next;
	mes "Sacerdote capo";
	mes "Per favore fai";
	mes "ti senti a tuo agio";
	mes "mentre sei qui.";
	mes "Onore ai guerrieri!";
	close;
}
