valkyrie,44,47,5	script	High Wizard#Valkyrie	735,{
	if (ADVJOB == 0 ||  Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Alto Mago]";
			mes "Congratulazioni.";
			mes "Onore ai guerrieri!";
			close;
		}
		mes "[Alto Mago]";
		mes "Noi Maghi Alti abbiamo";
		mes "la responsabilità di";
		mes "usando la nostra magia distruttiva";
		mes "per i giusti scopi.";
		next;
		mes "[Alto Mago]";
		mes "Una vita di addestramento";
		mes "è richiesto prima di diventare";
		mes "uno Stregone Alto. Riesci a immaginare";
		mes "cosa succederebbe se la nostra energia";
		mes "è stato messo nelle mani sbagliate?!";
		close;
	}
	else if (ADVJOB == Job_High_Wizard && Class == Job_Mage_High && JobLevel > 39) {
		mes "[Alto Mago]";
		mes "È ora.";
		mes "E Midgard ha";
		mes "bisogno di coloro che possono";
		mes "wieldare la magia più potente...";
		next;
		mes "[Alto Mago]";
		mes "Vorresti";
		mes "diventare un Alto Mago?";
		next;
		if (select("No.:Sì.") == 1) {
			mes "[Alto Mago]";
			mes "Quando sei pronto,";
			mes "sentiti libero di tornare.";
			mes "Onore ai guerrieri!";
			close;
		}
		if (SkillPoint) {
			mes "[Alto Mago]";
			mes "È ancora possibile che tu impari più abilità. Per favore usa";
			mes "tutti i tuoi Punti Abilità rimanenti prima di tornare da me.";
			close;
		}
		jobchange Job_high_Wizard;
		set ADVJOB,0;
		mes "[Alto Mago]";
		mes "Congratulazioni.";
		mes "Come Alto Mago,";
		mes "Spero usare tu usi";
		mes "i tuoi poteri di portare";
		mes "pace agli oppressi.";
		close;
	}
	mes "[Alto Mago]";
	mes "Benvenuto";
	mes "a Valhalla,";
	mes "la Sala d'Onore.";
	next;
	mes "[Alto Mago]";
	mes "Per favore fai";
	mes "ti senti a tuo agio";
	mes "mentre sei qui.";
	mes "Onore ai guerrieri!";
	close;
}
