valkyrie,44,39,5	script	Lord Knight#Valkyrie	56,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Signore Cavaliere]";
			mes "Congratulazioni.";
			mes "Onore ai guerrieri!";
			close;
		}
		mes "[Signore Cavaliere]";
		mes "Noi Cavalieri abbiamo un";
		mes "fantastico responsabilità...";
		mes "Servire e proteggere.";
		next;
		mes "[Signore Cavaliere]";
		mes "Anche al costo";
		mes "delle nostre vite,";
		mes "dobbiamo salvaguardare il";
		mes "il benessere dei nostri compagni.";
		close;
	}
	else if (ADVJOB == Job_Lord_Knight && Class == Job_Swordman_High && JobLevel > 39) {
		mes "[Signore Cavaliere]";
		mes "È arrivato il tuo momento!";
		mes "Il mondo ha ancora bisogno di te.";
		mes "Continua la tua vita";
		mes "come un eroe con un nuovo aspetto.";
		next;
		mes "[Signore Cavaliere]";
		mes "Vorresti";
		mes "diventare un Cavaliere Signore?";
		next;
		if (select("No.:Sì.") == 1) {
			mes "[Signore Cavaliere]";
			mes "Quando sei pronto,";
			mes "sentiti libero di tornare.";
			mes "Onore ai guerrieri!";
			close;
		}
		if (SkillPoint) {
			mes "[Signore Cavaliere]";
			mes "È ancora possibile che tu impari più abilità. Per favore usa";
			mes "tutti i tuoi Punti Abilità rimanenti prima di tornare da me.";
			close;
		}
		jobchange Job_Lord_Knight;
		set ADVJOB,0;
		mes "[Signore Cavaliere]";
		mes "Congratulazioni!";
		mes "Come Cavaliere Signore,";
		mes "Spero che tu sia";
		mes "in prima linea di battaglia,";
		mes "e guida i tuoi alleati alla vittoria!";
		close;
	}
	mes "[Signore Cavaliere]";
	mes "Benvenuto";
	mes "a Valhalla,";
	mes "la Sala d'Onore.";
	next;
	mes "[Signore Cavaliere]";
	mes "Per favore fai";
	mes "ti senti a tuo agio";
	mes "mentre sei qui.";
	mes "Onore ai guerrieri!";
	close;
}
