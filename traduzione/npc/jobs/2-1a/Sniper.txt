valkyrie,44,55,5	script	Sniper#Valkyrie	727,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Sniper]";
			mes "Congratulazioni.";
			mes "Onore ai guerrieri!";
			close;
		}
		mes "[Sniper]";
		mes "Un colpo.";
		mes "Un colpo di grazia.";
		mes "Non è così difficile";
		mes "una volta che sviluppi il";
		mes "visione per quello stile";
		mes "di combattere.";
		close;
	}
	else if (ADVJOB == Job_Sniper && Class == Job_Archer_High && JobLevel > 39) {
		mes "[Sniper]";
		mes "Il mondo è in";
		mes "bisogno di potenti arcieri";
		mes "come te. Sei pronto per";
		mes "la fantastica responsabilità?";
		next;
		mes "[Sniper]";
		mes "Sei disposto a";
		mes "fa il passo successivo e";
		mes "diventare un cecchino?";
		next;
		if (select("No.:Sì.") == 1) {
			mes "[Sniper]";
			mes "Quando sei pronto,";
			mes "sentiti libero di tornare.";
			mes "Onore ai guerrieri!";
			close;
		}
		if (SkillPoint) {
			mes "[Sniper]";
			mes "È ancora possibile che tu impari più abilità. Per favore usa";
			mes "tutti i tuoi Punti Abilità rimanenti prima di tornare da me.";
			close;
		}
		jobchange Job_Sniper;
		set ADVJOB,0;
		mes "[Sniper]";
		mes "Congratulazioni!";
		mes "Come Snaiper, spero";
		mes "che i minion del male";
		mes "non sarà mai sicuro quindi";
		mes "finché sono in";
		mes "il tuo sguardo!";
		close;
	}
	mes "[Sniper]";
	mes "Benvenuto";
	mes "a Valhalla,";
	mes "la Sala d'Onore.";
	next;
	mes "[Sniper]";
	mes "Per favore fai";
	mes "ti senti a tuo agio";
	mes "mentre sei qui.";
	mes "Onore ai guerrieri!";
	close;
}
