valkyrie,44,50,5	script	MasterSmith#Valkyrie	731,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[MasterSmith]";
			mes "Congratulazioni.";
			mes "Onore ai guerrieri!";
			close;
		}
		mes "[MasterSmith]";
		mes "Ah...";
		mes "Il culmine";
		mes "di artigianato.";
		mes "È il lavoro di";
		mes "un MaestroFabbro.";
		next;
		mes "[MasterSmith]";
		mes "Una volta che sei il/la";
		mes "il meglio del meglio,";
		mes "inizi a forgiare con";
		mes "lo scopo di scoprire";
		mes "nuovi e migliori modi di";
		mes "creando...";
		close;
	}
	else if (ADVJOB == Job_WhiteSmith && Class == Job_Merchant_High && JobLevel > 39) {
		mes "[MasterSmith]";
		mes "È arrivato il momento!";
		mes "Il nostro mondo ha bisogno di coraggiosi,";
		mes "avventurieri laboriosi";
		mes "come te...";
		next;
		mes "[MasterSmith]";
		mes "Vorresti";
		mes "diventare un MaestroFabbro?";
		next;
		if (select("No.:Sì.") == 1) {
			mes "[MasterSmith]";
			mes "Quando sei pronto,";
			mes "sentiti libero di tornare.";
			mes "Onore ai guerrieri!";
			close;
		}
		if (SkillPoint) {
			mes "[MasterSmith]";
			mes "È ancora possibile che tu impari più abilità. Per favore usa";
			mes "tutti i tuoi Punti Abilità rimanenti prima di tornare da me.";
			close;
		}
		jobchange Job_Whitesmith;
		set ADVJOB,0;
		mes "[MasterSmith]";
		mes "Congratulazioni!";
		mes "Come MaestroFabbro,";
		mes "Spero che tu forgerai";
		mes "un percorso verso un futuro più luminoso";
		mes "futuro per Midgard.";
		close;
	}
	mes "[MasterSmith]";
	mes "Benvenuto";
	mes "a Valhalla,";
	mes "la Sala d'Onore.";
	next;
	mes "[MasterSmith]";
	mes "Per favore fai";
	mes "ti senti a tuo agio";
	mes "mentre sei qui.";
	mes "Onore ai guerrieri!";
	close;
}
