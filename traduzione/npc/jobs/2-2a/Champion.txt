valkyrie,53,42,3	script	Champion#Valkyrie	52,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "Campione";
			mes "Congratulazioni.";
			mes "Onore ai guerrieri!";
			close;
		}
		mes "Campione";
		mes "Abilità";
		mes "Velocità.";
		mes "Forza.";
		mes "Agilità.";
		next;
		mes "Campione";
		mes "Un campione può";
		mes "beneficiare di tutto";
		mes "queste cose. Ma";
		mes "si può padroneggiare solo";
		mes "costanto nella vita...";
		close;
	}
	else if (ADVJOB == Job_Champion && Class == Job_Acolyte_High && JobLevel > 39) {
		mes "Campione";
		mes "È ora.";
		mes "Tempo per grandi eroi";
		mes "opporsi a la";
		mes "forze del male che affliggono";
		mes "il mondo di Midgard!";
		next;
		mes "Campione";
		mes "Vorresti";
		mes "diventare un Campione?";
		next;
		if (select("No.:Sì.") == 1) {
			mes "Campione";
			mes "Quando sei pronto,";
			mes "sentiti libero di tornare.";
			mes "Onore ai guerrieri!";
			close;
		}
		if (SkillPoint) {
			mes "Campione";
			mes "È ancora possibile che tu impari più abilità. Per favore usa";
			mes "tutti i tuoi Punti Abilità rimanenti prima di tornare da me.";
			close;
		}
		jobchange Job_Champion;
		set ADVJOB,0;
		mes "Campione";
		mes "Congratulazioni!";
		mes "Vivi come un campione,";
		mes "e portare luce in";
		mes "il mondo attraverso il";
		mes "forza dei tuoi pugni.";
		close;
	}
	mes "Campione";
	mes "Benvenuto";
	mes "a Valhalla,";
	mes "la Sala d'Onore.";
	next;
	mes "Campione";
	mes "Per favore fai";
	mes "ti senti a tuo agio";
	mes "mentre sei qui.";
	mes "Onore ai guerrieri!";
	close;
}
