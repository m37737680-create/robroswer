valkyrie,53,54,3	script	Minstrel#Valkyrie	741,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "Minestrello";
			mes "Congratulazioni.";
			mes "Onore ai guerrieri!";
			close;
		}
		mes "Minestrello";
		mes "Vuoi";
		mes "canti una canzone con me?";
		mes "Sha la la la la~";
		close;
	}
	else if (ADVJOB == Job_Clown && Class == Job_Archer_High && JobLevel > 39) {
		mes "Minestrello";
		mes "Il mondo cupo";
		mes "di mortali ha bisogno";
		mes "di una canzone più allegra.";
		mes "Lo porterai loro";
		mes "e cambiare le sorti nel";
		mes "battaglia contro il male?";
		next;
		mes "Minestrello";
		mes "Lo farai?";
		mes "per Midgard...";
		mes "Come un miniatore?";
		next;
		if (select("No.:Sì.") == 1) {
			mes "Minestrello";
			mes "Quando sei pronto,";
			mes "sentiti libero di tornare.";
			mes "Onore ai guerrieri!";
			close;
		}
		if (SkillPoint) {
			mes "Minestrello";
			mes "È ancora possibile che tu impari più abilità. Per favore usa";
			mes "tutti i tuoi Punti Abilità rimanenti prima di tornare da me.";
			close;
		}
		jobchange Job_Clown;
		set ADVJOB,0;
		mes "Minestrello";
		mes "Congratulazioni!";
		mes "Come Minestrello, il tuo";
		mes "le tue canzoni porteranno";
		mes "augurio ai tuoi alleati, e";
		mes "disperazione ai tuoi nemici.";
		close;
	}
	mes "Minestrello";
	mes "Benvenuto";
	mes "a Valhalla,";
	mes "la Sala d'Onore.";
	next;
	mes "Minestrello";
	mes "Per favore fai";
	mes "ti senti a tuo agio";
	mes "mentre sei qui.";
	mes "Onore ai guerrieri!";
	close;
}
