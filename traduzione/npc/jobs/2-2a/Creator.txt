valkyrie,53,50,3	script	Biochemist#Valkyrie	122,{
	if (Class == Job_Creator && bioeth == 13 && getskilllv("AM_BIOETHICS") == 0) {
		mes "Biochimico";
		mes "Ah, sei venuto a";
		mes "recuperare i ricordi";
		mes "perso per te? Sì, tu";
		mes "deve essere qui per il";
		mes "segreti della vita che";
		mes "un tempo tuoi...";
		next;
		if (select("Sì:No") == 1) {
			mes "Biochimico";
			mes "Chiudi gli occhi e";
			mes "metti la mente a tacere.";
			mes "ritorneremo al tuo";
			mes "passato per ricordare il";
			mes "frammenti del tuo perduto";
			mes "ricordi.";
			next;
			mes "Biochimico";
			mes "Quando apri gli occhi,";
			mes "ricorderai chiaramente";
			mes "il segreto della vita. Tu vorrai";
			mes "anche ricordare il peso di";
			mes "responsabilità nell'uso di questi";
			mes "segreti per le giuste estremità...";
			next;
			skill "AM_BIOETHICS",1,SKILL_PERM;
			mes "Biochimico";
			mes "Apri gli occhi...";
			mes "Ora che hai";
			mes "ricordare come";
			mes "creare vita artificiale, io solo";
			mes "chiedere che tu tratti tutto il tuo";
			mes "creazioni con rispetto.";
			close;
		}
		mes "Biochimico";
		mes "Se desideri";
		mes "recupera il tuo perso";
		mes "ricordi, per favore";
		mes "ritornare da me.";
		mes "Il segreto per creare";
		mes "la vita non è una cosa banale...";
		close;
	}
	else if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "Biochimico";
			mes "Congratulazioni.";
			mes "Onore ai guerrieri!";
			close;
		}
		mes "Biochimico";
		mes "È strano che";
		mes "qualcuno come me è qui.";
		mes "Ma anche qualcuno abile";
		mes "nei modi della scienza";
		mes "riuscire ad essere un eroe.";
		next;
		mes "Biochimico";
		mes "In questo caso,";
		mes "non è necessariamente";
		mes "il significa che ho usato, ma";
		mes "per cui ho";
		mes "lottato che mi ha guadagnato";
		mes "un posto a Valhalla...";
		close;
	}
	else if (ADVJOB == Job_Creator && Class == Job_Merchant_High && JobLevel > 39) {
		mes "Biochimico";
		mes "Sì...";
		mes "È ora.";
		mes "Ci servono più geni";
		mes "come te su Midgard.";
		next;
		mes "Biochimico";
		mes "Vorresti";
		mes "diventare un biochimico?";
		next;
		if (select("No.:Sì.") == 1) {
			mes "Biochimico";
			mes "Quando sei pronto,";
			mes "sentiti libero di tornare.";
			mes "Onore ai guerrieri!";
			close;
		}
		if (SkillPoint) {
			mes "Biochimico";
			mes "È ancora possibile che tu impari più abilità. Per favore usa";
			mes "tutti i tuoi Punti Abilità rimanenti prima di tornare da me.";
			close;
		}
		jobchange Job_Creator;
		if (bioeth == 13)
			skill "AM_BIOETHICS",1,SKILL_PERM;
		set ADVJOB,0;
		mes "Biochimico";
		mes "Congratulazioni!";
		mes "Come Biochimico,";
		mes "spero che tu usi il tuo";
		mes "vasta conoscenza per il";
		mes "giusti scopi.";
		close;
	}
	else {
		mes "Biochimico";
		mes "Benvenuto";
		mes "a Valhalla,";
		mes "la Sala d'Onore.";
		next;
		mes "Biochimico";
		mes "Per favore fai";
		mes "ti senti a tuo agio";
		mes "mentre sei qui.";
		mes "Onore ai guerrieri!";
		close;
	}
}
