		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "Zigeuner";
			mes "Congratulazioni.";
			mes "Onore ai guerrieri!";
			close;
		}
		mes "Zigeuner";
		mes "Vai a sinistra,";
		mes "spostati a destra!";
		mes "E passo...!";
		mes "Danzare può essere";
		mes "più di un hobby.";
		mes "Per me, è uno stile di vita~";
		close;
	}
	else if (ADVJOB == Job_Gypsy && Class == Job_Archer_High && JobLevel > 39) {
		mes "Zigeuner";
		mes "La terra di Midgard";
		mes "ha bisogno di donne talentuose";
		mes "per cambiare sottilmente i saldi";
		mes "nella battaglia tra il bene";
		mes "e malvagio.";
		next;
		mes "Zigeuner";
		mes "Sei pronto";
		mes "assumere questo ruolo,";
		mes "e diventare un Gitano?";
		next;
		if (select("No.:Sì.") == 1) {
			mes "Zigeuner";
			mes "Quando sei pronto,";
			mes "sentiti libero di tornare.";
			mes "Onore ai guerrieri!";
			close;
		}
		if (SkillPoint) {
			mes "Zigeuner";
			mes "È ancora possibile che tu impari più abilità. Per favore usa";
			mes "tutti i tuoi Punti Abilità rimanenti prima di tornare da me.";
			close;
		}
		jobchange Job_Gypsy;
		set ADVJOB,0;
		mes "Zigeuner";
		mes "Congratulazioni!";
		mes "Come zingaro, so";
		mes "le tue performance";
		mes "smuovere i cuori di tutti";
		mes "coloro che guarderanno...";
		close;
	}
	mes "Zigeuner";
	mes "Benvenuto";
	mes "a Valhalla,";
	mes "la Sala d'Onore.";
	next;
	mes "Zigeuner";
	mes "Per favore fai";
	mes "ti senti a tuo agio";
	mes "mentre sei qui.";
	mes "Onore ai guerrieri!";
	close;
}
