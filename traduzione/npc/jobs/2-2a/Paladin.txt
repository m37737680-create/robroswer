valkyrie,53,39,3	script	Paladin#Valkyrie	752,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Paladino]";
			mes "Congratulazioni.";
			mes "Onore ai guerrieri!";
			close;
		}
		mes "[Paladino]";
		mes "Non dimenticare";
		mes "che la Guerra Santa";
		mes "si avvicina rapidamente!";
		mes "Dobbiamo prepararci!";
		mes "Che la luce della giustizia";
		mes "illuminare sempre il nostro cammino!";
		close;
	}
	else if (ADVJOB == Job_Paladin && Class == Job_Swordman_High && JobLevel > 39) {
		mes "[Paladino]";
		mes "La Guerra Santa";
		mes "che ci sia prima che noi";
		mes "lo sai. Più che mai,";
		mes "abbiamo bisogno di uomini forti";
		mes "e le donne per combattere per cosa";
		mes "è buono e giusto.";
		next;
		mes "[Paladino]";
		mes "Combatterai su";
		mes "il lato della rettitudine";
		mes "come un Paladino?";
		next;
		if (select("No.:Sì.") == 1) {
			mes "[Paladino]";
			mes "Quando sei pronto,";
			mes "sentiti libero di tornare.";
			mes "Onore ai guerrieri!";
			close;
		}
		if (SkillPoint) {
			mes "[Paladino]";
			mes "È ancora possibile che tu impari più abilità. Per favore usa";
			mes "tutti i tuoi Punti Abilità rimanenti prima di tornare da me.";
			close;
		}
		jobchange Job_Paladin;
		set ADVJOB,0;
		mes "[Paladino]";
		mes "Congratulazioni.";
		mes "Come Paladino, spero";
		mes "proteggerai quelli";
		mes "più debole di te, e porta";
		mes "la vittoria degli Stati Uniti nel prossimo";
		mes "guerra tra bene e male.";
		close;
	}
	mes "[Paladino]";
	mes "Benvenuto";
	mes "a Valhalla,";
	mes "la Sala d'Onore.";
	next;
	mes "[Paladino]";
	mes "Per favore fai";
	mes "ti senti a tuo agio";
	mes "mentre sei qui.";
	mes "Onore ai guerrieri!";
	close;
}
