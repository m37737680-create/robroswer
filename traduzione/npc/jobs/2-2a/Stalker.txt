valkyrie,53,58,3	script	Stalker#Valkyrie	747,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Stalker]";
			mes "Congratulazioni.";
			mes "Onore ai guerrieri!";
			close;
		}
		mes "[Stalker]";
		mes "Ehi...";
		mes "È difficile";
		mes "essere un eroe";
		mes "e essendo misterioso,";
		mes "inaffidabile,";
		mes "furtivo";
		next;
		mes "[Stalker]";
		mes "Ma quando il";
		mes "diventa difficile";
		mes "i miei amici sanno che";
		mes "puoi contare su di me.";
		mes "ne ho bisogno e";
		mes "hanno bisogno di me.";
		close;
	}
	else if (ADVJOB == Job_Stalker && Class == Job_Thief_High && JobLevel > 39) {
		mes "[Stalker]";
		mes "Questo mondo ha bisogno di";
		mes "altri eroi che sono";
		mes "disposto a camminare sul filo";
		mes "tra l'ordine e la legge selvaggia.";
		next;
		mes "[Stalker]";
		mes "Sei pronto";
		mes "entrare nei ranghi";
		mes "dei più furtivi di";
		mes "guerrieri? Sei pronto";
		mes "diventare uno stalker?";
		next;
		if (select("No.:Sì.") == 1) {
			mes "[Stalker]";
			mes "Quando sei pronto,";
			mes "sentiti libero di tornare.";
			mes "Onore ai guerrieri!";
			close;
		}
		if (SkillPoint) {
			mes "[Stalker]";
			mes "È ancora possibile che tu impari più abilità. Per favore usa";
			mes "tutti i tuoi Punti Abilità rimanenti prima di tornare da me.";
			close;
		}
		jobchange Job_Stalker;
		set ADVJOB,0;
		mes "[Stalker]";
		mes "Congratulazioni!";
		mes "Come Stalker, spero";
		mes "tu pugnaloni le persone giuste";
		mes "dietro. Bandisci il";
		mes "disgustoso usando i propri";
		mes "metodi malvagio!";
		close;
	}
	mes "[Stalker]";
	mes "Benvenuto";
	mes "a Valhalla,";
	mes "la Sala d'Onore.";
	next;
	mes "[Stalker]";
	mes "Per favore fai";
	mes "ti senti a tuo agio";
	mes "mentre sei qui.";
	mes "Onore ai guerrieri!";
	close;
}
