//callfunc "F_CoolEventCorp","<menu>","<location>","<save map>",<x>,<y>,"<warp map>";
function	script	F_CoolEventCorp	{
	cutin "zonda_01",2;
	mes "[Cool Event Corp. Personale]";
	mes "Benvenuti a Cool Event Corp.";
	mes "Il nostro personale lavora sempre";
	mes "superare le vostre aspettative";
	mes "per un servizio di qualità. Quindi come";
	mes "Posso assisterla oggi?";
	next;
	switch(select(getarg(0))) {
	case 1:
		mes "[Cool Event Corp. Personale]";
		mes "Il tuo punto di riapparizione";
		mes "è stato salvato qui";
		mes getarg(1)+".";
		mes "Grazie per aver utilizzato il";
		mes "Servizio di Cool Event Corp.~";
		savepoint getarg(2),getarg(3),getarg(4),1,1;
		close2;
		break;
	case 2:
		mes "[Cool Event Corp. Personale]";
		if(!callfunc("F_CanOpenStorage")){
			mes "Mi dispiace, ma tu";
			mes "ho bisogno del novizio";
			mes "Livello di abilità base 6 a";
			mes "usa il servizio di archiviazione.";
			close2;
			break;
		}
		if(Zeny < 40){
			mes "Mi dispiace, ma tu non";
			mes "avere abbastanza Zeny per usare";
			mes "il servizio di archiviazione. Il nostro";
			mes "La tariffa di accesso allo storage è di 40 Zeny.";
			close2;
			break;
		}
		set Zeny, Zeny - 40;
		mes "Lasciami aprire il tuo personale";
		mes "archiviazione per te subito.";
		mes "Grazie per aver supportato Cool";
		mes "Event Corp. utilizzando il nostro";
		mes "servizi. Buona giornata~";
		callfunc("F_CheckKafCode");	//check your storage password, if set
		close2;
		openstorage;
		break;
	case 3:
		mes "Scegliere";
		mes "la tua destinazione.";
		next;
		if (VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE))
			set .@cost,4400;
		else
			set .@cost,2200;
		.@destination$ = getarg(5);
		if (select(.@destination$ +" -> "+ .@cost +" z:Annulla") == 1) {
			if ((countitem(7060) != 0) || (Zeny >= .@cost)) {
				if (countitem(7060) != 0) {
					delitem 7060,1; //Warp_Free_Ticket
				}
				else {
					set Zeny,Zeny-.@cost;
				}
				if (.@destination$ == "Veins") warp "veins",205,101;
				else if (.@destination$ == "Rachel") warp "rachel",115,125;
				cutin "",255;
				end;
			}
			else {
				mes "[Cool Event Corp. Personale]";
				mes "Mi dispiace, ma non hai";
				mes "abbastanza zeny per il Teletrasporto";
				mes "Servizio. La tariffa per teletrasportare";
				mes "a "+ .@destination$ +" è "+.@cost+" zeny.";
			}
		}
		close2;
		break;
	case 4:
		mes "[Cool Event Corp. Personale]";
		if(BaseClass != Job_Merchant){
			mes "Mi dispiace, ma il";
			mes "Servizio di noleggio carretti";
			if (checkre(0))
				mes "disponibile solo per le classi Merchant.";
			else {
				mes "disponibile solo ai Mercanti,";
				mes "Fucinieri, Smith Bianchi,";
				mes "Alchimisti e Creatori.";
			}
			close2;
			break;
		}
		if(getskilllv("MC_PUSHCART") == 0){
			mes "Puoi noleggiare un carrello solo dopo";
			mes "imparare l'abilità del carretto.";
			close2;
			break;
		}
		if(checkcart() == 1){
			mes "Hai già";
			mes "Equipato con un carretto.";
			mes "Purtroppo non possiamo";
			mes "noleggiare più di uno a";
			mes "un cliente alla volta.";
			close2;
			break;
		}
		if(countitem(7061)){
			delitem 7061,1; //Cart_Free_Ticket
			setcart;
			close2;
			break;
		}
		mes "Il noleggio del carretto";
		mes "la tariffa è di 800 Zeny. Vorrebbe";
		mes "Ti piace affittare un carretto?";
		next;
		if(select("Noleggiare un carretto:Annulla") == 1){
			if(Zeny < 800){
				mes "[Cool Event Corp. Personale]";
				mes "Mi dispiace, ma tu";
				mes "non ho abbastanza";
				mes "Zeny per pagare il carretto";
				mes "canone di locazione di 800 Zeny.";
			} else {
				set Zeny, Zeny - 800;
				setcart;
			}
		}
		close2;
		break;
	case 5:
		callfunc("F_SetKafCode","[Cool Event Corp. Staff]","Cool Event Corp.");
		//the function closes msg and removes cutins
	case 6:
	default:
		mes "[Cool Event Corp. Personale]";
		mes "Cool Event Corp. è sempre";
		mes "cercare di fornire il meglio";
		mes "servizi per i nostri clienti.";
		mes "Aiutaci a diventare i migliori con";
		mes "fornirci le vostre opinioni";
		mes "e un feedback onesto. Grazie.";
		close2;
	}
	cutin "", 255;
	end;
}

	callfunc "F_CoolEventCorp",
		"Save:Use Storage::Rent a Pushcart:Storage Password Service:Cancel",
		"in the town of Lighthalzen","lighthalzen",95,240;
	end;
}
