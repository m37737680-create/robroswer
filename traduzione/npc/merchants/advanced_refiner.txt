payon,157,146,6	script	Suhnbi#cash	85,{
	disable_items;
	mes "[Suhnbi]";
	mes "Sono l'Armaiolo";
	mes "Posso affinare ogni tipo di arma,";
	mes "armatura e attrezzature, quindi lasciatemi";
	mes "sai cosa vuoi raffinare.";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(set .@i,1; .@i<=10; set .@i,.@i+1) {
		if (getequipisequiped(.@indices[.@i])) {
			set .@menu$, .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			set .@equipped,1;
		}
		set .@menu$, .@menu$ + ":";
	}
	if (.@equipped == 0) {
		mes "[Suhnbi]";
		mes "Non credo di poter raffinare nessun articolo che hai...";
		close;
	}
	set .@part, .@indices[ select(.@menu$) ];

	if (!getequipisequiped(.@part)) //custom check
		close;
	if (!getequipisenableref(.@part)) {
		mes "[Suhnbi]";
		mes "Vai a trovare un altro fabbro. Non puoi raffinare questa cosa.";
		close;
	}
	if (getequiprefinerycnt(.@part) >= 10) {
		mes "[Suhnbi]";
		mes "Hmm... Qualcuno l'ha già perfezionato. Non credo di poterci lavorare ulteriormente.";
		close;
	}

	.@refineitemid = getequipid(.@part); // save id of the item
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	.@price = getequiprefinecost(.@part, REFINE_COST_ENRICHED, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_ENRICHED, REFINE_MATERIAL_ID);
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );

	// Make sure you have the necessary items and Zeny to refine your items
	// Determines chance of failure and verifies that you want to continue.
	callsub S_RefineValidate,.@itemtype,.@material,.@price,.@part,.@refineitemid,.@refinerycnt;

	mes "[Suhnbi]";
	mes "Clang! Clang! Clang!";
	if (getequippercentrefinery(.@part, true) > rand(100)) {
		successrefitem .@part;
		next;
		emotion ET_BEST;
		mes "[Suhnbi]";
		mes "Eccolo! È fatto.";
		mes "È passato un po' di tempo da quando ho fatto una così bella "+((.@itemtype == IT_WEAPON)?"arma":"armatura")+"Devi essere felice perché è diventato più forte!";
		close;
	}
	failedrefitem .@part;
	next;
	emotion (!rand(5))?ET_MONEY:ET_HUK;
	mes "[Suhnbi]";
	mes "Uuuuuuuuuummmmmph!!!";
	next;
	mes "[Suhnbi]";
	mes "...";
	mes ".....";
	mes ".......Uhuhuhuhu~";
	mes "........È stata la tua scelta e la mia capacità, nessun rimpianto.";
	close;

S_RefineValidate:
	.@itemtype = getarg(0);
	.@item_req = getarg(1);
	.@price = getarg(2);
	.@part = getarg(3);
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);

	if( .@itemtype == IT_ARMOR ){
		.@equip_lv = getequiparmorlv( .@part );

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if (VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE)) {
			switch(.@equip_lv) {
				case 1:
					set .@price, .@price * 10;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else if( .@itemtype == IT_WEAPON ){
		.@equip_lv = getequipweaponlv( .@part );

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if (VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE)) {
			switch( .@equip_lv ){
				case 1:
					set .@price, .@price * 40;
					break;
				case 2:
					set .@price, .@price * 50;
					break;
				case 3:
					set .@price, .@price * 2;
					break;
				case 4:
					set .@price, .@price * 2;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else{
		// TODO:
		close;
	}

	mes "[Suhnbi]";
	if (.@itemtype == IT_WEAPON)
		mes "Vuoi perfezionare un livello "+ .@equip_lv +" arma?";
	mes "Per affinare questo, avrai bisogno di un ^ff9999"+ getitemname(.@item_req) +"^000000 e "+ .@price +" zeny.";
	mes "Vuoi continuare?";
	next;
	if(select("Sì:No") == 1) {
		if (getequippercentrefinery(.@part) < 100) {
			if (.@itemtype == IT_WEAPON) {
				mes "[Suhnbi]";
				mes "Wow!!";
				mes "Questa arma probabilmente";
				mes "sembra che sia stato raffinato...";
				mes "molte volte...";
				mes "Potrebbe rompersi se";
				mes "lo migliori di nuovo.";
				next;
				mes "E se si rompe,";
				mes "non puoi più usarlo!";
				mes "Tutte le carte che ci sono e le proprietà ^ff0000saranno perse^000000!";
				mes "^ff0000Inoltre, l'attrezzatura si romperà!^000000";
				mes "Sei sicuro di voler continuare?";
				next;
				if(select("Sì:No") == 2) {
					mes "[Suhnbi]";
					mes "Bene.";
					mes "Perché se l'arma si rompe per un raffinamento irragionevole, allora anche io mi sento di cattivo umore.";
					close;
				}
			} else {
				mes "[Suhnbi]";
				mes "Ridacchiare. Ridacchiare. Oh, hai coraggio, osare raffinare questo.";
				mes "Sai che è piuttosto rischioso, non è vero?";
				next;
				mes "Se il tuo equipaggiamento difensivo è rotto, non potrai più usarlo mai più.";
				mes "Anche le tue carte e le tue modifiche svaniranno completamente ^ff0000disapparire^000000.";
				//mes "Everything will disappear. As in... GONE!";
				mes "Vuoi davvero continuare?";
				next;
				if(select("Sì:No") == 2) {
					mes "[Suhnbi]";
					mes "Che sciocchezze. Sprechi il mio prezioso tempo.";
					mes "Vattene a quel paese, punk.";
					close;
				}
			}
		}
		if (countitem(.@item_req) > 0 && Zeny > .@price) {
			delitem .@item_req,1;
			set Zeny, Zeny - .@price;

			// anti-hack
			if (callfunc("F_IsEquipIDHack", .@part, getarg(4)) ||
				callfunc("F_IsEquipRefineHack", .@part, getarg(5)) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3])) {
				mes "[Suhnbi]";
				emotion ET_FRET;
				mes "Aspetta un secondo...";
				mes "Pensi che sono stupido?!";
				mes "Hai cambiato l'oggetto mentre non guardavo! Via da qui!";
				close;
			}

			return;
		}
		mes "[Suhnbi]";
		mes "È tutto quello che hai?";
		mes "Mi dispiace molto, ma non posso fare nulla senza tutti i materiali. Inoltre, merito dei pagamenti per il mio lavoro, non è vero?";
		close;
	}
	mes "[Suhnbi]";
	mes "Non posso farci nulla anche se non te ne fa nulla...";
	close;
}
