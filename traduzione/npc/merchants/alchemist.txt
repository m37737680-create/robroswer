alde_alche,24,188,3	script	Guild Dealer	740,{
	if (checkweight(1201,1) == 0) {
		mes "Aspetta un minuto!";
		mes "Attualmente stai portando";
		mes "troppi oggetti con te.";
		mes "Per favore torna di nuovo";
		mes "dopo che hai memorizzato alcuni elementi nello storage kafra.";
		close;
	}
	mes "[Gever Al Sharp]";
	mes "Benvenuto/a al";
	mes "Unione degli Alchimisti.";
	mes "Come posso assisterti oggi?";
	next;
	switch(select("Acquistare materiali.:Acquistare un manuale di produzione.:Basta.")) {
	case 1:
		mes "[Gever Al Sharp]";
		mes "Cosa vorresti?";
		next;
		if (select("Ciotola di Medicina - 8 Zeny:Annulla.") == 1) {
			mes "[Gever Al Sharp]";
			mes "Quanti ne vuoi?";
			mes "Inserisci '0' se vuoi uscire.";
			next;
			while(1) {
				input .@input,0,2001;
				if (.@input == 0) {
					mes "[Gever Al Sharp]";
					mes "L'accordo è stato annullato.";
					mes "La prossima volta.";
					close;
				}
				else if (.@input < 1 || .@input > 2000) {
					mes "[Gever Al Sharp]";
					mes "Il numero deve";
					mes "meno di 2000.";
					next;
				}
				else {
					break;
				}
			}
			set .@sell,.@input * 8;
			set .@item_weight,.@input * 10;
			if (Zeny < .@sell) {
				mes "[Gever Al Sharp]";
				mes "Non lo fai";
				mes "a abbastanza zeny.";
				mes "Controlla quanti zeny";
				mes "a te la prima.";
				close;
			}
			if ((MaxWeight - Weight) < .@item_weight) {
				mes "[Gever Al Sharp]";
				mes "Non sembra che";
				mes "puoi portare tutto.";
				mes "Controlla lo spazio";
				mes "nel tuo inventario.";
				close;
			}
			set Zeny, Zeny-.@sell;
			getitem 7134,.@input; //Medicine_Bowl
			mes "[Gever Al Sharp]";
			mes "Grazie.";
			mes "Ancora.";
			close;
		}
		mes "[Gever Al Sharp]";
		mes "Bene allora,";
		mes "di nuovo quando";
		mes "devi acquistare";
		mes "materiali relativi a";
		mes "Alchimia, va bene?";
		close;
	case 2:
		mes "[Gever Al Sharp]";
		mes "Di cosa hai bisogno?";
		mes "I manuali costano generalmente 100.000 zeny. Ma ci sono un paio di manuali speciali che costeranno di più.";
		next;
		switch(select("Guida alla Creazione di Pozioni:Guida alla Creazione di Alcolici:Guida alla Creazione di Granate Artigianali:Guida alla Creazione di Bottiglie d'Acido:Guida alla Creazione di Bottiglie per Piante:Guida alla Creazione di Bottiglie Marine Sphere:Guida alla Creazione di Rivestimenti Lucidi:Guida condensata alla creazione di pozioni:Annulla Affare.")) {
		case 1:
			callsub S_SellManual,7144,100000; //Normal_Potion_Book
			break;
		case 2:
			callsub S_SellManual,7127,100000; //Alcol_Create_Book
			break;
		case 3:
			callsub S_SellManual,7128,100000; //FireBottle_Create_Book
			break;
		case 4:
			callsub S_SellManual,7129,100000; //Acid_Create_Book
			break;
		case 5:
			callsub S_SellManual,7130,100000; //Plant_Create_Book
			break;
		case 6:
			callsub S_SellManual,7131,100000; //Mine_Create_Book
			break;
		case 7:
			callsub S_SellManual,7132,100000; //Coating_Create_Book
			break;
		case 8:
			callsub S_SellManual,7133,240000; //Slim_Potion_Create_Book
			break;
		case 9:
			mes "[Gever Al Sharp]";
			mes "Beh allora...";
			mes "Torna se tu";
			mes "mai aver bisogno di comprare";
			mes "un manuale di produzione.";
			close;
		}
	case 3:
		mes "[Gever Al Sharp]";
		mes "Allora,";
		mes "ti auguro una buona giornata.";
		close;
	}

S_SellManual:
	.@item_id = getarg(0);
	.@zeny_req = getarg(1);

	if (Zeny < .@zeny_req) {
		mes "[Gever Al Sharp]";
		mes "Non lo fai";
		mes "a abbastanza zeny.";
		mes "Controlla quanti zeny";
		mes "a te la prima.";
		close;
	}
	mes "[Gever Al Sharp]";
	mes getitemname(.@item_id) +"?";
	mes "Saranno "+ .@zeny_req +" zeny.";
	next;
	if (select("Acquisto.:Basta.") == 1) {
		Zeny = Zeny - .@zeny_req;
		getitem .@item_id,1;
		mes "[Gever Al Sharp]";
		mes "Grazie per";
		mes "il vostro patrocinio.";
		close;
	}
	mes "[Gever Al Sharp]";
	mes "Ancora";
	mes "la prossima volta.";
	close;
}
