// Black Marketeer (Buy Licenses - Non-Merchant Classes) :: qskill_buy_store
//============================================================ 
que_job01,68,84,1	script	Black Marketeer#Buying	881,{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2400) {
		mes "[Signor Jass]";
		mes "Sei troppo avido, anche rispetto a qualcuno come me!";
		mes "Perché non vai a alleggerire la tua borsa prima?";
		close;
	}
	if (getskilllv("ALL_BUYING_STORE") == 1) {
		mes "[Signor Jass]";
		mes "Ehi, hai già fatto un contratto con Hugh.";
		mes "Non ho nessun affare con te.";
		close;
	}
	mes "[Signor Jass]";
	mes "Devi aver bisogno di qualcosa disperatamente per venire a trovarmi.";
	mes "Cosa vuoi?";
	next;
	switch(select("Licenza di negozio per acquirenti all'ingrosso:Chi sei?:Niente, niente!")) {
	case 1:
		mes "[Signor Jass]";
		mes "Lo sapevo!";
		mes "Certo, posso farlo per te.";
		mes "Il mio assomiglia esattamente a quello autentico che i mercanti prendono da quel bastardo Mr. Hugh!";
		next;
		mes "[Signor Jass]";
		mes "E la mia licenza è migliore perché non hai bisogno di imparare alcuna abilità.";
		mes "Quanti ne vuoi?";
		next;
		mes "[Signor Jass]";
		mes "Solo per farvi sapere che posso farne solo piccole quantità, fino a 10 alla volta.";
		mes "Costerà 500 zeny ciascuno.";
		next;
		while(1) {
			input .@input;
			mes "[Signor Jass]";
			if (.@input == 0) {
				mes "Non ti servono quelle licenze?";
				close;
			} else if (.@input > 10) {
				mes "Ne posso fare al massimo 10 alla volta, sai.";
				next;
			} else {
				mes "Costerà "+(.@input*500)+" zeny.";
				if (Zeny < (.@input*500)) {
					mes "ma non hai abbastanza soldi.";
					next;
					mes "[Signor Jass]";
					mes "Non conosci le basi del commercio? Tutto ha un prezzo.";
					mes "Se vuoi qualcosa, devi pagarlo.";
				} else {
					mes "Ah... Ah ah ah!";
					mes "Signor Hugh, mi occuperò del suo affare di licenze. Vedrà!";
					mes "*Risatina Risatina*";
					getitem 12548,.@input; //Buy_Market_Permit2
					set Zeny,Zeny-(.@input*500);
				}
				close;
			}
		}
	case 2:
		mes "[Signor Jass]";
		mes "Ho lasciato la mia città natale da molto tempo.";
		mes "È inutile chiedere chi sono perché tutto quello che mi è rimasto ora è il mio odio.";
		next;
		mes "[Signor Jass]";
		mes "...";
		mes "Hugh è un mercante corrotto senza alcun senso di etica commerciale.";
		mes "Il mio unico scopo nella vita è distruggere Hugh.";
		next;
		mes "[Signor Jass]";
		mes "Ah, ho bevuto troppo... (*Hic*)";
		mes "È solo l'alcol che parla.";
		mes "Per favore, dimentica qualsiasi cosa abbia detto.";
		close;
	case 3:
		mes "[Signor Jass]";
		mes "Va bene, va bene! Non devi urlare.";
		mes "Lasciami stare se non hai affari con me.";
		close;
	}
}

// Purchasing Team (Learn Skill - Merchant Classes)

alberta_in,58,52,4	script	Purchasing Team#Buying	59,{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2400) {
		mes "Non puoi conversare o eseguire la missione perché stai portando troppi oggetti.";
		close;
	}
	mes "[Signor Hugh]";
	if (BaseClass == Job_Merchant && getskilllv("MC_VENDING") >= 1) {
		if (getskilllv("ALL_BUYING_STORE") == 1) {
			mes "Sono Hugh del Team Acquisti.";
			mes "Come posso aiutarla oggi?";
			next;
			if(select("Acquista Licenza Negozio per Grandi Acquisti:Smetti") == 2) {
				mes "[Signor Hugh]";
				mes "Sentiti libero di chiedermelo se hai bisogno di licenze per negozi di grossisti.";
				mes "Ancora?";
				close;
			}
			mes "[Signor Hugh]";
			mes "Sono 200 zeny per ciascuna Licenza di Negozio per Grandi Acquirenti, e puoi acquistarne fino a 50 alla volta.";
			mes "Quante licenze ti servono?";
			next;
			while(1) {
				input .@input;
				mes "[Signor Hugh]";
				if (.@input == 0) {
					mes "Hai annullato l'operazione.";
					mes "Buona giornata.";
					close;
				} else if (.@input > 50) {
					mes "Inserisci un valore di 50 o inferiore.";
					next;
				} else {
					mes "Costerà "+(.@input*200)+" zeny per "+.@input+" licenze.";
					if (Zeny < (.@input*200)) {
						mes "ma non sembri avere abbastanza soldi.";
					} else {
						mes "Grazie per la vostra fiducia.";
						getitem 6377,.@input; //Buy_Market_Permit
						set Zeny,Zeny-(.@input*200);
					}
					close;
				}
			}
		} else {
			mes "Sono Hugh del Team Acquisti della Alberta Merchant Guild.";
			mes "Sei...";
			next;
			mes "[Signor Hugh]";
			mes "Sapevi che? La nostra gilda ha emesso una licenza per consentire agli individui di acquistare beni da altri.";
			next;
			select("Non ho mai avuto problemi ad acquistare articoli...");
			mes "[Signor Hugh]";
			mes "Hai ragione, ma pensaci:";
			mes "non hai avuto difficoltà ad acquistare all'ingrosso?";
			mes "Dovresti trovare e parlare con tutti coloro che hanno un oggetto che desideri.";
			mes "Molto scomodo, non è vero?";
			next;
			mes "[Signor Hugh]";
			mes "Poiché l'acquisto all'ingrosso è un problema importante per noi Mercanti,";
			mes "Ho proposto un piano innovativo alla nostra gilda, basato sui miei 10 anni di esperienza nell'effettuare acquisti.";
			next;
			mes "[Signor Hugh]";
			mes "Lasciare che gli individui si aprano";
			mes "Un negozio per acquirenti all'ingrosso!";
			mes "Questo è il titolo della mia proposta.";
			mes "Vedrai, quando lo leggerai...";
			next;
			mes "[Signor Hugh]";
			mes "...";
			mes "(Il signor Hugh continua a chiacchierare sui dettagli.)";
			next;
			select("Qual è il tuo punto?");
			mes "[Signor Hugh]";
			mes "Oh, sì. Riassumendo,";
			mes "puoi acquistare determinati articoli all'ingrosso tramite Vending.";
			next;
			mes "[Signor Hugh]";
			mes "Per aprire un Negozio di Acquisto all'Ingrosso, è necessaria una licenza emessa dalla Gilda dei Mercanti.";
			next;
			mes "[Signor Hugh]";
			mes "Te ne serve ogni volta che apri il negozio. Ci aspettiamo un aumento significativo dei profitti grazie a questo nuovo tipo di licenza.";
			next;
			mes "[Signor Hugh]";
			mes "Questo è il punto della mia proposta!";
			mes "Il nostro presidente era così felice di sapere che faremo un sacco di soldi!";
			next;
			mes "["+strcharinfo(0)+"]";
			mes "Per favore vai dritto al punto!";
			next;
			mes "[Signor Hugh]";
			mes "Non essere così impaziente, va bene?";
			mes "Il mio punto è che possiamo farti aprire il Negozio per Grandi Acquirenti se hai imparato l'Automazione.";
			next;
			mes "[Signor Hugh]";
			mes "Certo, addebitiamo 10.000 zeny come tassa di registrazione una tantum.";
			mes "Avrai bisogno della licenza ^4A4AFFBulk Buyer Shop License^000000 ogni volta che apri il negozio.";
			next;
			mes "[Signor Hugh]";
			mes "Vorresti registrarti ora?";
			mes "Se lo fai, ti insegnerò come aprire il negozio per acquirenti all'ingrosso.";
			next;
			if(select("Impara come aprire un negozio per acquirenti all'ingrosso:Annulla") == 2) {
				mes "[Signor Hugh]";
				mes "Mann, è deludente!";
				mes "Usare questa opzione per acquirenti all'ingrosso può giovare alla tua attività in molti modi, ma è una scelta tua.";
				mes "Sono sempre disponibile per consulenze!";
				close;
			}
			mes "[Signor Hugh]";
			if (Zeny < 10000) {
				mes "La tassa di registrazione è di 10.000 zeny.";
				mes "Per favore prepara prima la tariffa.";
				close;
			} else {
				mes "Hai preso una buona decisione.";
				mes "Per favore, dammi la quota di registrazione e firma proprio qui....";
				next;
				input .@name$;
				mes "[Signor Hugh]";
				mes .@name$+"....";
				mes "Mi piace la tua calligrafia.";
				mes "Va bene, ora sei autorizzato ad aprire il Negozio per Grandi Acquisti.";
				set Zeny,Zeny-10000;
				getitem 6377,5; //Buy_Stall_Permit
				skill "ALL_BUYING_STORE",1,SKILL_PERM_GRANT;
				next;
				mes "[Signor Hugh]";
				mes "Attualmente, solo gli oggetti normali ^8C2121TRANNOS^000000 l'equipaggiamento, alcune pozioni e gli oggetti artigianali possono essere acquistati all'ingrosso, ma questo può comunque esserti molto utile, a seconda di come lo usi.";
				next;
				mes "[Signor Hugh]";
				mes "E devi avere almeno uno degli articoli che vuoi comprare nel tuo inventario perché devi mostrarlo agli altri attraverso il tuo negozio.";
				next;
				mes "[Signor Hugh]";
				mes "La tua abilità dovrebbe ora essere registrata nella tua finestra delle abilità. Se non riesci a vederla, probabilmente devi minimizzare la tua Lista Abilità e controllare la terza scheda Lavoro.";
				next;
				mes "[Signor Hugh]";
				mes "Ti ho dato 5 licenze di negozio per grossisti per la tua prova.";
				mes "Buono shopping!";
				close;
			}
		}
	}
	mes "Sono Hugh del Team Acquisti della Alberta Merchant Guild.";
	mes "Mi piacerebbe chiacchierare, ma al momento sono troppo impegnato/a.";
	next;
	mes "[Signor Hugh]";
	mes "Il mio tempo è dedicato esclusivamente ai nostri clienti nel settore dei commercianti.";
	close;
}
