function Dyes {
	setarray .@DyeName$[0],"the default color","Orange","Violet","Red","Black","Green","Blue","White","Yellow";
	setarray .@DyeItems[0],0,980,981,975,983,979,978,982,976;
	set .@menu$,"- Default:";
	for(set .@i,0; .@i<getargcount(); set .@i,.@i+2)
		set .@menu$, .@menu$+"- "+.@DyeName$[getarg(.@i)]+":";
	set .@menu$, .@menu$+"- ^777777Cancel^000000";
	set .@s, select(.@menu$)-1;
	if (.@s == (getargcount()/2)+1) goto L_End;
	mes .n$;
	if (.@s == 0) {
		mes "Posso cambiare il tuo colore di tintura a";
		mes "il predefinito è gratuito.";
		mes "Sei sicuro?";
		next;
		switch(select("Sì:No")) {
		case 1:
			mes .n$;
			setlook 7,0;
			mes "I tuoi vestiti sono stati tinti "+.@DyeName$[0]+".";
			close;
		case 2:
			goto L_End;
		}
	}
	if (!countitem(.@DyeItems[getarg((.@s-1)*2)])) {
		mes "Per tingere i tuoi vestiti, mi serviranno i coloranti appropriati. Per favore controlla la mia lista prezzi per le informazioni.";
		close;
	}
	if (Zeny < 10000) {
		mes "Mi dispiace, ma non hai abbastanza soldi.";
		close;
	}
	delitem .@DyeItems[getarg((.@s-1)*2)],1;
	set Zeny, Zeny-10000;
	setlook 7, ((.@s)?(getarg((.@s-1)*2+1)):0);
	mes "I tuoi vestiti sono stati tinti "+((.@s)?.@DyeName$[getarg((.@s-1)*2)]:.@DyeName$[0])+".";
	close;
}
