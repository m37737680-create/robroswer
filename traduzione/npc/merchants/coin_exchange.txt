// Manuk & Splendide
//============================================================
-	script	::merchant_13_2	-1,{
	set .@n$, "["+strnpcinfo(1)+"]";
	mes .@n$;
	if (isequipped(2782) && ep13_2_rhea == 100) {
		if (!checkweight(1201,1)) {
			mes "Sembra che stia portando troppe cose.";
			mes "Perché non mettere alcuni dei tuoi oggetti in deposito e tornare?";
			close;
		}
		mes "Ciao.";
		mes "Cosa stai cercando?";
		next;
		if (strnpcinfo(1) == "Merchant of Manuk") {
			setarray .@Items[2],12342,12343,12348;
			set .@Book,11019; //Manuk_Selling_Item
			set .@Coin,6080; //Manuk_Coin
			set .@i, select("Visualizza descrizione articolo:L'opportunità di Manuk:Il coraggio di Manuk:La Fede di Manuk:Annulla");
		}
		else {
			setarray .@Items[2],12344,12345,12349;
			set .@Book,11018; //Splendide_Selling_Item
			set .@Coin,6081; //Splendide_Coin
			set .@i, select("Visualizza descrizione articolo:Compra il marmellata di frutta di Pinguicula:Compra il marmellata di miele di Luciola:Compra Lacrime di Cornus:Non fare niente");
		}
		mes .@n$;
		if (.@i == 1) {
			mes "Ecco le descrizioni degli articoli.";
			close2;
			readbook .@Book,1;
			end;
		}
		else if (.@i < 5) {
			mes "Posso venderti 3 "+getitemname(.@Items[.@i])+" per ^3131FF1 moneta^000000.";
			next;
			switch(select("Acquisto:Non acquistare")) {
			case 1:
				if (countitem(.@Coin)) {
					mes .@n$;
					mes "Grazie per essere venuto/a.";
					delitem .@Coin,1; 
					getitem .@Items[.@i],3;
					close;
				}
				else {
					mes .@n$;
					mes "Non hai abbastanza monete.";
					close;
				}
			case 2:
				mes .@n$;
				break;
			}
		}
		mes "Ripassiscilo se cambi idea.";
		close;
	}
	else {
		if (strnpcinfo(1) == "Merchant of Manuk") {
			mes "Rtt od d";
			mes "Cosa è un irr";
		}
		else {
			mes "BurWehAla";
			mes "tasnarAndu Ie Ru";
		}
		close;
	}
}
man_in01,286,16,1	duplicate(merchant_13_2)	Merchant of Manuk	454
spl_in01,110,326,5	duplicate(merchant_13_2)	Merchant of Splendide	439
