payon,173,238,5	script	Jade#pay	754,{
	if (checkweight(1201,1) == 0) {
		mes "^3355FFUn attimo! In questo momento, stai portando troppi oggetti con te. Per favore torna dopo aver messo alcuni dei tuoi effetti nello Stoccaggio di Kafra.^000000";
		close;
	}
	mes "[Giada]";
	mes "Portami due";
	mes "Gioielli di";
	mes "stesso colore, e io";
	mes "trasformali in Pietre Preziose";
	mes "di un colore diverso.";
	next;
	switch(select("Gemme blu in quelle rosse!:Gemme rosse in quelle gialle!:Gemme gialle in gemme blu!")) {
	case 1: callsub S_TradeGems,717,716;
	case 2: callsub S_TradeGems,716,715;
	case 3: callsub S_TradeGems,715,717;
	}

S_TradeGems:
	.@item_req = getarg(0);
	.@item_id = getarg(1);

	if (countitem(.@item_req) < 2) {
		mes "[Giada]";
		mes "Ah...!";
		mes "Stai scherzando, vero?";
		mes "Non posso fornirti questo";
		mes "servizio se non";
		mes "almeno dammi";
		mes "2 "+ getitemname(.@item_req) +"s!";
		close;
	}
	else {
		.@gems = countitem(.@item_req) /2;
		mes "[Giada]";
		mes "Credo di poter creare";
		mes "un totale di "+ .@gems +" "+ getitemname(.@item_id) +"s";
		mes "usando il "+ getitemname(.@item_req) +"s";
		mes "che hai attualmente.";
		mes "Cosa vuoi fare?";
		next;
		switch(select("Dammi quanti ne puoi.:Voglio impostare l'importo.:Ho lasciato.")) {
		case 1:
			delitem .@item_req,.@gems * 2;
			getitem .@item_id,.@gems;
			break;
		case 2:
			mes "[Giada]";
			mes "Quanti";
			mes "vuoi?";
			mes "Il numero massimo";
			mes "che puoi inserire è 100.";
			next;
			while(1) {
				input .@input,0,101;
				if (.@input == 0) {
					mes "[Giada]";
					mes "Niente affatto?";
					mes "Penso che tu";
					mes "cambiato idea...";
					close;
				}
				else if (.@input > 100) {
					mes "[Giada]";
					mes "Errm...";
					mes "Ti ho chiesto di inserire";
					mes "un importo non superiore a";
					mes "più di 100, ricordi...?";
					next;
				}
				else if (.@gems < .@input) {
					// Custom dialogue
					mes "[Giada]";
					mes "Errm...";
					mes "Non ce l'hai";
					mes "molte gemme da scambiare...";
					next;
				}
				else break;
			}
			delitem .@item_req,.@input * 2;
			getitem .@item_id,.@input;
			break;
		case 3:
			mes "[Giada]";
			mes "Certo, nessun problema.";
			mes "Torna pure quando vuoi.";
			close;
		}
		mes "[Giada]";
		mes "Eccolo/a.";
		mes "Sentiti libero di venire";
		mes "tornare in qualsiasi momento.";
		mes "Hmm, cosa sta cercando?";
		mes "C'è qualcosa sul mio viso?";
		close;
	}
}
