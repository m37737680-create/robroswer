-	script	Ice Cream Maker::IceCreamer	-1,{
	mes "Gelatiera";
	mes "Gelato fresco fatto con neve di Lutie!";
	mes "Goditelo ora, non sarà in saldo per molto tempo!";
	mes "^3355FF100 Zeny^000000 Gelato,";
	mes "Gelato!";
	next;
	if(select("Dammi il gelato!:Annulla Transazione") == 2) {
		mes "Sei sicuro di non volerne nessuno?";
		mes "Non lo venderò per molto tempo,";
		mes "e quando ne avrò finito,";
		mes "non ce ne saranno più!!!";
		close;
	}
	mes "Gelatiera";
	mes "Gelato fresco fatto con neve di Lutie!";
	mes "Tutti vogliono il nostro delizioso gelato, ";
	mes "ma abbiamo una quantità limitata,";
	mes "quindi puoi acquistare solo 5 alla volta!!";
	next;
	while(true) {
		input .@input;
		if(.@input < 1) {
			mes "Gelatiera";
			mes "Se non vuoi comprare nessuno,";
			mes "potresti lasciare il prossimo cliente";
			mes "fare un acquisto?";
			mes "Grazie.";
			close;
		}
		if(.@input > 5) {
			mes "Gelatiera";
			mes "Ahi";
			mes "Te lo aspetti troppo.";
			mes "Gentile cliente,";
			mes "Se mangi più di 5 gelati,";
			mes "Se potresti dover fare molti viaggi";
			mes "in bagno stasera.";
			next;
			continue;
		}
		break;
	}
	if(Zeny<.@input*100) {
		mes "Gelatiera";
		mes "Caro cliente, il tuo portafoglio sembra leggero.";
		mes "Il prezzo è di ^3355FF100 Zeny^000000 per gelato.";
		close;
	}
	if(checkweight(536,.@input) == 0) {
		mes "Gelatiera";
		mes "Caro cliente, sembri portare molto.";
		mes "Il gelato va bene,";
		mes "ma devi considerare il tuo peso";
		mes "prima di effettuare un acquisto.";
		close;
	}
	set Zeny,Zeny-100*.@input;
	getitem 536,.@input;	// Ice_Cream
	close;
}

// Alberta
alberta,120,45,2	duplicate(IceCreamer)	Ice Cream Maker#1	85
