lhz_in02,230,284,4	script	Hotel Employee#01	86,{
	mes "Dipendente dell'hotel";
	mes "Benvenuto a";
	mes "il Drago Reale,";
	mes "dove puoi trovare il";
	mes "migliori alloggiamenti";
	mes "e il miglior servizio.";
	next;
	switch(select("Punto di salvataggio:Riposo - 5.000 zeni:Annulla")) {
	case 1:
		savepoint "lhz_in02",209,275,1,1;
		mes "Dipendente dell'hotel";
		mes "Grazie, tuo";
		mes "Punto di riapparizione ha";
		mes "salvato qui in";
		mes "Il Drago Reale.";
		close;
	case 2:
		if(Zeny < 5000) {
			mes "Dipendente dell'hotel";
			mes "Mi dispiace, ma";
			mes "hai bisogno di 5.000 zeny";
			mes "per effettuare il check-in.";
			close;
		}
		set Zeny,Zeny - 5000;
		mes "Dipendente dell'hotel";
		mes "Grazie~";
		mes "Spero ti piaccia";
		mes "il tuo soggiorno nel";
		mes "Drago Reale.";
		close2;
		warp "lhz_in02",219,150;
		percentheal 100,100;
		npcskill "AL_BLESSING",10,99,99;
		end;
	case 3:
		mes "Dipendente dell'hotel";
		mes "Grazie e";
		mes "ti auguro una buona giornata.";
		close;
	}
}

//======================= Inn Function ==============================
function	script	F_InnMaid	{
	.@npc_name$ = getarg(0);

	mes .@npc_name$;
	mes "Benvenuto a";
	mes getarg(1) + ".";
	mes "Come posso aiutarla?";
	next;
	switch(select("Salva:Riposare -> 5000 zeny:Annulla")) {
	case 1:
		mes .@npc_name$;
		mes "Il tuo punto di riapparizione";
		mes "è stato salvato.";
		mes "Grazie,";
		mes "per favore torna.";
		savepoint getarg(2),getarg(3),getarg(4),1,1;
		close;
	case 2:
		mes .@npc_name$;
		if(Zeny < 5000){
			mes "Mi dispiace, ma la tassa di servizio è di 5.000 zeni. Per favore, assicurati di avere abbastanza soldi per il check-in la prossima volta, ok?";
			close;
		}
		mes "Grazie.";
		mes "Spero che tu";
		mes "goditi il tuo riposo~";
		close2;
		set Zeny,Zeny - 5000;
		percentheal 100,100;
		return;
	case 3:
		close;
	}
}
