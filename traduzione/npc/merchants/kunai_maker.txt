que_ng,72,29,3	script	Kunai Merchant Kashin	83,{
	if(BaseClass == Job_Ninja) {
		mes "[Kashin]";
		mes "Sono Kashin, distributore";
		mes "di Kunai per Ninja. Prendi";
		mes "un'occhiata e lasciami";
		mes "sai se sei interessato";
		mes "in qualsiasi mio articolo.";
		next;
		switch(select("Kunai Veleno di 10 Stelle:10 Kunai a ghiacciolo:Kunai al vento forte 10:10 Kunai di Terra Nera:Kunai Onda di Calore 10:Annulla")) {
		case 1: callfunc "Kunai_Trade",13250,20,7524,1,13259; break;
		case 2: callfunc "Kunai_Trade",13251,8,7522,2,13255; break;
		case 3: callfunc "Kunai_Trade",13252,4,7523,2,13257; break;
		case 4: callfunc "Kunai_Trade",13253,2,7524,1,13256; break;
		case 5: callfunc "Kunai_Trade",13254,1,7521,2,13258; break;
		case 6:
			mes "[Kashin]";
			mes "Bene allora, grazie";
			mes "per visitare il mio negozio.";
			mes "Per favore vieni da me quando";
			mes "devi comprare alcuni";
			mes "Ciao. Ci vediamo per ora~";
			close;
		}
	}
	mes "[Kashin]";
	mes "Sono Kashin, distributore";
	mes "di Kunai per Ninja. Se tu";
	mes "hai amici che sono";
	mes "Ninja, poi potresti";
	mes "voglio parlarvi di me.";
	close;
}

function	script	Kunai_Trade	{
	setarray .@item_req[0], getarg(0), getarg(2);
	setarray .@req_amount[0], getarg(1), getarg(3);
	.@item_id = getarg(4);

	mes "[Kashin]";
	mes "Puoi scambiare";
	mes .@req_amount[0] +" "+ getitemname(.@item_req[0]) +" and";
	mes .@req_amount[1] +" "+ getitemname(.@item_req[1]) +" for every";
	mes "set di 10 "+ getitemname(.@item_id) +".";
	next;
	mes "[Kashin]";
	mes "Posso darti un massimo di 500 set di Kunai alla volta.";
	mes "Se vuoi annullare l'operazione, inserisci '0.'. Quanti";
	mes "Set di kunai che vorresti?";
	next;
	input .@amount;
	if(.@amount < 1 || .@amount > 500) {
		mes "[Kashin]";
		mes "Eh? Mi dispiace, ma";
		mes "Non posso darti quello";
		mes "molti set di kunai. Per favore";
		mes "inserisci un valore inferiore a 500.";
		close;
	}
	if (countitem(.@item_req[0]) < .@amount*.@req_amount[0] || countitem(.@item_req[1]) < .@amount*.@req_amount[1]) {
		mes "[Kashin]";
		mes "Mmh, tu non hai";
		mes "abbastanza articoli per questo";
		mes "Scambio di kunai. Per favore";
		mes "ricontrolla i tuoi oggetti";
		close;
 	}
	if (checkweight(.@item_id, .@amount*10) == 0) {
		mes "[Kashin]";
		mes "Mmm, sembra che il tuo";
		mes "Inventario non ha";
		mes "abbastanza spazio per memorizzare";
		mes "altri elementi. Meglio";
		mes "libera un po' di spazio prima.";
		close;
	}
	mes "[Kashin]";
	mes "Ottimo, è tutto";
	mes "ordine, quindi procediamo";
	mes "e completa questa transazione.";
	mes "Sono sicuro che sarai";
	mes "piuttosto soddisfatto di";
	mes "questi kunai.";
	delitem .@item_req[0],.@req_amount[0]*.@amount;
	delitem .@item_req[1],.@req_amount[1]*.@amount;
	getitem .@item_id,10*.@amount;
	close;
}
