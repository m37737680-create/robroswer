prontera,123,102,5	script	Merchant#pron	85,{
	if (MaxWeight - Weight < 6301 || checkweight(1201,1) == 0) {
		mes "[Mercante]";
		mes "Ahah!";
		mes "Sei un superuomo?";
		mes "Porti così tante cose!";
		mes "Meglio che tu metta un po' di quello";
		mes "in Archiviazione Kafra~";
		close;
	}
	mes "[Mercante]";
	mes "Buongiorno!";
	mes "Hai raccolto qualche";
	mes "articoli come Conchiglie o Piuma?";
	mes "E i gelatini? Oh sì,";
	mes "Ne ho bisogno per qualcosa.";
	next;
	mes "[Mercante]";
	mes "Certo, non te lo chiederò";
	mes "darmi quella roba gratis. E se scambio una Pozione Rossa con 5 Conchiglie, 10 Effiluvio,";
	mes "o 10 caramelle?";
	next;
	switch(select("Certo!:Penso che sia una copia!:No, grazie.")) {
	case 1:
		mes "[Mercante]";
		mes "Quindi quale articolo devo";
		mes "vuoi portarmi?";
		mes "Conchiglie, Piumoni o Gelatine?";
		next;
		switch(select("Gusci:Gelatinei:Piuma:Annulla")) {
		case 1: callfunc "F_PotExchange",935,5;
		case 2: callfunc "F_PotExchange",909,10;
		case 3: callfunc "F_PotExchange",914,10;
		case 4:
			mes "[Mercante]";
			mes "Certo~";
			mes "Nessun problema.";
			close;
		}
	case 2:
		mes "[Mercante]";
		mes "Una fregatura...?";
		mes "Se controlli il valore di mercato";
		mes "degli articoli scambiati, sono in realtà io a ricevere";
		mes "svenduto qui.";
		close;
	case 3:
		mes "[Mercante]";
		mes "Va bene,";
		mes "nessun problema.";
		mes "Ma torna da me";
		mes "se cambi idea.";
		close;
	}
}

function	script	F_PotExchange	{
	.@item_req = getarg(0);
	.@req_amount = getarg(1);

	if (countitem(.@item_req) < .@req_amount) {
		mes "[Mercante]";
		mes "Ehi, dove sono";
		mes "tutti quelli "+ getitemname(.@item_req) +"";
		mes "quello che avevi promesso?";
		mes "Dammi "+ getitemname(.@item_req) +"!";
		close;
	}
	else {
		mes "[Mercante]";
		mes "Ok, lascio controllare";
		mes "quanti "+ getitemname(.@item_req) +" tu";
		mes "su di te. Mmh...";
		next;
		mes "[Mercante]";
		mes "Hai";
		mes "un totale di "+ countitem(.@item_req) +" "+ getitemname(.@item_req) +".";
		mes "Posso darti un totale";
		mes "di "+ (countitem(.@item_req)/.@req_amount) +" Pozioni rosse per quelli.";
		next;
		mes "[Mercante]";
		mes "Cosa dici?";
		mes "Abbiamo un accordo?";
		next;
		if (select("Affare.:Nessun accordo.") == 1) {
			mes "[Mercante]";
			mes "Sai l'esatto";
			mes "numero di Pozioni Rosse";
			mes "vuoi ricevere per";
			mes "quelli "+ getitemname(.@item_req) +"non è vero?";
			next;
			mes "[Mercante]";
			mes "Ora, posso scambiare con te";
			mes "almeno 1 Pozione Rossa";
			mes "e un massimo di 100 Pozioni Rosse";
			mes "un tempo. Se cambi il tuo";
			mes "Mente, inserisci solo '0' per annullare.";
			next;
			while(1) {
				input .@input,0,101;
				if (.@input == 0) {
					mes "[Mercante]";
					mes "Cos'è...?";
					mes "Per l'amor del cielo perché tu";
					mes "nemmeno il pensiero di parlarmi?";
					mes "Molto indeciso, non è vero?";
					close;
				}
				else if (.@input > 100) {
					mes "[Mercante]";
					mes "Non posso darti altro";
					mes "più 100 Pozioni Rosse";
					mes "immediatamente. Proviamo";
					mes "di nuovo.";
					next;
				}
				else {
					break;
				}
			}
			.@put_out_item = .@input * .@req_amount;
			if (countitem(.@item_req) < .@put_out_item) {
				mes "[Mercante]";
				mes "Oh no, il numero che hai inserito non sembra corretto. Ti conviene ricontrollare quanti Pozioni Rosse puoi scambiare di nuovo.";
				close;
			}
			mes "[Mercante]";
			mes "Eccolo/a!";
			mes "Quante Pozioni Rosse";
			mes "Te l'ho dato, dovrebbe essere buono. Grazie, è stato un buon affare~";
			delitem .@item_req,.@put_out_item; //Shell
			getitem 501,.@input; //Red_Potion
			close;
		}
		mes "[Mercante]";
		mes "Eh~";
		mes "Va bene.";
		mes "Anche se non sono";
		mes "Pozioni rosse più utili";
		mes "a un avventuriero come te?";
		close;
	}
}
