alberta_in,16,28,4	script	Pharmacist	61,{
	if (checkweight(1201,1) == 0) {
		mes "Aspetta un momento!";
		mes "Attualmente stai portando";
		mes "troppi oggetti con te.";
		mes "Per favore torna più tardi";
		mes "dopo che hai messo alcuni oggetti nello stoccaggio di kafra.";
		close;
	}
	mes "Vecchio farmacista";
	mes "Ummm...";
	mes "Cosa ti porta qui...?";
	next;
	switch(select("Prepara Pozione:Parla.:Mescolare informazioni:Annulla")) {
	case 1:
		if (MaxWeight - Weight < 5000) {
			mes "Vecchio farmacista";
			mes "Perché porti così tanti!";
			mes "Non essere avaro, porta solo ciò di cui hai bisogno!";
			next;
			mes "Vecchio farmacista";
			mes "Sei troppo pesante per ricevere pozioni da me...";
			mes "Vai a riporre degli oggetti nel tuo magazzino prima!";
			close;
		} else {
			mes "Vecchio farmacista";
			mes "Hai tutto pronto, vero? Quale vorresti?";
			next;
			switch(select("Pozione Rossa.:Pozione Arancione.:Pozione Gialla.:Pozione Bianca.:Pozione Blu.:Pozione Verde.:In realtà, non voglio niente.")) {
			case 1:
				callsub L_Making,507,3,501;
			case 2:
				mes "Vecchio farmacista";
				if (countitem(507) < countitem(713)) {
					set .@max,countitem(507);
				}
				else if (countitem(508) < countitem(713)) {
					set .@max,countitem(508);
				}
				else {
					set .@max,countitem(713);
				}
				if ((countitem(507) < 1) || (countitem(508) < 1) || (countitem(713) == 0)) {
					mes "Sciocco! Cosa ti aspettavi?! Venire qui con niente. Tsk!";
					mes "Vattene!";
					close;
				}
				if (Zeny < 3) {
					mes "Sciocco! Cosa ti aspettavi?! Venire qui con niente. Tsk!";
					mes "Vattene!";
					close;
				}
				mes "Quanti?";
				next;
				switch(select("Quanti ne posso fare:Voglio scegliere un importo.:In realtà, non voglio niente.")) {
				case 1:
					if ((countitem(507) < .@max) || (countitem(508) < .@max) || (countitem(713) < .@max) || (Zeny < .@max*3)) {
						mes "Vecchio farmacista";
						mes "Sciocco! Non hai nemmeno tutti i materiali e vuoi che ti faccia delle pozioni?!";
						close;
					}
					set Zeny,Zeny-(.@max*5);
					delitem 507,.@max; //Red_Herb
					delitem 508,.@max; //Yellow_Herb
					delitem 713,.@max; //Empty_Bottle
					getitem 502,.@max; //Orange_Potion
					break;
				case 2:
					mes "Vecchio farmacista";
					mes "Poi scegli un numero inferiore a 100. Se non ne vuoi nessuno, inserisci semplicemente '0'. Con i materiali che hai, puoi fare circa "+.@max+" pozioni.";
					next;
					input .@amount,0,101;
					if (.@amount == 0) {
						mes "Vecchio farmacista";
						mes "Decidi già!";
						close;
					}
					if (.@amount > 100) {
						mes "Vecchio farmacista";
						mes "Sei sordo? Ho detto meno di 100!";
						close;
					}
					if ((countitem(507) < .@amount) || (countitem(508) < .@amount) || (countitem(713) < .@amount) || (Zeny < .@amount*3)) {
						mes "Vecchio farmacista";
						mes "Sciocco! Non hai nemmeno tutti i materiali e vuoi che ti faccia delle pozioni?!";
						close;
					}
					next;
					set Zeny,Zeny-(.@amount*5);
					delitem 507,.@amount; //Red_Herb
					delitem 508,.@amount; //Yellow_Herb
					delitem 713,.@amount; //Empty_Bottle
					getitem 502,.@amount; //Orange_Potion
					break;
				case 3:
					mes "Vecchio farmacista";
					mes "Cosa?!";
					mes "Grrr...";
					mes "Boh!";
					mes "Vattene!";
					close;
				}
				mes "Vecchio farmacista";
				mes "Eccolo qui. È tutto fatto così puoi prenderlo. Ma ricorda! Abusare di medicina non è buono.";
				close;
			case 3:
				callsub L_Making,508,10,503;
			case 4:
				callsub L_Making,509,20,504;
			case 5:
				callsub L_Making,510,30,505;
			case 6:
				callsub L_Making,511,3,506;
			case 7:
				mes "Vecchio farmacista";
				mes "Cosa?!";
				mes "Grrr...";
				mes "Boh!";
				mes "Vattene!";
				close;
			}
		}
	case 2:
		mes "Vecchio farmacista";
		mes "Con i medicinali, si può aumentare la capacità di rigenerazione di una persona. Ma sono efficaci solo fino a un certo punto. *Sospiro* Sto iniziando a pensare ai giorni in cui ero giovane. Devo stare invecchiando.";
		next;
		mes "Vecchio farmacista";
		mes "Comunque, una pozione è semplicemente una pozione. Niente di più e niente di meno.";
		close;
	case 3:
		mes "Vecchio farmacista";
		mes "Hrrrmm...";
		mes "Voi giovani potete essere piuttosto fastidiosi. Ma, dato che l'avete chiesto, vi spiego.";
		next;
		mes "Vecchio farmacista";
		mes "Le erbe funzionano bene da sole, ma se usi le mie tecniche e abilità speciali per creare pozioni con esse, l'effetto è molto molto maggiore.";
		next;
		mes "Vecchio farmacista";
		mes "Se lo chiedi con entusiasmo e cortesia, te li farò. Ma non gratis... Non preoccuparti però, chiedo solo una piccola tariffa, quindi non è così costoso.";
		next;
		mes "Vecchio farmacista";
		mes "Pozione Rossa - ^0098E52 Erbe Rosse, 1 Bottiglia Vuota, 2 zeni tassa.^000000";
		mes "Pozione Arancione - ^0098E51 Erba Rossa, 1 Erba Gialla, 1 Bottiglia Vuota, 5 zeni tassa.^000000";
		mes "Pozione Gialla - ^0098E52 Erbe Gialle, 1 Bottiglia Vuota, 10 zeni.^000000";
		next;
		mes "Vecchio farmacista";
		mes "Pozione Bianca - ^0098E52 Erbe Bianche, 1 Bottiglia Vuota, 20 zeny tassa.^000000";
		mes "Pozione Blu - ^0098E52 Erbe Blu, 1 Bottiglia Vuota, tassa di 30 zeni.^000000";
		mes "Pozione Verde - ^0098E52 Erbe Verdi, 1 Bottiglia Vuota, 3 zeni tassa.^000000";
		close;
	case 4:
		mes "Vecchio farmacista";
		mes "Che persona noiosa. Se hai qualcosa da dire, perché non lo dici?!";
		close;
	}
	
L_Making:
	.@item_req = getarg(0);
	.@req_amount = getarg(1);

	mes "Vecchio farmacista";
	if ((countitem(.@item_req)/2) < countitem(713)) {
		.@max = countitem(.@item_req) /2;
	} else {
		.@max = countitem(713);
	}
	if (countitem(.@item_req) < 2 || countitem(713) == 0) {
		mes "Sciocco! Cosa ti aspettavi?! Venire qui con niente. Tsk!";
		mes "Vattene!";
		close;
	}
	if (Zeny < 3) {
		mes "Sciocco! Cosa ti aspettavi?! Venire qui con niente. Tsk!";
		mes "Vattene!";
		close;
	}
	mes "Quanti?";
	next;
	switch(select("Quanti ne posso fare:Voglio scegliere un importo.:In realtà, non voglio niente.")) {
	case 1:
		if ((countitem(.@item_req) < .@max*2) || countitem(713) < .@max || (Zeny < .@max*.@req_amount)) {
			mes "Vecchio farmacista";
			mes "Sciocco! Non hai nemmeno tutti i materiali e vuoi che ti faccia delle pozioni?!";
			close;
		}
		Zeny = Zeny - (.@max*.@req_amount);
		delitem .@item_req,.@max*2;
		delitem 713,.@max; //Empty_Bottle
		getitem getarg(2),.@max; 
		break;
	case 2:
		mes "Vecchio farmacista";
		mes "Poi scegli un numero inferiore a 100. Se non ne vuoi nessuno, inserisci semplicemente '0'. Con i materiali che hai, puoi fare circa "+.@max+" pozioni.";
		next;
		input .@amount;
		if (.@amount == 0) {
			mes "Vecchio farmacista";
			mes "Decidi già!";
			close;
		}
		if (.@amount > 100) {
			mes "Vecchio farmacista";
			mes "Sei sordo? Ho detto meno di 100!";
			close;
		}
		if ((countitem(.@item_req) < .@amount*2) || countitem(713) < .@amount || (Zeny < .@amount*.@req_amount)) {
			mes "Vecchio farmacista";
			mes "Sciocco! Non hai nemmeno tutti i materiali e vuoi che ti faccia delle pozioni?!";
			close;
		}
		Zeny = Zeny - (.@max*.@req_amount);
		delitem .@item_req,.@amount*2;
		delitem 713,.@amount; //Empty_Bottle
		getitem getarg(2),.@amount; 
		break;
	case 3:
		mes "Vecchio farmacista";
		mes "Cosa?!";
		mes "Grrr...";
		mes "Boh!";
		mes "Vattene!";
		close;
	}
	mes "Vecchio farmacista";
	mes "Eccolo qui. È tutto fatto così puoi prenderlo. Ma ricorda! Abusare di medicina non è buono.";
	close;
}
