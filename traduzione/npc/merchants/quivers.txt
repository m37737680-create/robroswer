payon_in01,5,134,5	script	Inventor Jaax	89,{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2000) {
		mes "[Inventore Jaax]";
		mes "Ehi, stai portando";
		mes "troppo roba. Perché non lo nascondi in Kafra Storage? Possiamo parlare dopo che l'hai fatto, giusto?";
		close;
	}
	mes "[Inventore Jaax]";
	mes "Mi chiamo Jaax.";
	mes "Senza ego, posso";
	mes "che sono forse il/la";
	mes "^663300greatest inventor of our time^000000.";
	next;
	mes "[Inventore Jaax]";
	mes "Questa volta, ho";
	mes "ho creato qualcosa";
	mes "veramente straordinario. Li chiamo... ^663300Frecoviali Magici^000000 !! Questo sarà ricordato nella storia come un";
	mes "rivoluzione freccia!";
	emotion ET_BEST;
	next;
	mes "[Inventore Jaax]";
	mes "Ho studiato magia e quiver per anni, lavorando giorno e notte finché non ho finalmente capito come condensare le frecce con la magia! Con i quiver magici, porterai più frecce, ma con meno peso!";
	next;
	mes "[Inventore Jaax]";
	mes "Ti piacerebbe provare uno dei miei sacche di frecce? Non ho dubbi che qualcuno come te possa apprezzare il mio genio!";
	next;
	switch(select("Fascio:Fascia di Frecce d'Acciaio:Fasce di frecce d'acciaio:Fanteria Oridecon Freccia-Custodia:Fanteria di frecce Fire Arrow:Fantereria Freccia d'Argento:Fasce di frecce a punta di vento:Fasce di frecce di pietra:Fasceviere di Freccia Cristallo:Fasce di Frecce Ombra:Fantereria di Frecce Immateriale:Fantereria di frecce arrugginite")) {
	case 1: callsub S_BuyQuiver,1750,500,500,12004;
	case 2: callsub S_BuyQuiver,1770,500,500,12005;
	case 3: callsub S_BuyQuiver,1753,500,500,12006;
	case 4: callsub S_BuyQuiver,1765,500,500,12007;
	case 5: callsub S_BuyQuiver,1752,500,500,12008;
	case 6: callsub S_BuyQuiver,1751,500,500,12009;
	case 7: callsub S_BuyQuiver,1755,500,500,12010;
	case 8: callsub S_BuyQuiver,1756,500,500,12011;
	case 9: callsub S_BuyQuiver,1754,500,500,12012;
	case 10: callsub S_BuyQuiver,1767,500,500,12013;
	case 11: callsub S_BuyQuiver,1757,500,500,12014;
	case 12: callsub S_BuyQuiver,1762,500,500,12015;
	}
	//This doesn't seem to be accessable...
	//mes "[Inventor Jaax]";
	//mes "Is there";
	//mes "anything you want?";
	//close;

S_BuyQuiver:
	.@item_id = getarg(0);
	.@amount = getarg(1);
	.@zeny_req = getarg(2);

	if (countitem(.@item_id) >= .@amount) {
		mes "[Inventore Jaax]";
		mes "Eccellente!";
		mes "Porti con te delle Frecce? Ti fornirò un sacchetto che può contenere "+ .@amount +" di tuo "+ getitemname(.@item_id) +"per solo ^FF3131"+ .@zeny_req +" Zeny^000000.";
		next;
		switch(select("Conservare tante Frecce negli archi quanto possibile:Acquista 1 quaderno:Annulla")) {
		case 1:
			.@arrows = countitem(.@item_id);
			set .@quiver,.@arrows / .@amount;
			set .@arrows_used,.@quiver * .@amount;
			.@arrow_zeny01 = .@quiver * .@zeny_req;
			mes "Numero di";
			mes "Frecce: ^3131FF"+.@arrows+" ^000000";
			mes "Numero Massimo";
			mes "di Acquisto";
			mes "Quiver: ^3131FF"+.@quiver+" ^000000";
			mes "Zeny richiesto: ^3131FF"+.@arrow_zeny01+" Zeny^000000";
			next;
			mes "[Inventore Jaax]";
			mes "Vorresti";
			mes "comprare tanti Fionardi";
			mes "come puoi per le Frecce";
			mes "stai portando attualmente?";
			next;
			if (select("Sì:Annulla") == 1) {
				if (.@arrow_zeny01 < Zeny) {
					mes "[Inventore Jaax]";
					mes "Eccolo/a!";
					mes "Ricorda solo che ^FF0000non potrai usare il Quiver quando il tuo peso trasportato è il 90% del tuo limite massimo di peso^000000.";
					set Zeny, Zeny-.@arrow_zeny01;
					delitem .@item_id,.@arrows_used;// Arrow
					getitem getarg(3),.@quiver; //Quiver
					next;
					mes "[Inventore Jaax]";
					mes "Quindi...";
					mes "Solo tieni traccia";
					mes "di quanto sei";
					mes "portare dal tempo";
					mes "a tempo e tu dovresti";
					mes "stai bene.";
					close;
				}
				else {
					mes "[Inventore Jaax]";
					mes "Mi dispiace, ma non hai abbastanza Zeny. Non posso semplicemente regalare queste cose dopo anni di lavoro";
					mes "sviluppare questa tecnologia rivoluzionaria!";
					close;
				}
			}
			mes "[Inventore Jaax]";
			mes "Hai cambiato idea?";
			mes "Quando la gloria di possedere";
			mes "un quaderno è così vicino?";
			close;
		case 2:
			if (Zeny > .@zeny_req) {
				mes "[Inventore Jaax]";
				mes "Eccolo/a!";
				mes "Ricorda solo che ^FF0000non potrai usare il Quiver quando il tuo peso trasportato è il 90% del tuo limite massimo di peso^000000.";
				Zeny = Zeny -.@zeny_req;
				delitem .@item_id,.@amount;// Arrow
				getitem getarg(3),1; //Quiver
				next;
				mes "[Inventore Jaax]";
				mes "Quindi...";
				mes "Solo tieni traccia";
				mes "di quanto sei";
				mes "portare dal tempo";
				mes "a tempo e tu dovresti";
				mes "stai bene.";
				close;
			}
			else {
				mes "[Inventore Jaax]";
				mes "Non nemmeno";
				mes "avere "+ .@zeny_req +" Zeny?";
				mes "Mi dispiace tanto. Non avevo";
				mes "l'idea che tu fossi così...";
				mes "^333333Destitute^000000.";
				close;
			}
		case 3:
			mes "[Inventore Jaax]";
			mes "Cos'è...?";
			mes "Non vedi che questo invento può cambiare per sempre il modo in cui vengono trasportate le Frecce?! Il futuro è adesso!";
			close;
		}
	}
	else {
		mes "[Inventore Jaax]";
		mes "Puoi portare un massimo di "+ .@amount +" Frecce all'interno di questo quiver. È stato fatto usando il mio metodo segreto,";
		mes "quindi il peso totale delle Frecce e del Porta-frecce è inferiore a quello di portare solo le Frecce.";
		next;
		mes "[Inventore Jaax]";
		mes "È un miracolo della scienza! Uno che puoi sperimentare per te stesso se mi porti almeno "+ .@amount +" Frecce e "+ .@zeny_req +" Zenny per ogni Fialora.";
		close;
	}
}
