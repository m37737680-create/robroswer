//============================================================
//= Main Refiner Function
//============================================================
//= To allow auto safe refining/multiple refining set the
//= second argument to '1' in the function call.
//= If you enable this function, be sure to edit the value of
//= .@safe to the max safe refine in refine.yml as well.
//============================================================
function	script	refinemain	{
	disable_items;
	.@npc_name$ = getarg(0);
	.@features = getarg(1);
	mes "["+ .@npc_name$ +"]";
	mes "Sono l'Armaiolo.";
	mes "Posso raffinare ogni tipo di armi, armature e equipaggiamento, quindi lasciatemi";
	mes "sai cosa voglio che tu migliori.";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(.@i = 1; .@i<getarraysize(.@indices); ++.@i) {
		if(getequipisequiped(.@indices[.@i])) {
			.@menu$ = .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			.@equipped = 1;
		}
		.@menu$ = .@menu$ + ":";
	}
	if (.@equipped == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "Non credo di poter raffinare nessun articolo che hai...";
		close;
	}
	.@part = .@indices[select(.@menu$)];

	if(!getequipisequiped(.@part)) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "Non indossi";
		mes "qualcosa lì che";
		mes "Posso affinare.";
		emotion ET_FRET;
		close;
	}
	//Check if the item is refinable...
	if(!getequipisenableref(.@part)) {
		mes "["+ .@npc_name$ +"]";
		mes "Non credo di potercela fare";
		mes "affinare questo elemento in qualsiasi modo...";
		close;
	}
	//Check to see if the items is already +10
	if(getequiprefinerycnt(.@part) >= 10) {
		mes "["+ .@npc_name$ +"]";
		mes "Non posso affinare questo";
		mes "ancora. Questo è come";
		mes "sempre più raffinato!";
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );

	if( .@itemtype == IT_ARMOR ){
		.@equip_lv = getequiparmorlv( .@part );

		switch( .@equip_lv ){
			case 1:
				.@safe = 4;
				break;
			default:
				// TODO:
				close;
		}

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if( VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE) ){
			switch( .@equip_lv ){
				case 1:
					.@price = .@price * 10;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else if( .@itemtype == IT_WEAPON ){
		.@equip_lv = getequipweaponlv( .@part );

		switch( .@equip_lv ){
			case 1:
				.@safe = 7;
				break;
			case 2:
				.@safe = 6;
				break;
			case 3:
				.@safe = 5;
				break;
			case 4:
				.@safe = 4;
				break;
			default:
				// TODO:
				close;
		}

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if( VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE) ){
			switch( .@equip_lv ){
				case 1:
					.@price = .@price * 40;
					break;
				case 2:
					.@price = .@price * 50;
					break;
				case 3:
					.@price = .@price * 2;
					break;
				case 4:
					.@price = .@price * 2;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else{
		.@safe = 4;
	}

	// TODO: What was this? Maybe shadow?
	// case 5: .@price = .@price * 10; break;

	if(.@features != 1) {
		mes "["+ .@npc_name$ +"]";
		mes "Per affinare questo ho bisogno";
		mes "uno ^003366"+getitemname(.@material)+"^000000 e";
		mes "una commissione di " + .@price + " Zeny";
		mes "Vuoi davvero continuare?";
		next;
		if(select("Sì:No") == 2){
			mes "["+ .@npc_name$ +"]";
			mes "Sì...";
			mes "Non c'è bisogno di";
			mes "Corri. Prenditi il tuo tempo.";
			close;
		}
		if(getequippercentrefinery(.@part) < 100) {
			mes "["+ .@npc_name$ +"]";
			mes "Oh no! Se continuo a";
			mes "risolverlo, c'è il rischio che possa";
			switch(.@material) {
			case 985:
				mes "be destroyed! That means that ^FF0000this equipment^000000, and ^FF0000any cards^000000 or special properties added to this armor, ^FF0000will be gone^000000.";
				break;
			default:
				mes "be destroyed, and you'd ^FF0000lose the weapon^000000, any ^FF0000cards in the weapon^000000,";
				mes "o qualsiasi proprietà speciale aggiunta.";
				break;
			}
			next;
			mes "["+getarg(0)+"]";
			mes "Non posso renderlo più chiaro.";
			mes "Una volta che un'arma è distrutta,";
			mes "non c'è modo di recuperarlo.";
			mes "Hai davvero la possibilità di";
			mes "^FF0000lose this weapon^000000 forever.";
			mes "Vuoi ancora affinare?";
			next;
			if(select("Sì:No") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "Sono completamente d'accordo...";
				mes "Potrei essere un ottimo rifinatore, ma a volte anche io faccio errori.";
				close;
			}
		}
		if((countitem(.@material) < 1) || (Zeny < .@price)) {
			mes "["+ .@npc_name$ +"]";
			mes "Non sembri avere";
			mes "abbastanza Zeny o "+getitemname(.@material)+"...";
			mes "Vai a prendere altri. Io sarò";
			mes "qui tutto il giorno se hai bisogno di me.";
			close;
		}
		Zeny = Zeny-.@price;
		delitem .@material,1;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
			mes "["+ .@npc_name$ +"]";
			emotion ET_FRET;
			mes "Aspetta un secondo...";
			mes "Pensi che sono stupido?!";
			mes "Hai cambiato l'oggetto mentre non guardavo! Via da qui!";
			close;
		}

		if(getequippercentrefinery(.@part) <= rand(100)) {
			failedrefitem .@part;
			mes "["+ .@npc_name$ +"]";
			emotion (!rand(5))?ET_MONEY:ET_HUK;
			.@lose = rand(1,3);
			if (.@lose == 1) {
				mes "MIO DIO!";
				mes "Accidenti! Non di nuovo!";
				mes "Mi dispiace moltissimo, ma sai che la pratica rende perfetti.";
				mes "Um, giusto? Eh eh...";
			} else if(.@lose == 2) {
				mes "Nooooooo!";
				mes "Si è rotto!";
				mes "Mi dispiace!";
			} else {
				mes "Cazzo!";
				mes "Non poteva sopportare";
				mes "molto più tempra!";
				mes "Scusa per questo...";
			}
			close;
		}
		mes "["+getarg(0)+"]";
		successrefitem .@part;
		emotion ET_SMILE;
		.@win = rand(1,3);
		if (.@win == 1) {
			mes "Perfetto!";
			mes "Ehi ehi!";
			mes "Ancora una volta,";
			mes "lavoro impeccabile";
			mes "dalla maestra~";
		} else if(.@win == 2) {
			mes "Successo!";
			mes "Ancora una volta, la mia incredibile";
			mes "il talento acceca veramente";
			mes "e brilla oggi.";
		} else {
			mes "Ehi ehi!";
			mes "Ho finito.";
			mes "Senza dubbio, il mio lavoro è";
			mes "a tuo piacimento.";
			mes "Perfezione pura e assoluta~";
		}
		close;
	}

// New Refining Functions ========================
	if (.@refinerycnt < .@safe) {
		mes "["+ .@npc_name$ +"]";
		mes "Posso affinare questo al limite di sicurezza o a un numero desiderato di volte. È la tua scelta.";
		next;
		.@menu2 = select("Al limite sicuro, per favore.","Deciderò quante volte.","Ho cambiato idea...");
	} else
		.@menu2 = 2;
	switch(.@menu2){
	case 1: 
		.@refinecnt = .@safe - .@refinerycnt;
		break;
	case 2:
		next;
		mes "["+ .@npc_name$ +"]";
		mes "Quante volte vorresti che io perfezionassi il tuo articolo?";
		next;
		input .@refinecnt;
		.@refinecheck = .@refinecnt + .@refinerycnt;
		if (.@refinecnt < 1 || .@refinecheck > 10) {
			mes "["+ .@npc_name$ +"]";
			mes "Non posso raffinare questo articolo così tante volte.";
			close;
		}
		if(.@refinecheck > .@safe) {
			.@refinecheck = .@refinecheck - .@safe;
			mes "["+ .@npc_name$ +"]";
			mes "Questo cercherà di raffinare l'attrezzatura " + .@refinecheck + " oltre il limite di sicurezza. Il tuo equipaggiamento potrebbe essere distrutto... va bene così?";
			next;
			if(select("Sì...","No...") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "Hai detto così... Sia fatto.";
				close;
			}
		}
		break;
	case 3:
		next;
		mes "["+ .@npc_name$ +"]";
		mes "Hai detto così... Sia fatto.";
		close;
	}
	.@fullprice = .@price * .@refinecnt;
	mes "["+ .@npc_name$ +"]";
	mes "Ti costerà " + .@refinecnt + " " + getitemname(.@material) + " e " + .@fullprice + " Va bene Zeny.";
	next;
	if(select("Sì","No...") == 2){
		mes "["+ .@npc_name$ +"]";
		mes "Hai detto così... Sia fatto.";
		close;
	}
	if(countitem(.@material) < .@refinecnt || Zeny < .@fullprice) {
		mes "["+ .@npc_name$ +"]";
		mes "È tutto quello che hai? Purtroppo non posso lavorare per te a un prezzo più basso. Prova a metterti nei miei panni.";
		close;
	}
	Zeny = Zeny - .@fullprice;
	delitem .@material,.@refinecnt;
	while(.@refinecnt){
		if (getequipisequiped(.@part) == 0) {
			mes "["+ .@npc_name$ +"]";
			mes "Guarda qui... non hai nessun articolo su...";
			close;
		}
		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
				callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt) || (.@menu2 == 1 && getequippercentrefinery(.@part) < 100)) {
			mes "["+ .@npc_name$ +"]";
			mes "Clang... No, ma ti sei immaginato che potessi essere così stupido?!";
			mes "L'hai cambiato...";
			mes "Vattene prima che ti stordisca con il mio Martello!!";
			close;
		} 
		mes "Clang, clang!!!";
		if(.@menu2 == 2 && getequippercentrefinery(.@part) <= rand(100)) {
			failedrefitem .@part;
			emotion ET_HUK;
			mes "["+ .@npc_name$ +"]";
			mes "WAHHHH!!! Mi dispiace tanto... Ti avevo avvertito che poteva succedere...";
			.@refinecnt = .@refinecnt - 1;
			if(.@refinecnt == 0) close;
			mes "Ecco lo Zeny e i materiali inutilizzati indietro...";
			getitem .@material,.@refinecnt;
			.@fullprice = .@refinecnt * .@price;
			Zeny = Zeny + .@fullprice;
			close;
		}
		successrefitem .@part;
		emotion ET_BEST;
		.@refinecnt = .@refinecnt - 1;
		.@refinerycnt = getequiprefinerycnt(.@part);
		next;
	}
	mes "["+ .@npc_name$ +"]";
	mes "Tutto finito... Ritorna presto.";
	close;
}

	if (checkweight(1201,1) == 0) {
		mes "Aspetta un minuto!!";
		mes "Attualmente stai portando";
		mes "troppi oggetti con te.";
		mes "Per favore riprova";
		mes "dopo che perderai un po' di peso.";
		close;
	}
	.@npc_name$ = getarg(0);
	mes "["+ .@npc_name$ +"]";
	mes "Vendo due tipi di metallo";
	mes "per temprare le armi.";
	mes "Ho ^007777Phracon^000000 per il Livello 1";
	mes "Armi, e ^007777Emveretarcon^000000";
	mes "per Armi di Livello 2.";
	next;
	switch(select("Phracon - 200 Zeny:Emveretarcon - 1000 Zeny:Chiedere di altri metalli")) {
	case 1:
		.@material = 1010;
		.@price = 200;
		break;
	case 2:
		.@material = 1011;
		.@price = 1000;
		break;
	case 3:
		mes "["+ .@npc_name$ +"]";
		mes "Altri metalli?";
		mes "Beh, avrai bisogno di metalli speciali per potenziare armi di livello superiore, o qualsiasi tipo di armatura. Ma sai, Oridecon e Elunio sono davvero";
		mes "difficile trovare...";
		close;
	}
	mes "["+ .@npc_name$ +"]";
	mes "Quanti ne desideri comprare?";
	mes "Se non ne vuoi nessuno, per favore inserisci il numero, '0.'";
	next;
	while(1) {
		input .@input;
		if (.@input == 0) {
			mes "["+ .@npc_name$ +"]";
			mes "L'accordo ha";
			mes "stato cancellato.";
			close;
		}
		else if (.@input < 0 || .@input > 500) {
			mes "["+ .@npc_name$ +"]";
			mes "Va bene, puoi";
			mes "acquistare fino a 500.";
			mes "Non più di quello,";
			mes "capito? Bene.";
			next;
		}
		else {
			break;
		}
	}
	.@sell = .@input * .@price;
	if (Zeny < .@sell) {
		mes "["+ .@npc_name$ +"]";
		mes "Ehm...";
		mes "Non hai";
		mes "Zeny abbastanza per comprare";
		mes ""+ .@input +" di loro.";
		close;
	}
	if (checkweight(.@material,.@input) == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "Mmm...";
		mes "Non posso darti nulla se non hai abbastanza spazio nel tuo inventario. Perché non metti le tue cose extra nello Stoccaggio di Kafra e riprovi?";
		close;
	}
	getitem .@material,.@input;
	Zeny = Zeny-.@sell;
	mes "["+ .@npc_name$ +"]";
	mes "Eccolo/a!";
	mes "Grazie per";
	mes "il vostro patrocinio.";
	close;
}

// Equipment Repair Function
//============================================================
function	script	repairmain	{
	.@repairprice = 5000;
	.@npc_name$ = getarg(0);
	mes "["+ .@npc_name$ +"]";
	mes "Ciao!";
	mes "Vuoi di me";
	mes "riparare qualsiasi articolo?";
	mes "Puoi contare su di me";
	mes "per riparazioni di articoli!";
	next;
	switch(select("In realtà, ho alcuni articoli...:Nessuno al momento.")) {
	case 1:
		.@checkitem = 1;
		while (1) {
			if (getbrokenid(.@checkitem) == 0) {
				break;
			}
			.@checkitem = .@checkitem+1;
		}
		.@checkitem = .@checkitem-1;
		if (!.@checkitem) {
			mes "["+ .@npc_name$ +"]";
			mes "Oh wow, questo è incredibile!";
			mes "Devi prenderti molto cura delle tue cose. Nessuno dei tuoi oggetti è danneggiato!";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "Se tutti sono come te, sarò disoccupato!! Haha~!";
			close;
		}
		mes "["+ .@npc_name$ +"]";
		mes "Mmm...";
		mes "Vediamo...";
		mes "Di tutti i tuoi articoli,";
		mes "" + .@checkitem + " sono danneggiati.";
		mes "Vuoi riparare?";
		next;
		.@totalcost = .@repairprice*.@checkitem;
		mes "["+ .@npc_name$ +"]";
		mes "Ogni riparazione costa " + .@repairprice + " Zeny. Quindi riparare tutti i tuoi oggetti danneggiati costerebbe " + .@totalcost + " Zeny! Vorresti riparare gli oggetti?";
		next;
		switch(select("Sì:No")) {
		case 1:
			if (Zeny < .@totalcost) {
				mes "["+ .@npc_name$ +"]";
				mes "Ehi ehi...";
				mes "Controlla il tuo portafoglio prima di ricevere la fattura delle riparazioni! Non posso riparare nulla perché non hai abbastanza Zeny.";
				close;
			}
			.@checkitem2 = 1;
			while (1) {
				if (getbrokenid(.@checkitem2) == 0) {
					break;
				}
				.@checkitem2 = .@checkitem2+1;
			}
			.@checkitem2 = .@checkitem2-1;
			if (.@checkitem == .@checkitem2) {
				Zeny = Zeny-.@totalcost;
				while (.@checkitem) {
					repair(.@checkitem);
					.@checkitem = .@checkitem-1;
				}
				mes "["+ .@npc_name$ +"]";
				mes "Va bene! Tutto fatto. Ora, prova ad essere un po' più attento. Anche gli oggetti hanno una vita, sai.";
				close;
			}
			else {
				mes "["+ .@npc_name$ +"]";
				mes "Mmh? Qualcosa non va. Aspetta... Equipa gli oggetti di cui hai bisogno per riparare e poi torna da me.";
				close;
			}
		case 2:
			mes "["+ .@npc_name$ +"]";
			mes "Beh, non è un problema per me, ma non è buono lasciare oggetti danneggiati. Dovresti farli riparare il prima possibile!";
			close;
		}
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "Hohoho...";
		mes "Non hai";
		mes "qualche affare con me";
		mes "se non hai nessun/a";
		mes "articoli da riparare.";
		close;
	}
}
