// PecoPeco Breeder (for Knights) :: breeder
//============================================================
prontera,55,350,5	script	Peco Peco Breeder#knt	105,{
	if(Upper==0) set .@price,2500; //Normal Peco - default price
	if(Upper==1) set .@price,2500; //Armored Peco
	if(Upper==2) set .@price,2500; //Baby Peco

	mes "[Allevatore di Peco Peco]";
	if (BaseJob == Job_Knight && Class < Job_Rune_Knight) {
		mes "Benvenuto.";
		mes "Onorevole Cavaliere,";
		mes "ti piacerebbe affittare";
		mes "un Peco Peco? L'affitto";
		mes "tassa è "+.@price+" zeny.";
		next;
		switch(select("Noleggiare Peco Peco:Annulla")) {
		case 1:
			if (Zeny < .@price) {
				mes "[Allevatore di Peco Peco]";
				mes "Non fai";
				mes "a abbastanza zeny.";
				mes "Sei...";
				mes "fallito?";
				close;
			}
			else if(getskilllv("KN_RIDING") == 0) {
				mes "[Allevatore di Peco Peco]";
				mes "Mi dispiace, ma sei";
				mes "non idoneo per questo";
				mes "servizio. Per favore vai a imparare";
				mes "la abilità Peco Peco Ride per prima.";
				close;
			}
			else if(checkriding()) {
				mes "[Allevatore di Peco Peco]";
				mes "Sei già";
				mes "montato su un";
				mes "Peco Peco.";
				close;
			}
			else if(ismounting()) {
				mes "[Allevatore di Peco Peco]";
				mes "Rimuovi il tuo deposito contante.";
				close;
			}
			set Zeny, Zeny-.@price;
			setriding;
			close;
		case 2:
			mes "[Allevatore di Peco Peco]";
			mes "Capisco";
			mes "Bene allora,";
			mes "ti auguro una buona giornata.";
			close;
		}
	} else {
		mes "Mi dispiace, ma questi";
		mes "I pecos sono solo";
		mes "disponibile per i Cavalieri";
		mes "e Cavalieri del Signore.";
		close;
	}
}

// Grand PecoPeco Breeder (for Crusaders)
//============================================================
prontera,232,318,3	script	Peco Peco Breeder#cru	105,{
	if(Upper==0) set .@price,3500; //Normal Peco - default price
	if(Upper==1) set .@price,3500; //Armored Peco
	if(Upper==2) set .@price,3500; //Baby Peco

	mes "[Allevatore di Peco Peco]";
	if (BaseJob == Job_Crusader && Class < Job_Rune_Knight) {
		if(Upper != 1 ) mes "Benvenuto, Crociato.";
		else mes "Benvenuto, Paladino.";
		mes "Abbiamo un speciale";
		mes "Peco Peco preparato";
		mes "per te. Noleggiarne uno";
		mes "costare "+.@price+" zeny.";
		next;
		switch(select("Noleggiare un PecoPeco:Smetti")) {
		case 1:
			if (Zeny < .@price) {
				mes "[Allevatore di Peco Peco]";
				mes "Non fai";
				mes "a abbastanza zeny.";
				mes "Se desideri";
				mes "un Peco Peco per favore";
				mes "portare "+.@price+" zeny...";
				close;
			}
			else if(getskilllv("KN_RIDING") == 0) {
				mes "[Allevatore di Peco Peco]";
				mes "Devi prima imparare";
				mes "andare su un PecoPeco prima";
				mes "Posso noglierne uno a te.";
				close;
			}
			else if(checkriding()) {
				mes "[Allevatore di Peco Peco]";
				mes "Sei già";
				mes "montato su un Peco Peco.";
				close;
			}
			else if(ismounting()) {
				mes "[Allevatore di Peco Peco]";
				mes "Rimuovi il tuo deposito contante.";
				close;
			}
			set Zeny, Zeny-.@price;
			setriding;
			close;
		case 2:
			mes "[Allevatore di Peco Peco]";
			mes "A presto.";
			close;
		}
	} else {
		mes "Cosa posso fare per te?";
		mes "Si prega di notare che";
		mes "questo noleggio Peco Peco";
		mes "servizio è strettamente per";
		mes "Crociati e Paladini.";
		close;
	}
}

// Falcon Master :: breeder

hu_in01,381,304,5	script	Falcon Breeder#hnt	105,{
	if(Upper==0) set .@price,2500; //Normal Falcon - default price
	if(Upper==1) set .@price,2500; //Scarf Falcon
	if(Upper==2) set .@price,2500; //Baby Falcon

	mes "[Allevatore di Falchi]";
	if (BaseJob == Job_Hunter) {
		if (checkwug()) {
			mes "Um...";
			mes "Non puoi affittare un Falco";
			mes "finché non smentisci il tuo warg per primo!";
			close;
		}
		mes "Hai bisogno di un Falco?";
		mes "Puoi affittare il tuo";
		mes "affidabile uccello predatore per un";
		mes "tassa di solo "+.@price+" zeny~";
		next;
		switch(select("Noleggiare Falco:Annulla")) {
		case 1:
			if (checkwug()) {
				mes "[Allevatore di Falchi]";
				mes "Um...";
				mes "Non puoi affittare un Falco";
				mes "finché non smentisci il tuo warg per primo!";
				close;
			}
			if (Zeny < .@price) {
				mes "[Allevatore di Falchi]";
				mes "Cos'è questo?";
				mes "Non hai";
				mes "abbastanza zeny?!";
				mes "Meglio che tu inizi";
				mes "cacciare soldi";
				mes "invece dei mostri~";
				close;
			}
			else if(getskilllv("HT_FALCON") == 0) {
				mes "[Allevatore di Falchi]";
				mes "Oh mio Dio~";
				mes "Vai a imparare come";
				mes "gestire un Falco";
				mes "per prima cosa! Non posso noleggiarne uno";
				mes "a te se non puoi";
				mes "gestiscilo, sai.";
				close;
			}
			else if(checkfalcon()) {
				mes "[Allevatore di Falchi]";
				mes "Um...";
				mes "Hai già";
				mes "un falco. È giusto";
				mes "lì, non lo vedi?";
				close;
			}
			set Zeny, Zeny-.@price;
			setfalcon;
			close;
		case 2:
			mes "[Allevatore di Falchi]";
			mes "Aspetta, dove sono";
			mes "Vai? Questi";
			mes "I falchi sono di prim'ordine,";
			mes "Lo garantisco! Dai, yo~";
			close;
		}
	} else {
		mes "Giovane sciocco!";
		mes "I falchi possono solo";
		mes "essere usato da Cacciatori";
		mes "e i cecchini, capito?";
		mes "Heh heh, geloso?";
		close;
	}
}
