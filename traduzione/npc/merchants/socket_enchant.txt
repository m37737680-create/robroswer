function	script	Func_Socket	{
	.@item_id = getarg(0);
	.@zeny_req = getarg(4);
	setarray .@item_req[0], getarg(5), getarg(7,0);
	setarray .@req_amount[0], getarg(6), getarg(8,0);

	disable_items;
	mes "[Seiyablem]";
	mes "Vuoi aggiungere uno Slot a un "+ getitemname(.@item_id) +"?";
	mes "Allora, per favore portami ^FF0000"+ .@req_amount[0] +" "+ getitemname(.@item_req[0]) + (.@req_amount[0] > 1 && .@item_req[0] != 999 ? "s":"") +"^000000, "+ (.@item_req[1] != 0 && .@req_amount[1] != 0 ? ("^FF0000"+ .@req_amount[1] +" "+ getitemname(.@item_req[1]) + (.@req_amount[1] > 1 && .@item_req[1] != 999 ? "s":"") + "^000000 ") : "") + "e il mio ^FF0000" + (.@zeny_req >= 1000 ? (.@zeny_req /1000 +",000") : .@zeny_req) +",000 zeni^000000 commissione di servizio.";
	mes "Ah, e non dimenticare di portare quello "+ getitemname(.@item_id) +"!";
	next;
	mes "[Seiyablem]";
	if (getiteminfo(.@item_id, ITEMINFO_LOCATIONS) & EQP_HAND_R) // weapon
	{
		mes "Posso provare ad aggiungere uno slot ora se hai gli oggetti e lo zeny richiesti.";
		mes "Tuttavia, devi sapere che c'è la possibilità che io possa fallire.";
		mes "Quindi, devo darti un avvertimento onesto...";
		next;
		mes "[Seiyablem]";
		mes "Se questo tentativo di aggiungere uno Slot alla tua Arma fallisce, allora l'Arma ^FF0000Weapon^000000, ^FF0000e qualsiasi Carta composta ad essa verrà distrutta^000000.";
	}
	else // armor
	{
		mes "Se hai tutti i materiali richiesti, la mia tariffa di servizio zeny e l'Armatura, allora possiamo procedere con il tentativo di Aggiunta Slot.";
		mes "Ma prima di quello, devo avvertirti del rischio.";
		next;
		mes "[Seiyablem]";
		mes "Se questo tentativo di aggiungere uno Slot alla tua Armatura fallisce, allora l'Armatura ^FF0000Armor^000000, ^FF0000i suoi potenziamenti^000000 ^FF0000e qualsiasi Carta composta ad essa verrà distrutta^000000.";
	}
	mes "Vuoi ancora provare ad aggiungere uno Slot?";
	next;
	switch(select("Tentativo di Aggiunta Slot:Annulla"))
	{
		case 1:
			if ((Zeny >= .@zeny_req*1000) && countitem(.@item_req[0]) >= .@req_amount[0] && countitem(getarg(7,512)) >= .@req_amount[1] && countitem(.@item_id) > 0)// if there's no getarg(7) and getarg(8) provided, check itemcount against 0 (we still need a valid item ID though)
			{
				mes "[Seiyablem]";
				mes "Allora, che il lavoro abbia inizio!";
				mes "Meglio che tu preghi per un risultato positivo.";
				next;
				set .@a, rand(1,100);
				if ((.@a > getarg(2)) && (.@a < getarg(3)))
				{
					specialeffect getarg(3) == 51 ? EF_LORD : EF_SANCTUARY; // only S Class items have Lord of Vermillion effect
					mes "[Seiyablem]";
					mes "Ottimo, sembra essere riuscito.";
					mes "Sembra piuttosto ben fatto. Congratulazioni!";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					getitem getarg(1),1;
					next;
					mes "[Seiyablem]";
					mes "A presto, amico!";
					close;
				}
				else
				{
					specialeffect EF_SUI_EXPLOSION;
					mes "[Seiyablem]";
					mes "Wah! ...Mi dispiace tanto, è fallito.";
					mes "Tuttavia, sono completamente innocente.";
					mes "Questa è la tua fortuna, ed è destinata da dio, ok?";
					mes "Non essere così deluso e riprova la prossima volta.";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					next;
					mes "[Seiyablem]";
					mes "Ti auguro buona fortuna la prossima volta!";
					close;
				}
			}
			else
			{
				mes "[Seiyablem]";
				mes "Vorrei procedere con questo tentativo di Aggiunta Slot, ma ti mancano alcune cose.";
				mes "Sei sicuro di avere l'attrezzatura, i materiali necessari e lo zeny?";
				close;
			}
		case 2:
			mes "[Seiyablem]";
			mes "Ho bisogno di tempo per pensarci, eh?";
			mes "Va bene, capisco.";
			mes "Ricorda solo che la vita non è divertente se giochi sempre sul sicuro~";
			close;
	}
}
