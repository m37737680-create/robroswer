prontera,98,121,4	script	Acolyte#Prtclear	90,{
	mes "[Keiki]";
	mes "Ciao avventuriero.";
	mes "Ho studiato la magia da tutto il Rune-Midgards per migliorare quello che credo sia una delle abilità più grandi disponibili alla classe di acolito.";
	next;

	if (is_function("VIP_iRO_Acolyte"))
		callfunc "VIP_iRO_Acolyte","[Keiki]","Juno";

	mes "[Keiki]";
	mes "Sono l'unico e unico Acolita che ha raggiunto la abilità Portale Warp di Livello 10!";
	emotion ET_THINK, playerattached();
	next;
	mes "[Keiki]";
	mes "È vero! E...";
	mes "Ti prometto che non dimentico le località che ho già memorizzato.";
	mes "Un giorno migliorerò le mie abilità per teletrasportarmi ovunque voglia~";
	next;
	mes "[Keiki]";
	mes "Sono disposto a trasportarti nei molti luoghi che ho memorizzato per una piccola tariffa.";
	mes "Vorresti usare questo servizio?";
	next;
	switch(select("Sì:No")) {
	case 1:
		mes "[Keiki]";
		mes "Dove vorresti andare?";
		mes "Ti auguro buona fortuna per il tuo viaggio.";
		next;

		setarray .@towns$[0], "Izlude", "Geffen", "Payon", "Morocc", "Alberta", "Al De Baran", "Comodo", "Umbala", "Juno";
		setarray .@cost[0],        600,     1200,    1200,     1200,      1800,          2200,     2200,     2200,   1800;

		set .@size, getarraysize(.@towns$);
		for(set .@i,0; .@i<.@size; set .@i,.@i+1)
			set .@menu$, .@menu$+.@towns$[.@i]+" -> "+.@cost[.@i]+"z:";
		set .@i, select(.@menu$+"Annulla")-1;
		if (.@i == .@size)
			close;
		if (Zeny < .@cost[.@i]) {
			mes "[Keiki]";
			mes "Mi dispiace, ma non hai";
			mes "abbastanza zeny per il Teletrasporto";
			mes "Servizio. La tariffa per teletrasportare";
			mes "a "+.@towns$[.@i]+" è "+.@cost[.@i]+" zeny.";
			close;
		}
		specialeffect2 EF_READYPORTAL;
		specialeffect2 EF_TELEPORTATION;
		specialeffect2 EF_PORTAL;
		next;
		set Zeny, Zeny - .@cost[.@i];
		switch(.@i) {
			case 0: if (checkre(0)) warp "izlude",128,98; else warp "izlude",91,105; break;
			case 1: warp "geffen",120,39; break;
			case 2: warp "payon",161,58; break;
			case 3: warp "morocc",156,46; break;
			case 4: warp "alberta",117,56; break;
			case 5: warp "aldebaran",168,112; break;
			case 6: warp "comodo",209,143; break;
			case 7: warp "umbala",100,154; break;
			case 8: warp "yuno",158,125; break;
		}
		close;
	case 2:
		close;
	}
}
