prt_are_in,103,11,3	script	Arena Point Manager	732,{
	mes "Gestore Punti Arena";
	mes "Spero che tu stia passando una buona";
	mes "tempo nell'Arena. Se hai";
	mes "guadagnati Punti Turbo Track a";
	mes "A Turbo Track di De Baran, io posso,";
	mes "trasformarli in Punti Arena.";
	next;
	switch(select("Controllo del punto:Convertire Punti:^660000Informazioni di conversione^000000")) {
	case 1:
		mes "Gestore Punti Arena";
		mes "" + strcharinfo(0) + ",";
		mes "hai attualmente";
		mes "" + arena_point + " Punti Arena";
		mes "e " + tt_point + " Punti Turbo Track.";
		close;
	case 2:
		mes "Gestore Punti Arena";
		mes "" + strcharinfo(0) + ",";
		mes "hai attualmente";
		mes "" + arena_point + " Punti Arena";
		mes "e " + tt_point + " Punti Turbo Track.";
		next;
		mes "Gestore Punti Arena";
		mes "Scegliere tra";
		mes "il Punto di Tracciamento al Punto dell'Arena";
		mes "conversioni. Tieni presente che";
		mes "quando si converte più di 10";
		mes "Track Points at one time, you can only convert in ^4D4DFFmultiples of 10^000000.";
		next;
		switch(select("2 TP -> 1 AP:4 TP -> 2 AP:6 TP -> 3 AP:8 TP -> 4 AP:10 TP e più:Annulla")) {
		case 1:
			callsub S_ExchangePoints,28999,2,1;
		case 2:
			callsub S_ExchangePoints,28998,4,2;
		case 3:
			callsub S_ExchangePoints,28997,6,2;
		case 4:
			callsub S_ExchangePoints,28996,8,4;
		case 5:
			mes "Gestore Punti Arena";
			mes "Inserisci il numero";
			mes "quante volte desideri convertire";
			mes "Punti di pista turbo 10 in";
			mes "Punti Arena. Il più grande";
			mes "il valore che puoi inserire è 20.";
			mes "Per annullare, inserire ''^3355FF0^000000.''";
			next;
			input .@input;
			if (.@input <= 0) {
				mes "Gestore Punti Arena";
				mes "Hai";
				mes "cancellato";
				mes "questo servizio.";
				close;
			}
			else if (.@input > 20) {
				mes "Gestore Punti Arena";
				mes "La tua richiesta supera";
				mes "il limite massimo. Per favore";
				mes "inserisci un valore non maggiore di 20.";
				close;
			}
			else {
				set .@want_point1,10 * .@input;
				set .@want_point,5 * .@input;
				set .@my_arena_all,arena_point + .@want_point;
				set .@my_turbo_all,tt_point - .@want_point1;
				if (.@my_arena_all > 28999) {
					mes "Gestore Punti Arena";
					mes "Supererai il";
					mes "massima quantità di";
					mes "Punti Arena se procediamo";
					mes "con questa conversione del tuo";
					mes "Punti Turbo Track. Non puoi";
					mes "a più di 29.000 Punti Arena.";
					next;
					mes "Gestore Punti Arena";
					mes "Per favore spendi un po' di";
					mes "i tuoi Punti Arena prima";
					mes "riutilizzando questo servizio di nuovo.";
					mes "Grazie per la vostra fiducia.";
					close;
				}
				if (.@my_turbo_all < 0) {
					mes "Gestore Punti Arena";
					mes "Mi dispiace, ma";
					mes "non hai abbastanza";
					mes "Punti Turbo Track a";
					mes "esegui questa Arena";
					mes "Conversione di punti.";
					close;
				}
				else {
					mes "Gestore Punti Arena";
					mes "Hai convertito";
					mes "Punti Tracciato Turbo 10";
					mes "in 5 Punti Arena " + .@input + " volte.";
					mes "Un totale di " + .@want_point1 + " I Punti Turbo Track sono stati convertiti in";
					mes "" + .@want_point +" Punti Arena.";
					set tt_point, .@my_turbo_all;
					set arena_point, .@my_arena_all;
					next;
					mes "Gestore Punti Arena";
					mes "" + strcharinfo(0) + ",";
					mes "ora hai";
					mes "^4682B4" + arena_point + "Punti dell'Arena ^000000";
					mes "e ^00688B" + tt_point + "^000000 Punti Tracciato Turbo";
					mes "Grazie per la vostra fiducia.";
					close;
				}
			}
		case 6:
			mes "Gestore Punti Arena";
			mes "Hai";
			mes "cancellato";
			mes "questo servizio.";
			close;
		}
	case 3:
		// Again. no dialog for this option.
		close;
	}
	close;

S_ExchangePoints:
	if (arena_point > getarg(0) ) {
		mes "Gestore Punti Arena";
		mes "Supererai il";
		mes "massima quantità di";
		mes "Punti Arena se procediamo";
		mes "con questa conversione del tuo";
		mes "Punti Turbo Track. Non puoi";
		mes "a più di 29.000 Punti Arena.";
		next;
		mes "Gestore Punti Arena";
		mes "Per favore spendi un po' di";
		mes "i tuoi Punti Arena prima";
		mes "riutilizzando questo servizio di nuovo.";
		mes "Grazie per la vostra fiducia.";
		close;
	}
	.@track_point = getarg(1);
	if (tt_point >= .@track_point) {
		.@arena_point = getarg(2);
		mes "Gestore Punti Arena";
		mes "" + strcharinfo(0) + ",";
		mes "hai convertito "+ .@track_point +" Tracciare";
		mes "Punti in "+ .@arena_point +" Punto Arena.";
		tt_point = tt_point - .@track_point;
		arena_point = arena_point + .@arena_point;
		next;
		mes "Gestore Punti Arena";
		mes "" + strcharinfo(0) + ",";
		mes "ora hai";
		mes "^4682B4" + arena_point + "Punti dell'Arena ^000000";
		mes "e ^00688B" + tt_point + "^000000 Punti Tracciato Turbo";
		mes "Grazie per la vostra fiducia.";
		close;
	}
	else {
		mes "Gestore Punti Arena";
		mes "Mi dispiace, ma non hai";
		mes "abbastanza Punti Turbo Track.";
		mes "Sono necessari almeno 2 Punti Turbo Track per questo servizio di conversione.";
		close;
	}
}
