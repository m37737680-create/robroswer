	set .@i, callfunc("F_GM_NPC",1357,0);
	if (.@i == -1) {
		mes "Gestore Arena";
		mes "Comando annullato.";
		close;
	} else if (.@i == 0) {
		mes "Password errata.";
		close;
	} else {
		mes "Gestore Arena";
		mes "Seleziona un'opzione.";
		next;
		switch(select("Riavviare l'arena:Riordina il Tempo di Classifica")) {
		case 1:
			switch(select("Liv 50:Livello 60:Livello 70:Livello 80:Modalità Festa")) {
			case 1:
				set .@arena,50;
				break;
			case 2:
				set .@arena,60;
				break;
			case 3:
				set .@arena,70;
				break;
			case 4:
				set .@arena,80;
				break;
			case 5:
				mes "Gestore Arena";
				mes "Attenzione";
				mes "Hai scelto di riavviare la fase dell'arena del party.";
				mes "Desideri procedere?";
				next;
				switch(select("Sì:No")) {
				case 1:
					donpcevent "Ponox::OnStart";
					mes "Gestore Arena";
					mes "Il palco dell'arena è stato riattivato con successo.";
					close;

				case 2:
					mes "Gestore Arena";
					mes "Comando annullato.";
					close;
				}
			}
			mes "Gestore Arena";
			mes "Attenzione";
			mes "Hai scelto di riavviare Lv "+.@arena+" palco dell'arena.";
			mes "Desideri procedere?";
			next;
			switch(select("Sì:No")) {
			case 1:
				donpcevent "Lv"+.@arena+" Waiting Room::OnStart";
				mes "Gestore Arena";
				mes "Il palco dell'arena è stato riattivato con successo.";
				close;
			case 2:
				mes "Gestore Arena";
				mes "Comando annullato.";
				close;
			}
		case 2:
			mes "Gestore Arena";
			mes "Attenzione";
			mes "^CE0000Hai scelto di riorganizzare il tempo della classifica. Assicurati che questa non sia una buona decisione, a meno che non sia successo qualcosa di serio!^000000";
			next;
			mes "Gestore Arena";
			mes "Sei sicuro di voler riorganizzare il tempo del ranking?";
			next;
			switch(select("No:Sì")) {
			case 1:
				mes "Gestore Arena";
				mes "Uffa~ :)";
				close;
			case 2:
				switch(select("Liv 50:Livello 60:Livello 70:Livello 80:Modalità Festa")) {
				case 1:
					set .@min$,"$top50min";
					set .@sec$,"$top50sec";

					break;

					set .@min$,"$top70min";

function	script	Func_Are_Rew	{
	.@item_id = getarg(0);
	.@amount = getarg(1);
	.@arena_point = getarg(2);

	mes "[Givu]";
	mes "Vorresti scambiare i tuoi punti dell'arena con "+ getitemname(.@item_id) +"?";
	mes "Puoi scambiare ^3131FF"+ .@arena_point +" punti dell'arena con "+ .@amount +" "+ getitemname(.@item_id) +"^000000.";
	mes "Se desideri annullare, inserisci 0. Se no, inserisci quante volte di punti dell'arena ^3131FFtimes^000000 che desideri spendere.";
	next;
	input .@reward;
	if (.@reward <= 0) {
		mes "[Givu]";
		mes "Hai annullato la tua richiesta.";
		close;
	} else if (.@reward > 1500) {
		mes "[Givu]";
		mes "Hai superato la capacità massima.";
		close;
	}
	if (arena_point < (.@reward * .@arena_point)) {
		mes "[Givu]";
		mes "Non hai abbastanza punti nell'arena.";
		mes "Controlla il totale dei punti dell'arena che hai.";
		close;
	}
	arena_point = arena_point - (.@reward * .@arena_point);
	getitem .@item_id,.@reward * .@amount;
	mes "[Givu]";
	mes "Grazie, per favore torna.";
	close;
}
