function	script	F_AuctionWarper	{
	mes "Guida all'Asta";
	mes "Ciao, vorresti";
	mes "vorrei entrare nel";
	mes "Sala d'asta?";
	next;
	if (select("Sì:No") == 1) {
		.@num = getarg(0);
		if (.@num == 1 || .@num == 4) {
			mes "Guida all'Asta";
			mes "Ottimo! Bene allora,";
			mes "Spero ti divertirai";
			mes "e goditi l'asta~";
		} else {
			mes "Guida all'Asta";
			mes "Buon appalto.";
		}
		close2;
		switch(.@num) {
		case 1:
			warp "auction_01",179,53; end;
		case 2:
			warp "auction_01",21,43; end;
		case 3:
			warp "auction_02",151,23; end;
		case 4:
			warp "auction_02",43,24; end;
		}
	}
	mes "Guida all'Asta";
	mes "Allora,";
	mes "a più tardi.";
	mes "Se cambi il tuo";
	mes "mente, per favore vieni";
	mes "e godetevi le aste~";
	close;
}
