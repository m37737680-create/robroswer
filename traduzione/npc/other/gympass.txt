payon,173,141,4	script	Ripped Cabus#GymPass	899,{
	mes "Strappato";
	mes "Ehi, ciao. La gente non è";
	mes "tanto fisicamente attivi quanto loro";
	mes "era abitudine. Anche se combatti";
	mes "per vivere, il tuo corpo potrebbe";
	mes "debole e molle in alcuni";
	mes "aree. Sai cosa intendo?";
	next;
	mes "Strappato";
	mes "Ehi, allenati con me e posso";
	mes "garantire che sarai in grado";
	mes "sollevare e portare più del tuo";
	mes "cose. Solo dammi il tuo";
	mes "^FF0000Pass palestra^000000 ogni volta,";
	mes "e saremo pronti per partire.";
	next;
	mes "Strappato";
	mes "Ma non ti emozionare troppo:";
	mes "non importa quanto addestramento";
	mes "Ti guido attraverso, puoi";
	mes "Esagerare. Hai mai sentito parlare di";
	mes "qualcuno che è troppo muscoloso?";
	mes "È perché sono morti. Vedi?";
	next;
	mes "Strappato";
	mes "Direi che sarebbe sicuro";
	mes "per te allenare seriamente";
	mes "con me e aumenta il tuo";
	mes "capacità di carico dell'articolo ^FF000010 volte^000000.";
	mes "Allora... Sei pronto a sudare?";
	next;
	switch(select("Sì:No:Ehm, i miei allenamenti hanno perso effetto.")) {
	case 1:
		if (gympassmemory < 10) {
			set .@add_carry,gympassmemory + 1;
			set .@remain_carry,10 - .@add_carry;
			if (countitem(7776) > 0) {
				mes "Strappato";
				mes "Oh, fantastico, ti vedo";
				mes "ho portato il tuo abbonamento alla palestra.";
				mes "Va bene, fai solo quello che faccio";
				mes "e prova a sentire il bruciore.";
				mes "Pronti? Facciamolo.";
				next;
				specialeffect2 EF_EARTHSPIKE;
				next;
				specialeffect2 EF_DEVIL;
				next;
				specialeffect2 EF_COIN;
				next;
				specialeffect2 EF_SIGHTRASHER;
				next;
				mes "Strappato";
				mes "Lì, dovresti essere in grado";
				mes "trasportare più cose con te.";
				mes "Vediamo, possiamo aumentare";
				mes "la capacità di carico del tuo articolo";
				mes "^FF00000" + .@remain_carry + "^000000 volte se continuiamo";
				mes "allenarsi insieme così.";
				delitem 7776,1; //Max_Weight_Up_Scroll
				set gympassmemory,.@add_carry;
				skill "ALL_INCCARRY",.@add_carry,SKILL_PERM_GRANT;
				close;
			}
			else {
				mes "Strappato";
				mes "Ehi, cosa ti ho detto?";
				mes "Devi portarmi il tuo";
				mes "^FF0000Gym Pass^000000 se vuoi";
				mes "elaborare e costruire il tuo";
				mes "muscoli portanti oggetti.";
				close;
			}
		}
		else {
			mes "Strappato";
			mes "Amico, non credo che possiamo";
			mes "costruisci il tuo trasporto di oggetti";
			mes "muscoli più di così.";
			mes "È troppo pericoloso per il tuo";
			mes "corpo se anche provassimo! Forza,";
			mes "Ti ho parlato dei limiti.";
			close;
		}
	case 2:
		mes "Strappato";
		mes "Ah, che peccato.";
		mes "Beh, torna se tu";
		mes "cambia idea. Dimmi";
		mes "i tuoi amici su di me:";
		mes "se sono molli, aiuto";
		mes "metterli in forma.";
		close;
	case 3:
		if (gympassmemory > 0) {
			mes "Strappato";
			mes "Cosa è successo?";
			mes "Lasciare il tuo oggetto portare";
			mes "i muscoli si atrofizzano semplicemente? Fortunato";
			mes "per te, c'è una cosa del genere";
			mes "come memoria muscolare. Non ci vorrà";
			mes "quanto tempo ci vorrà per ricostruirli...";
			next;
			specialeffect2 EF_EARTHSPIKE;
			next;
			specialeffect2 EF_DEVIL;
			next;
			specialeffect2 EF_COIN;
			next;
			specialeffect2 EF_SIGHTRASHER;
			next;
			mes "Strappato";
			mes "Come sta?";
			mes "Il tuo oggetto portando";
			mes "i muscoli sono ricresciuti,";
			mes "proprio così! Cerca di non";
			mes "annullare di nuovo, ok?";
			skill "ALL_INCCARRY",gympassmemory,SKILL_PERM_GRANT;
			close;
		}
		else {
			mes "Strappato";
			mes "Ehm...";
			mes "Non ha funzionato";
			mes "insieme prima.";
			mes "Ne sono sicuro/a.";
			close;
		}
	}
}
