-	script	Mercenary Manager#main	-1,{
	setarray .@name$, 	"Spear", 			"Sword", 			"Bow";
	setarray .@faith$, 	"SPEAR_MERC_GUILD",	"SWORD_MERC_GUILD",	"ARCH_MERC_GUILD";
	setarray .@item, 	12182,				12172,				12162;
	.@npc$ = strnpcinfo(2);
	for (.@size = getarraysize(.@name$); .@type < .@size; .@type++)
		if( compare( .@npc$,.@name$[.@type] ) ) break;
	.@faith_merc = mercenary_get_faith(getd(.@faith$[.@type]));
	mes "[Manager di Mercenari]";
	mes "Benvenuto/a al";
	mes .@name$[.@type] + " Mercenary Guild.";
	mes "Cosa posso fare per te?";
	next;
	switch(select("Assumere mercenario:Info Mercenario:Nulla:Mercenari di decima elementare")) {
		case 1:
			mes "[Manager di Mercenari]";
			mes "Vuoi assumere un/una";
			mes .@name$[.@type] + " Mercenary?";
			mes "Quale grado eri";
			mes "interessato ad assumere?";
			next;
			for (.@i = 1; .@i <= 9; .@i++)
				.@menu$ = .@menu$ + callfunc("F_GetNumSuffix",.@i) + " Grado " + .@name$[.@type] + " Mercenario:";
			.@Grade = select(.@menu$);
			.@BaseLevel = 5 + (.@Grade * 10);
			.@BaseLevel = (.@val > 90)? 90 : .@BaseLevel;
			.@ZenyCost = 7 * .@Grade;
			setarray .@FaithCost[7], 50,100,300;
			.@FaithCost = .@FaithCost[.@Grade];
			mes "[Manager di Mercenari]";
			mes "Quindi vuoi assumere un " + callfunc("F_GetNumSuffix",.@Grade);
			mes "Grado " + .@name$[.@type] + " Mercenario?";
			mes "Devi aver raggiunto";
			mes "Livello Base " + .@BaseLevel + " o superiore, e";
			mes "deve pagare il " + .@ZenyCost + "tassa di 000 zeny.";
			next;
			if(select("Sì:No") == 2) {
				mes "[Manager di Mercenari]";
				mes "Oh, davvero? Ebbene, ora";
				mes "potrebbe non essere un buon momento";
				mes "per te considerare di assumere";
				mes "un mercenario, ma per favore senti";
				mes "libero di tornare se il tuo";
				mes "ha bisogno di cambiare. Grazie~";
			}
			else if(.@FaithCost && .@faith_merc < .@FaithCost) {
				mes "[Manager di Mercenari]";
				mes "Oh... Il tuo punteggio di fedeltà";
				mes "con il " + .@name$[.@type] + " Mercenario";
				mes "La gilda non è abbastanza alta per";
				mes "Assumi questo mercenario. Per favore";
				mes "ritornare dopo aver guadagnato";
				mes "" + .@FaithCost + " o più fedeltà con noi.";
			}
			else if (BaseLevel < .@BaseLevel) {
				mes "[Manager di Mercenari]";
				mes "Mi dispiace, ma il tuo Base";
				mes "Livello non abbastanza alto";
				mes "assumere questo mercenario.";
				mes "Torna da me";
				mes "una volta che raggiungi il Livello Base " + .@BaseLevel + ".";
			}
			else if (Zeny < .@ZenyCost * 1000) {
				mes "[Manager di Mercenari]";
				mes "Mi dispiace, ma tu";
				mes "non ho abbastanza zeny";
				mes "assumere questo mercenario.";
				mes "La commissione di assunzione è " + .@ZenyCost + "zeny.";
			}
			else {
				mes "[Manager di Mercenari]";
				mes "Ottimo! I nostri mercenari";
				mes "sinceri e devoti";
				mes "proteggendo i loro clienti.";
				mes "Mercenari convocati";
				mes "offrire il loro supporto a te per";
				mes "Trenta minuti. Prenditi cura di te ora.";
				Zeny = Zeny - (.@ZenyCost * 1000);
				getitem .@item[.@type] - 10 + .@Grade, 1;
			}
			close;
		case 2:
			mes "[Manager di Mercenari]";
			mes "I mercenari sono soldati";
			mes "che combatterà al tuo fianco";
			mes "sul campo di battaglia, ma lì";
			mes "sono alcuni termini e condizioni";
			mes "devi assumere loro.";
			next;
			mes "[Manager di Mercenari]";
			mes "Devi raggiungere un livello";
			mes "requisito e paga un zeny";
			mes "tassa per assumere un mercenario.";
			mes "Mercenari di grado superiore";
			mes "richiederà anche che tu";
			mes "costruisci un sistema di fidelizzazione con noi.";
			next;
			mes "[Manager di Mercenari]";
			mes "I contratti di mercenari non possono essere";
			mes "trasferito ad altre persone,";
			mes "e permettiamo solo una differenza di 5 Livello Base tra il Mercenario";
			mes "e cliente così non puoi assumere nessuno più forte di te.";
			next;
			mes "[Manager di Mercenari]";
			mes "Beh, puoi capire il";
			mes "dettagli quando tu effettivamente formi";
			mes "un contratto con uno dei nostri";
			mes "Mercenari, e ricevere";
			mes "lo Scrollo di Invocazione che";
			mes "chiamare un mercenario dalla tua parte.";
			next;
			mes "[Manager di Mercenari]";
			mes "Non puoi dare questo rotolo";
			mes "a chiunque altro, e il";
			mes "Mercenario rimarrà solo";
			mes "con te per 30 minuti dopo";
			mes "lo richiami. Non dimenticare";
			mes "riguardo il limite di tempo, ok?";
			close;
		case 3:
			mes "[Manager di Mercenari]";
			mes "No? Non avevi bisogno di nessun";
			mes "aiuto? Beh, sentiti libero di";
			mes "mi chiedi se ne ho";
			mes "domande sui mercenari.";
			close;
		case 4:
			mes "[Manager di Mercenari]";
			mes "Mercenari di decima elementare sono";
			mes "il meglio che possiamo offrire,";
			mes "e usiamo criteri diversi";
			mes "affinché i nostri clienti li assumano.";
			mes "Non c'è commissione di zeny, ma tu";
			mes "deve avere 500 Fedeltà.";
			next;
			mes "[Manager di Mercenari]";
			mes "Una volta che fai un contratto";
			mes "con un mercenario di decima elementare,";
			mes "il tuo punteggio di fedeltà sarà";
			mes "diminuito di 400. In altri";
			mes "parole, paghi 400 Fedeltà";
			mes "assumere un mercenario di decimo grado.";
			next;
			mes "[Manager di Mercenari]";
			mes "Devi anche essere a";
			mes "Livello Base 90 o superiore per";
			mes "assumi un mercenario di decimo grado.";
			mes "Sei ancora interessato a";
			mes "formando questo contratto?";
			next;
			if (select("Sì:No") == 2) {
				mes "[Manager di Mercenari]";
				mes "Capisco... Ci vuole";
				mes "lavoro duro e sacrificio per";
				mes "anche arrivare al punto dove";
				mes "puoi assumere un decimo anno";
				mes "Mercenario. Hai considerato";
				mes "assumere un mercenario di grado inferiore?";
			}
			else if (.@faith_merc < 500) {
				mes "[Manager di Mercenari]";
				mes "Mi dispiace, ma il tuo";
				mes "Il punteggio di fedeltà è troppo";
				mes "basso per assumere un decimo anno";
				mes "Mercenario. Devi aver";
				mes "500 o più Fedeltà a";
				mes "formare un contratto con uno.";
			}
			else if (BaseLevel < 90) {
				mes "[Manager di Mercenari]";
				mes "Mi dispiace, ma devi";
				mes "essere al Livello Base 90 o";
				mes "più alto per formare un contratto";
				mes "con un mercenario di decima elementare.";
			}
			else {
				mes "[Manager di Mercenari]";
				mes "Congratulazioni! Sembra";
				mes "come se stessi pianificando di prendere";
				mes "su un lavoro molto pericoloso";
				mes "dato che stai assumendo un decimo";
				mes "Grado Mercenario. Ti auguro";
				mes "tanta fortuna con lui.";
				mercenary_set_faith getd(.@faith$[.@type]),-400;
				getitem .@item[.@type],1;
			}
			close;
	}
}

-	script	Mercenary Merchant#dummy::MercMerchant	-1,{
	mes "Banchiere di merci mercenarie";
	mes "Ciao, vendo beni";
	mes "che i mercenari possono";
	mes "usare. C'è qualcosa";
	mes "in particolare che";
	mes "cercando?";
	next;
	setarray .@item, 12184,	12185,	12241,	12242,	12243;
	setarray .@cost, 2500,	5000,	800,	1500,	3000;
	for (.@size = getarraysize(.@item); .@i < .@size; .@i++)
		.@menu$ = .@menu$ + getitemname(.@item[.@i]) + ":";
	.@m = select(.@menu$)-1;
	.@str$ = .@cost[.@m] + "";
	.@len = getstrlen(.@str$);
	.@str_cost$ = (.@len <= 3)? .@str$ : insertchar(.@str$,",",.@len-3);
	mes "Banchiere di merci mercenarie";
	mes getitemname(.@item[.@m]);
	mes "ciascun costo " + .@str_cost$ + " zeny.";
	mes "Quanti ne vorresti?";
	next;
	input .@input;
	if (!.@input) {
		mes "Banchiere di merci mercenarie";
		mes "Hai cambiato idea?";
		mes "Va bene, sentiti libero di venire";
		mes "ritornare da me quando vuoi";
		mes "per comprare qualsiasi Pozione di Mercenario.";
		close;
	}
	else if (.@input < 0 || .@input > 10000) {
		mes "Banchiere di merci mercenarie";
		mes "Mi dispiace, ma tu";
		mes "si può comprare fino a";
		mes "diecimila di queste pozioni";
		mes "Uno alla volta. Si prega di inserire";
		mes "un numero da 1 a 10,000.";
		close;
	}
	else if (set(.@total_po, .@input * .@cost[.@m]) > Zeny) {
		mes "Banchiere di merci mercenarie";
		mes "Mi dispiace, ma tu non";
		mes "a abbastanza zeny per questo";
		mes "molte pozioni. Beh, eccomi qui";
		mes "sii qui quando sarai pronto";
		mes "acquistare qualcosa";
		mes "per i tuoi Mercenari.";
		close;
	}
	else if (!checkweight(.@item[.@m], .@input)) {
		mes "Banchiere di merci mercenarie";
		mes "Se ti desse così tanti";
		mes "pozioni, tu non saresti";
		mes "in grado di portarli con te.";
		mes "Per favore torna dopo";
		mes "liberi un po' di spazio";
		mes "nel tuo Inventario.";
		close;
	}
	mes "Banchiere di merci mercenarie";
	if (.@input == 1) 
		mes "Ecco il tuo " + getitemname(.@item[.@m]) + ".";
	else {
		mes "Eccolo qui, questo è esattamente";
		mes callfunc("F_InsertPlural",.@input,getitemname(.@item[.@m])) + ".";
	}
	mes "Grazie, e per favore vieni";
	mes "di nuovo quando hai bisogno di più";
	mes "pozioni per i tuoi Mercenari.";
	Zeny = Zeny - .@total_po;
	getitem .@item[.@m], .@input;
	close;
}

-	script	Mercenary_Switch	-1,{

	setarray .@name$, "Spear","Sword","Bow";
	.@npc$ = strnpcinfo(2);
	for (.@size = getarraysize(.@name$); .@type < .@size; .@type++)
		if (.@npc$ == .@name$[.@type]) break;

	mes "[Controllore]";
	mes "Inserisci una password.";
	next;
	.@i = callfunc("F_GM_NPC",1854,0,0,10000);
	mes "[Controllore]";
	if (.@i == -2)
		mes "Errore.";
	else if (.@i == 0)
		mes "Numero sbagliato";
	else if (.@i == 1) {
		mes "Seleziona.";
		next;
		switch( select( "Spegnere NPC Mercenario","Accendi NPC Mercenario" ) ) {
		case 1:
			mes "Gli NPC sono spenti.";
			disablenpc "Mercenary Manager#" + .@name$[.@type];
			disablenpc "Mercenary Merchant#" + .@name$[.@type];
			break;
		case 2:
			mes "Gli NPC sono accesi.";
			enablenpc "Mercenary Manager#" + .@name$[.@type];
			enablenpc "Mercenary Merchant#" + .@name$[.@type];
			break;
		}
	}
	close;
}
