aldebaran,139,63,4	script	Soldier#alde	105,{
	cutin "prt_soldier",2;
	mes "[Al De Baran Guardia]";
	mes "Sono solo un guardia qualunque";
	mes "che potresti trovare in qualsiasi altra città. Non credo nemmeno di avere un nome...";
	next;
	mes "[Al De Baran Guardia]";
	mes "Sono responsabile delle Guide di Servizio della Caserma Al De Baran. Lasciatemi guidarvi";
	mes "attraverso la nostra città!";
	next;
	if (select("Guida alla posizione.:Termina conversazione.") == 1) {
		viewpoint 1,61,229,0,0xFF6633;
		viewpoint 1,72,197,1,0x0000FF;
		viewpoint 1,223,222,2,0x00FFFF;
		viewpoint 1,233,105,3,0x515151;
		viewpoint 1,197,70,4,0x3355FF;
		viewpoint 1,60,60,5,0xFF5555;
		mes "^FF6633+^000000 -> Ufficio Centrale di Kafra ";
		mes "^0000FF+^000000 -> Negozio di armi ";
		mes "^00FFFF+^000000 -> Gilda di Stregoni (Chiusa)";
		mes "^515151+^000000 -> Pub";
		mes "^3355FF+^000000 -> Negozio di Oggetti";
		mes "^FF5555+^000000 -> Gilda degli Alchimisti";
		close2;
		cutin "prt_soldier",255;
		end;
	}
	mes "[Al De Baran Guardia]";
	mes "Siamo giurati di proteggere Al De Baran! Che le forze del male vengano sempre schiacciate da";
	mes "pugno giusto del bene!";
	close2;
	cutin "prt_soldier",255;
	end;
}

aldebaran,133,108,4	script	Soldier#2alde	105,{
	cutin "prt_soldier",2;
	mes "[Al De Baran Guardia]";
	mes "Sono solo un/una";
	mes "guardia ordinaria,";
	mes "il tipo che puoi";
	mes "trova in qualsiasi altra città.";
	next;
	mes "[Al De Baran Guardia]";
	mes "Quando non sono troppo impegnato";
	mes "proteggere l'Al De Baran";
	mes "popolace, sono qui per dare indicazioni agli avventurieri";
	mes "come te stesso.";
	next;
	switch(select("Ufficio Centrale di Kafra :Negozio di armi :Gilda dei Maghi :Pubblico :Negozio di oggetti :Gilda degli Alchimisti :Fine Conversazione ")) {
	case 1:
		viewpoint 1,61,229,0,0xFF6633;
		mes "^FF6633+^000000 -> Ufficio Centrale di Kafra ";
		break;
	case 2:
		viewpoint 1,72,197,1,0x0000FF;
		mes "^0000FF+^000000 -> Negozio di armi ";
		break;
	case 3:
		viewpoint 1,223,222,2,0x00FFFF;
		mes "^00FFFF+^000000 -> Gilda di Stregoni (Chiusa)";
		break;
	case 4:
		viewpoint 1,233,105,3,0x515151;
		mes "^515151+^000000 -> Pub ";
		break;
	case 5:
		viewpoint 1,197,70,4,0x3355FF;
		mes "^3355FF+^000000 -> Negozio di Oggetti ";
		break;
	case 6:
		viewpoint 1,60,60,5,0xFF5555;
		mes "^FF5555+^000000 -> Gilda degli Alchimisti";
		break;
	case 7:
		mes "[Al De Baran Guardia]";
		mes "Siamo giurati di";
		mes "proteggi Al De Baran!";
		mes "Che le forze del bene";
		mes "sempre prevalere sul male";
		break;
	}
	close2;
	cutin "prt_soldier",255;
	end;
}
