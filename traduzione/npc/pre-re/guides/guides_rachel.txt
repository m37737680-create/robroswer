rachel,138,146,5	script	Rachel Guide	934,{
	mes "[Guida di Rachel]";
	mes "Benvenuti nella capitale di";
	mes "Arunafeltz, Rachel dove un caldo";
	mes "il respiro della dea Freya arriva.";
	mes "Se è la prima volta per te";
	mes "per usare i servizi guida, perché";
	mes "non controlli prima il menù \'Notice\'?";
	set .@loop1,1;
	while(.@loop1) {
		next;
		switch(select("Guida al villaggio:Rimuovi i segni dalla mini-mappa:Avviso:Annulla")) {
		case 1:
			mes "[Guida di Rachel]";
			mes "Posso dirti la posizione di qualsiasi edificio purché sia a Rachel.";
			mes "Allora dove vuoi andare?";
			if (.@compass_check == 0) {
				mes "Vorresti che io";
				mes "segnalare le posizioni";
				mes "sulla tua mini-mappa?";
				next;
				if (select("Sì:No") == 1) set .@compass_check,1;
			}
			set .@loop2,1;
			while(.@loop2) {
				if (.@wait_button_chk == 0) set .@wait_button_chk,1;
				else next;

				switch(select("Cheshrumnir:Dentro:Negozio di armi:Laboratorio di strumenti:Aeroporto:Annulla")) {
				case 1:
					mes "[Guida di Rachel]";
					mes "Cheshrumnir è un luogo sacro dove risiede Pope, l'incarnazione della dea Freya.";
					mes "Prendi la strada verso nord per trovare l'edificio.";
					if (.@compass_check == 1) viewpoint 1,150,249,1,0xFF0000;
					break;
				case 2:
					mes "[Guida di Rachel]";
					mes "Puoi riposare la tua stanchezza del viaggio nell'osteria.";
					mes "L'edificio sulla sinistra accanto a me è l'Inn of Rachel.";
					if (.@compass_check == 1) viewpoint 1,115,149,2,0xFF00FF;
					break;
				case 3:
					mes "[Guida di Rachel]";
					mes "Vuoi dare un'occhiata alle armi che vengono vendute a Rachel?";
					mes "Il negozio di armi è situato vicino al cancello occidentale.";
					if (.@compass_check == 1) viewpoint 1,42,87,3,0x99FFFF;
					break;
				case 4:
					mes "[Guida di Rachel]";
					mes "Il negozio di pozioni di Rachel vende le pozioni della migliore qualità.";
					mes "Si trova vicino al cancello occidentale.";
					if (.@compass_check == 1) viewpoint 1,83,78,4,0x0000FF;
					break;
				case 5:
					mes "[Guida di Rachel]";
					mes "L'aeroporto è situato fuori dal cancello orientale.";
					if (.@compass_check == 1) viewpoint 1,273,125,5,0x00FF00;
					break;
				case 6:
					mes "[Guida di Rachel]";
					mes "Se ti piace eliminare tutti i marcatori di posizione sulla tua Mappa Minima,";
					mes "ri chiedimi e scegli il menù \'Rimuovi Segni dalla Mini-Mappa\'";
					set .@loop2,0;
					break;
				}
			}
			break;

		case 2:
			viewpoint 2,150,249,1,0xFF0000;
			viewpoint 2,115,149,2,0xFF00FF;
			viewpoint 2,42,87,3,0x99FFFF;
			viewpoint 2,83,78,4,0x0000FF;
			viewpoint 2,273,125,5,0x00FF00;
			mes "[Guida di Rachel]";
			mes "Va bene, sono spariti ora. Se hai altre località da chiedere, fammelo sapere.";
			break;
		case 3:
			mes "[Guida di Rachel]";
			mes "Quando stai usando il menù 'Guida del Villaggio',";
			mes "assicurati che le ubicazioni degli edifici vengano contrassegnate sulla tua minimaappa nell'angolo superiore destro del tuo schermo.";
			mes "Se non riesci a vedere la tua minamappa, usa il tasto di scelta rapida 'ctrl+tab' o premi il pulsante 'Mappa' nelle tue finestre di informazioni base, ok?";
			mes "E puoi anche ingrandire la tua minamappa utilizzando il pulsante “-” nel caso non riesci a vedere l'intera mappa del villaggio.";
			break;
		case 4:
			mes "[Guida di Rachel]";
			mes "Spero che tu abbia un viaggio meraviglioso";
			mes "ad Arunafeltz.";
			close2;
			set .@loop1,0;
			break;
		}
	}
	close;
}
