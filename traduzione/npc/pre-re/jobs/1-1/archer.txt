payon_in02,64,71,4	script	Archer Guildsman#archer	85,{
	if (Upper == 1) {
		if (Class == Job_Novice_High && (ADVJOB == Job_Sniper || ADVJOB == Job_Clown || ADVJOB == Job_Gypsy)) {
			mes "[Arciere Gilda]";
			mes "Ehi, ti conosco.";
			mes "Hai fatto questo test";
			mes "prima, non lo facevi?";
			next;
			mes "[Arciere Gilda]";
			mes "Devi essere stato";
			mes "a Valhalla e rinato.";
			mes "Wow, è così impressionante!";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Arciere Gilda]";
				mes "Ehm...";
				mes "Meglio che tu impari tutte le abilità di base prima di poter diventare un Arciere.";
				next;
				mes "[Arciere Gilda]";
				mes "A dopo.";
				close;
			}
			mes "[Arciere Gilda]";
			mes "Beh allora. Io non";
			mes "qualcos'altro che dire.";
			mes "So che sarai un grande Arciere...";
			next;
			skill "NV_TRICKDEAD",0,SKILL_PERM;
			jobchange Job_Archer_high;
			skill "AC_MAKINGARROW",1,SKILL_PERM;
			skill "AC_CHARGEARROW",1,SKILL_PERM;
			mes "[Arciere Gilda]";
			mes "Sebbene non ci sia un particolare";
			mes "un premio per te questa volta, spero che tu capisca. Prenditi cura di te.";
			close;
		}
		else {
			mes "[Arciere Gilda]";
			mes "Oh...?";
			mes "Ehi, quali sono";
			mes "cosa fai qui...?";
			next;
			mes "[Arciere Gilda]";
			mes "Posso dire che non sei fatto per essere un Arciere. Sembra quasi che tu sia destinato a fare";
			mes "qualcos'altro...";
			close;
		}
	}
	mes "[Arciere Gilda]";
	mes "Buongiorno. Come posso aiutarla?";
	next;
	switch(select("Voglio essere un arcere.:Ho bisogno dei requisiti, per favore.:Niente, grazie.")) {
	case 1:
		if (BaseJob == Job_Archer) {
			mes "[Arciere Gilda]";
			mes "Sei già diventato un Arciere...";
			close;
		}
		else if (BaseJob != Job_Archer && BaseJob != Job_Novice) {
			mes "[Arciere Gilda]";
			mes "Mmm...";
			mes "Non sembri affatto un Novizio...";
			next;
			mes "[Arciere Gilda]";
			mes "Comunque, qualunque cosa tu sia, non puoi scegliere un lavoro da Arciere perché hai già un lavoro.";
			close;
		}
		if (job_archer_q == 0) {
			mes "[Arciere Gilda]";
			mes "Vuoi essere un Arciere?";
			mes "Se è così, devi compilare questo modulo di domanda.";
			next;
			if (select("Applicare.:Annulla") == 1) {
				set job_archer_q,1;
				setquest 1004;
				mes "[Arciere Gilda]";
				mes "Va bene, firma qui. Va bene, ehm, ti promuoverò quando rispetterai i requisiti.";
				next;
				mes "[Arciere Gilda]";
				mes "Se pensi di averli già incontrati, possiamo verificarlo ora.";
				mes "Sei pronto?";
				next;
				if (select("Sì, lo sono.:No, non ancora.") == 1) {
					mes "[Arciere Gilda]";
					mes "Va bene, lascio controllare.";
					next;
				}
				else {
					mes "[Arciere Gilda]";
					mes "Capisco. Sentiti libero di dare un'occhiata ai requisiti.";
					close;
				}
			}
			else {
				mes "[Arciere Gilda]";
				mes "Beh, va bene.";
				mes "A presto.";
				close;
			}
		}
		mes "[Arciere Gilda]";
		mes "Sei..." + strcharinfo(0) + "?";
		next;
		if (!callfunc("F_CanChangeJob")) {
			mes "[Arciere Gilda]";
			mes "Beh, non sei al livello di lavoro giusto. Per favore ricontrolla i requisiti.";
			next;
			mes "[Arciere Gilda]";
			mes "Il tuo livello di lavoro deve essere almeno 10 e non dimenticare che dovresti imparare tutte le Abilità Base. Una volta fatto questo, torna indietro.";
			close;
		}
		if (job_archer_q == 1) {
			set .@archer_item1,countitem(1066) * 5;
			set .@archer_item2,countitem(1067) * 3;
			set .@archer_item3,countitem(1068) * 2;
			set .@archer_item4,countitem(1019);
			set .@total_archer,.@archer_item1 + .@archer_item2 + .@archer_item3 + .@archer_item4;
			set .@total_archer2,(((.@archer_item2 + .@archer_item3) * 2) + .@archer_item4);
			mes "[Arciere Gilda]";
			mes "Eccellente!";
			mes "Ora allora,";
			mes "vediamo...";
			next;
			mes "[Arciere Gilda]";
			mes "Valuterò il valore dei vari tipi di Bauli, necessari per produrre un Arco, che hai portato.";
			next;
			if (countitem(1066) == 0 && countitem(1067) == 0 && countitem(1068) == 0 && countitem(1019) == 0) {
				mes "[Arciere Gilda]";
				mes "Um...";
				mes "Purtroppo non hai portato nessuno degli articoli richiesti. Non c'è nulla che io possa periziare.";
				close;
			}
			mes "[Arciere Gilda]";
			if (countitem(1066) != 0) mes " Livello S : " + countitem(1066) + " ea, Grado: " + .@archer_item1 + " . ";
			if (countitem(1067) != 0) mes " Grado A : " + countitem(1067) + " ea, Grado : " + .@archer_item2 + " . ";
			if (countitem(1068) != 0) mes " Grado B : " + countitem(1068) + " ea, Grado : " + .@archer_item3 + " . ";
			if (countitem(1019) != 0) mes " Grado C : " + countitem(1019) + " ea, Grado : " + .@archer_item4 + " . ";
			if (.@total_archer < 25) {
				mes "Votazioni Totali: ^FF0000" + .@total_archer + "^000000 / 40";
				next;
				mes "[Arciere Gilda]";
				mes "Meno di 25!? Devi prendere un voto di almeno 25! Forza, prova più duramente!";
				close;
			}
			else {
				mes "Votazioni Totali: ^0000FF" + .@total_archer + "^000000 / 40";
				next;
				mes "[Arciere Gilda]";
				if (.@total_archer > 40) {
					mes "Wow! Più di 40!";
					mes "Eccellente! Congratulazioni!";
				}
				else if (.@total_archer > 30) {
					mes "Più di 30! Ottimo lavoro!";
					mes "Congratulazioni!";
				}
				else {
					mes "*Suspiro* Beh, sei passato/a per un pelo... Comunque, complimenti.";
				}
			}
			next;
			mes "[Arciere Gilda]";
			mes "Trasferirò questi Trunks al nostro Reparto Produzione Arco. Ora che hai soddisfatto i requisiti, lasciami promuoverti subito!";
			if (countitem(1066) != 0) delitem 1066,countitem(1066); //Tree_Of_Archer_1
			if (countitem(1067) != 0) delitem 1067,countitem(1067); //Tree_Of_Archer_2
			if (countitem(1068) != 0) delitem 1068,countitem(1068); //Tree_Of_Archer_3
			if (countitem(1019) != 0) delitem 1019,countitem(1019); //Wooden_Block
		}
		next;
		callfunc "Job_Change",Job_Archer;
		callfunc "F_ClearJobVar";
		completequest 1004;
		mes "[Arciere Gilda]";
		mes "Congratulazioni!";
		mes "Sei ora un Arciere!";
		next;
		mes "[Arciere Gilda]";
		mes "Certo, ci aspettiamo che contribuirai al futuro della Gilda degli Arcieri con i tuoi sforzi.";
		next;
		mes "[Arciere Gilda]";
		mes "Ah, il tuo arco è arrivato dal Reparto di Produzione Archi. Eccolo qui, prendilo! È tuo~";
		getitem 1702,1; //Bow_
		getitem 1750,.@total_archer2; //Arrow
		next;
		mes "[Arciere Gilda]";
		mes "Ora, via'. Caccia con orgoglio, sapendo di essere stato addestrato da uno dei migliori!";
		close;
	case 2:
		mes "[Arciere Gilda]";
		mes "Spiegherò i requisiti per essere un Arciere.";
		if (BaseJob != Job_Novice) {
			if (BaseJob == Job_Archer) {
				next;
				mes "[Arciere Gilda]";
				mes "Ma...";
				mes "Sei già un Arciere. Dovresti saperlo già...";
			}
			else {
				next;
				mes "[Arciere Gilda]";
				mes "Aspetta un secondo. Hai già scelto un altro lavoro. Non hai bisogno di saperlo~";
			}
			mes "Quindi...Sì...nessun vero motivo per dirti i requisiti...";
		}
		next;
		mes "[Arciere Gilda]";
		mes "Innanzitutto, devi avere il Livello di Lavoro 9 come Principiante e conoscere tutte le Abilità Base.";
		next;
		mes "[Arciere Gilda]";
		mes "Un arcere ha bisogno di una concentrazione e dei riflessi estremamente elevati, quindi non accettiamo coloro che hanno poca pazienza.";
		next;
		mes "[Arciere Gilda]";
		mes "Devi anche raccogliere ^FF0000Trunks^000000. Ci sono 4 tipi diversi di Trunks, ognuno con una qualità diversa. Ti verranno assegnati gradi diversi per i tuoi Trunks, a seconda della loro qualità.";
		next;
		mes "[Arciere Gilda]";
		mes "Per diventare un Arciere, devi ottenere un voto di almeno ^0000FF25^000000 punti su 40. Puoi prendere Trunks da 'Willow', l'albero. Attento, però. Possono essere mostri difficili.";
		close;
	case 3:
		close;
	}
}
