que_ng,187,149,3	script	Magazine Dealer Kenny::mdk	4_M_01,{
	mes "[Kenny]";
	if (BaseClass == Job_Gunslinger) {
		mes "Benvenuti nel mio negozio di riviste.";
		mes "Come saprai, grandi numeri";
		mes "di proiettili possono essere trasportati di più";
		mes "facilmente quando sono nelle Riviste. Ora, posso interessarti a";
		mes "qualcosa di particolare?";
		next;
		switch(select("Pacchetto Sfera del Vento:Pacchetto Sfera Ombra:Pacchetto Sfera Veleno:Pacchetto Sfera Acqua:Pacchetto Sfera di Fuoco:Cartuccia:Cartuccia di sangue:Cartuccia d'argento:Annulla")) {
		case 1: callsub S_Casing,13204,12144;// <bullet id>, <pack id>
		case 2: callsub S_Casing,13206,12145;
		case 3: callsub S_Casing,13205,12146;
		case 4: callsub S_Casing,13207,12147;
		case 5: callsub S_Casing,13203,12148;
		case 6: callsub S_Casing,13200,12149;
		case 7: callsub S_Casing,13202,12150;
		case 8: callsub S_Casing,13201,12151;
		case 9:
			mes "[Kenny]";
			mes "Beh, se mai trovi";
			mes "che hai troppi";
			mes "pallini, vieni a vedermi.";
			mes "È un'idea intelligente salvare";
			mes "bulletini con le mie riviste.";
			close;
		}
	}
	mes "Benvenuti nel mio negozio.";
	mes "Qui fornisco Riviste";
	mes "e Cartucce per Pistolieri.";
	mes "Scusa, ma non sembra che";
	mes "i miei servizi sarebbero di qualsiasi";
	mes "usavi tu, avventuriero.";
	next;
	mes "[Kenny]";
	mes "Ehm, ma se per caso";
	mes "conosci qualche pistolero, invia";
	mes "alla mia maniera. Non potrai mai";
	mes "troppi punti elenco.";
	close;

S_Casing:
	.@bullet_id = getarg(0);
	.@pack_id = getarg(1);

	mes "[Kenny]";
	mes "Ora puoi scambiare";
	mes callfunc("F_InsertPlural",500,getitemname(.@bullet_id));
	mes "e 500 zeny per 1 "+getitemname(.@pack_id)+", so make sure";
	mes "hai abbastanza proiettili";
	mes "e zeny per questo scambio.";
	next;
	mes "[Kenny]";
	mes "Puoi scambiare "+callfunc("F_InserisciPlurale",500,getitemname(.@bullet_id))+" e 500 zeny";
	mes "con 1 "+getitemname(.@pack_id)+".";
	next;
	mes "[Kenny]";
	mes "Ricorda che non posso dare";
	mes "più di 50 riviste";
	mes "alla volta. Ora si prega di inserire";
	mes "il numero di riviste che tu";
	mes "vuoi ricevere. Se vuoi annullare, inserisci semplicemente '0.'";
	next;
	input .@amount;
	mes "[Kenny]";
	if (.@amount > 50 || .@amount == 0) {
		mes "Ehi, non posso darti";
		mes "tanti riviste.";
		mes "Per favore riprova, e";
		mes "inserisci un numero no";
		mes "maggiore di 50.";
	}
	else if (countitem(.@bullet_id) >= (500*.@amount)) {
		if (Zeny >= (500*.@amount)) {
			if (checkweight(.@pack_id,.@amount) == 0) {
				mes "Ehi, hai un sacco di";
				mes "di spazzatura accatastata nel tuo";
				mes "Inventario. Libera un po'";
				mes "spazio, e poi torna indietro";
				mes "e scambiare le vostre palle per";
				mes "dopo alcuni riviste, ok?";
			} else {
				mes "Allora, eccoci";
				mes "le tue riviste. Grazie";
				mes "per visitare il mio negozio, e";
				mes "Spero che tu usi tutto";
				mes "de i tuoi munizioni con saggezza.";
				Zeny = Zeny -500*.@amount;
				delitem .@bullet_id,(500*.@amount);
				getitem .@pack_id,.@amount;
			}
		} else {
			mes "Scusa, ma non";
			mes "a abbastanza zeny per";
			mes "scambio di riviste.";
			mes "Torna nel mio negozio";
			mes "dopo che hai risparmiato";
			mes "più soldi.";
		}
	} else {
		mes "Scusa, ma non";
		mes "avere abbastanza proiettili per";
		mes "scambio di riviste.";
		mes "Forse dovresti raddoppiare";
		mes "controlla il tuo inventario, e";
		mes "poi torna da me più tardi.";
	}
	close;
}
