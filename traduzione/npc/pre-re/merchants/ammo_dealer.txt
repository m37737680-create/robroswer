alberta,118,160,3	script	Bullet Dealer Tony#alb::bdt	4_M_04,{
	mes "[Tony]";
	if (BaseClass == Job_Gunslinger) {
		mes "Sono Tony, il venditore di proiettili.";
		mes "Vieni da me quando sei";
		mes "scarsso di munizioni. Portami solo";
		mes "i materiali, e farò";
		mes "tu le pallottole di cui hai bisogno.";
		next;
		switch(select("Sfera Veleno:Sfera di Flare:Sfera luminosa:Sfera Cieca:Sfera ghiacciata:Annulla")) {
		case 1: callsub S_BulletTrade,937,10,13205;// <item required>, <amount req>, <bullet id>
		case 2: callsub S_BulletTrade,7097,2,13203;
		case 3: callsub S_BulletTrade,7053,3,13204;
		case 4: callsub S_BulletTrade,1024,5,13206;
		case 5: callsub S_BulletTrade,7054,2,13207;
		case 6:
			mes "[Tony]";
			mes "Cambio idea?";
			mes "Beh, se avrai bisogno";
			mes "nessun punto elenco, ci sarò subito";
			mes "qui. Torna quando vuoi";
			mes "pensi che ne avrai bisogno di più";
			mes "munizioni, Gunslinger.";
			close;
		}
	}
	mes "Ciao, sono Tony. Sono in";
	mes "accusa di distribuire";
	mes "e fare punti elenco per";
	mes "Sciabannieri. È solo";
	mes "come piace alla nostra gilda";
	mes "fare cose.";
	next;
	mes "[Tony]";
	mes "Mi dispiace se sei venuto/a";
	mes "qui per comprare dei proiettili.";
	mes "Posso fare affari solo con";
	mes "Gunslinger completi.";
	close;

S_BulletTrade:
	.@required_id = getarg(0);
	.@required_amount = getarg(1);
	.@bullet_id = getarg(2);

	mes "[Tony]";
	mes "Per ogni insieme di";
	mes callfunc("F_InsertPlural",30,getitemname(.@bullet_id))+",";
	mes "devi darmi";
	mes "Emveretarcon,";
	mes "Phracon, e";
	mes callfunc("F_InsertPlural",.@required_amount,getitemname(.@required_id))+".";
	next;
	mes "[Tony]";
	mes "Ricorda che posso dare";
	mes "un massimo di 500 set di";
	mes "30 proiettili alla volta. Per favore.";
	mes "inserisci il numero di set di punti elenco";
	mes "che desideri. Se vuoi";
	mes "cancellare, poi inserire solo '0.'";
	next;
	input .@amount;
	mes "[Tony]";
	if (.@amount < 1 || .@amount > 500) {
		mes "Ehi, non posso darti";
		mes "quanti proiettili. Non";
		mes "dimenticare di inserire un numero";
		mes "non è più alto di 500";
		mes "se vuoi scambiare il tuo";
		mes "articoli per alcuni proiettili.";
	}
	else if (countitem(1010) >= .@amount && countitem(1011) >= .@amount && countitem(.@required_id) >= (.@amount*.@required_amount)) {
		if (checkweight(.@bullet_id,.@amount * 30) == 0) {
			mes "Eh? Il tuo inventario non";
			mes "abbastanza spazio per questo";
			mes "molti punti elenco. Torna più tardi";
			mes "dopo che fai più spazio";
			mes "disponibile. Prova a mettere alcuni di";
			mes "le tue cose in Kafra Storage.";
		} else {
			mes "Ottimo, tutto sembra";
			mes "essere in ordine. Lasciami prendere";
			mes "questi materiali, e ecco";
			mes "i tuoi proiettili. È un piacere";
			mes "fare affari con voi";
			delitem 1010,.@amount; //Phracon
			delitem 1011,.@amount; //Emveretarcon
			delitem .@required_id,.@amount * .@required_amount;
			getitem .@bullet_id,.@amount * 30;
		}
	} else {
		mes "Eh? Assomigli a";
		mes "non ho abbastanza materiali";
		mes "per tanti proiettili. Beh,";
		mes "non è un problema. Vieni solo";
		mes "tornare dopo aver raccolto tutto";
		mes "che ti serve, ok?";
	}
	close;
}

que_ng,187,156,3	duplicate(bdt)	Bullet Dealer Tony#ng	86
