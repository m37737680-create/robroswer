prt_in,243,168,4	script	Jovovich	91,{
	mes "[Parrucchiere Jovovich]";
	mes "Benvenuto!";
	mes "Come posso aiutarla?";
	next;
	mes "[Parrucchiere Jovovich]";
	if (Sex == SEX_MALE) mes "Oh, no! I tuoi capelli sono rovinati. Sembra che tu possa aver bisogno di un trattamento professionale. Vieni a sederti qui, per favore. Vieni.";
	else {
		if (rand(20) > 11) {
			mes "Ehi!? Oh mio Dio! Oh no no no no! I tuoi capelli sono così rovinati! Non è buono se lasci i tuoi capelli così.";
			next;
			mes "[Parrucchiere Jovovich]";
			mes "Mi lasceresti curare i tuoi capelli? Per favore?";
		}
		else mes "Wow! I tuoi capelli sarebbero perfetti una volta tinti~ Che ne dici di tingerti i capelli per cambiare?";
	}
	next;
	while(1) {
		switch(select("Tinte per capelli:Suggerimenti e Informazioni:Annulla")) {
		case 1:
			mes "[Parrucchiere Jovovich]";
			mes "Sì sì, buona scelta~";
			mes "Allora, questa è la tua occasione per un makeover, la tua occasione di sbocciare in bellezza! Scegli il colore che desideri.";
			next;
			while(1) {
				if (.@choose_success == 1) {
					mes "[Parrucchiere Jovovich]";
					mes "Cosa ne pensi? Volevi un colore diverso?";
					next;
					if (select("Sì:No") == 1) {
						mes "[Parrucchiere Jovovich]";
						mes "Ok! Scegli il colore che desideri.";
						next;
					}
					else  {
						mes "[Parrucchiere Jovovich]";
						mes "Mmm, sono un po' deluso. Volevo fare meglio. Ma prometto che la prossima volta lo farò meglio. Per favore, venici ancora~";
						close;
					}
				}

				while(1) {
					switch(select("Rosso, per favore.:Giallo, per favore.:Violetta, per favore.:Arancione, per favore.:Verde, per favore.:Blu, per favore.:Bianco, per favore.:Nero, per favore.:In realtà, mi piace i miei capelli così come sono.")) {
					case 1: set .@headpalette,8; break;
					case 2: set .@headpalette,1; break;
					case 3: set .@headpalette,2; break;
					case 4: set .@headpalette,3; break;
					case 5: set .@headpalette,4; break;
					case 6: set .@headpalette,5; break;
					case 7: set .@headpalette,6; break;
					case 8: set .@headpalette,7; break;
					case 9:
						if (.@choose_success != 0) {
							mes "[Parrucchiere Jovovich]";
							mes "Devi amare il colore dei tuoi capelli~";
							close;
						}
						mes "[Parrucchiere Jovovich]";
						mes "Eeeh~? Non ti colorerai i capelli? Sono un po' triste...";
						close;
					}

					if (.@headpalette == getlook(VAR_HEADPALETTE)) {
						mes "[Parrucchiere Jovovich]";
						mes "Eh? Ma è il colore di capelli che hai già. Per favore scegli un altro colore.";
						next;
					}
					else {
						switch(.@headpalette) {
						case 1: callsub S_NoDye,976,"yellow"; break;
						case 2: callsub S_NoDye,978,"violet"; break;
						case 3: callsub S_NoDye,980,"orange"; break;
						case 4: callsub S_NoDye,979,"green"; break;
						case 5: callsub S_NoDye,981,"blue"; break;
						case 6: callsub S_NoDye,982,"white"; break;
						case 7: callsub S_NoDye,983,"black"; break;
						case 8: callsub S_NoDye,975,"red"; break;
						}
						if (Zeny < 1000) {
							mes "[Parrucchiere Jovovich]";
							mes "La tariffa è di 1000 zeny. Non hai abbastanza...?";
							close;
						}

						switch(.@headpalette) {
						case 1: delitem 976,1; break; //Lemon_Dyestuffs
						case 2: delitem 978,1; break; //Violet_Dyestuffs
						case 3: delitem 980,1; break; //Orange_Dyestuffs
						case 4: delitem 979,1; break; //Darkgreen_Dyestuffs
						case 5: delitem 981,1; break; //Cobaltblue_Dyestuffs
						case 6: delitem 982,1; break; //White_Dyestuffs
						case 7: delitem 983,1; break; //Black_Dyestuffs
						case 8: delitem 975,1; break; //Scarlet_Dyestuffs
						}
						set Zeny, Zeny-1000;
						setlook VAR_HEADPALETTE,.@headpalette;
						set .@choose_success,1;
						break;
					}

				}

			}

		case 2:
			mes "[Parrucchiere Jovovich]";
			mes "Quando ti senti giù, quando vieni lasciato/a, quando vuoi impressionare qualcuno, o anche solo quando vuoi distinguerti...";
			next;
			mes "[Parrucchiere Jovovich]";
			mes "Per quel luogo e momento speciali, non vorresti un'acconciatura tutta tua? Finché hai i coloranti appropriati, farò sembrare meravigliosi i tuoi capelli.";
			next;
			mes "[Parrucchiere Jovovich]";
			mes "E prova a non preoccuparti troppo della tariffa. Acquisire la bellezza è come acquisire tutto. Hehe~";
			next;
			mes "[Parrucchiere Jovovich]";
			mes "* Commissioni e Informazioni *";
			mes "- 1 articolo di tintura del colore dei capelli che desideri.";
			mes "- Tassa di 1000 zeny.";
			close;

		case 3:
			mes "[Parrucchiere Jovovich]";
			mes "Uomini o Donne...";
			mes "Ognuno ha il diritto e l'obbligo di essere bello.";
			close;
		}

	}

S_NoDye:
	if (countitem(getarg(0)) == 0) {
		mes "[Parrucchiere Jovovich]";
		mes "Eh?! Ma ti serve l'oggetto '"+getitemname(getarg(0))+"per tingerti i capelli "+getarg(1)+"...";
		close;
	}
	return;
}
