alberta,26,229,0	script	Kafra Employee#bunny	83,{
	if (BUNYBND == 1) {
		mes "Dipendente Kafra";
		mes "Ciao!";
		mes "Come ti piacerebbe";
		mes "partecipare a Kafra";
		mes "Specialità della corporazione";
		mes "^529DFFEvento Banda dei Conigli^000000?";
		next;
		switch(select("Certo, ho portato gli articoli.:Informazioni sull'evento:Annulla")) {
		case 1:
			mes "Dipendente Kafra";
			mes "Va bene~";
			mes "Lasciami controllare per";
			mes "vedi se hai portato";
			mes "tutti gli articoli...";
			next;
			setarray .@Items[0],949,100,706,1,722,1,2213,1;
			for(set .@i,0; .@i<8; set .@i,.@i+2)
				if (countitem(.@Items[.@i]) < .@Items[.@i+1]) {
					mes "Dipendente Kafra";
					mes "Oh, mi dispiace";
					mes "ma devi";
					mes "almeno portare";
					mes callfunc("F_InsertPlural",.@Items[.@i+1],getitemname(.@Items[.@i]))+".";
					close;
				}
			mes "Dipendente Kafra";
			mes "Ottimo, vedo che";
			mes "hai raccolto";
			mes "tutto ciò di cui ho bisogno per";
			mes "crea la Banda del Coniglio.";
			mes "Per favore, aspetti un momento";
			mes "mentre lo monto...";
			next;
			for(set .@i,0; .@i<8; set .@i,.@i+2)
				if (countitem(.@Items[.@i]) < .@Items[.@i+1]) {
					mes "Dipendente Kafra";
					mes "Eh? Mi dispiace,";
					mes "ma in realtà non posso";
					mes "fallo adesso. Tu";
					mes "bisogno "+callfunc("F_InserisciPlurale",.@Items[.@i+1],getitemname(.@Items[.@i]));
					mes "affinché io possa mettere questo";
					mes "I conigli si riuniscono...";
					close;
				}
			delitem 949,100; //Feather
			delitem 706,1; //Four_Leaf_Clover
			delitem 722,1; //Scarlet_Jewel
			delitem 2213,1; //Cat_Hairband
			mes "Dipendente Kafra";
			mes "Ah, eccolo/eccola~";
			mes "La perfetta Banda di Coniglietti!";
			mes "Spero che ti piaccia.";
			getitem 2214,1; //Bunny_Band
			set BUNYBND,0;
			next;
			mes "Dipendente Kafra";
			mes "Grazie per";
			mes "partecipare a questo";
			mes "evento speciale e tuo";
			mes "patrocinio continuato";
			mes "di Kafra Services~";
			close;
		case 2:
			goto L_Info;
		case 3:
			close;
		}
	}
	mes "Dipendente Kafra";
	mes "Ciao! Ti piacerebbe";
	mes "per unirsi al nostro speciale ^529DFFBunny";
	mes "Evento di Band^000000 ospitato da Kafra";
	mes "Corporazione e sponsorizzata da";
	mes "la Gilda dei Mercanti dell'Alberta?";
	next;
	switch(select("Unisciti all'evento:Informazioni sull'evento:Annulla")) {
	case 1:
		set BUNYBND,1;
		mes "Dipendente Kafra";
		mes "Ottimo! Grazie per";
		mes "partecipando! Se tu";
		mes "non ho già sentito";
		mes "devi raccogliere questi";
		mes "articoli se vuoi che metta un";
		mes "Bunny Band insieme per te...";
		next;
		mes "Dipendente Kafra";
		mes "100 Piume,";
		mes "Trifoglio a quattro foglie,";
		mes "Kitty Band e";
		mes "Un perla. È tutto!";
		mes "Buona fortuna e io sarò";
		mes "in attesa di te qui~";
		close;
	case 2:
		break;
	case 3:
		close;
	}
L_Info:
	mes "Dipendente Kafra";
	mes "Per ringraziare i nostri stimati";
	mes "clienti, Kafra Corporation";
	mes "ha preparato un evento speciale";
	mes "dove si riunirà il dipendente di Kafra";
	mes "Braccialetti da coniglio per avventurieri";
	mes "chi porta gli articoli richiesti.";
	next;
	mes "Dipendente Kafra";
	mes "Per questo speciale,";
	mes "oggetto unico";
	mes "porta 100 Piume,";
	mes "Trifoglio a quattro foglie,";
	mes "Kitty Band e";
	mes "Perla.";
	next;
	mes "Dipendente Kafra";
	mes "Quando sei pronto,";
	mes "tornare e portare";
	mes "que articoli a me.";
	mes "(Sponsorizzato da il";
	mes "(Gilda dei Mercanti dell'Alberta).";
	close;
}
