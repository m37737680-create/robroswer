			mes "alla volta, solo per sicurezza.";
			next;
		}
		else break;
	}
	set .@total_cost,.@sell * .@item_cost;
	set .@total_weight,.@sell * .@item_weight;

		mes "abbastanza spazio nel tuo";
		mes "Inventario per questo molti";
		mes getitemname(.@item_id) +"...";
		close;
