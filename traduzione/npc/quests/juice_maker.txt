payon_in03,188,146,5	script	Marx Hansen#juice	86,{
	if (checkweight(1201,1) == 0) {
		mes "Aspetta un momento!";
		mes "Attualmente stai portando";
		mes "troppi oggetti con te.";
		mes "Per favore torna più tardi";
		mes "dopo che hai messo alcuni oggetti nello stoccaggio di kafra.";
		close;
	}
	if (MISC_QUEST&1) {
		mes "[Mercante Marx Hansen]";
		mes "Benvenuto.";
		mes "Sei venuto a";
		mes "processare anche frutta?";
		next;
		switch(select("Prepara il succo.:Parlare e ottenere informazioni sulla lavorazione dei frutti.:Annulla.")) {
		case 1:
			mes "[Mercante Marx Hansen]";
			mes "Che tipo di succo di frutta vorresti preparare?";
			next;
			switch(select("Succo di mele:Succo di banana:Succo di carota:Succo d'uva:Annulla")) {
			case 1:
				set .@fruit,512;
				set .@juice,531;
				break;
			case 2:
				set .@fruit,513;
				set .@juice,532;
				break;
			case 3:
				set .@fruit,515;
				set .@juice,534;
				break;
			case 4:
				set .@fruit,514;
				set .@juice,533;
				break;
			case 5:
				mes "[Mercante Marx Hansen]";
				mes "Beh allora...";
				mes "A presto.";
				close;
			}
			if (countitem(.@fruit) == 0 || countitem(713) == 0 || Zeny < 3) {
				mes "[Mercante Marx Hansen]";
				mes "Oh no...";
				mes "Non hai tutti i materiali necessari per fare "+getitemname(.@juice)+", Ho bisogno di 1 "+getitemname(.@fruit)+" e 1 bottiglia vuota. Avrò anche bisogno di una tassa di 3 zeny.";
				next;
				mes "[Mercante Marx Hansen]";
				mes "Quando avrai preparato tutto, frullerò la frutta per darti un succo delizioso.";
				close;
			}

			mes "[Mercante Marx Hansen]";
			mes "Quanti ne vorresti?";
			next;
			switch(select("Quanti ne posso.:Voglio una certa quantità.:Annulla.")) {
			case 1:
				set .@make,countitem(.@fruit);
				if (countitem(713) < .@make) set .@make, countitem(713);
				if (Zeny/3 < .@make) set .@make, Zeny/3;
				break;
			case 2:
				mes "[Mercante Marx Hansen]";
				mes "Scegli un numero inferiore a 100. Se non vuoi, metti '0'. Puoi fare fino a " + countitem(.@fruit) + " bottiglie di succo.";
				next;
				while(1) {
					input .@input;
					if (.@input == 0) {
						mes "[Mercante Marx Hansen]";
						mes "Beh allora...";
						mes "Ancora.";
						close;
					}
					else if (.@input > 100) {
						mes "[Mercante Marx Hansen]";
						mes "Più di 100 bottiglie è impossibile. Scegli una quantità diversa.";
						next;
					}
					else {
						break;
					}
				}
				set .@make,.@input;
				break;
			case 3:
				mes "[Mercante Marx Hansen]";
				mes "Beh allora...";
				mes "Ancora.";
				close;
			}

			set .@total_zeny,3 * .@make;

			if (countitem(.@fruit) < .@make || countitem(713) < .@make || Zeny < .@total_zeny) {
				mes "[Mercante Marx Hansen]";
				mes "Oh no...";
				mes "Non hai tutti i materiali necessari. Non posso aiutare in una situazione del genere. Immagino che tu raccolga quello di cui hai bisogno.";
				close;
			}
			delitem .@fruit,.@make;
			delitem 713,.@make;
			set Zeny, Zeny-.@total_zeny;
			getitem .@juice,.@make;

			mes "[Mercante Marx Hansen]";
			mes "Eccolo! Succo fresco e delizioso come promesso. Dovrebbe essere molto rinfrescante e gradevole.";
			next;
			mes "[Mercante Marx Hansen]";
			mes "Beh allora...";
			mes "Ancora.";
			close;
		case 2:
			mes "[Mercante Marx Hansen]";
			mes "Prima che gli esseri umani potessero sviluppare società così vaste, raccoglievano frutta dagli alberi per sopravvivere. La frutta potrebbe essere stata la benedizione della natura che ci ha permesso di esistere nel mondo.";
			next;
			mes "[Mercante Marx Hansen]";
			mes "Dato che la vita è diventata così prospera, la generazione più giovane sembra non mangiare più frutta. Così, ho iniziato a pensare a modi per rendere la frutta più facile da mangiare.";
			next;
			mes "[Mercante Marx Hansen]";
			mes "Ho realizzato che quando fai il succo di frutta è più comodo da mangiare e ha un sapore molto migliore.";
			next;
			mes "# Informazioni sui Succhi di Frutta #";
			mes "^CC4E5C- Succo di mele -^000000";
			mes "Mela x 1 ea, Bottiglia vuota x 1 ea, 3 zeny.";
			mes "^E3CF57- Succo di banana -^000000";
			mes "Banana x 1 ea, Bottiglia Vuota x 1 ea, 3 zeny.";
			mes "^ED9121- Succo di carota -^000000";
			mes "Carota x 1 unità, Bottiglia Vuota x 1 unità, 3 zeni.";
			mes "^CC00FF- Succo d'uva -^000000";
			mes "Uva x 1 ea, Bottiglia vuota x 1 ea, 3 zeny.";
			close;
		case 3:
			mes "[Mercante Marx Hansen]";
			mes "Ciao!";
			mes "Se visiti";
			mes "qualcuno, parlaci!";
			close;
		}
	}
	else {
		mes "[Mercante Marx Hansen]";
		mes "Benvenuto.";
		mes "Come posso aiutarla?";
		next;
		if (select("Parla:Annulla") == 1) {
			mes "[Mercante Marx Hansen]";
			mes "Prima che gli esseri umani potessero sviluppare società così vaste, raccoglievano frutta dagli alberi per sopravvivere. La frutta potrebbe essere stata la benedizione della natura che ci ha permesso di esistere nel mondo.";
			next;
			mes "[Mercante Marx Hansen]";
			mes "Dato che la vita è diventata così prospera, la generazione più giovane sembra non mangiare più frutta. Così, ho iniziato a pensare a modi per rendere la frutta più facile da mangiare.";
			next;
			mes "[Mercante Marx Hansen]";
			mes "Ho realizzato che quando fai il succo di frutta è più comodo da mangiare e ha un sapore molto migliore.";
			close;
		}
		mes "[Mercante Marx Hansen]";
		mes "Ciao!";
		mes "Se visiti";
		mes "qualcuno, parlaci!";
		close;
	}
