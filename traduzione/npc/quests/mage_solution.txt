pay_arche,122,100,0	script	Dollshoi	88,{
	mes "[Gilda Maghi]";
	mes "Vuoi una Soluzione? Mmh, dammi 50 Zeny e almeno un Tubetto da Prova Vuoto.";
	next;
	if (select("Va bene, affare fatto.:No, dimenticalo.") == 1) {
		mes "[Gilda Maghi]";
		if (Zeny < 50) {
			mes "Ehi! Non hai abbastanza soldi per coprire la mia spesa di 50 Zeny.";
			close;
		}
		if (countitem(1092) == 0) {
			mes "Non puoi trasportare soluzioni senza un flacone! Portami un Tubetto Vuoto.";
			close;
		}
		delitem 1092,1; //Empty_Cylinder
		set Zeny, Zeny-50;
		getitem 1089,1; //Payon_Potion
	}
	close;
}
