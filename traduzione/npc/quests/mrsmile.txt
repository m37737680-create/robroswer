	mes "[Ragazza Sorridente]";
	mes "Ciao ~ Ciao ~";
	mes "Questo è Smile Assistance.";
	mes "Come posso aiutarla?";
	next;
	switch(select("^3355FFSignor Sorriso^000000 ?:Costruire ^3355FFMr. Smile^000000 :Smetti")) {
	case 1:
		mes "[Ragazza Sorridente]";
		mes "Evento nazionale tenuto dal comando di ^5577FFLa maestà Tristram III^000000,";
		mes "che intende incoraggiare la nazione del Regno di Rune-Midgarts";
		mes " per un'atmosfera più piacevole!";
		mes "Sono ^3355FF' Assistenza Sorrisi '^000000,";
		mes "chi guida l'evento nazionale con il nome di ";
		mes "^3355FFSmile throughout the Rune-Midgarts Kingdom~^000000";
		next;
		mes "[Ragazza Sorridente]";
		mes "Con articoli semplici e facili da trovare,";
		mes "Posso fornirti";
		mes "^3355FF' Signor Smile '^000000.'";
		mes "I requisiti sono i seguenti.";
		next;
		mes "[Ragazza Sorridente]";
		mes "^3355FF10 Jellopy^000000";
		mes "^3355FF10 Fluff^000000";
		mes "^3355FF10 Clover^000000";
		next;
		mes "[Ragazza Sorridente]";
		mes "Con questo evento";
		mes "Tutti saranno felici e sorrideranno,";
		mes "ritrovarsi con altre persone,";
		mes "E cercheremo di rendere Ragnarok il gioco più divertente del mondo.";
		close;
	case 2:
		if (countitem(909) > 9 && countitem(914) > 9 && countitem(705) > 9) {
			mes "[Ragazza Sorridente]";
			mes "Congratulazioni!";
			mes "Ora per favore prendi questo Mr.Smile.";
			next;
			delitem 909,10; //Jellopy
			delitem 914,10; //Fluff
			delitem 705,10; //Clover
			getitem 2278,1; //Mr_Smile
			mes "[Ragazza Sorridente]";
			mes "Sua maestà, Tristram III";
			mes "ha promesso di fare del suo meglio per rendere Ragnarok migliore e più divertente.";
			next;
			mes "Il fatto menzionato sopra";
			mes "annunciato da";
			mes "l'Ufficio di Informazioni Pubbliche del Regno di Rune-Midgards.";
			close;
		}
		else {
			mes "[Ragazza Sorridente]";
			mes "Oh - sfortunatamente";
			mes "Non hai portato";
			mes "abbastanza articoli per Mr. Smile.";
			mes "^3355FF10 Jellopy^000000";
			mes "^3355FF10 Fluff^000000";
			mes "^3355FF10 Clover^000000";
			mes "Si prega di verificare i requisiti sopra.";
			close;
		}
	case 3:
		mes "[Ragazza Sorridente]";
		mes "Grazie per averci visitato.";
		mes "Noi ..";
		mes "Il Regno di Rune-Midgard";
		mes "sempre cerca di fare Ragnarok";
		mes "gioco migliore e più divertente.";
		next;
		mes "[Ragazza Sorridente]";
		mes "Ti chiediamo sinceramente";
		mes "cooperare.";
		mes "Il fatto menzionato sopra";
		mes "fu annunciato dal Ufficio di Informazione Pubblica del Regno di Rune-Midgards.";
		close;
	}
}
