		close;
	}
}

// Number Match Game :: gef_yagu
//============================================================
geffen,122,77,4	script	Nia#yagu	700,{
	mes "[Nia]";
	mes "Ciao, sono Nia, la quinta";
	mes "discepolo di Mawong, il";
	mes "maestro del gioco. Ora,";
	mes "come vorresti giocare un";
	mes "Gioco di abbinamento numeri con me?";
	next;
	switch(select("Gioco di abbinamento numeri?:Pratica Abbinamento Numerico:Gioca a Abbina Numeri:Detentore del record:Annulla")) {
	case 1:
		mes "[Nia]";
		mes "Ah, beh, l'obiettivo di";
		mes "il gioco del Numero Match consiste nel";
		mes "indovina il numero che sto pensando";
		mes "di. Certo, ce ne sono alcuni";
		mes "regole che limitano i numeri";
		mes "che posso scegliere tra.";
		next;
		mes "[Nia]";
		mes "Tutti i numeri che faccio";
		mes "deve essere composto da 3 cifre, non posso";
		mes "usa 0 per qualsiasi delle cifre";
		mes "e non posso usare lo stesso";
		mes "numero per qualsiasi delle cifre";
		mes "più di una volta.";
		next;
		mes "[Nia]";
		mes "Quindi, per esempio, non posso";
		mes "usa i numeri 103, 112,";
		mes "252, o 701. Quei numeri";
		mes "o hanno uno 0, o usano";
		mes "lo stesso numero nelle cifre";
		mes "più di una volta. Ho capito ora?";
		next;
		mes "[Nia]";
		mes "Ora, hai solo 5 tentativi,";
		mes "e dopo ogni ipotesi, ti dirò";
		mes "quanti numeri hai indovinato";
		mes "correttamente, seguito dal numero di cifre indovinate correttamente in";
		mes "il posto decimale corretto.";
		next;
		mes "[Nia]";
		mes "Per esempio, diciamo che il";
		mes "il numero che mi viene in mente è 168.";
		mes "Se la tua prima ipotesi è 678, allora hai indovinato due delle cifre,";
		mes "Sei e otto, correttamente. Tuttavia,";
		mes "solo l'8 è al posto giusto.";
		next;
		mes "[Nia]";
		mes "Sai, sarebbe una buona idea";
		mes "se provi la Modalità Pratica";
		mes "prima. Poi, quando sei più";
		mes "comodo, gioca il vero";
		mes "Gioco di abbinamento numeri. Non dimenticare che la velocità conta per il tuo punteggio!";
		next;
		mes "[Nia]";
		mes "Ah, se per caso";
		mes "creare un nuovo record in il";
		mes "Gioco di abbinamento numeri, tu";
		mes "far registrare il tuo nome,";
		mes "quindi non perdere questa occasione per";
		mes "fama e forse anche fortuna~";
		close;
	case 2:
		mes "[Nia]";
		mes "Ottimo, facciamo una pratica";
		mes "Gioco di abbinamento numeri. Per favore";
		mes "Prova a indovinare il mio numero, ok?";
		mes "Se vuoi annullare, inserisci un numero maggiore di 1.000.";
		set .@practice,1;
		break;
	case 3:
		mes "[Nia]";
		mes "Ottimo, giochiamo ad Abbinare i Numeri!";
		mes "Avrai 5 possibilità per indovinare il numero che sto pensando.";
		mes "Se vuoi annullare, inserisci un numero maggiore di 1.000.";
		set .@practice,0;
		break;
	case 4:
		set .@name_record$, $050908_minus1_yagu$;
		if (.@name_record$ == "") //custom
			set .@name_record$,"(null)";
		set .@score_record, $050908_minus1_yagu;
		set .@score_min, .@score_record / 60;
		set .@score_sec, .@score_record % 60;
		mes "[Nia]";
		mes "^ff0000" + .@name_record$ + "^000000 è il";
		mes "detentore del record per il Numero";
		mes "Gioco di abbinamento con un tempo di";
		mes "^ff0000" + .@score_min + " minuti, " + .@score_sec + " secondi^000000.";
		close;
	case 5:
		mes "[Nia]";
		mes "Prenditi cura di te";
		mes "avventuriero. Se mai";
		mes "mi va di giocare a un Numero";
		mes "Gioco di abbinamento, vieni e parla";
		mes "a me, va bene? Ci vediamo~";
		close;
	}
	next;

	// Generate 3 digits of a number.
	while(1) {
		set .@yagu100, rand(1,9);
		set .@yagu10, rand(1,9);
		set .@yagu1, rand(1,9);
		if ((.@yagu100 != .@yagu10) && (.@yagu100 != .@yagu1) && (.@yagu10 != .@yagu1)) {
			// Store the actual number (unofficial).
			set .@yagu, (100*.@yagu100) + (10*.@yagu10) + .@yagu1;
			break;
		}
	}

	// The official time calculation is based on ServerTime,
	// an int HHMMSS equivalent to atoi(gettimestr("%H%M%S",7)).
	// UNIX time is really a better way to do this.
	set .@start_time, gettimetick(2);

	while(1) {

		// Guess a number that meets all conditions.
		while(1) {
			if (input(.@input,100,999) != 0) {
				mes "[Nia]";
				mes "Hai annullato il";
				mes "Gioco di abbinamento numeri.";
				close;
			}
			set .@input100, .@input / 100;
			set .@input10, (.@input % 100) / 10;
			set .@input1, .@input % 10;
			if ((.@input100 > 0) && (.@input10 > 0) && (.@input1 > 0)) {
				if ((.@input100 != .@input10) && (.@input100 != .@input1) && (.@input10 != .@input1))
					break;
			}
			mes "[Nia]";
			mes "Hai inserito uno dei";
			mes "cifre come 0, o inserisci il";
			mes "stesso valore numerico in";
			mes "le cifre più di una volta.";
			mes "Inserisci un altro tentativo.";
			next;
		}

		set .@retry, .@retry+1;
		mes "[Nia]";
		mes "^ff0000"+strcharinfo(0)+"^000000,";
		mes "la tua ipotesi è ^0000ff"+.@input+"^000000.";
		mes "Dammi un momento per";
		mes "Genera i tuoi risultati.";
		next;

		// Correct digits in correct locations.
		set .@strike,0;
		if (.@yagu100 == .@input100)
			set .@strike, .@strike+1;
		if (.@yagu10 == .@input10)
			set .@strike, .@strike+1;
		if (.@yagu1 == .@input1)
			set .@strike, .@strike+1;

		// Correct digits in incorrect locations.
		set .@ball,0;
		if ((.@yagu100 == .@input10) || (.@yagu100 == .@input1))
			set .@ball, .@ball+1;
		if ((.@yagu10 == .@input100) || (.@yagu10 == .@input1))
			set .@ball, .@ball+1;
		if ((.@yagu1 == .@input100) || (.@yagu1 == .@input10))
			set .@ball, .@ball+1;

		if (.@strike == 3) {
			if (.@practice) {
				mes "[Nia]";
				mes "Sì, il numero che";
				mes "Pensai fosse ^ff0000"+.@yagu+"^000000!";
				mes "Congratulazioni, tu";
				mes "ho appena vinto questa pratica";
				mes "gioco! Ora dovresti essere";
				mes "pronto per il vero~";
				close;
			} else {

				// Calculate final time and check against the previous record.
				set .@end_time, gettimetick(2);
				if (.@end_time < .@start_time)
					set .@end_time, .@end_time + 3600;
				set .@total_time, .@end_time - .@start_time;
				set .@total_min, .@total_time / 60;
				set .@total_sec, .@total_time % 60;
				set .@name_record$, $050908_minus1_yagu$;
				if (.@name_record$ == "") //custom
					set .@name_record$,"(null)";
				set .@score_record, $050908_minus1_yagu;
				set .@score_min, .@score_record / 60;
				set .@score_sec, .@score_record % 60;

				if (.@score_record >= .@total_time || !.@score_record) { //custom (second condition)
					set $050908_minus1_yagu$, strcharinfo(0);
					set $050908_minus1_yagu, .@total_time;
					mes "[Nia]";
					mes "Ottimo! Hai indovinato";
					mes "il mio numero, che era";
					mes "^ff0000" + .@yagu + "^000000. Hai finito in";
					mes "^ff0000" + .@total_min + " minuti, " + .@total_sec + " secondi^000000,";
					mes "superando il precedente record";
					mes "di ^ff0000" + .@score_min + " minuti, " + .@score_sec + " secondi^000000.";
					next;
					mes "[Nia]";
					mes "^ff0000" + .@name_record$ + "^000000";
					mes "metti quel vecchio disco,";
					mes "ma ora sei il";
					mes "nuova persona da battere nel";
					mes "Gioco di abbinamento. Congratulazioni";
					mes "per stabilire un nuovo record!";
					close;
				}
				mes "[Nia]";
				mes "Ottimo lavoro, hai indovinato";
				mes "il mio numero! La risposta, di";
				mes "corso, era ^ff0000" + .@yagu + "^000000. Ci sei riuscito";
				mes "indovinare questo numero correttamente";
				mes "in ^ff0000" + .@total_min + " minuti, " + .@total_sec + " secondi^000000.";
				next;
				mes "[Nia]";
				mes "^ff0000" + .@name_record$ + "^000000 è";
				mes "il corrente Numero Match";
				mes "detentore del record con un tempo";
				mes "di ^ff0000" + .@score_min + " minuti, " + .@score_sec + " secondi^000000.";
				mes "Buona fortuna e spero che tu";
				mes "potrei rompere questo record un giorno~";
				close;
			}
		} else {
			if (.@practice) {
				mes "[Nia]";
				mes "La tua ipotesi ha ^ff0000" + .@ball + "di ^000000";
				mes "numeri corretti. ^ff0000" + .@strike + "di ^000000";
				mes "le cifre del tuo indovinello hanno";
				mes "il numero corretto in il";
				mes "collocazione corretta dei numeri";
				mes "Allora, riprova~";
				next;
			} else {
				mes "[Nia]";
				mes "Il tuo ultimo tentativo aveva ^ff0000" + .@ball + "^000000";
				mes "dei numeri corretti,";
				mes "e ^ff0000" + .@strike + "^000000 cifre nella tua ipotesi";
				mes "aveva il numero corretto in";
				mes "il corretto posizionamento del dito.";
				switch(.@retry) {
					case 1: mes "Ti rimangono quattro tentativi."; break;
					case 2: mes "Ti rimangono tre tentativi."; break;
					case 3: mes "Ti rimangono due tentativi."; break;
					case 4: mes "Ti resta solo un tentativo..."; break;
					case 5: mes "La risposta era ^ff0000" + .@yagu + "^000000."; close;
				}
				next;
			}
		}
	}
}
