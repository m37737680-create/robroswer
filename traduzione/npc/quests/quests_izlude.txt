// Edgar's Offer :: iz_npc
//============================================================
-	script	::Edgar_izlude	-1,{
	if (MISC_QUEST & 16) {
		mes "[Edgar]";
		mes "Allora stai andando di nuovo in Alberta? Lasciami darti lo stesso sconto e addebitare solo 250 Zeny, proprio come l'ultima volta.";
		mes "Come suona?";
		next;
		if (select("Va bene~!:Perché mi stai essendo così gentile?!") == 1) {
			if (Zeny < 250) {
				mes "[Edgar]";
				mes "Um...";
				mes "Questo non è";
				mes "soldi sufficienti.";
				mes "Perché non vai";
				mes "prendere più contanti?";
				close ;
			}
			set Zeny, Zeny-250;
			warp "alberta",195,164;
			end;
		}
		mes "[Edgar]";
		mes "È semplicemente il modo in cui sono. Questo, e il tuo sorriso diavolesco mi ricordano il mio amato figlio biondo che ha lasciato casa anni fa per diventare marinaio sulla propria nave. Che Dio lo benedica, ovunque lui sia.";
		close;
	}
	mes "[Edgar]";
	mes "La mia città, Izlude, è collegata all'Alberta dal porto ad Ovest. C'è così tanto traffico tra noi che quasi divento un albertiano. Hehehe~";
	next;
	mes "[Edgar]";
	mes "C'è questo tizio che conosco abbastanza bene, Phelix, che vive in Alberta. Quel tizio è davvero avaro... Costa per tutto!";
	next;
	mes "[Edgar]";
	mes "Ma è davvero un bravo ragazzo e ama aiutare le persone. Ha un buon cuore e ti darà il suo supporto se rispetti il suo prezzo.";
	next;
	mes "[Edgar]";
	mes "Di recente, le persone dell'Alberta dicono che sta davvero cercando di aiutare la gente e che la sua richiesta di Jellopies è solo una copertura. Beh, dovresti dare un'occhiata a quello che ha da offrire.";
	next;
	if (select("Mi puoi dire come arrivare in Alberta?:Fine Conversazione") == 1) {
		mes "[Edgar]";
		mes "Eh? Beh, puoi usare i piedi e semplicemente camminare. Ma se hai dei soldi, vorrei suggerirti di prendere una nave.";
		next;
		if (select("Ok, ho capito.:Ma sono stanco di camminare e non ho soldi!") == 1) {
			mes "[Edgar]";
			mes "Va bene, prenditi cura di te~";
			close;
		}
		mes "[Edgar]";
		mes "Va bene...";
		mes "Non vuoi camminare E non hai soldi, ma vuoi comunque andare lì? Oh cielo...";
		next;
		mes "[Edgar]";
		mes "Va bene va bene. Io, essendo il capitano di una nave, posso portarti lì a un prezzo più basso. Come suona 250 Zeny?";
		next;
		if (select("Va bene~!:Bah, che truffa!!") == 1) {
			set MISC_QUEST,MISC_QUEST | 16;
			if (Zeny < 250) {
				mes "[Edgar]";
				mes "Um...";
				mes "Questo non è";
				mes "soldi sufficienti.";
				mes "Vai e prendi";
				mes "altri";
				close;
			}
			set Zeny, Zeny-250;
			warp "alberta",195,164;
			end;
		}
		mes "[Edgar]";
		mes "Ragazzo oh ragazzo,";
		mes "se pensi";
		mes "è una copia sgradita...";
		close;
	}
	mes "[Edgar]";
	mes "Sì, va bene.";
	mes "A dopo~";
	close;
}
