prt_church,173,23,4	script	Cleric	79,{
	mes "[Acolita Klift]";
	mes "Ah. Fratelli! Il compito di";
	mes "prendersi cura del nostro agnello smarrito peso";
	mes "e ti?";
	mes "Sono qui per assisterti.";
	next;
	switch (select("^3355FF Su l'abilità nascosta dell'accolito ^000000 ..:^3355FF Luce sacra ^000000 addestramento:Termina conversazione")) {
	case 1:
		mes "[Acolita Klift]";
		mes "I nostri membri del clero";
		mes "imparare naturalmente una competenza come";
		mes "man mano che maturano. Man mano che si avvicinano ";
		mes "i loro anni come acoliti più anziani";
		mes "questa abilità speciale.";
		next;
		mes "[Acolita Klift]";
		mes "Di fronte a un grande pericolo e";
		mes "una lotta senza fine contro il male,";
		mes "i nostri fratelli hanno molto bisogno di assistenza.";
		mes "^3355FF' Luce Sacra '^000000 è quell'abilità.";
		mes "Per acquisire questa abilità per te stesso,";
		mes "richiede un po' di lavoro.";
		next;
		mes "[Acolita Klift]";
		mes "Per acquisire la capacità, devi trovare questi oggetti.";
		mes "^FF33551 Opal^000000";
		mes "^FF33551 Crystal Blue^000000";
		mes "^FF33551 Rosario^000000 ";
		next;
		mes "[Acolita Klift]";
		mes "Per poter anche solo usare questa abilità,";
		mes "si deve avere sufficiente esperienza.";
		mes "^FF3355 Livello di Lavoro 30^000000";
		mes "è necessario imparare questa abilità.";
		next;
		mes "[Acolita Klift]";
		mes "Per quanto riguarda un prete, a causa dei loro";
		mes "già un'ampia esperienza,";
		mes "sono in grado di acquisire questa abilità in qualsiasi";
		mes "livello di lavoro.";
		close;
	case 2:
		if (getskilllv("AL_HOLYLIGHT") == 1) {
			mes "[Acolita Klift]";
			mes "Fratello, tu già possiedi";
			mes "la capacità di ` Holy Light '";
			mes "Non puoi acquisire una capacità che";
			mes "già possedere";
			mes "Prego che tu stia usando";
			mes "questa abilità per il lavoro di bene";
			close;
		}
		if ((countitem(727) > 0) && (countitem(991) > 0) && (countitem(2608) > 0) && (JobLevel > 29 || (BaseJob == Job_Priest || BaseJob == Job_Monk)) && (BaseClass == Job_Acolyte)) {
			mes "[Acolita Klift]";
			mes "La tua fede si è rivelata degna";
			mes "per che tu ottenga la capacità ' Luce Sacra '.";
			mes "La tua abilità è adeguata";
			mes "per usare questa abilità.";
			mes "Usalo con saggezza. . .";
			next;
			delitem 727,1; //White_Jewel
			delitem 991,1; //Crystal_Blue
			delitem 2608,1; //Rosary
			skill "AL_HOLYLIGHT",1,SKILL_PERM;
			mes "[Acolita Klift]";
			mes "Ora conosci ' Luce Sacra '";
			mes "Puoi usare questa abilità solo nel";
			mes "migliore condotta . . . . .";
			close;
		}
		mes "[Acolita Klift]";
		mes "Oh, è chiaro. . .";
		mes "Non sei ancora pronto a";
		mes "ricevere l'abilità ' Luce Sacra '";
		next;
		mes "[Acolita Klift]";
		mes "Devi ascoltare attentamente quello che";
		mes "è necessario per questa abilità.";
		mes "Se ascolti attentamente,";
		mes "puoi imparare ciò che ti manca.";
		mes "Affinché tu possa migliorare te stesso.";
		close;
	case 3:
		mes "[Acolita Klift]";
		mes ". . . . .";
		mes "Comprendo il tuo zelo.";
		mes "Hai ancora molto tempo per";
		mes "praticare e acquisire esperienza.";
		mes "Benedizioni su di te . . . . .";
		close;
	}
}
