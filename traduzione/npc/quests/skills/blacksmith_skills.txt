			mes "diventa troppo pieno per posizionare";
			mes "un oggetto di peso 500 dentro ad esso.";
			mes "Quando finisci questo compito,";
			mes "ritorna e riportami.";
			close;
		}
		mes "[Goodman]";
		mes "Tornato già, eh?";
		mes "Va bene, lascio controllare";
		mes "le cose che stai portando.";
		mes "Vediamo, Inventario, Inventario...";
		next;
		mes "[Goodman]";
		mes "Ottimo lavoro: sei passato.";
		mes "Ti rendi conto ora che il";
		mes "tempo ed esperienza spesi in";
		mes "raccogliere questi oggetti vale";
		mes "più dei loro prezzi in zeny?";
		next;
		mes "[Goodman]";
		if (Upper == 1)
			mes "Credo che i Maestri Smith";
		else
			mes "Credo che i fabbri";
		mes "dovrebbe prima e soprattutto essere";
		mes "artigiani. Essere ricchi e famosi";
		mes "è semplicemente un effetto collaterale di quello.";
		if (Upper == 1)
			mes "Gli onorevoli Maestri Smith sanno che il valore sta nell'artigianato...";
		else
			mes "Gli onorevoli fabbri sanno che il valore sta nell'artigianato...";
		next;
		if (Upper == 1) {
			mes "[Goodman]";
			mes "Ora, se non metti troppo";
			mes "molta importanza su un articolo's";
			mes "quanto vale in termini di zeny, tu";
			mes "essere in grado di lasciarli andare molto";
			mes "più facilmente. Guardami fare questo.";
			mes "Vedi? Ora anche tu puoi farlo.";
			next;
		}
		mes "[Goodman]";
		mes "La capacità che hai imparato è";
		mes "chiamato Avidità, ma spero";
		mes "lo usi in un modo che";
		mes "è diverso dal suo nome.";
		mes "Non dimenticare che, e spero";
		mes "crei veri capolavori.";
		skill "BS_GREED",1,SKILL_PERM;
		if (Upper == 1)
			BLACK_SK2 = 4;	// note: official script is using another variable when a transcendent character is doing the quest for the first time
		else
			BLACK_SK2 = 2;
		close;
	}
	mes "[Goodman]";
	mes "Come hai usato";
	mes "la abilità dell'Avarizia? Speriamo.";
	mes "è stato comodo in";
	mes "i tuoi sforzi per forgiare beni";
	mes "di maggiore qualità per tutti";
	mes "dei tuoi compagni.";
	if (getskilllv("BS_GREED") == 0)	// debug
		skill "BS_GREED",1,SKILL_PERM;
	close;
}
