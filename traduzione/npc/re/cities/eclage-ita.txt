ecl_in02,164,56,3	script	Receptionist#Laphine	835,{
	mes "[Receptionist]";
	mes "Benvenuti alla Clinica Eclage.";
	next;
	switch (select("Salva posizione:Riposo:Termina la conversazione")) {
	case 1:
		mes "[Receptionist]";
		mes "La clinica è stata impostata come punto di salvataggio. Stai attento però~";
		savepoint "ecl_in02",162,50,1,1;
		close;
	case 2:
		mes "[Receptionist]";
		mes "Ripristina la tua mente e il tuo corpo nella nostra clinica. È pieno dell'energia pulita di Eclage.";
		mes "Normalmente addebitiamo 1 Splendide Coin, ma offriamo uno sconto speciale per i viaggiatori. Il tuo totale è 5.000 zeny. ";
		next;
		switch (select("Paga con Splendide Coin:Paga con Zeny:Lasci perdere")) {
		case 1:
			if (!countitem(6081)) {
				mes "[Receptionist]";
				mes "Sembra che tu non abbia abbastanza monete. Vuoi controllare di nuovo il tuo account, per favore?";
				close;
			}
			mes "[Receptionist]";
			mes "Sentiti a casa~";
			close2;
			delitem 6081,1; //6081
			percentheal 100,100;
			warp "ecl_in02",167,49;
			end;
		case 2:
			if (Zeny < 5000) {
				mes "[Receptionist]";
				mes "Sembra che tu non disponga di risorse sufficienti per procedere. Ti dispiace controllare di nuovo il tuo account?";
				close;
			}
			mes "[Receptionist]";
			mes "Sentiti a casa~";
			close2;
			Zeny = Zeny - 5000;
			percentheal 100,100;
			warp "ecl_in02",167,49;
			end;
		case 3:
			close;
		}
	case 3:
		close;
	}
}
