ecl_in02,164,56,3	script	Receptionist#Laphine	835,{
	mes "[Receptionista]";
	mes "Benvenuti alla Clinica Eclage.";
	next;
	switch (select("Salva posizione:Riposo:Finisci la conversazione")) {
	case 1:
		mes "[Receptionista]";
		mes "La clinica è stata impostata come punto di salvataggio. Fai attenzione però~";
		savepoint "ecl_in02",162,50,1,1;
		close;
	case 2:
		mes "[Receptionista]";
		mes "Ripristina mente e corpo presso la nostra clinica. È piena dell'energia pulita di Eclage.";
		mes "Ci addebito normalmente 1 Splendide Coin, ma offriamo uno sconto speciale per i viaggiatori. Il tuo totale è di 5.000 zeny. ";
		next;
		switch (select("Paga con Splendide Coin:Paga con Zeny:Dimenticalo")) {
		case 1:
			if (!countitem(6081)) {
				mes "[Receptionista]";
				mes "Sembra che non abbia abbastanza monete. Potrebbe controllare nuovamente il suo conto, per favore?";
				close;
			}
			mes "[Receptionista]";
			mes "Senti a tuo agio~";
			close2;
			delitem 6081,1; //6081
			percentheal 100,100;
			warp "ecl_in02",167,49;
			end;
		case 2:
			if (Zeny < 5000) {
				mes "[Receptionista]";
				mes "Sembra che non abbia abbastanza risorse per procedere. Le dispiacerebbe controllare nuovamente il suo account?";
				close;
			}
			mes "[Receptionista]";
			mes "Senti a tuo agio~";
			close2;
			Zeny = Zeny - 5000;
			percentheal 100,100;
			warp "ecl_in02",167,49;
			end;
		case 3:
			close;
		}
	case 3:
		close;
	}
}
