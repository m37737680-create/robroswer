// Channel Warper :: channel
//============================================================
function	script	F_IzludeChannel	{
	mes "[Channel Warp ufficiale]";
	mes "Izlude è un posto molto vivace, vero?";
	mes "Sono qui per aiutarti a far sì che Izlude si senta più a suo agio!";
	mes "Posso inviarti un'altra copia di Izlude.";
	mes "Vuoi andare a un'altra copia?";
	next;
	set .@i, select(getarg(0)+":Non importa");
	if (.@i < 6) {
		setarray .@maps$[1],"izlude","izlude_a","izlude_b","izlude_c","izlude_d";
		savepoint .@maps$[.@i],128,98,1,1;
		warp .@maps$[.@i],128,98;
		end;
	}
	close;
}
