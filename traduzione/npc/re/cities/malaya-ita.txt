// Jeepneys
//============================================================
function	script	F_Malaya_Jeepney	{
	.@mapName$        = getarg(0);
	.@passengers      = getarg(1);

	// set the other messages of varying amount)
	for (.@i = 5; .@i < getargcount(); .@i++) {
		setd(".@msgJeepneyInfo$[" + (.@i - 5) + "]", getarg(.@i));
	}

	if (malaya_hi < 10) {
		mes "[Autista della Jeepney]";
		mes getarg(2);
		close;
	} else if ((malaya_hi >= 10) && (malaya_hi < 20)) {
		mes "[Autista della Jeepney]";
		mes getarg(3);
		close;
	} else {
		mes "[Autista della Jeepney]";
		mes getarg(4);
		next;
	}
	switch(select("Consiglio [Passeggero " + getmapusers(.@mapName$) + "/" + .@passengers +"]:Jeepney?:Ci sono altri Jeepney?:Ah... sì...")) {
	case 1:
		if (getmapusers(.@mapName$) >= .@passengers) {
			mes "[Autista della Jeepney]";
			mes "Temo che la Jeepney sia piena.";
			mes "Mi dispiace, ma che ne dici di qualche altro Jeepney?";
			close;
		} else {
			mes "[Autista della Jeepney]";
			mes "Buona giornata.";
			close2;
			warp .@mapName$,29,24;
			end;
		}
	case 2:
		// iterate through all the jeepney information for this specific NPC
		for (.@i = 0; .@i < getarraysize(.@msgJeepneyInfo$); .@i++) {
			mes "[Autista della Jeepney]";
			mes getd(".@msgJeepneyInfo$[" + .@i + "]");

			if (.@i < (getarraysize(.@msgJeepneyInfo$) - 1)) {
				next;
			}
		}
		close;
	case 3:
		mes "[Autista della Jeepney]";
		mes "OH! Naturalmente sono operativi anche altri Jeepney.";
		next;
		mes "[Autista della Jeepney]";
		mes "A Port Malaya ci sono Jeepney da 12, 30 e 60 passeggeri, di cui 3 operativi ciascuno.";
		next;
		mes "[Autista della Jeepney]";
		mes "Vuoi conoscere la posizione di altre auto?";
		next;
		switch (select("Vorrei sapere dov'è l'autovettura 12.:Vorrei sapere dov'è l'autovettura 30.:Vorrei sapere dov'è l'autovettura 60.")) {
		case 1:
			viewpoint 1, 237, 240, 1, 0xF7E009;
			viewpoint 1, 67 , 44 , 2, 0xF7E009;
			viewpoint 1, 282, 129, 3, 0xF7E009;
			set .@zif_in, 12;
			break;
		case 2:
			viewpoint 1, 134, 250, 4, 0xF7E009;
			viewpoint 1, 341, 153, 5, 0xF7E009;
			viewpoint 1, 293, 290, 6, 0xF7E009;
			set .@zif_in, 30;
			break;
		case 3:
			viewpoint 1, 242, 221, 7, 0xF7E009;
			viewpoint 1, 62 , 245, 8, 0xF7E009;
			viewpoint 1, 257, 58 , 9, 0xF7E009;
			set .@zif_in, 60;
			break;
		}
		mes "[Autista della Jeepney]";
		mes .@zif_in+" passenger Jeepney's location has been marked on your map.";
		next;
		break;
	}
	mes "[Autista della Jeepney]";
	mes "Buon viaggio.";
	close;
}
