moscovia,161,76,3	script	Moscovia Guide#01mosc	959,{
	mes "Guida a Mosca";
	mes "Un paradiso sul mare,";
	mes "belle onde e caldo sole.";
	mes "^8B4513Moscovia^000000 ti aspetta.";
	mes "Sto supportando per offrire informazioni su qui.";
	mes "Se vuoi chiedere qualsiasi cosa sentiti libero di dirmelo.";
	F_Navi("[Moscovia Guide]");
	set .@str$,"Is there anything else I can do for you?";
	while (1) {
		next;
		switch(select("[ Kafra Dipendente ]:Palazzo:[ Cappella ]:Negozio di Armi:Negozio di armature:Negozio di strumenti:[ Locanda ]:Ritorno all'Alberta:Rimuovi i segni dalla mini-mappa:Annulla")) {
		case 1:
			mes "Guida a Mosca";
			mes "Il ^0000FFKafra Employee^000000 ha";
			mes "segnalato sulla tua minipeta";
			mes "Controlla la tua minamappa.";
			mes "Hai bisogno di qualcos'altro?";
			viewpoint 1,223,191,0,0x0A82FF;
			break;
		case 2:
			mes "Guida a Mosca";
			mes "Nel "+F_Navi("Palazzo","Mosca,255,140","^006400")+" troverai il nostro Signore lo Zar e altri servi.";
			mes .@str$;
			viewpoint 1,255,140,1,0x0A82FF;
			break;
		case 3:
			mes "Guida a Mosca";
			mes "Il "+F_Navi("Cappella","Mosca,253,179","^008080")+" si trova qui.";
			mes .@str$;
			viewpoint 1,253,179,2,0xAAFF00;
			break;
		case 4:
			mes "Guida a Mosca";
			mes "Puoi comprare armi dai migliori artigiani di Moscovia se vai al "+F_Navi("Negozio di armi","Mosca,186,191","^FF1493")+".";
			mes .@str$;
			viewpoint 1,186,191,3,0x008080;
			break;
		case 5:
			mes "Guida a Mosca";
			mes "Troverai solo l'attrezzatura migliore alla Moscovia "+F_Navi("Negozio di armature","Mosca,204,169","^8B4513")+".";
			mes .@str$;
			viewpoint 1,204,169,4,0xB200FF;
			break;
		case 6:
			mes "Guida a Mosca";
			mes "Il "+F_Navi("Laboratorio di strumenti","Mosca,221,176","^FF1493")+" si trova appena a sud del centro città.";
			mes "Puoi trovare ogni tipo di cose di cui hai bisogno per i tuoi viaggi.";
			mes .@str$;
			viewpoint 1,221,176,5,0xFF1493;
			break;
		case 7:
			mes "Guida a Mosca";
			mes "L'osteria "+F_Navi("Albero di Erbe Appiccicose","Mosca,227,206")+" è appena a nord dal centro città.";
			mes "Se hai bisogno di riposare, non c'è posto migliore dove soggiornare.";
			mes .@str$;
			viewpoint 1,227,206,6,0x8B4513;
			break;
		case 8:
			mes "Guida a Mosca";
			mes "Puoi trovare il";
			mes F_Navi("Moscovia P.R. Officer","moscovia,166,53")+" at the docks if";
			mes "vuoi tornare in Alberta.";
			viewpoint 1,166,53,7,0x00BFFF;
			break;
		case 9:
			mes "Guida a Mosca";
			mes "Rimuoverò tutti i segni dalla tua minima mappa.";
			mes .@str$;
			viewpoint 2,1,1,0,0xFFFFFF;
			viewpoint 2,1,1,1,0xFFFFFF;
			viewpoint 2,1,1,2,0xFFFFFF;
			viewpoint 2,1,1,3,0xFFFFFF;
			viewpoint 2,1,1,4,0xFFFFFF;
			viewpoint 2,1,1,5,0xFFFFFF;
			viewpoint 2,1,1,6,0xFFFFFF;
			viewpoint 2,1,1,7,0xFFFFFF;
			break;
		case 10:
			mes "Guida a Mosca";
			mes "Buon viaggio!";
			close;
		}
	}
}
