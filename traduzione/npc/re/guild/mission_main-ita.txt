function brief;
function mob_brief;
function calcexp;
function calcjobexp;

	if (!checkweight(1201,3)){
		mes "Hai troppi oggetti per procedere con la missione.";
		mes "Per favore, alleggerisci il carico e riprova.";
		close;
	}

	.@zone$ = strnpcinfo(2);
	if(.@zone$ == "gqaldeg"){
		.@zoneid = 3;
		.@edition = 1;
	} else if(.@zone$ == "gqaru"){
		.@zoneid = 4;
		.@edition = 2;
	} else if(.@zone$ == "gqgefg"){
		.@zoneid = 2;
		.@edition = 1;
	} else if(.@zone$ == "gqpayg"){
		.@zoneid = 1;
		.@edition = 1;
	} else if(.@zone$ == "gqprtg"){
		.@zoneid = 0;
		.@edition = 1;
	} else if(.@zone$ == "gqsch"){
		.@zoneid = 5;
		.@edition = 2;
	} else if(.@zone$ == "gqtealde"){
		.@zoneid = 6;
		.@edition = 3;
	} else if(.@zone$ == "gqteprt"){
		.@zoneid = 7;
		.@edition = 3;
	} else {
		// Custom error message.
		debugmes "Guild mission script called from an unknown agit region [" + .@zone$ + "].";
		end;
	}
	.@npcname$ = .npcname$[.@zoneid];

	mes .@npcname$;
	mes "Forniamo missioni e missioni giornaliere che puoi completare nella regione di " + .region$[.@zoneid] + ".";
	mes "Ci sono semplici missioni giornaliere e missioni che devi completare solo entro le ore dell'assedio. Quindi, pensaci~";
	next;
	switch(select("Conquistare " + .region$[.@zoneid], "Missioni giornaliere del dungeon della gilda", "A proposito di missioni", "Non ne ho bisogno.")){
		case 1: // Conquering.
			.@gid = getcharid(2);
			if(.@gid && is_guild_leader() == true){
				.@time_check1 = checkquest(.conquer_delay_questid[.@zoneid], PLAYTIME);
				if(.@time_check1 == 0){
					mes .@npcname$;
					mes "È passata meno di un'ora dall'ultima volta che abbiamo controllato!";
					mes "Torna quando il tempo di attesa è finito.";
					next;
					mes .@npcname$;
					mes "Se hai la fortuna dalla tua parte, avrai buoni risultati.";
					close;
				} else if(.@time_check1 == 2){
					erasequest .conquer_delay_questid[.@zoneid];
					mes .@npcname$;
					mes "Ho confermato che il tempo di attesa per la conquista delle missioni è terminato.";
					mes "Eliminerò i record non necessari per te.";
					close;
				} else {
					.@siege_check = (.@edition == 1 ? agitcheck() : .@edition == 2 ? agitcheck2() : agitcheck3());
					.@time = atoi(gettimestr("%H%M",4));
					.@after_time = ((.weekday[.@zoneid] == gettime(4)) && ((.@time > .start_time[.@zoneid]) && (.@time < .end_time[.@zoneid])));

					if(.@siege_check && .@after_time){
						mes .@npcname$;
						mes "In base a ^4d4dffil numero di fortezze che la tua gilda ha conquistato in questo preciso momento,^000000 ti darò una ricompensa.";
						mes "Vuoi controllare adesso? Sei consapevole che ^4d4dffil numero può cambiare anche mentre parliamo,^000000 vero?";
						next;
						if(select("Tornerò più tardi.","Controlliamo subito!") == 1){
							mes .@npcname$;
							mes "Giusto. La vita è tutta una questione di tempismo.";
							mes "Torna quando splende di più.";
							close;
						} else {
							.@total = 0;
							for(.@i = 0; .@i < 5; ++.@i){
								.@total += (.@gid == getcastledata(.castle$[.@i+.@zoneid*5],CD_GUILD_ID));
							}

							if(!.@total){
								mes .@npcname$;
								mes "Sfortunatamente, non esiste alcuna fortezza che la tua gilda abbia conquistato.";
								mes "Riprova!";
								close;
							} else {
								mes .@npcname$;
								if(.@total == 5)
									mes "Incredibile.";
								else
									mes "Attualmente, il numero totale di fortezze conquistate dalla tua gilda";
								for(.@i = 0; .@i < 5; ++.@i){
									mes getcastlename(.castle$[.@i+.@zoneid*5]) + (.@gid == getcastledata(.castle$[.@i+.@zoneid*5],CD_GUILD_ID) ? ": ^4d4dffOccupied^000000" : "");
								}
								if(.@total == 5)
									mes "La tua gilda ha vinto " + .region$[.@zoneid] + "!";
								else
									mes "È " + .@total + "!";
								next;
								mes .@npcname$;
								mes "Ti darò una ricompensa di conseguenza. Spero che ne farai buon uso.";
								mes "Hai un'ora di attesa fino al prossimo rapporto. Tienilo a mente e riprova.";
								setquest .conquer_delay_questid[.@zoneid];
								getitem 6489,.@total;
								getitem 6615,.@total;
								getitem 12888,.@total;
								close;
							}
						}
					} else {
						mes .@npcname$;
						mes "Mi dispiace ma puoi completare questa missione solo durante le ore dell'assedio. Temo di non poter fare niente qui.";
						close;
					}
				}
			} else {
				mes .@npcname$;
				mes "Mi dispiace ma le missioni di conquista sono riservate ai maestri di gilda.";
				mes "I membri dovrebbero invece provare le missioni giornaliere dei dungeon di gilda.";
				close;
			}
			break;
		case 2: // Daily Quests
			mes .@npcname$;
			mes "Queste sono missioni che puoi completare nei sotterranei della gilda " + .region$[.@zoneid] + " zona.";
			mes "Puoi cancellarli se vuoi. Sentiti libero di sfidare te stesso.";
			next;
			.@cancel_index = (.@edition == 1 ? 3 : 2);
			for(.@i = 0; .@i < 3; ++.@i){
				if(.daily_hunting_questid[.@zoneid*3 + .@i])
					.@menu$ = .@menu$ + .region$[.@zoneid] + " Missione " + (.@i+1) + (isbegin_quest(.daily_hunting_questid[.@zoneid*3 + .@i]) ? " ^4d4dffControlla il risultato^000000" : "") + ":";

			.@sel = select(.@menu$) - 1;
			.@index = .@zoneid*3 + .@sel;
			if(.@sel == .@cancel_index){
				mes .@npcname$;
				mes "Non devi avere fretta poiché queste missioni si svolgono nei sotterranei della gilda.";
				mes "Sentiti libero di provarlo ogni volta che vuoi.";
				close;
			}
			if(BaseLevel < .level_req[.@index]){
				mes .@npcname$;
				mes "Hmm, sfortunatamente, questa missione è limitata agli utenti con livello ^4d4dff " + .level_req[.@index] + "Solo ^000000 o superiore.";
				mes "Il tuo livello non è abbastanza alto.";
				close;
			}
			.@hunting = checkquest(.daily_hunting_questid[.@index], HUNTING);
			.@playtime = checkquest(.daily_delay_questid[.@index], PLAYTIME);
			if(.@hunting > -1){
				if(.@hunting == 2){
					mes .@npcname$;
					mes "Oh, hai fatto un buon lavoro.";
					mes "Non è molto, ma spero che possa essere utile.";
					setquest .daily_delay_questid[.@index];
					erasequest .daily_hunting_questid[.@index];
					getexp calcexp(.@zoneid,.@index),calcjobexp(.@edition);

				} else {
					mes .@npcname$;
					mes "Hmm? Sembra che tu non abbia ancora completato la missione..";
					mes "C'è qualche problema?";
					next;

						mes .@npcname$;
						mes "Non sono sicuro per quanto tempo potrai utilizzare il dungeon della gilda, ma buona fortuna.";
						close;
					} else {

						mes "Beh, nessuno ti obbliga.";
						mes "Fai quello che ti piace fare.";
						mes "Allora annullerò la missione.";
						erasequest .daily_hunting_questid[.@index];
						close;
					}
				}
			} else {
				if(.@playtime == 0 || .@playtime == 1){
					mes .@npcname$;
					mes "La regola è impostata qui in modo che ti sia consentito completare solo una missione ogni giorno.";
					mes "Vorrei che tornassi quando il tempo di attesa sarà finito.";
					close;
				} else if(.@playtime == 2){
					erasequest .daily_delay_questid[.@index];
					mes .@npcname$;
					mes "Ho confermato che il tempo di attesa per questa missione è finito.";
					mes "Eliminerò i record utilizzati per la conferma.";
					mes "Ora prova a completare la missione!";
					close;
				} else {
					mes .@npcname$;
					mes "Questo è il " + .ordinal$[.@sel] + " missione dentro " + .region$[.@zoneid] + ".";
					brief(.daily_hunting_questid[.@index]);
					next;
					mes .@npcname$;
					if(mob_brief(.daily_hunting_questid[.@index]))
						mes "Ti ricompenserò di conseguenza una volta completata questa missione.";
					else
						mes "Ti darò punti esperienza in cambio.";
					next;
					if(select("Accettalo.","Rifiutalo.") == 1){
						mes .@npcname$;
						mes "L'ingresso al dungeon della gilda è limitato. Ma non è nemmeno che tu non riesca a trovare un modo per arrivarci.";
						mes "Pregherò per il tuo successo nella lotta contro quei mostri.";
						setquest .daily_hunting_questid[.@index];
						close;
					} else {
						mes .@npcname$;
						mes "Bene, fai come desideri.";
						close;
					}
				}
			}
			break;
		case 3: // About quests
			mes .@npcname$;
			mes "Non è affatto difficile.";
			mes "Offro una ricompensa ^4d4dffa seconda di quante fortezze hai conquistato.^000000";
			mes "^4d4dffIl tipo e la qualità della ricompensa dipendono dal numero di fortezze conquistate nel momento stesso in cui controllo.^000000";
			next;
			mes .@npcname$;
			mes "E ovviamente controllo solo durante le ore di assedio.";
			mes "^4d4dffLa ricezione di un premio è seguita da un'ora di attesa. Quindi faresti meglio a capire anche il momento giusto.^000000";
			next;
			mes .@npcname$;
			mes "Se sei il maestro di una gilda che sta attualmente partecipando a battaglie d'assedio, dovresti provarci.";
			next;
			mes .@npcname$;
			mes "Le missioni giornaliere dei dungeon di gilda consistono in missioni di combattimento che puoi completare nei dungeon di gilda.";
			mes "È semplice.";
			close;
		case 4: // I don't need it.
			mes .@npcname$;
			mes "Sei davvero qualcos'altro~!";
			close;
	}
	end;

	function calcexp {
		.@zoneid = getarg(0);
		.@index = getarg(1);
		.@maxlevel = .max_level[.@index];
		.@isboss = (.@zoneid < 4 && !((.@index+1)%3));
		
		if(!.@isboss && BaseLevel > .@maxlevel){
			return 1000 * (100 + ((.@maxlevel - 100) * 3));
		} else {
			return 1000 * (100 + ((BaseLevel - 100) * 3)) * (1 + .@isboss);
		}
	}
	
	function calcjobexp {
		.@edition = getarg(0);
		
		if(.@edition == 3){
			if(jobcanentermap("te_prtcas01")){
				if (JobLevel > 50)
					return 1000 * (50 + ((JobLevel - 50) * 3));
				else
					return 5000;
			} else {
				debugmes "Unexpected case for Job: " + Class + "(" + jobname(Class) + ") for gld_mission#core::calcjobexp";
				return 0;
			}
		} else if(.@edition == 2 ) {
			if(JobLevel > 50)
				return 1000 * (50 + ((JobLevel - 50) * 3));
			else
				return 3000;
		} else {
			return 5000 * JobLevel;
		}
		return 0;
	}
