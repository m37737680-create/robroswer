function mob_brief;
function calcexp;
function calcjobexp;

	if (!checkweight(1201,3)){
		mes "Hai troppi oggetti per procedere con la missione.";
		mes "Alleggerisci il tuo carico e riprova.";
		close;
	}

	.@zone$ = strnpcinfo(2);
	if(.@zone$ == "gqaldeg"){
		.@zoneid = 3;
		.@edition = 1;
	} else if(.@zone$ == "gqaru"){
		.@zoneid = 4;
		.@edition = 2;
	} else if(.@zone$ == "gqgefg"){
		.@zoneid = 2;
		.@edition = 1;
	} else if(.@zone$ == "gqpayg"){
		.@zoneid = 1;
		.@edition = 1;
	} else if(.@zone$ == "gqprtg"){
		.@zoneid = 0;
		.@edition = 1;
	} else if(.@zone$ == "gqsch"){
		.@zoneid = 5;
		.@edition = 2;
	} else if(.@zone$ == "gqtealde"){
		.@zoneid = 6;
		.@edition = 3;
	} else if(.@zone$ == "gqteprt"){
		.@zoneid = 7;
		.@edition = 3;
	} else {
		// Custom error message.
		debugmes "Guild mission script called from an unknown agit region [" + .@zone$ + "].";
		end;
	}
	.@npcname$ = .npcname$[.@zoneid];

	mes .@npcname$;
	mes "Forniamo missioni e questre giornaliere che puoi completare nella regione di " + .region$[.@zoneid] + ".";
	mes "Ci sono anche semplici missioni giornaliere e missioni che dovete completare solo durante le ore del sacco. Quindi, pensaci~";
	next;
	switch(select("Conquistare " + .region$[.@zoneid], "Missioni Giornaliere della Gilda Sotterraneo", "Sulle quest", "Non ne ho bisogno.")){
		case 1: // Conquering.
			.@gid = getcharid(2);
			if(.@gid && is_guild_leader() == true){
				.@time_check1 = checkquest(.conquer_delay_questid[.@zoneid], PLAYTIME);
				if(.@time_check1 == 0){
					mes .@npcname$;
					mes "È passato meno di un'ora dall'ultima volta che abbiamo controllato!";
					mes "Torna quando il tempo di attesa è finito.";
					next;
					mes .@npcname$;
					mes "Se hai fortuna dalla tua parte, avrai buoni risultati.";
					close;
				} else if(.@time_check1 == 2){
					erasequest .conquer_delay_questid[.@zoneid];
					mes .@npcname$;
					mes "Ho confermato che il tempo di attesa per le missioni di conquista è terminato.";
					mes "Eliminerò i record non necessari per te.";
					close;
				} else {
					.@siege_check = (.@edition == 1 ? agitcheck() : .@edition == 2 ? agitcheck2() : agitcheck3());
					.@time = atoi(gettimestr("%H%M",4));
					.@after_time = ((.weekday[.@zoneid] == gettime(4)) && ((.@time > .start_time[.@zoneid]) && (.@time < .end_time[.@zoneid])));

					if(.@siege_check && .@after_time){
						mes .@npcname$;
						mes "Secondo ^4d4dffil numero di fortezze che la vostra gilda ha conquistato in questo preciso momento, ^000000 vi darò una ricompensa.";
						mes "Vuoi controllare adesso? Sei consapevole che ^4d4dffil numero può cambiare anche mentre parliamo,^000000 giusto?";
						next;
						if(select("Tornerò più tardi.","Controlliamo subito!") == 1){
							mes .@npcname$;
							mes "Giusto. La vita è tutta una questione di tempismo.";
							mes "Torna quando brilla di più.";
							close;
						} else {
							.@total = 0;
							for(.@i = 0; .@i < 5; ++.@i){
								.@total += (.@gid == getcastledata(.castle$[.@i+.@zoneid*5],CD_GUILD_ID));
							}

							if(!.@total){
								mes .@npcname$;
								mes "Sfortunatamente, non c'è nessuna fortezza che la vostra gilda abbia conquistato.";
								mes "Riprova!";
								close;
							} else {
								mes .@npcname$;
								if(.@total == 5)
									mes "Incredibile.";
								else
									mes "Attualmente, il numero totale di fortezze che la vostra gilda ha conquistato";
								for(.@i = 0; .@i < 5; ++.@i){
									mes getcastlename(.castle$[.@i+.@zoneid*5]) + (.@gid == getcastledata(.castle$[.@i+.@zoneid*5],CD_GUILD_ID) ? ": ^4d4dffOccupied^000000" : "");
								}
								if(.@total == 5)
									mes "La tua gilda ha conquistato " + .region$[.@zoneid] + "!";
								else
									mes "è " + .@total + "!";
								next;
								mes .@npcname$;
								mes "Ti darò un premio di conseguenza. Spero che tu ne faccia buon uso.";
								mes "Hai un'ora di tempo di attesa fino al prossimo rapporto. Tieni questo a mente e riprova.";
								setquest .conquer_delay_questid[.@zoneid];
								getitem 6489,.@total;
								getitem 6615,.@total;
								getitem 12888,.@total;
								close;
							}
						}
					} else {
						mes .@npcname$;
						mes "Mi dispiace ma puoi completare questa missione solo durante le ore del sacco. Temo di non poter fare nulla qui.";
						close;
					}
				}
			} else {
				mes .@npcname$;
				mes "Mi dispiace ma le missioni di conquista sono solo per i maestri di gilda.";
				mes "I membri dovrebbero provare le missioni quotidiane della dungeon del gilda invece.";
				close;

			break;
		case 2: // Daily Quests
			mes .@npcname$;
			mes "Queste sono missioni che puoi completare nelle dungeon di gilda di " + .region$[.@zoneid] + " area.";
			mes "Puoi annullarli se vuoi. Sentiti libero di sfidarti.";
			next;
			.@cancel_index = (.@edition == 1 ? 3 : 2);
			for(.@i = 0; .@i < 3; ++.@i){

					.@menu$ = .@menu$ + .region$[.@zoneid] + " Missione " + (.@i+1) + (isbegin_quest(.daily_hunting_questid[.@zoneid*3 + .@i]) ? " ^4d4dffCheck the result^000000" : "") + ":";

			.@index = .@zoneid*3 + .@sel;
			if(.@sel == .@cancel_index){

				mes "Non devi avere fretta dato che queste missioni si svolgono nelle dungeon di gilda.";
				mes "Sentiti libero di provarci quando vuoi.";
				close;
			}

				mes .@npcname$;
				mes "Mmh, purtroppo, questa missione è limitata agli utenti con livello ^4d4dfflevel " + .level_req[.@index] + "^000000 o solo superiore.";
				mes "Il tuo livello non è abbastanza alto.";
				close;
			}
			.@hunting = checkquest(.daily_hunting_questid[.@index], HUNTING);

			if(.@hunting > -1){
				if(.@hunting == 2){
					mes .@npcname$;
					mes "Hai fatto un buon lavoro.";
					mes "Non è molto, ma spero che sarà utile.";

					mes .@npcname$;

					mes "C'è qualche problema?";

						mes "Non sono sicuro per quanto tempo potrai usare la dungeon della gilda, ma buona fortuna.";
						close;

						mes .@npcname$;
						mes "Beh, nessuno ti sta forzando.";
						mes "Fai quello che vuoi fare.";
						mes "Annullerò la missione allora.";
						erasequest .daily_hunting_questid[.@index];
						close;
					}
				}
			} else {
				if(.@playtime == 0 || .@playtime == 1){
					mes .@npcname$;
					mes "La regola è stabilita qui affinché tu sia autorizzato a completare una missione al giorno.";
					mes "Vorrei che tu tornassi quando il tempo di attesa sarà finito.";
					close;
				} else if(.@playtime == 2){
					erasequest .daily_delay_questid[.@index];
					mes .@npcname$;
					mes "Ho confermato che il tempo di attesa per questa missione è finito.";
					mes "Eliminerò i record utilizzati per la conferma.";
					mes "Ora, prova a completare la missione!";
					close;
				} else {
					mes .@npcname$;
					mes "Questo è il " + .ordinal$[.@sel] + " missione in " + .region$[.@zoneid] + ".";
					brief(.daily_hunting_questid[.@index]);
					next;
					mes .@npcname$;
					if(mob_brief(.daily_hunting_questid[.@index]))
						mes "Ti ricompenserò di conseguenza al completamento di questa missione.";
					else
						mes "Ti darò punti esperienza in cambio.";
					next;
					if(select("Accettalo.","Rifiutalo.") == 1){
						mes .@npcname$;
						mes "L'ingresso alla dungeon della gilda è ristretto. Ma non è che tu non possa trovare un modo per arrivarci, neanche.";
						mes "Pregherò per il tuo successo nel combattere quei mostri.";
						setquest .daily_hunting_questid[.@index];
						close;
					} else {
						mes .@npcname$;
						mes "Beh, fai come vuoi.";
						close;
					}
				}
			}
			break;
		case 3: // About quests
			mes .@npcname$;
			mes "Non è affatto difficile.";
			mes "Ti do una ricompensa ^4d4dffa seconda di quante fortezze hai conquistato.^000000";
			mes "^4d4dffIl tipo e la qualità del premio dipendono dal numero di fortezze conquistate nel momento esatto in cui controllo.^000000";
			next;
			mes .@npcname$;
			mes "E naturalmente, controllo solo durante le ore del assedio.";
			mes "^4d4dffReceiving a reward is followed by an hour of waiting time. So you'd better figure out the right timing as well.^000000";
			next;
			mes .@npcname$;
			mes "Se sei un maestro di una gilda che sta partecipando a battaglie d'assedio, dovresti provarci.";
			next;
			mes .@npcname$;
			mes "Le missioni giornaliere della gilda sono composte da missioni di combattimento che puoi completare nelle dungeon della gilda.";
			mes "È semplice.";
			close;
		case 4: // I don't need it.
			mes .@npcname$;
			mes "Sei davvero un altro!";
			close;
	}
	end;

	function calcexp {
		.@zoneid = getarg(0);
		.@index = getarg(1);
		.@maxlevel = .max_level[.@index];
		.@isboss = (.@zoneid < 4 && !((.@index+1)%3));
		
		if(!.@isboss && BaseLevel > .@maxlevel){
			return 1000 * (100 + ((.@maxlevel - 100) * 3));
		} else {
			return 1000 * (100 + ((BaseLevel - 100) * 3)) * (1 + .@isboss);
		}
	}
	
	function calcjobexp {
		.@edition = getarg(0);
		
		if(.@edition == 3){
			if(jobcanentermap("te_prtcas01")){
				if (JobLevel > 50)
					return 1000 * (50 + ((JobLevel - 50) * 3));
				else
					return 5000;
			} else {
				debugmes "Unexpected case for Job: " + Class + "(" + jobname(Class) + ") for gld_mission#core::calcjobexp";
				return 0;
			}
		} else if(.@edition == 2 ) {
			if(JobLevel > 50)
				return 1000 * (50 + ((JobLevel - 50) * 3));
			else
				return 3000;
		} else {
			return 5000 * JobLevel;
		}
		return 0;
	}
