dali02,142,81,3	script	Dimensional Device#air1	PORTAL,{
	if (checkquest(15050,PLAYTIME) == 2) {
		erasequest 15050;// Memorial Dungeon: Airship Assault
		erasequest 15051;
	}
	else if (checkquest(15051,PLAYTIME) == 2) {// Memorial Dungeon: Airship Assault
		mes "^ff0000Non è possibile rientrare nel Memorial Dungeon entro 23 ore dall'ultimo ingresso.^000000";
		close;
	}
	.@party_id = getcharid(1);
	.@p_name$ = getpartyname(.@party_id);
	.@md_name$ = "Assault on the Airship";

	if (BaseLevel < 125) {
		mes "^ff0000Gli utenti di livello inferiore a 125 non possono entrare in questo dungeon^000000";
		close;
	}
	if (.@party_id < 1) {
		mes "^ff0000Per partecipare, devi creare o unirti a un gruppo di 1 o più membri.^000000";
		close;
	}
	if (is_party_leader() == false) {
		mes "^ff0000Questo Memorial Dungeon è solo per 1 persona. Può entrare solo il leader del gruppo.^000000";
		close;
	}
	.@s = select( "Accendi il dispositivo", "Usa il dispositivo", "Cancellare" );
	if (is_party_leader(.@party_id) == false) {
		mes "^ff0000Questo Memorial Dungeon è solo per 1 persona. Può entrare solo il leader del gruppo.^000000";
		close;
	}
	switch(.@s) {
	case 1:
		if (instance_create(.@md_name$) < 0) {
			mes "Nome del partito: " + getpartyname( getcharid(1) );
			mes "Nome del personaggio: " + strcharinfo(0);
			mes "^0000ff" + .@md_name$ + "^000000 - Creazione fallita!";
			close;
		}
		mes "^ff0000Dopo che il dispositivo di quotatura si è formato^000000";
		mes "^ff0000fai clic per accedere nuovamente.^000000";
		close;
	case 2:
		switch( instance_enter(.@md_name$) ) {
		case IE_OK:
			if (isbegin_quest(15050) == 0) {
				setquest 15050;// Memorial Dungeon: Airship Assault
				setquest 15051;// Memorial Dungeon: Airship Assault
			}
			mapannounce "dali02", """ + strcharinfo(0) + ", member of the party " + getpartyname( getcharid(1) ) + " entered the instance " + .@md_name$ + ".", bc_map,"0x00ff99";
			specialeffect2 EF_WARP;
			// warp "1@air1",244,73;
			end;
		case IE_NOMEMBER:
			mes "^ff0000Per partecipare, devi creare o unirti a un gruppo di 1 o più membri.^000000";
			close;
		case IE_NOINSTANCE:
			mes "^ff0000Il leader del gruppo non ha utilizzato il dispositivo.^000000";
			close;
		case IE_OTHER:
			mes "Si è verificato un errore sconosciuto.";// Custom
			close;
		}
	case 3:
		end;
	}
}
