ma_scene01,174,179,4	script	Taho	541,{
	disable_items;
	if (checkweight(1201,1) == 0) {
		mes "Stai trasportando troppi oggetti. Torna dopo averne tolti alcuni.";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "Stai trasportando troppo peso. Torna dopo aver ridotto il peso.";
		close;
	}
	if (BaseLevel < 140) {
		mes "[Taho]";
		mes "EHI. Questo è un posto troppo pericoloso per te. Non puoi mantenerti al sicuro. Esci da qui e vai subito al villaggio!";
		close;
	}
	if (malaya_bakona1 < 7 && malaya_bakona2 < 15) {
		mes "[Taho]";
		mes "Verremo uccisi tutti da quel Bakonawa. Ah, dobbiamo ucciderlo ma sfortunatamente non siamo preparati.";
		close;
	}

	set .@party_id,getcharid(1);
	set .@md_name$, "Bakonawa Lake";

	set .@baku_time, checkquest(12278,PLAYTIME);
	if (.@baku_time == -1) {
		if (checkquest(12279) == -1) setquest 12279;
	}
	if (!.@party_id) {
		mes "[Taho]";
		mes "Non c'è nessuna compagnia che può venire con te? O anche se vuoi andare da solo, vieni dopo aver organizzato una festa.";
		close;
	}
	if (.@baku_time == -1) {
		if (is_party_leader() == true) {
			mes "[Taho]";
			mes "Sei il capo degli assassini Bakonawa? La strada non è buona quindi dobbiamo scendere dal dirupo con una corda, va bene?";
			next;
			set .@i, select("Per favore, intreccia una corda.:Ora andrò giù.:Cancellare.");
		} else {
			mes "[Taho]";
			mes "Sei venuto per uccidere Bakonawa. Il capo del tuo gruppo deve aiutarmi a tessere la corda per scendere.";
			next;
			set .@i, select(":Ora andrò giù.:Cancellare.");
		}
		switch(.@i) {
		case 1:
			if (instance_create(.@md_name$) < 0) {
				mes "[Taho]";
				mes "Oh, la corda è caduta. Devo farne uno nuovo.";
				close;
			}
			mes "[Taho]";
			mes "Adesso sto tessendo, così potrai scendere quando avrò finito.";
			close;
		case 2:
			switch(instance_enter(.@md_name$)) {
			case IE_OTHER:
				mes "[Taho]";
				mes "Si è verificato un errore sconosciuto.";
				close;
			case IE_NOINSTANCE:
				mes "[Taho]";
				mes "La corda non è stata ancora tessuta. Aspetta un po'.";
				close;
			case IE_NOMEMBER:
				mes "[Taho]";
				mes "I civili non possono entrare.";
				close;
			case IE_OK:
				mapannounce "ma_scene01",""Un membro del partito, "+strcharinfo(0)+" of the party "+getpartyname(.@party_id)+" is entering the dungeon, "+.@md_name$+".",bc_map,"0x00ff99"; //FW_NORMAL 12 0 0
				setquest 12278;
				//warp "1@ma_b",64,51;
				close;
			}
		case 3:
			close;
		}
	} else if (.@baku_time == 0 || .@baku_time == 1) {
		mes "[Taho]";
		mes "La corda usata per arrivare al Lago è già rotta. Ora stiamo ripulendo l'ambiente circostante, quindi torna tra un po'.";
		close;
	} else if (.@baku_time == 2) {
		mes "[Taho]";
		mes "È tutto ripulito qui intorno. Ora si può scendere nuovamente dopo aver recuperato una corda.";
		erasequest 12278;
		close;
	}
}
