ma_fild02,312,317,5	script	Guard#buwaya_cave	570,{
	if (BaseLevel < 130) {
		mes "[Guardia]";
		mes "Persone sotto il ^ff0000livello 130^000000";
		mes "non posso entrare in questo posto";
		mes "Questo posto è pericoloso. Per favore, torna indietro.";
		close;
	}

	set .@party_id,getcharid(1);
	set .@md_name$,"Buwaya Cave";

	if (!.@party_id) {
		mes "[Guardia]";
		mes "Buwaya è pericoloso.";
		mes "Per favore, torna dopo aver formato a";
		mes "squadra con altre persone.";
		close;
	}
	if (is_party_leader() == true) {
		mes "[Guardia]";
		mes "Questo posto è pericoloso. Per favore, torna indietro.";
		mes "Se non hai affari qui, per favore torna indietro.";
		next;
		while(1) {
			switch(select("Perché è così pericoloso?:Sono qui per dare la caccia a Buwaya.:Termina la conversazione.")) {
			case 1:
				mes "[Guardia]";
				mes "Recentemente c'è stato un caso";
				mes "dove gli abitanti dei villaggi sono scomparsi.";
				mes "Dopo aver indagato sulle tracce, ";
				mes "abbiamo scoperto il mostro gigante Buwaya";
				mes "era responsabile.";
				next;
				mes "[Guardia]";
				mes "Sono stati inviati soldati e mercenari ";
				mes "dare la caccia a Buwaya, ma non lo sono mai stati";
				mes "completamente riuscito.";
				next;
				mes "[Guardia]";
				mes "Buwaya sta ancora rapindo";
				mes "gli abitanti del villaggio fino ad oggi.";
				mes "Per favore, riunisci alcune persone per sbarazzarti di Buwaya.";
				next;
				break;
			case 2:
				set .@playtime, checkquest(4229,PLAYTIME);
				if (.@playtime == -1) {
					// fall through
				} else if (.@playtime == 0 || .@playtime == 1) {
					mes "[Guardia]";
					mes "Buwaya è ancora nascosto.";
					mes "Anche se entri adesso, Buwaya non uscirà. ";
					mes "Per favore, torna più tardi.";
					close;
				} else {
					erasequest 4229;
					// fall through
				}
				if (instance_create(.@md_name$) < 0) {
					mes "[Guardia]";
					mes "Il nome del partito è... "+getpartyname(.@party_id)+".";
					mes "Il leader del partito è... "+strcharinfo(0);
					mes "^0000ff"+.@md_name$+"^000000 non può essere aperto adesso.";
					mes "Per favore riprova un attimo più tardi.";
					close;
				}
				mes "[Guardia]";
				mes "Aprirò il tunnel a ^0000ff"+.@md_name$+"^000000.";
				mes "Ti auguro buona fortuna.";
				close;
			case 3:
				mes "[Guardia]";
				mes "Questo posto è pericoloso. Non puoi entrare.";
				close;
			}
		}
	} else {
		mes "[Guardia]";
		mes "Fammi parlare con il leader del partito.";
		mes "Per favore, richiamate il leader.";
		close;
	}
}

ma_fild02,315,323,0	script	Cave Entrance#buwaya	45,2,2,{
	end;
OnTouch:
	if (BaseLevel < 130) {
		mes "[Guardia]";
		mes "Persone sotto il ^ff0000livello 130^000000";
		mes "non posso entrare in questo posto";
		mes "Questo posto è pericoloso. Per favore, torna indietro.";
		close;
	}
	set .@playtime, checkquest(4229,PLAYTIME);
	if (.@playtime == -1) {
		// fall through
	} else if (.@playtime == 0 || .@playtime == 1) {
		mes "[Guardia]";
		mes "Buwaya è ancora nascosto.";
		mes "Anche se entri adesso, Buwaya non uscirà.";
		mes "Per favore, torna più tardi.";
		close;
	} else {
		erasequest 4229;
		// fall through
	}
	switch(select("Entra.:Torna indietro.")) {
	case 1:
		switch(instance_enter("Buwaya Cave")) {
		case IE_OTHER:
			mes "[Guardia]";
			mes "Oh, adesso non è un buon momento.";
			mes "Per favore riprova un attimo dopo.";
			close;
		case IE_NOINSTANCE:
		case IE_NOMEMBER:
			mes "[Guardia]";
			mes "Questo posto è pericoloso. Per favore non entrare.";
			close;
		case IE_OK:
			mapannounce "ma_fild02",getpartyname(getcharid(1))+" party's "+strcharinfo(0)+" member began hunting Buwaya in Buwaya Cave.",bc_map,"0x00ff99"; //FW_NORMAL 12 0 0
			setquest 4229;
			//warp "1@ma_c",35,57;
			end;
		default:
			mes "- Impossibile entrare a causa di una forza non identificata. -";
			close;
		}
	case 2:
		mes "[Guardia]";
		mes "Devi dare valore alla tua vita.";
		mes "È meglio pensare attentamente prima di agire.";
		close;
	}
}
