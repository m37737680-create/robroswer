			mes "[Macchina]";

			next;
			mes "[Macchina]";
			mes "^ff0000Non è possibile accedere a questa macchina per 20 ore dopo l'ultimo accesso.^000000";

			mes "^ff0000Solo un membro del gruppo può entrare nel Memorial Dungeon.^000000";
			close;
		case IE_OK:
			setquest 13185;
			specialeffect2 EF_READYPORTAL2;

			end;

		end;
