		close;

			mes "Macchina";

			next;
			mes "Macchina";
			mes "^ff0000Questa macchina non può essere acceduta per 20 ore dopo il tuo ultimo accesso.^000000";

			close;

			setquest 13185;
			specialeffect2 EF_READYPORTAL2;
			specialeffect2 EF_PORTAL2;

			end;
