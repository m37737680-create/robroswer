	case 1:
		instance_create(.@md_name$);
		mes "[Sposare Jay]";
		mes "Il Memorial Dungeon è stato creato.";
		mes "Ora puoi entrare.";
		close;

		switch( instance_enter(.@md_name$) ) {
		case IE_NOMEMBER:
			if (is_party_leader() == true)
				end;
			mes "Solo i membri del gruppo possono entrare in questo dungeon commemorativo.";
			close;
		case IE_NOINSTANCE:
			if (is_party_leader() == true) {
				mes "Non è registrato alcun Memorial Dungeon.";
				close;
			}
			mes "Il Friday Memorial Dungeon non esiste.";
			mes "Il leader del tuo gruppo non ha ancora creato il Dungeon commemorativo.";
			close;
		case IE_OTHER:
			mes "Si è verificato un errore sconosciuto.";
			close;
		case IE_OK:
			mes "[Sposare Jay]";
			mes "Via, vai!";
			mapannounce "gef_tower", """ + strcharinfo(0) + " of the party, " + getpartyname( getcharid(1) ) + ", is entering the " + .@md_name$ + ".", bc_map, 0xFF99;
			if (isbegin_quest(12379) < 1)
				setquest 12379;
			end;
		}
		end;
	case 3:
		mes "[Sposare Jay]";
		mes "Non ti sei preparato?";
		mes "Non posso davvero farci niente, immagino";
		close;
	}
	end;

		'rand_chest++;

	}
	sleep 10000;
	monster 'map_md_gef$,0,0, getarg(1), getarg(0),1, instance_npcname("md_gef_mobs_spawn") + "::" + getarg(2);
	end;

	'step = 0;
	'hard = 0;
	'count = 0;
	'rand_chest = 0;
	'freestones = 0;
	'boss_actv = 0;
	'map_md_gef$ = instance_mapname("1@md_gef");
	end;

			delitem 25235,10; //Catalyst
			Zeny -= 100000;

			// anti-hack
			if (callfunc("F_IsEquipIDHack", .@slot, .@equip_id) || callfunc("F_IsEquipCardHack", .@slot, .@card[0], .@card[1], .@card[2], .@card[3]))
				end;

			delequip .@slot;
			.@enchant = F_enchant(1);
			specialeffect2 EF_REPAIRWEAPON;
			mes "[Collezionista amatoriale]";
			mes "IL " + mesitemlink(.@equip_id, false) + " è stato incantato " + mesitemlink(.@enchant, false) + ".";
			getitem2 .@equip_id,1,1,0,0,.@equip_card[0],.@equip_card[1],.@equip_card[2],.@enchant;
			close;
		}
		end;
	}
	else {	// reset
		if (.@equip_id == 28483) {
			.@pos = 2;
			.@success = 80;	// success royal
		} else {
			.@pos = 3;
			.@success = 20;	// success normal
		}
		if (.@equip_card[.@pos] == 0) {
			mes "[Collezionista amatoriale]";
			mes "IL " + mesitemlink(.@equip_id, false) + " non è al suo massimo incanto.";
			mes "Non riesco a reimpostare questo accessorio.";
			close;
		}
		mes "[Collezionista amatoriale]";
		mes "Tutti gli effetti incantati verranno rimossi dal " + mesitemlink(.@equip_id, false) + "?";
		mes "^FF0000C'è un " + .@success + "% di possibilità di successo. L'accessorio verrà distrutto se il ripristino fallisce!^000000";
		next;
		if (select( "Torno più tardi", "Per favore procedi") == 1)
			end;
		if (countitem(25235) < 10) {
			mes "[Collezionista amatoriale]";
			mes "Per favore, portamene 10 volte " + mesitemlink(25235, false) + "";
			close;
		}
		if (Zeny < 100000) {
			mes "[Collezionista amatoriale]";
			mes "Mi servono 100.000 zeny!";
			close;
		}
		delitem 25235,10; //Catalyst
		Zeny -= 100000;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@slot, .@equip_id) || callfunc("F_IsEquipCardHack", .@slot, .@card[0], .@card[1], .@card[2], .@card[3]))
			end;

		delequip .@slot;
		if (rand(100) > .@success) {
			specialeffect2 EF_SUI_EXPLOSION;
			mes "[Collezionista amatoriale]";
			mes "Maledizione...";
			mes "Ho fallito e l'accessorio è stato distrutto. Mi dispiace tanto.";
			close;
		}
		specialeffect2 EF_REPAIRWEAPON;
		mes "[Collezionista amatoriale]";
		mes "Ci sono riuscito.";
		mes "IL " + mesitemlink(.@equip_id, false) + " è stato inizializzato.";
		getitem2 .@equip_id,1,1,0,0,.@equip_card[0],.@equip_card[1],0,0;
		close;
	}
	end;

function	F_equip_menu {
	.@l$ = getitemname(getequipid(EQI_ACC_L));
	.@r$ = getitemname(getequipid(EQI_ACC_R));
	.@acc_l$ = .@l$ == "null" ? "" : "Left Acc - " + .@l$;
	.@acc_r$ = .@r$ == "null" ? "" : "Right Acc - " + .@r$;

	switch( select( .@acc_l$, .@acc_r$ ) ) {
	case 1:
		.@slot = EQI_ACC_L;
		break;
	case 2:
		.@slot = EQI_ACC_R;
		break;
	}
	return .@slot;
}
