moro_cav,50,64,5	script	Yellow Seed#bios	CLEAR_NPC,{
	if (BaseLevel < 160) {
		mes "^ff0000I personaggi di livello inferiore a 160 non possono entrare in questo dungeon.^000000";
		close;
	}
	if (isbegin_quest(15006) == 0) {
		mes "^ff0000Devi prima incontrare il Vecchio Errante per avere accesso a questo dungeon.^000000";
		close;
	}
	.@party_id = getcharid(1);
	if (.@party_id < 1) {
		mes "^ff0000Devi prima formare/unirti a un gruppo di almeno 1 persona per entrare in questo dungeon.^000000";
		close;
	}
	switch( checkquest(15005,PLAYTIME) ) {
	case -1:
		break;
	case 0:
	case 1:
		if (checkquest(15007,PLAYTIME) == 2) {// 5 mins to re-enter
			mes "^ff0000Non è possibile accedere a questo Memorial Dungeon per 23 ore dopo la tua ultima visita.^000000";
			close;
		}
		switch( checkquest(15008,PLAYTIME) ) {
		case -1:
			break;
		case 0:
		case 1:
			mes "^0000ffNon è possibile accedere a questo Memorial Dungeon per 23 ore dopo la tua ultima visita.^000000";
			close;
		case 2:
			break;
		}
		break;
	case 2:
		erasequest 15005;// Memorial Dungeon: Isle of Bios
		erasequest 15007;// Memorial Dungeon: Isle of Bios
		if (isbegin_quest(15008) > 0)
			erasequest 15008;// Memorial Dungeon: Isle of Bios
		break;
	}
	if (is_party_leader() == true)
		.@create$ = "Create Memorial dungeon.";
	switch( select( .@create$, "Entra nell'Isola di Bios.", "Cancellare." ) ) {
	case 1:
		instance_create("Isle of Bios");
		mes "^ff0000Creato il Dungeon commemorativo.^000000";
		mes "^ff0000Fai clic e premi nuovamente Invio.^000000";
		close;
	case 2:
		switch( instance_enter("Isle of Bios") ) {
		case IE_NOMEMBER:
		case IE_NOINSTANCE:
		case IE_OTHER:
			if (is_party_leader(.@party_id) == false) {
				mes "^ff0000Tieni presente che non è possibile accedere a questo Memorial Dungeon per 23 ore dopo la tua ultima visita.^000000";
				next;
			}
			mes "^ff0000Il leader del tuo gruppo non ha ancora creato un Crack of Time.^000000";
			close;
		case IE_OK:
			mapannounce "moro_cav", """ + strcharinfo(0) + " of the party, " + getpartyname( getcharid(1) ) + ", is entering Isle of Bios.",bc_map,"0x00FF99";
			if (isbegin_quest(15005) == 0) {
				setquest 15005;// Memorial Dungeon: Isle of Bios
				setquest 15007;// Memorial Dungeon: Isle of Bios
			}
			end;
		}
	case 3:
		end;
	}
	end;

OnInit:
	while(true) {
		sleep 500;
		specialeffect EF_ENHANCE;
	}
	end;
}
