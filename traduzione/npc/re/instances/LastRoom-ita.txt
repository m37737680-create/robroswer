				mes "[Segno]";
				mes "...mi sento molto sospettoso!";
				mes "È un po' pericoloso restare soli. Si consiglia di fare squadra!";

					setquest 11380;// Final Room
				end;
			}
			end;
		case 3:
			close3;
		}
		end;
	case 0:
	case 1:
		mes "[Segno]";
		cutin "bu_mark1.bmp",0;
		mes "...Sembri stanco.";
		mes "Visto che sei andato lì dovresti prenderti un giorno per riposarti.";
		close3;
	case 2:
		mes "[Segno]";
		cutin "bu_mark1.bmp",0;
		mes "Non posso credere che tu voglia entrare di nuovo in un posto come questo...";
		erasequest 11379;// Final Room
		close3;
	}
}
