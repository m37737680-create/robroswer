		}
		mes "[Runane]";

		mes "^ff0000Per favore, aspetti.^000000";
		close3;
	case 2:
		switch( instance_enter(.@md_name$) ) {
		case IE_NOMEMBER:
			close3;

			if (is_party_leader() == false) {

				mes "Teletrasporto...";

				mes "[Runane]";

			mes "^ff0000Your party leader has not yet created a Time Crack.^000000";
			close3;

			mes "Si è verificato un errore sconosciuto.";
			close3;
		case IE_OK:
			setquest 13182;// Nightmarish Jitterbug: Waiting

			specialeffect2 EF_PORTAL2;
			// warp "1@jtb",16,17;
			end;
		}
	case 3:
		cutin "",255;
		end;
	}
