dali,139,118,4	script	Dimensional Device#sara	10007,{
	if (BaseLevel < 99) {
		mes "[L'avventuriero Leon]";
		mes "EHI! Sei troppo giovane per giocarci! Torna quando avrai raggiunto almeno il ^0000FFLivello 99^000000!";
		close;
	}
	if ((saramemory < 1) || (saramemory == 2) || (saramemory == 3) || (saramemory == 5)) {
		mes "[L'avventuriero Leon]";
		mes "Prima di giocare con quello; Perché non vieni qui e mi parli?";
		close;
	}

	// 20 hour cooldown
	set .@ins_sara_time,checkquest(15002,PLAYTIME);
	if (.@ins_sara_time == -1) {

		set .@party_id,getcharid(1);
		set .@p_name$,getpartyname(.@party_id);
		set .@md_name$,"Sara's Memories";

		if (!.@party_id) {
			mes "[L'avventuriero Leon]";
			mes "Prima di entrare bisogna organizzare una festa! Non sai come? Digita /organizzaNOMEPARTEQUI.";
			close;
		}

		if (is_party_leader() == true)
			set .@menu$, "Boot up the dimensional device.:Use the dimensional device.:Cancel.";
		else
			set .@menu$, ":Use the dimensional device.:Cancel.";

		switch(select(.@menu$)) {
		case 1:
			if (instance_create(.@md_name$) < 0) {
				mes "Nome della festa: "+.@p_name$;
				mes "Leader del partito: "+strcharinfo(0);
				mes "^0000ff"+.@md_name$+" ^000000- Prenotazione fallita!";
				close;
			}
			mes "^FF0000Il dispositivo dimensionale si avvia in modo pulito. Usa il dispositivo per entrare nella fessura nello spazio e nel tempo.^000000";
			close;
		case 2:
			switch(instance_enter(.@md_name$)) {
			case IE_OTHER:
				mes "Si è verificato un errore sconosciuto.";
				close;
			case IE_NOINSTANCE:
				mes "La prigione commemorativa "+.@md_name$+" non esiste.";
				mes "Il leader del gruppo non ha ancora generato il dungeon.";
				close;
			case IE_NOMEMBER:
				mes "Solo i membri registrati possono accedere all'istanza "+.@md_name$+".";
				close;
			case IE_OK:
				mapannounce "dali",strcharinfo(0)+", member of the party "+.@p_name$+" entered the instance "+.@md_name$+".",bc_map,"0x00ff99";
				setquest 15002;
				end;
			}
		case 3:
			mes "^FF0000Il leader del gruppo non ha avviato il dispositivo dimensionale.^000000";
			close;
		}
	} else if (.@ins_sara_time == 0 || .@ins_sara_time == 1) {
		mes "[L'avventuriero Leon]";
		mes "Penso che tra circa 20 ore dall'ultima volta che sei entrato nella spaccatura, sarà abbastanza stabile da permetterti di attraversarla di nuovo.";
		close;
	}
	else {
		erasequest 15002;
		mes "^0000FF[ Ricordi di Sara ] Il tempo di recupero è stato cancellato. Ora è possibile utilizzare nuovamente il dispositivo dimensionale.^000000";
		close;
	}
}
