dali,139,118,4	script	Dimensional Device#sara	10007,{
	if (BaseLevel < 99) {
		mes "[Avventuriero Leon]";
		mes "Ehi! Sei troppo giovane per giocare con quello! Torna quando avrai almeno il livello ^0000FFLevel 99^000000!";
		close;
	}
	if ((saramemory < 1) || (saramemory == 2) || (saramemory == 3) || (saramemory == 5)) {
		mes "[Avventuriero Leon]";
		mes "Prima che tu vada a giocare con quello; Perché non vieni qui e mi parli?";
		close;
	}

	// 20 hour cooldown
	set .@ins_sara_time,checkquest(15002,PLAYTIME);
	if (.@ins_sara_time == -1) {

		set .@party_id,getcharid(1);
		set .@p_name$,getpartyname(.@party_id);
		set .@md_name$,"Sara's Memories";

		if (!.@party_id) {
			mes "[Avventuriero Leon]";
			mes "Prima di entrare devi organizzare una festa! Non lo sai fare? Digita /organize NOMEFESTAQUI.";
			close;
		}

		if (is_party_leader() == true)
			set .@menu$, "Boot up the dimensional device.:Use the dimensional device.:Cancel.";
		else
			set .@menu$, ":Use the dimensional device.:Cancel.";

		switch(select(.@menu$)) {
		case 1:
			if (instance_create(.@md_name$) < 0) {
				mes "Nome della festa: "+.@p_name$;
				mes "Leader del Partito: "+strcharinfo(0);
				mes "^0000ff"+.@md_name$+" ^000000- Prenotazione fallita!";
				close;
			}
			mes "^FF0000Il dispositivo dimensionale si avvia correttamente. Usa il dispositivo per entrare nella crepa nello spazio e nel tempo.^000000";
			close;
		case 2:
			switch(instance_enter(.@md_name$)) {
			case IE_OTHER:
				mes "Si è verificato un errore sconosciuto.";
				close;
			case IE_NOINSTANCE:
				mes "La dungeon commemorativa "+.@md_name$+" non esiste.";
				mes "Il capo del partito non ha ancora generato il dungeon.";
				close;
			case IE_NOMEMBER:
				mes "Solo i membri registrati possono accedere all'istanza "+.@md_name$+".";
				close;
			case IE_OK:
				mapannounce "dali",strcharinfo(0)+", member of the party "+.@p_name$+" entered the instance "+.@md_name$+".",bc_map,"0x00ff99";
				setquest 15002;
				end;
			}
		case 3:
			mes "^FF0000The party leader did not boot up the dimensional device.^000000";
			close;
		}
	} else if (.@ins_sara_time == 0 || .@ins_sara_time == 1) {
		mes "[Avventuriero Leon]";
		mes "Penso che tra circa 20 ore dopo il tuo ultimo passaggio nel varco, sarà abbastanza stabile per potervi attraversare di nuovo.";
		close;
	}
	else {
		erasequest 15002;
		mes "^0000FF[ I Ricordi di Sara ] Il tempo di ricarica è stato azzerato. È ora possibile usare nuovamente il dispositivo dimensionale.^000000";
		close;
	}
}
