		// player is leader of the party
		.@create$ = "Apply for entry to " + .@md_name$ + "";

	switch( select( .@create$, "Entra " + .@md_name$ + "" ) ) {

		cutin "ep18_maram_01.png",2;
		mes "[Maram]";
		mes "Quando l'applicazione è completa, parlami di nuovo e seleziona ^0000CD" + .@md_name$ + " entry^000000";
		if (instance_create(.@md_name$) < 0) {

			mes "Leader: " + strcharinfo(0) + "";
			mes "^0000ff" + .@md_name$ + " ^000000- Errore sconosciuto";

		}

			mes "^ff0000Si è verificato un errore sconosciuto.^000000";

			mes "[Maram]";
			mes "Sembra che tu non abbia ancora creato il dungeon. Per favore, riprova tra un momento.";
			close3;
		case IE_NOMEMBER:
			mes "[Maram]";
			mes "^ff0000Only a member of the party can enter the Memorial Dungeon.^000000";
			close;
		case IE_OK:
			mapannounce "que_thr", """ + strcharinfo(0) + " of the party, " + getpartyname( getcharid(1) ) + ", is entering " + .@md_name$ + ".", bc_map, 0xFF99;
			// warp 1@tcamp,103,237;
			end;
		}
	}
	end;

OnInit:
	questinfo( QTYPE_QUEST2, QMARK_YELLOW, "isbegin_quest(16577) == 1" );
