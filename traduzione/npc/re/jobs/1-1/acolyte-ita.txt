prt_church,184,41,4	script	Cleric#aco	60,{
	if (Upper == 1) {
		if (ADVJOB == Job_High_Priest || ADVJOB == Job_Champion) {
			if (Class == Job_Novice_High) {
				mes "[Padre Mareusis]";
				mes "Ah, sento che hai resistito";
				mes "un'esperienza di vita passata. Devi aver imparato molte cose prima di entrare nel Valhalla.";
				next;
				if (!callfunc("F_CanChangeJob")) {
					mes "[Padre Mareusis]";
					mes "Sfortunatamente, non penso che tu sia ancora pronto per diventare un Accolito. Per favore, finisci prima di apprendere tutte le competenze di base.";
					next;
					mes "[Padre Mareusis]";
					mes "Intanto,";
					mes "Aspetterò finché";
					mes "sei pronto.";
					mes "Che Dio sia";
					mes "con te.";
					close;
				}
				mes "[Padre Mareusis]";
				mes "Bene, ti do il benvenuto";
				mes "di ritorno dal Valhalla e";
				mes "ti auguro buona fortuna per il tuo";
				mes "il viaggio di una nuova vita.";
				next;
				skill "NV_TRICKDEAD",0,SKILL_PERM;
				jobchange Job_Acolyte_High;
				skill "AL_HOLYLIGHT",1,SKILL_PERM;
				mes "[Padre Mareusis]";
				mes "Ora avventurati e cerca coloro che hanno bisogno del tuo aiuto. Possa Dio illuminare il tuo cammino.";
				close;
			}else{
				mes "[Padre Mareusis]";
				mes "Ora avventurati alla ricerca di persone che hanno bisogno del tuo aiuto. Possa Dio illuminare la tua strada.";
				close;
			}
		}else{
			mes "[Padre Mareusis]";
			mes "Sento che hai sopportato un'esperienza di vita passata. Devi aver imparato molte cose prima di entrare nel Valhalla.";
			next;
			mes "[Padre Mareusis]";
			mes "Tuttavia, posso dire che non sei adatto per essere un Accolito. Per favore, ricorda chi eri nella tua vita passata e trova la tua strada.";
			close;
		}
	}
	mes "[Padre Mareusis]";
	mes "Cos'è che cerchi?";
	next;
	switch(select("Cambia il tuo lavoro in accolito.:Chiedi i requisiti per essere un accolito.:Smettila.")) {
	case 1:
		mes "[Padre Mareusis]";
		if(BaseJob != Job_Novice) {
			if(BaseJob == Job_Acolyte)
				mes "Ti senti bene oggi? Posso dire dal tuo abbigliamento che sei già un Accolito. Non stai scherzando, vero?";
			else
				mes "Mi dispiace ma sembra che tu abbia già il tuo lavoro, vero?";
			close;
		}else{
			mes "Desideri veramente diventare un servitore di Dio?";
			mes "Vediamo se sei pronto o no... Hmm...";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Padre Mareusis]";
				mes "Oh mio Dio?! Non hai ancora completato la pratica base?! Hai molta strada da fare! Torna di nuovo dopo aver aumentato il tuo livello di lavoro!";
				close;
			}
			mes "[Padre Mareusis]";
			mes "Hmm... il tuo livello lavorativo è sufficiente...";
			mes "Bene. Adesso ti darò la qualifica per diventare Accolito.";
			next;
			mes "[Padre Mareusis]";
			mes "Ricordatevi sempre di essere grati a Dio, che si prende cura di noi in ogni momento. Nel caos e nei momenti di difficoltà, affronta le tue difficoltà con fede incrollabile.";
			callfunc "Job_Change",Job_Acolyte;
			getitem 1545,1;	// N_Mace
			next;
			mes "[Padre Mareusis]";
			mes "Desidero infine congratularmi sinceramente con voi per aver perseverato nella prova della penitenza.";
			close;
		}
	case 2:
		mes "[Padre Mareusis]";
		mes "Desideri diventare un Accolito?";
		mes "Quindi, devi soddisfare i seguenti requisiti pensando che queste siano le pratiche date da Dio.";
		next;
		mes "[Padre Mareusis]";
		mes "Innanzitutto, devi raggiungere almeno il livello 9 del lavoro per principianti e apprendere tutte le abilità di base.";
		mes "Questa è la cosa più elementare da fare, quindi devi considerarla come il modo di allenarti.";
		next;
		mes "[Padre Mareusis]";
		mes "Quando pensi di aver soddisfatto questo requisito, torna di nuovo da me. Allora avrai un lavoro santo in cui potrai diffondere la volontà di Dio.";
		close;
	case 3:
		close;
	}
}
