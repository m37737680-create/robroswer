prt_church,184,41,4	script	Cleric#aco	60,{
	if (Upper == 1) {
		if (ADVJOB == Job_High_Priest || ADVJOB == Job_Champion) {
			if (Class == Job_Novice_High) {
				mes "[Padre Mareusis]";
				mes "Ah, sento che hai sopportato";
				mes "un'esperienza di vita passata. Devi aver imparato molte cose prima di entrare a Valhalla.";
				next;
				if (!callfunc("F_CanChangeJob")) {
					mes "[Padre Mareusis]";
					mes "Purtroppo, non credo che tu sia pronto per diventare un Acolita ancora. Per favore, finisci di imparare tutte le Abilità Base prima.";
					next;
					mes "[Padre Mareusis]";
					mes "Nel frattempo,";
					mes "Aspetterò finché";
					mes "sei pronto.";
					mes "Che Dio sia";
					mes "con te.";
					close;
				}
				mes "[Padre Mareusis]";
				mes "Beh, ti do il benvenuto";
				mes "tornato da Valhalla e";
				mes "ti auguro buona fortuna per il tuo";
				mes "il viaggio di una nuova vita.";
				next;
				skill "NV_TRICKDEAD",0,SKILL_PERM;
				jobchange Job_Acolyte_High;
				skill "AL_HOLYLIGHT",1,SKILL_PERM;
				mes "[Padre Mareusis]";
				mes "Ora, avventurati e cerca coloro che hanno bisogno del tuo aiuto. Che Dio illumini il tuo cammino.";
				close;
			}else{
				mes "[Padre Mareusis]";
				mes "Ora, avventurati a cercare persone che hanno bisogno del tuo aiuto. Che Dio illumini la tua strada.";
				close;
			}
		}else{
			mes "[Padre Mareusis]";
			mes "Sento che hai subito un'esperienza di vita passata. Devi aver imparato molte cose prima di entrare a Valhalla.";
			next;
			mes "[Padre Mareusis]";
			mes "Tuttavia, posso dirti che non sei adatto a essere un Acolita. Ricorda chi eri nella tua vita passata e trova il tuo sentiero.";
			close;
		}
	}
	mes "[Padre Mareusis]";
	mes "Cosa cerchi?";
	next;
	switch(select("Cambia il tuo lavoro in acolita.:Chiedere che i requisiti siano quelli di un acolita.:Smettila.")) {
	case 1:
		mes "[Padre Mareusis]";
		if(BaseJob != Job_Novice) {
			if(BaseJob == Job_Acolyte)
				mes "Ti senti bene oggi? Posso capire dal tuo abbigliamento che sei già un Acolita. Non stai scherzando, vero?";
			else
				mes "Mi dispiace ma sembra che tu abbia già il tuo lavoro, non è vero?";
			close;
		}else{
			mes "Desideri veramente diventare un servo di Dio?";
			mes "Vediamo se sei pronto o no... Mmh...";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Padre Mareusis]";
				mes "Oh mio Dio?! Non hai ancora completato la pratica di base?! Hai ancora molta strada da fare! Torna dopo aver aumentato il livello del tuo lavoro!";
				close;
			}
			mes "[Padre Mareusis]";
			mes "Mmh... il tuo livello lavorativo è sufficiente.";
			mes "Bene. Ora ti darò la qualifica per diventare un Acolita.";
			next;
			mes "[Padre Mareusis]";
			mes "Ricorda sempre di essere grato a Dio, che si prende cura di noi tutto il tempo. Nel caos e nei momenti di difficoltà, affronta le tue avversità con fede incrollabile.";
			callfunc "Job_Change",Job_Acolyte;
			getitem 1545,1;	// N_Mace
			next;
			mes "[Padre Mareusis]";
			mes "Infine, voglio congratularmi sinceramente con te per aver resistito alla tua prova di penitenza.";
			close;
		}
	case 2:
		mes "[Padre Mareusis]";
		mes "Desideri diventare un Acolita?";
		mes "Allora, devi adempiere ai seguenti requisiti pensando che queste siano le pratiche date da Dio.";
		next;
		mes "[Padre Mareusis]";
		mes "Innanzitutto, devi raggiungere almeno il Livello di Lavoro Principiante 9 e imparare tutte le Abilità Base.";
		mes "Questa è la cosa più basilare da fare, quindi devi considerarla il modo di allenarti.";
		next;
		mes "[Padre Mareusis]";
		mes "Quando pensi di aver adempiuto questo requisito, allora torna da me di nuovo. Allora avrai un lavoro sacro nel quale puoi diffondere la volontà di Dio.";
		close;
	case 3:
		close;
	}
}
