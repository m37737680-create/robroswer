	if (Upper == 1) {
		if (Class == Job_Novice_High && (ADVJOB == Job_Sniper || ADVJOB == Job_Clown || ADVJOB == Job_Gypsy)) {
			mes "[Arciere Gilda]";
			mes "Ehi, ti conosco.";
			mes "Hai fatto questo test";
			mes "prima, non lo facevi?";
			next;
			mes "[Arciere Gilda]";
			mes "Devi essere stato";
			mes "a Valhalla e rinato.";
			mes "Wow, è così impressionante!";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Arciere Gilda]";
				mes "Ehm...";
				mes "Meglio che tu impari tutte le abilità di base prima di poter diventare un Arciere.";
				next;
				mes "[Arciere Gilda]";
				mes "A dopo.";
				close;
			}
			mes "[Arciere Gilda]";
			mes "Beh allora. Io non";
			mes "qualcos'altro che dire.";
			mes "So che sarai un grande Arciere...";
			next;
			skill "NV_TRICKDEAD",0,SKILL_PERM;
			jobchange Job_Archer_high;
			skill "AC_MAKINGARROW",1,SKILL_PERM;
			skill "AC_CHARGEARROW",1,SKILL_PERM;
			mes "[Arciere Gilda]";
			mes "Sebbene non ci sia un particolare";
			mes "un premio per te questa volta, spero che tu capisca. Prenditi cura di te.";
			close;
		}else{
			mes "[Arciere Gilda]";
			mes "Oh...?";
			mes "Ehi, quali sono";
			mes "cosa fai qui...?";
			next;
			mes "[Arciere Gilda]";
			mes "Posso dire che non sei fatto per essere un Arciere. Sembra quasi che tu sia destinato a fare";
			mes "qualcos'altro...";
			close;
		}
	}
	mes "[Arciere Gilda]";
	mes "Piacere di conoscerti. Come posso aiutarti?";
	next;
	switch(select("Voglio essere un arcere.:Ho bisogno dei requisiti, per favore.:Niente, grazie.")) {
	case 1:
		mes "[Arciere Gilda]";
		if(BaseJob != Job_Novice) {
			if(BaseJob == Job_Archer)
				mes "Ma dai, stai scherzando...";
			else {
				mes "Mi dispiace ma solo i principianti possono cambiare lavoro.";
				mes "Hai già un lavoro decente, vero?";
			}
			close;
		}else{
			mes "Sei... "+strcharinfo(0)+", giusto?";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Arciere Gilda]";
				mes "Beh, tu non sei al livello di abilità giusto.";
				mes "Il tuo livello di incarico deve essere almeno ^4A4AFF10^000000 e il tuo livello di Abilità Base dovrebbe raggiungere ^4A4AFFlivello 9";
				next;
				mes "[Arciere Gilda]";
				mes "Poiché un Arciere ha bisogno di una concentrazione estremamente alta, non accettiamo coloro che hanno poca pazienza.";
				close;
			}
			mes "[Arciere Gilda]";
			mes "La tua abilità di base è ora sufficiente..";
			mes "....Hmm~~ quindi sei ora pronto per essere un Arciere. Farò il passo subito.";
			next;
			mes "[Arciere Gilda]";
			mes "Congratulazioni! Ora sei un Arciere! Inoltre, speriamo che tu partecipi attivamente a molti programmi per la rinascita della Gilda degli Arcieri.";
			mes "Ah, gli articoli sono arrivati dal Reparto Produzione. Eccoli qui, prendi questi! Sono tutti tuoi!";
			callfunc "Job_Change",Job_Archer;
			getitem 1742,1;	// N_Composite_Bow
			getitem 12004,1;	// Arrow_Container
			getitem 12009,1;	// Silver_Arrow_Container
			getitem 12008,1;	// Fire_Arrow_Container
			next;
			mes "[Arciere Gilda]";
			mes "Avere un arco e delle frecce, ora sei un vero Arciere.";
			mes "Se apri il contenitore delle frecce, ci sono frecce dentro e poi puoi equipaggiarle.";
			mes "Beh, mi aspetto di sentire notizie migliori da te. È ora di dire addio.";
			mes "Ciao.";
			close;
		}
	case 2:
		mes "[Arciere Gilda]";
		mes "Un arcere ha abilità nell'uso di un arco e diversi talenti.";
		mes "La più grande abilità di un Arciere è attaccare i nemici da una lunga distanza.";
		next;
		mes "[Arciere Gilda]";
		mes "Sebbene un Arciere abbia meno PV, può sparare ai nemici da una lunga distanza,";
		mes "quindi un arcere è più sicuro in una battaglia vera.";
		next;
		mes "[Arciere Gilda]";
		mes "Sebbene un Arciere in Ragnarok abbia meno PV, ha alta accuratezza e tasso di attacco in modo che l'arciere possa uccidere i mostri prima che si avvicinino a un Arciere.";
		next;
		mes "[Arciere Gilda]";
		mes "^8C2121An Archer can change jobs to a Hunter.^000000";
		mes "^8C2121Other than Hunter, if you are a man, you can change your job to Bard and if you are a woman, you can change your job to Dancer.^000000";
		close;
	case 3:
		mes "[Arciere Gilda]";
		mes "Se hai domande, sentiti libero di venire a chiederle.";
		close;
	}
}
