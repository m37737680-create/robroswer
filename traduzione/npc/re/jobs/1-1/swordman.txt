izlude_in,74,172,4	script	Swordman#swd	119,{
	if (Upper == 1) {
		if (Class == Job_Novice_High && (ADVJOB == Job_Lord_Knight || ADVJOB == Job_Paladin)) {
			mes "[Spadaccino]";
			mes "È...";
			mes "Non può essere...";
			mes "Sei rinato, vero?";
			next;
			mes "[Spadaccino]";
			mes "Vedo che stai ripercorrendo il sentiero del Spadaccino! Una volta abituato a brandire una spada, non potrai più tornare indietro!!";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Spadaccino]";
				mes "Mh? Ah, devi prima padroneggiare le Abilità di Base per essere pronto a diventare uno Spadaccino.";
				next;
				mes "[Spadaccino]";
				mes "Torna da me quando avrai finito di imparare le abilità base per principianti.";
				close;
			}
			mes "[Spadaccino]";
			mes "Eccellente! Lasciami promuoverti subito a Spadaccino!";
			next;
			skill "NV_TRICKDEAD",0,SKILL_PERM;
			jobchange Job_Swordman_High;
			skill "SM_MOVINGRECOVERY",1,SKILL_PERM;
			skill "SM_FATALBLOW",1,SKILL_PERM;
			skill "SM_AUTOBERSERK",1,SKILL_PERM;
			mes "[Spadaccino]";
			mes "Hmm... Sembri uno Spadaccino molto esperto. Comunque, sono sicuro che devi allenarti per migliorare le tue abilità e guadagnare forza!";
			close;
		}else{
			mes "[Spadaccino]";
			mes "Hmm...?";
			mes "Sei rinato/a";
			mes "guerriero, non è vero?";
			next;
			mes "[Spadaccino]";
			mes "Hmmm...";
			mes "Sembra che essere";
			mes "uno Svordato non fa parte";
			mes "di il tuo destino. Mi dispiace,";
			mes "ma sembra che non ci sia niente";
			mes "Posso fare per te.";
			close;
		}
	}
	mes "[Spadaccino del Gilda]";
	mes "Questa è la Gilda del Spadaccino.";
	mes "Perché sei qui?";
	next;
	switch(select("Parlami di essere un Spadaccino.:Voglio essere un Spadaccino.:Nulla.")) {
	case 1:
		mes "[Spadaccino del Gilda]";
		mes "Quindi desideri saperne di più dello potente Spadaccino? Va bene!";
		next;
		mes "[Spadaccino del Gilda]";
		mes "La caratteristica più distintiva del Spadaccino è che lo Spadaccino può mostrarci le sue vere abilità nel combattimento ravvicinato.";
		mes "Ci sono tre motivi!";
		next;
		mes "[Spadaccino del Gilda]";
		mes "Innanzitutto, lo Swordman ha più PV degli altri lavori.";
		mes "Secondo, eccetto archi e bastoni, lo Swordman può usare tutte le altre armi così possono combattere con la loro capacità ottimale.";
		mes "E terzo, la maggior parte delle abilità del Spadaccino conferiscono potenti attacchi fisici.";
		next;
		mes "[Spadaccino del Gilda]";
		mes "Sebbene ti abbia dato una semplice spiegazione, credo che tu comprenda il significato fondamentale di cosa sia essere un Spadaccino.";
		mes "Secondo me, Swordman è la migliore classe di tutte!";
		close;
	case 2:
		mes "[Spadaccino del Gilda]";
		if(BaseJob != Job_Novice) {
			if(BaseJob == Job_Swordman) {
				mes "Sei già un eccellente Spadaccino, non è vero?";
				mes "Devotirovi ad essere un grande Spadaccino.";
			}else{
				mes "Hai già un altro lavoro, vero?";
				mes "Hai esagerato con quella battuta.";
			}
			close;
		}else{
			if (!callfunc("F_CanChangeJob")) {
				mes "Mi dispiace dirtelo ma per essere un Spadaccino, devi raggiungere almeno il livello di lavoro ^4A4AFFJob Level 10^000000.";
				mes "e ^4A4AFFLivello di Abilità Base 9^000000.";
				next;
				mes "[Spadaccino del Gilda]";
				mes "Vuoi essere un Spadaccino senza avere il requisito minimo?";
				mes "Pensi che essere un Spadaccino sia così facile?";
				close;
			}
			mes "Mmm, sia il tuo Livello di Lavoro che il Livello Base di Abilità sono corretti.";
			mes "Bene. Vuoi essere un Spadaccino subito?";
			next;
			if(select("Sì, lo faccio.:Ci ci penserò ancora.") == 2) {
				mes "[Spadaccino del Gilda]";
				mes "Sì. È necessaria una decisione prudente per scegliere un lavoro.";
				mes "Ma mi dispiace... che tu lo consideri di nuovo dopo aver superato tutte le difficoltà....";
				close;
			}
			mes "[Spadaccino del Gilda]";
			mes "Congratulazioni! Da ora in poi, vivrai la vita di un Spadaccino!";
			mes "Facciamolo subito!";
			next;
			mes "[Spadaccino del Gilda]";
			mes "Congratulazioni ancora per essere un Spadaccino e spero che tu partecipi a molte attività per la rinascita della nostra gilda.";
			callfunc "Job_Change",Job_Swordman;
			getitem 13415,1; // N_Falchion
			close;
		}
	case 3:
		close;
	}
}
