moc_prydb1,39,129,2	script	Thief Guide#thief	69,{
	if (Upper == 1) {
		if (ADVJOB == Job_Assassin_Cross || ADVJOB == Job_Stalker) {
			if (Class == Job_Novice_High) {
				mes "Guida del Ladro";
				mes "E? Ti conosco? È inquietante che tu sembri così familiare. Non hai un gemello, vero?";
				next;
				if (!callfunc("F_CanChangeJob")) {
					mes "Guida del Ladro";
					mes "Cosa, vuoi essere un Ladro? Mi dispiace, ma sembri aver bisogno di più addestramento.";
					next;
					mes "Guida del Ladro";
					mes "Prenditi il tuo tempo e impara tutte le abilità di base, va bene? Allora, ci vediamo più tardi~!";
					close;
				}
				mes "Guida del Ladro";
				mes "Beh, ho questa sensazione che tu abbia vissuto una vita di combattimenti, quindi ti promuovo a Ladro proprio in questo momento. Meglio darti ai tipi duri quello che vuoi...";
				next;
				skill "NV_TRICKDEAD",0,SKILL_PERM;
				jobchange Job_Thief_High;
				skill "TF_SPRINKLESAND",1,SKILL_PERM;
				skill "TF_BACKSLIDING",1,SKILL_PERM;
				skill "TF_PICKSTONE",1,SKILL_PERM;
				skill "TF_THROWSTONE",1,SKILL_PERM;
				mes "Guida del Ladro";
				mes "Dato che sei diventato un Ladro, vivi come un Ladro. Ora, vai e fallo! Prossimo~";
				close;
			}else{
				mes "Guida del Ladro";
				if (Sex == SEX_MALE)
					mes "Ehi, amico.";
				else
					mes "Ehi, tesoro~";
				close;
			}
		}else{
			mes "Guida del Ladro";
			if (Sex == SEX_MALE)
				mes "Ehi, amico.";
			else
				mes "Ehi, tesoro.";
			mes "Ehi! Sembri troppo angelico per voler essere un Ladro!! Via da qui, sono occupato. Avanti!";
			close;
		}
	}
	if(q_job_thief == 1) {
		mes "Bbandito di gilda";
		mes "Eh... Sembra un amato "+(Sex == SEX_MALE?"figlio":"figlia")+" di una ricca famiglia, ma perché sei qui in questo posto sporco?";
		next;
		mes "Bbandito di gilda";
		mes "Beh, io non sono responsabile di trasformarti in un Ladro. Io mi limito ad accettare candidature, capito?";
		mes "Se vuoi diventare un Ladro, chiedi al tizio dall'occhio acuto accanto a me.";
		close;
	}
	mes "Bbandito di gilda";
	if(BaseJob != Job_Novice) {
		if(BaseJob == Job_Thief) {
			mes "Ehi~ se hai qualche problema, dimmelo quando vuoi, eh?";
			close;
		}else{
			mes "Che diavolo...?";
			if(Sex == SEX_MALE) {
				mes "Ehm... tu sei un "+jobname(Class)+"testardo!";
				next;
				mes "Bbandito di gilda";
				mes "Ehi, fratello.";
			}else{
				next;
				mes "Bbandito di gilda";
				mes "Ehi, signora.";
			}
			mes "Perché sei qui? Torna al tuo posto~ torna~~";
			close;
		}
	}
	mes "Eh... Sembra un amato "+(Sex == SEX_MALE?"figlio":"figlia")+" di una ricca famiglia, ma perché sei qui in questo posto sporco?";
	next;
	if(select("Voglio essere un ladro.:Nulla.") == 2) {
		mes "Bbandito di gilda";
		mes "Sai che non puoi essere un ladro senza una domanda...";
		mes "Cosa ti passa per la mente...?";
		close;
	}
	mes "Bbandito di gilda";
	mes "Beh, ne sei così orgoglioso?";
	mes "Mi stai dicendo con tanto orgoglio che vuoi essere un Ladro! Perché non vai in tutti i villaggi e ti fai pubblicità per essere un ladro?";
	mes "Ah ah ah! Vai e metti un cartello che dice 'Sarò un ladro orgoglioso che ruba le cose degli altri'.";
	next;
	mes "Bbandito di gilda";
	mes "Vuoi tanto essere un ladro?";
	next;
	switch(select("Sì.:No.:E tu?")) {
	case 1:
		mes "Bbandito di gilda";
		mes "Oh, lo fai...? Uhm... beh... vivo e imparo a vedere persone strane come te.";
		next;
		break;
	case 2:
		mes "Bbandito di gilda";
		mes "Allora perché sei qui? Pensi di poterti trasformare in un ladro così facilmente?";
		next;
		break;
	case 3:
		mes "Bbandito di gilda";
		mes "Eh..? io? io?";
		mes "Beh... mi sono adattato bene ad essere un ladro... caratteristicamente... non mi dispiace questa sciocca faccenda.";
		next;
		break;
	}
	mes "Bbandito di gilda";
	mes "Comunque, nel mondo esterno, non dire mai che vuoi essere un Ladro!!";
	next;
	mes "Bbandito di gilda";
	mes "Quindi, vuoi candidarti per essere un Ladro?";
	next;
	if(select("Sì, lo faccio.:No.") == 2) {
		mes "Bbandito di gilda";
		mes "Beh... fai quello che vuoi fare~ Vai per la tua strada~";
		close;
	}
	mes "Bbandito di gilda";
	mes "Allora, dimmi il tuo nome. Eh? Cosa?";
	mes ""+strcharinfo(0)+". Allora "+strcharinfo(0)+", vero...?";
	next;
	if (!callfunc("F_CanChangeJob")) {
		mes "Bbandito di gilda";
		mes "Posso vedere la tua forte volontà di diventare un Ladro......";
		mes "Ma solo con la tua volontà, non puoi farcela in un vero combattimento, vero?";
		mes "Quindi vai e raggiungi almeno il Livello di Abilità Base 9.";
		close;
	}
	mes "Bbandito di gilda";
	mes "Mmm.... Ho trovato i tuoi interessanti precedenti penali dai dati dell'agenzia investigativa. Beh, sembri avere quello che serve per essere un bravo ladro.";
	mes ""+strcharinfo(0)+"Dichiaro ora che hai superato il colloquio di lavoro!";
	next;
	mes "Bbandito di gilda";
	mes "Beh, io non sono responsabile di trasformarti in un Ladro. Io mi limito ad accettare candidature, capito?";
	mes "Se vuoi diventare un Ladro, chiedi al ragazzo accanto a me.";
	set q_job_thief,1;
	close;
}
