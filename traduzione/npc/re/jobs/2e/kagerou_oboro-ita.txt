function	script	F_KO_Survival_dice	{

//Return: 1:win, 0:draw, -1:lose
function	script	F_KO_Survival_rps	{
	setarray .@rps$[1], "Scissors", "Rock", "Paper";
	setarray .@emote[1], ET_SCISSOR, ET_ROCK, ET_WRAP;  //might be iRO custom(?)

	mes "Roccia! Carta! Forbici!";
	next;
	set .@pc, select("Dai le forbici.:Regala roccia.:Dai carta.");
	set .@npc, rand(1,3);
	emotion .@emote[.@pc], playerattached();
	emotion .@emote[.@npc];
	mes "=Risultati del gioco=";
	mes "Macchina: " + .@rps$[.@npc] + ".";
	mes "Voi: " + .@rps$[.@pc] + ".";
	next;
	if ((.@npc == 1 && .@pc == 2) || (.@npc == 2 && .@pc == 3) || (.@npc == 3 && .@pc == 1)) {
		mes "Hai vinto";
		return 1;
	} else if (.@pc == .@npc) {
		mes "Disegno";
		return 0;
	} else if ((.@npc == 1 && .@pc == 3) || (.@npc == 2 && .@pc == 1) || (.@npc == 3 && .@pc == 2)) {
		mes "Perdi";
		return -1;
	} else {
		mes "Si è verificato un errore sconosciuto.";
		mes "Si prega di contattare il servizio clienti.";
		close;
	}
}
