function	script	F_KO_Survival_dice	{

		end;

	set .@dice, rand(1,6);
	emotion ET_OTL + .@dice, playerattached();

		return .@dice;

		mes "Premi Chiudi per andare avanti " + .@dice + " blocchi.";
		callfunc "F_KO_Survival_warp", .@dice + getarg(0);
		end;
	}
}

//Return: 1:win, 0:draw, -1:lose
function	script	F_KO_Survival_rps	{
	setarray .@rps$[1], "Scissors", "Rock", "Paper";
	setarray .@emote[1], ET_SCISSOR, ET_ROCK, ET_WRAP;  //might be iRO custom(?)

	mes "Forza! Penniche! Forbici!";
	next;
	set .@pc, select("Dai le forbici.:Dai la roccia.:Dare carta.");
	set .@npc, rand(1,3);
	emotion .@emote[.@pc], playerattached();
	emotion .@emote[.@npc];
	mes "=Risultati dei Giochi=";
	mes "Macchina: " + .@rps$[.@npc] + ".";
	mes "Tu: " + .@rps$[.@pc] + ".";
	next;
	if ((.@npc == 1 && .@pc == 2) || (.@npc == 2 && .@pc == 3) || (.@npc == 3 && .@pc == 1)) {
		mes "Hai vinto";
		return 1;
	} else if (.@pc == .@npc) {
		mes "Disegna";
		return 0;
	} else if ((.@npc == 1 && .@pc == 3) || (.@npc == 2 && .@pc == 1) || (.@npc == 3 && .@pc == 2)) {
		mes "Hai perso";
		return -1;
	} else {
		mes "Si è verificato un errore sconosciuto.";
		mes "Contattare il servizio clienti.";
		close;
	}
}
