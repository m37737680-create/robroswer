			mes "È possibile impostare solo un numero compreso tra 1 e 24.";
		} else {
			set job_rune_edq, .@input;
			mes "Completato";
		}
		close;
	case 3:
		mes "1~5";
		input .@input;
		next;
		if (.@input < 1 || .@input > 5) {
			mes "Imposta l'oggetto per modificare lo Stregone";
			mes "È possibile impostare solo un numero compreso tra 1 e 5.";
		} else {
			set job_soc, .@input;
			mes "Completato";
		}
		close;
	}
