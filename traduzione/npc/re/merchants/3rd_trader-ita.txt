job3_guil01,91,93,3	script	Rare Herb Collector	1_M_03,{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2000) {
		mes "- Apetta un minuto !! -";
		mes "- Attualmente stai portando con te -";
		mes "- troppi oggetti con te. -";
		mes "- Per favore riprova -";
		mes "- dopo aver perso peso. -";
		close;
	}
	if (Class == Job_Guillotine_Cross || Class == Job_Guillotine_Cross_T || Class == Job_Baby_Guillotine_Cross) {
		mes "[Rara collezionista di erbe velenose]";
		mes "Vago per il mondo e raccolgo rare erbe velenose. Recentemente ho iniziato a commerciare l'erba chiamata Izidor. Se sei interessato, puoi acquistarli.";
		next;
		.@i = select("Come posso acquistarli?:Scambiatelo con sangue animale:Scambiatelo con un'erba amara:Scambialo con un'erba mortale nociva:Scambialo con una rosa ghiacciata:Scambiatelo con Ment:Scambiatelo con Hinalle")-2;
		if (.@i == -1) {
			mes "[Rara collezionista di erbe velenose]";
			mes "Puoi acquistarli per 10.000 zeny e uno di questi oggetti rari, Sangue animale, Erba amara, Erba mortale nociva, Rosa ghiacciata, Menta o un Hinalle.";
			next;
			mes "[Rara collezionista di erbe velenose]";
			mes "Perché chiedo soldi? Questo è solo per portare un po' di rispetto ad un collezionista come me. Ah ah ah...";
			close;
		} else {
			setarray .@exchange[0],702,621,631,749,605,703;
			.@item = .@exchange[.@i];
			.@price = 10000;

			mes "[Rara collezionista di erbe velenose]";
			mes "Quanti ne vuoi?";
			mes "Puoi acquistare solo un massimo di '2000' ciascuno.";
			mes "Inserisci '0' se desideri annullare.";
			next;
			input .@amount;
			if (.@amount == 0) {
				mes "[Rara collezionista di erbe velenose]";
				mes "Hai annullato lo scambio.";
				close;
			}
			if (.@amount > 2000) {
				mes "[Rara collezionista di erbe velenose]";
				mes "Il numero deve essere inferiore a 2.000!";
				close;
			}
			.@total = .@amount * .@price;
			if (countitem(.@item) < .@amount || Zeny < .@total) {
				mes "[Rara collezionista di erbe velenose]";
				mes "Ehi, torna quando avrai tutti i requisiti per lo scambio.";
				close;
			}
			mes "[Rara collezionista di erbe velenose]";
			mes "Bene. Ho ricevuto il denaro e l'oggetto speciale.";
			delitem .@item, .@amount;
			Zeny = Zeny - .@total;
			getitem 709, .@amount; //Izidor
			close;
		}
	}
	mes "[Rara collezionista di erbe velenose]";
	mes "Vago per il mondo e raccolgo rare erbe velenose. Ma non ho voglia di vendere le mie erbe a una persona come te... ah ah ah...";
	close;
}
