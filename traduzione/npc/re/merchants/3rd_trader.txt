job3_guil01,91,93,3	script	Rare Herb Collector	1_M_03,{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2000) {
		mes "Aspetta un minuto!!";
		mes "Attualmente stai portando";
		mes "troppi oggetti con te.";
		mes "Per favore riprova";
		mes "dopo che perderai un po' di peso.";
		close;
	}
	if (Class == Job_Guillotine_Cross || Class == Job_Guillotine_Cross_T || Class == Job_Baby_Guillotine_Cross) {
		mes "Collezionista di Erbe Velenose Rare";
		mes "Viaggio per il mondo e raccolgo rare erbe velenose. Recentemente, ho iniziato a commerciare con l'erba chiamata Izidor. Se siete interessati, potete comprarle.";
		next;
		.@i = select("Come posso comprarli?:Scambialo con Sangue Animale:Scambialo con un Erba Amara:Scambialo con un Erba Letale e Nociva:Scambialo con una Rosa Congelata:Scambialo con Ment:Scambialo con Hinalle")-2;
		if (.@i == -1) {
			mes "Collezionista di Erbe Velenose Rare";
			mes "Puoi comprarli per 10.000 zeny e uno di questi rari oggetti, Sangue Animale, Erba Amara, Erba Nociva Letale, Rosa Congelata, Ment o un Hinalle.";
			next;
			mes "Collezionista di Erbe Velenose Rare";
			mes "Perché chiedo soldi? È solo per rendere un piccolo omaggio a un collezionista come me. Ah ah ah...";
			close;
		} else {
			setarray .@exchange[0],702,621,631,749,605,703;
			.@item = .@exchange[.@i];
			.@price = 10000;

			mes "Collezionista di Erbe Velenose Rare";
			mes "Quanti ne vuoi?";
			mes "Puoi comprare un massimo di '2000' ea.";
			mes "Inserisci '0' se vuoi annullare.";
			next;
			input .@amount;
			if (.@amount == 0) {
				mes "Collezionista di Erbe Velenose Rare";
				mes "Hai annullato l'operazione.";
				close;
			}
			if (.@amount > 2000) {
				mes "Collezionista di Erbe Velenose Rare";
				mes "Il numero deve essere minore di 2,000!";
				close;
			}
			.@total = .@amount * .@price;
			if (countitem(.@item) < .@amount || Zeny < .@total) {
				mes "Collezionista di Erbe Velenose Rare";
				mes "Torna quando avrai tutti i requisiti per lo scambio.";
				close;
			}
			mes "Collezionista di Erbe Velenose Rare";
			mes "Bene. Ho ricevuto i soldi e l'oggetto speciale.";
			delitem .@item, .@amount;
			Zeny = Zeny - .@total;
			getitem 709, .@amount; //Izidor
			close;
		}
	}
	mes "Collezionista di Erbe Velenose Rare";
	mes "Vagabondo per il mondo e raccolgo rare erbe velenose. Ma non mi va di vendere le mie erbe a una persona come te... ah ah ah...";
	close;
}
