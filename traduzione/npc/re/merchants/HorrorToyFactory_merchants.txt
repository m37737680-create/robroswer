		break;
	default:
		break;
	}
	if (.@enchant == 9) {
		mes "C'è un problema, per favore torna di nuovo!";// custom
		close;
	}
	if (.@card[3] == 0 && .@sc_type < 4)
		.@index_slot = 3;
	else if (.@card[2] == 0 && .@sc_type < 3)
		.@index_slot = 2;
	else if (.@card[1] == 0 && .@sc_type < 2)
		.@index_slot = 1;
	else {
		mes "Non so cosa sia questo, ma penso che questa non sia una cosa che potrei gestire.";// custom
		close;
	}
	if (countitem(7642) > 14) {// Bloody_Coin
		delitem 7642,15;// Bloody_Coin

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@eq_num, .@item_id) || callfunc("F_IsEquipCardHack", .@eq_num, .@card[0], .@card[1], .@card[2], .@card[3]) || callfunc("F_IsEquipRefineHack", .@eq_num, .@refine))
			close;

		delequip .@eq_num;
		if (.@enchant == 0) {
			mes "Ahi! Povera di te, l'incantesimo è fallito e la tua attrezzatura è rotta, Accidenti!";
			close2;
			specialeffect2 EF_LORD;
			end;
		}
		.@card[ .@index_slot ] = .@enchant;
		getitem2 .@item_id,1,1,.@refine,0,.@card[0],.@card[1],.@card[2],.@card[3];
		mes "Mmm... È stato un successo. Dai un'occhiata.";
		specialeffect2 EF_REPAIRWEAPON;
		close;
	}
	mes "Hmm... A proposito, devi ricaricare le Monete Sangue. Non puoi nemmeno pagare la tariffa.";
	close;
}
