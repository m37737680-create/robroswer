Purchase:
	next;
	switch( select( "Datemene uno", "Non credo" )) {
		case 1:
			if (!checkweight(22980,1) || (MaxWeight - Weight) < 1000) {
				mes "Non puoi procedere con la conversazione perché hai una grande quantità di articoli.";
				mes "Organizza i tuoi elementi e riprova.";
				close;
			}
			else if (Zeny < 20000) {
				mes "[Rovina il Commerciante Nero]";
				mes "Sembra che tu non abbia più soldi. Una pergamena costa 20.000 Zeny.";
				close;
			}
			Zeny -= 20000;
			getitem 22980,1;
			mes "[Rovina il Commerciante Nero]";
			mes "È stato un buon affare. Hai bisogno di più?";
			goto Purchase;
		case 2:
			mes "[Rovina il Commerciante Nero]";
			mes "Torna quando vuoi";
			close;
	}
}

// Equipment Shop
cmd_fild07,57,275,5	script	Artifact Appraiser#pa0829_01	1_F_02,{
	if (!checkweight(1201,1) || (MaxWeight - Weight) < 1000) {
		mes "Non puoi procedere con la conversazione perché hai una grande quantità di articoli.";
		mes "Organizza i tuoi elementi e riprova.";
		close;
	}
	.@stone_id = 6905;
	mes "[ Perito artefatto ]";
	mes "Scegli il tipo di attrezzatura che desideri acquistare. Puoi acquistarne quanti ne vuoi, purché ne hai " + getitemname(.@stone_id) + ".";
	next;
	switch (select("Cancellare:Armi:Armature")) {
		case 1:
			mes "[ Perito artefatto ]";
			mes "Per favore, torna quando vuoi~";
			close;

		case 2:
			setarray .@equip_id,1994,1938,13323,13126,28703,2024,16038,21014,28105,18128;
			.@price = 50;
			break;

		case 3:
			setarray .@equip_id,15141,22075,20779,19033;
			.@price = 50;
			break;
	}
	.@menu$ = "Cancellare:";
	for (.@i = 0; .@i < getarraysize(.@equip_id); .@i++)
		.@menu$ += getitemname(.@equip_id[.@i]) + ":";
	.@s = select(.@menu$) - 1;
	switch (.@s) {
		case 0:
			mes "[ Perito artefatto ]";
			mes "Per favore, torna quando vuoi~";
			close;

		default:
			.@s--;
			mes "[ Perito artefatto ]";
			mes "Hai bisogno di ^0000FF" + .@price + "^000000 " + getitemname(.@stone_id) + " per acquistare il " + getitemname(.@equip_id[.@s]) + "~";
			next;
			if (select("Cancellare:Acquistare") == 1) {
				mes "[ Perito artefatto ]";
				mes "Per favore, torna quando vuoi~";
				close;
			}
			if (countitem(.@stone_id) < .@price) {
				mes "[ Perito artefatto ]";
				mes "Non ne hai abbastanza " + getitemname(.@stone_id) + " per acquistare questo articolo.";
				close;
			}
			mes "[ Perito artefatto ]";
			mes "Grazie per la fiducia. Per favore, torna la prossima volta.";
			delitem .@stone_id,.@price;
			getitem .@equip_id[.@s],1;
			close;
	}
}

// Equipment Enchanter
cmd_fild07,60,275,3	script	Artifact Enhancer#pa0829_01	4_F_JOB_BLACKSMITH,{
	if (!checkweight(1201,1) || (MaxWeight - Weight) < 1000) {
		mes "Non puoi procedere con la conversazione perché hai una grande quantità di articoli.";
		mes "Organizza i tuoi elementi e riprova.";
		close;
	}
	disable_items;
	.@stone_id = 6905;
	function equip_check;
	mes "[Potenziamento artefatti]";
	mes "Se stai cercando di rafforzare l'equipaggiamento che hai ottenuto nello Spazio Infinito, sei nel posto giusto.";
	mes "Ne hai qualcuno? " + getitemname(.@stone_id) + "?";
	next;
	switch (select("Come incanto il mio equipaggiamento?:Incanta l'equipaggiamento.:Inizializza l'incantesimo dell'equipaggiamento.")) {
		case 1:
			mes "[Potenziamento artefatti]";
			mes "Potrai ottenere " + getitemname(.@stone_id) + ", se esplori lo spazio sotto il faro di Paros.";
			next;
			mes "[Potenziamento artefatti]";
			mes "Per quel materiale, posso rinforzare l'attrezzatura che hai ottenuto dallo Spazio Infinito.";
			next;
			mes "[Potenziamento artefatti]";
			mes "Posso incantare solo il 3° e il 4° slot.";
			next;
			mes "[Potenziamento artefatti]";
			mes "Non c'è alcuna possibilità che l'equipaggiamento venga distrutto durante il processo di incantamento, ma c'è un'alta probabilità che venga distrutto durante l'inizializzazione degli incantamenti.";
			next;
			mes "[Potenziamento artefatti]";
			mes "Allora vieni a trovarmi quando vuoi incantare l'equipaggiamento.";
			break;

		case 2:
			.@stone_id = 6905;
			.@fee = 20;
			mes "[Potenziamento artefatti]";
			mes "Seleziona l'equipaggiamento che vuoi incantare.";
			next;
			switch (select("Cancellare:Arma:Armatura:Scarpe:Indumento:Timone")) {
				case 1:
					mes "[Potenziamento artefatti]";
					mes "Allora vieni a trovarmi quando vuoi incantare l'equipaggiamento.";
					close;

				case 2:
					.@part = EQI_HAND_R;
					break;

				case 3:
					.@part = EQI_ARMOR;
					break;

				case 4:
					.@part = EQI_SHOES;
					break;

				case 5:
					.@part = EQI_GARMENT;
					break;

				case 6:
					.@part = EQI_HEAD_TOP;
					break;
			}
			.@equip_id = getequipid(.@part);
			.@refine = getequiprefinerycnt(.@part);
			equip_check(.@equip_id);
			for (.@i = 0;.@i < 4;.@i++) {
				.@card[.@i] = getequipcardid(.@part,.@i);
				.@check[.@i] = getequipcardid(.@part,.@i);
			}
			if (.@card[2] > 0) {
				mes "[Potenziamento artefatti]";
				mes "Posso incantare solo fino al terzo slot. Il tuo equipaggiamento non può essere ulteriormente incantato.";
				close;
			}
			switch (.@part) {
				case EQI_HAND_R:
					setarray .@enchant$,
					"4700,4701:4710,4711:4720,4721",
					"4811,4810,4809,4808,4820,4821,4822,4823:4815,4814,4813,4812,4826,4827,4828,4829:4832,4833,4834,4835,4836,4837,4838,4839";
					break;

				case EQI_ARMOR:
				case EQI_SHOES:
					setarray .@enchant$,
					"4700,4701,4702,4703:4710,4711,4712,4713:4720,4721,4722,4723",
					"4795,4796,4797:4870,4871,4800:4870,4871,4800";
					break;

				case EQI_GARMENT:
				case EQI_HEAD_TOP:
					setarray .@enchant$,
					"4700,4701,4702,4703:4710,4711,4712,4713:4720,4721,4722,4723",
					"4861,4862,4867,4868,4900:4861,4862,4867,4868,4900:4861,4862,4867,4868,4900";
					break;
			}
			.@slot = (.@card[3] > 0 ? 2 : 3);
			.@index = (.@slot == 3 ? 0 : 1);
			mes "[Potenziamento artefatti]";
			mes "Puoi scegliere 3 tipi di incantesimo e la tariffa di incantesimo è " + .@fee + " " + getitemname(.@stone_id) + ".";
			mes "Mi assicurerò che venga incantato senza problemi.";
			next;
			.@type = select("Esentato:Fisico:Magico:Allineare") - 2;
			mes "[Potenziamento artefatti]";
			if (.@slot == 3)
				mes "Ok, iniziamo il primo incantesimo.";
			else
				mes "Ok, procediamo con il secondo incantesimo.";
			next;
			if (select("Tornerò più tardi.:Per favore procedi.") == 1) {
				mes "[Potenziamento artefatti]";
				mes "Se cambi idea, torna da me.";
				close;
			}
			if (countitem(.@stone_id) < .@fee) {
				mes "[Potenziamento artefatti]";
				mes "Uhm. A proposito, non penso che stessi ascoltando, non ne hai abbastanza " + getitemname(.@stone_id) + ".";
				close;
			}
			explode(.@T$,.@enchant$[.@index],":");
			explode(.@TT$,.@T$[.@type],",");
			.@enchant = atoi(.@TT$[rand(getarraysize(.@TT$))]);
			.@card[.@slot] = .@enchant;
			delitem .@stone_id,.@fee;
			if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipRefineHack", .@part, .@refine) || callfunc("F_IsEquipCardHack", .@part, .@check[0], .@check[1], .@check[2], .@check[3]))
				close;
			delequip .@part;
			getitem2 .@equip_id,1,1,.@refine,0,.@card[0],.@card[1],.@card[2],.@card[3];
			specialeffect2 EF_REPAIRWEAPON;
			mes "[Potenziamento artefatti]";
			mes "Uhm. È stato ben fatto. Controlla subito la tua attrezzatura.";
			break;
		
		case 3:
			.@stone_id = 6905;
			.@fee = 30;
			.@break_chance = 30;
			mes "[Potenziamento artefatti]";
			mes "Seleziona l'equipaggiamento che vuoi incantare.";
			next;
			switch (select("Cancellare:Arma:Armatura:Scarpe:Indumento:Timone")) {
				case 1:
					mes "[Potenziamento artefatti]";
					mes "Allora vieni a trovarmi quando vuoi incantare l'equipaggiamento.";
					close;

				case 2:
					.@part = EQI_HAND_R;
					break;

				case 3:
					.@part = EQI_ARMOR;
					break;

				case 4:
					.@part = EQI_SHOES;
					break;

				case 5:
					.@part = EQI_GARMENT;
					break;

				case 6:
					.@part = EQI_HEAD_TOP;
					break;
			}
			.@equip_id = getequipid(.@part);
			.@refine = getequiprefinerycnt(.@part);
			equip_check(.@equip_id);
			for (.@i = 0; .@i < 4; .@i++) {
				.@card[.@i] = getequipcardid(.@part,.@i);
				.@check[.@i] = .@card[.@i];
			}
			if (.@card[3] == 0) {
				mes "[Potenziamento artefatti]";
				mes "Il tuo equipaggiamento non ha alcun incantesimo su di esso.";
				close;
			}
			mes "[Potenziamento artefatti]";
			mes "Esiste la possibilità che la tua attrezzatura spaziale infinita venga distrutta durante il processo di inizializzazione. Sei ancora disposto a procedere?";
			next;
			if (select("Esentato:Procedere") == 1) {
				mes "[Potenziamento artefatti]";
				mes "Se cambi idea, torna da me.";
				close;
			}
			if (countitem(.@stone_id) < .@fee) {
				mes "[Potenziamento artefatti]";
				mes "Uhm. A proposito, non penso che stessi ascoltando, non ne hai abbastanza " + getitemname(.@stone_id) + ".";
				close;
			}
			delitem .@stone_id,.@fee;
			if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipRefineHack", .@part, .@refine) || callfunc("F_IsEquipCardHack", .@part, .@check[0], .@check[1], .@check[2], .@check[3]))
				close;
			delequip .@part;
			if (rand(1,100) > .@chance) {
				getitem2 .@equip_id,1,1,.@refine,0,.@card[0],.@card[1],0,0;
				specialeffect2 EF_REPAIRWEAPON;
				mes "[Potenziamento artefatti]";
				mes "Uhm. È stato ben fatto. Controlla subito la tua attrezzatura.";
			} else {
				specialeffect2 EF_REFINEFAIL;
				mes "[Potenziamento artefatti]";
				mes "Beh, ti avevo avvertito. Non sei fortunato, eh?";
			}
			break;
	}
	close;

	function	equip_check	{
		setarray .@equip_id,1994,1938,13323,13126,28703,2024,16038,21014,28105,18128,15141,22075,20779,19033;
		if (!getarg(0)) {
			mes "[Potenziamento artefatti]";
			mes "Ti sei tolto l'attrezzatura?";
			close;
		}
		if (inarray(.@equip_id,getarg(0)) == -1) {
			mes "[Potenziamento artefatti]";
			mes "Questo equipaggiamento non è adatto per incantare. Non dimenticare che solo le serie infinite di equipaggiamenti possono essere incantate.";
			close;
		}
		return;
	}
