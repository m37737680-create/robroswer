cmd_fild07,375,167,1	script	Ruins Black Trader#pa0829_01	4_F_JOB_HUNTER,{
	mes "[Rovine Mercante Nero]";
	mes "Ehi, non è difficile entrare e uscire da qui ogni volta? Per soli ^0000ff20,000^000000 Zeny, puoi avere uno scrollo che ti porterà direttamente all'ingresso delle rovine, cosa ne pensi?";
Purchase:
	next;
	switch( select( "Dammi uno", "Non credo" )) {
		case 1:
			if (!checkweight(22980,1) || (MaxWeight - Weight) < 1000) {
				mes "Non puoi procedere con la conversazione perché hai una grande quantità di elementi.";
				mes "Per favore, organizza i tuoi oggetti e riprova.";
				close;
			}
			else if (Zeny < 20000) {
				mes "[Rovine Mercante Nero]";
				mes "Non sembri avere più soldi. Uno scrollo costa 20.000 Zeny.";
				close;
			}
			Zeny -= 20000;
			getitem 22980,1;
			mes "[Rovine Mercante Nero]";
			mes "Era un buon affare. Ne vuoi di più?";
			goto Purchase;
		case 2:
			mes "[Rovine Mercante Nero]";
			mes "Torna pure quando vuoi";
			close;
	}
}

// Equipment Shop
cmd_fild07,57,275,5	script	Artifact Appraiser#pa0829_01	1_F_02,{
	if (!checkweight(1201,1) || (MaxWeight - Weight) < 1000) {
		mes "Non puoi procedere con la conversazione perché hai una grande quantità di elementi.";
		mes "Per favore, organizza i tuoi oggetti e riprova.";
		close;
	}
	.@stone_id = 6905;
	mes "[ Perito di Artefatti ]";
	mes "Scegli il tipo di attrezzatura che vuoi acquistare. Puoi acquistarne quante ne vuoi, purché tu abbia " + getitemname(.@stone_id) + ".";
	next;
	switch (select("Annulla:Armi:Corazze")) {
		case 1:
			mes "[ Perito di Artefatti ]";
			mes "Torna quando vuoi~";
			close;

		case 2:
			setarray .@equip_id,1994,1938,13323,13126,28703,2024,16038,21014,28105,18128;
			.@price = 50;
			break;

		case 3:
			setarray .@equip_id,15141,22075,20779,19033;
			.@price = 50;
			break;
	}
	.@menu$ = "Annulla:";
	for (.@i = 0; .@i < getarraysize(.@equip_id); .@i++)
		.@menu$ += getitemname(.@equip_id[.@i]) + ":";
	.@s = select(.@menu$) - 1;
	switch (.@s) {
		case 0:
			mes "[ Perito di Artefatti ]";
			mes "Torna quando vuoi~";
			close;

		default:
			.@s--;
			mes "[ Perito di Artefatti ]";
			mes "Hai bisogno di ^0000FF" + .@price + "^000000 " + getitemname(.@stone_id) + " per acquistare il " + getitemname(.@equip_id[.@s]) + "~";
			next;
			if (select("Annulla:Acquisto") == 1) {
				mes "[ Perito di Artefatti ]";
				mes "Torna quando vuoi~";
				close;
			}
			if (countitem(.@stone_id) < .@price) {
				mes "[ Perito di Artefatti ]";
				mes "Non hai abbastanza " + getitemname(.@stone_id) + " per acquistare questo articolo.";
				close;
			}
			mes "[ Perito di Artefatti ]";
			mes "Grazie per la tua fiducia. Ti preghiamo di tornare la prossima volta.";
			delitem .@stone_id,.@price;
			getitem .@equip_id[.@s],1;
			close;
	}
}

// Equipment Enchanter
cmd_fild07,60,275,3	script	Artifact Enhancer#pa0829_01	4_F_JOB_BLACKSMITH,{
	if (!checkweight(1201,1) || (MaxWeight - Weight) < 1000) {
		mes "Non puoi procedere con la conversazione perché hai una grande quantità di elementi.";
		mes "Per favore, organizza i tuoi oggetti e riprova.";
		close;
	}
	disable_items;
	.@stone_id = 6905;
	function equip_check;
	mes "[ Potenziatore di Artefatti ]";
	mes "Se stai cercando di rafforzare l'equipaggiamento che hai ottenuto nello Spazio Infinito, sei nel posto giusto.";
	mes "Hai qualche " + getitemname(.@stone_id) + "?";
	next;
	switch (select("Come incanto la mia attrezzatura?:Incantare l'equipaggiamento.:Inizializzare l'incantesimo dell'equipaggiamento.")) {
		case 1:
			mes "[ Potenziatore di Artefatti ]";
			mes "Sarai in grado di ottenere " + getitemname(.@stone_id) + ", se esplori lo spazio sotto il Faro di Paros.";
			next;
			mes "[ Potenziatore di Artefatti ]";
			mes "Per quel materiale, posso rinforzare l'equipaggiamento che hai preso dallo Spazio Infinito.";
			next;
			mes "[ Potenziatore di Artefatti ]";
			mes "Posso incantare solo lo slot 3 e lo slot 4.";
			next;
			mes "[ Potenziatore di Artefatti ]";
			mes "Non c'è alcuna possibilità che l'attrezzatura venga distrutta durante il processo di incantesimo, ma c'è un'alta probabilità che venga distrutta quando si inizializzano gli incantesimi.";
			next;
			mes "[ Potenziatore di Artefatti ]";
			mes "Vieni a trovarmi quando vuoi incantare l'equipaggiamento.";
			break;

		case 2:
			.@stone_id = 6905;
			.@fee = 20;
			mes "[ Potenziatore di Artefatti ]";
			mes "Seleziona l'equipaggiamento che vuoi incantare.";
			next;
			switch (select("Annulla:Arma:Armatura:Scarpe:Indumento:Timone")) {
				case 1:
					mes "[ Potenziatore di Artefatti ]";
					mes "Vieni a trovarmi quando vuoi incantare l'equipaggiamento.";
					close;

				case 2:
					.@part = EQI_HAND_R;
					break;

				case 3:
					.@part = EQI_ARMOR;
					break;

				case 4:
					.@part = EQI_SHOES;
					break;

				case 5:
					.@part = EQI_GARMENT;
					break;

				case 6:
					.@part = EQI_HEAD_TOP;
					break;
			}
			.@equip_id = getequipid(.@part);
			.@refine = getequiprefinerycnt(.@part);
			equip_check(.@equip_id);
			for (.@i = 0;.@i < 4;.@i++) {
				.@card[.@i] = getequipcardid(.@part,.@i);
				.@check[.@i] = getequipcardid(.@part,.@i);
			}
			if (.@card[2] > 0) {
				mes "[ Potenziatore di Artefatti ]";
				mes "Posso incantare solo fino al terzo slot. Il tuo equipaggiamento non può essere ulteriormente incantato.";
				close;
			}
			switch (.@part) {
				case EQI_HAND_R:
					setarray .@enchant$,
					"4700,4701:4710,4711:4720,4721",
					"4811,4810,4809,4808,4820,4821,4822,4823:4815,4814,4813,4812,4826,4827,4828,4829:4832,4833,4834,4835,4836,4837,4838,4839";
					break;

				case EQI_ARMOR:
				case EQI_SHOES:
					setarray .@enchant$,
					"4700,4701,4702,4703:4710,4711,4712,4713:4720,4721,4722,4723",
					"4795,4796,4797:4870,4871,4800:4870,4871,4800";
					break;

				case EQI_GARMENT:
				case EQI_HEAD_TOP:
					setarray .@enchant$,
					"4700,4701,4702,4703:4710,4711,4712,4713:4720,4721,4722,4723",
					"4861,4862,4867,4868,4900:4861,4862,4867,4868,4900:4861,4862,4867,4868,4900";
					break;
			}
			.@slot = (.@card[3] > 0 ? 2 : 3);
			.@index = (.@slot == 3 ? 0 : 1);
			mes "[ Potenziatore di Artefatti ]";
			mes "Puoi scegliere 3 tipi di incantesimo, e la tariffa per l'incantesimo è " + .@fee + " " + getitemname(.@stone_id) + ".";
			mes "Mi assicurerò che venga incantato senza problemi.";
			next;
			.@type = select("Smetti:Fisico:Magico:Intervallo") - 2;
			mes "[ Potenziatore di Artefatti ]";
			if (.@slot == 3)
				mes "Va bene, cominciamo il primo incantesimo.";
			else
				mes "Va bene, procediamo con il secondo incantesimo.";
			next;
			if (select("Tornerò più tardi.:Procedi.") == 1) {
				mes "[ Potenziatore di Artefatti ]";
				mes "Se cambi idea, torna semplicemente da me.";
				close;
			}
			if (countitem(.@stone_id) < .@fee) {
				mes "[ Potenziatore di Artefatti ]";
				mes "Hm. A proposito, non credo che tu stessi ascoltando, non ne hai abbastanza " + getitemname(.@stone_id) + ".";
				close;
			}
			explode(.@T$,.@enchant$[.@index],":");
			explode(.@TT$,.@T$[.@type],",");
			.@enchant = atoi(.@TT$[rand(getarraysize(.@TT$))]);
			.@card[.@slot] = .@enchant;
			delitem .@stone_id,.@fee;
			if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipRefineHack", .@part, .@refine) || callfunc("F_IsEquipCardHack", .@part, .@check[0], .@check[1], .@check[2], .@check[3]))
				close;
			delequip .@part;
			getitem2 .@equip_id,1,1,.@refine,0,.@card[0],.@card[1],.@card[2],.@card[3];
			specialeffect2 EF_REPAIRWEAPON;
			mes "[ Potenziatore di Artefatti ]";
			mes "Hm. È stato ben fatto. Controlla immediatamente il tuo equipaggiamento.";
			break;
		
		case 3:
			.@stone_id = 6905;
			.@fee = 30;
			.@break_chance = 30;
			mes "[ Potenziatore di Artefatti ]";
			mes "Seleziona l'equipaggiamento che vuoi incantare.";
			next;
			switch (select("Annulla:Arma:Armatura:Scarpe:Indumento:Timone")) {
				case 1:
					mes "[ Potenziatore di Artefatti ]";
					mes "Vieni a trovarmi quando vuoi incantare l'equipaggiamento.";
					close;

				case 2:
					.@part = EQI_HAND_R;
					break;

				case 3:
					.@part = EQI_ARMOR;
					break;

				case 4:
					.@part = EQI_SHOES;
					break;

				case 5:
					.@part = EQI_GARMENT;
					break;

				case 6:
					.@part = EQI_HEAD_TOP;
					break;
			}
			.@equip_id = getequipid(.@part);
			.@refine = getequiprefinerycnt(.@part);
			equip_check(.@equip_id);
			for (.@i = 0; .@i < 4; .@i++) {
				.@card[.@i] = getequipcardid(.@part,.@i);
				.@check[.@i] = .@card[.@i];
			}
			if (.@card[3] == 0) {
				mes "[ Potenziatore di Artefatti ]";
				mes "Il tuo equipaggiamento non ha nessun incantesimo su di esso.";
				close;
			}
			mes "[ Potenziatore di Artefatti ]";
			mes "C'è la possibilità che il tuo equipaggiamento di spazio infinito venga distrutto durante il processo di inizializzazione. Sei ancora disposto a procedere?";
			next;
			if (select("Smetti:Procedi") == 1) {
				mes "[ Potenziatore di Artefatti ]";
				mes "Se cambi idea, torna semplicemente da me.";
				close;
			}
			if (countitem(.@stone_id) < .@fee) {
				mes "[ Potenziatore di Artefatti ]";
				mes "Hm. A proposito, non credo che tu stessi ascoltando, non ne hai abbastanza " + getitemname(.@stone_id) + ".";
				close;
			}
			delitem .@stone_id,.@fee;
			if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipRefineHack", .@part, .@refine) || callfunc("F_IsEquipCardHack", .@part, .@check[0], .@check[1], .@check[2], .@check[3]))
				close;
			delequip .@part;
			if (rand(1,100) > .@chance) {
				getitem2 .@equip_id,1,1,.@refine,0,.@card[0],.@card[1],0,0;
				specialeffect2 EF_REPAIRWEAPON;
				mes "[ Potenziatore di Artefatti ]";
				mes "Hm. È stato ben fatto. Controlla immediatamente il tuo equipaggiamento.";
			} else {
				specialeffect2 EF_REFINEFAIL;
				mes "[ Potenziatore di Artefatti ]";
				mes "Beh, ti avevo avvertito. Non sei fortunato, eh?";
			}
			break;
	}
	close;

	function	equip_check	{
		setarray .@equip_id,1994,1938,13323,13126,28703,2024,16038,21014,28105,18128,15141,22075,20779,19033;
		if (!getarg(0)) {
			mes "[ Potenziatore di Artefatti ]";
			mes "Hai tolto l'attrezzatura?";
			close;
		}
		if (inarray(.@equip_id,getarg(0)) == -1) {
			mes "[ Potenziatore di Artefatti ]";
			mes "Questa attrezzatura non è adatta per l'incantesimazione. Non dimenticare che solo la serie di equipaggiamento infinita può essere incantata.";
			close;
		}
		return;
	}
