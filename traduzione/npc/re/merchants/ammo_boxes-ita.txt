que_ng,187,149,3	script	Magazine Dealer Kenny::mdk	4_M_01,{
	mes "[Kenny]";
	mes "Le borse per proiettili ti aiuteranno a trasportare i proiettili molto più facilmente.";
	mes "Vieni a dare un'occhiata~";
	next;
	switch(select("Vari casi di proiettili:Scambio di sfere:Scambio della Vecchia Sfera:Scambio di lumache:Cancellare")) {
	case 1:
		switch(select("Custodia per proiettili:Caso di proiettile insanguinato:Custodia a proiettile d'argento:Custodia per proiettili AP per munizioni:Custodia per proiettili con munizioni ardenti:Custodia per proiettili per congelamento munizioni:Custodia per proiettili per munizioni Lightning:Custodia per proiettili in pietra magica per munizioni:Custodia per proiettili purificante per munizioni:Custodia per proiettili Flare:Custodia per proiettili fulminei:Custodia per proiettili di ghiaccio:Caso di proiettili avvelenati:Caso di proiettili ciechi:Cancellare")) {
			case 1: callsub S_Casing,13200,12149;
			case 2: callsub S_Casing,13222,22737;
			case 3: callsub S_Casing,13221,22738;
			case 4: callsub S_Casing,13215,22744;
			case 5: callsub S_Casing,13216,22745;
			case 6: callsub S_Casing,13217,22746;
			case 7: callsub S_Casing,13218,22747;
			case 8: callsub S_Casing,13219,22748;
			case 9: callsub S_Casing,13220,22749;
			case 10: callsub S_Casing,13228,23123;
			case 11: callsub S_Casing,13229,23124;
			case 12: callsub S_Casing,13230,23125;
			case 13: callsub S_Casing,13231,23126;
			case 14: callsub S_Casing,13232,23127;
			case 15: callsub S_Close;
		}
	case 2:
		switch(select("Proiettile di fuoco:Proiettile del vento:Proiettile congelante:Proiettile avvelenato:Proiettile cieco:Cancellare")) {
		case 1: callsub S_Sphere_Exchange,13223,13228;
		case 2: callsub S_Sphere_Exchange,13224,13229;
		case 3: callsub S_Sphere_Exchange,13227,13230;
		case 4: callsub S_Sphere_Exchange,13225,13231;
		case 5: callsub S_Sphere_Exchange,13226,13232;
		case 6: callsub S_Close;
		}
	case 3:
		switch(select("Sfera di fuoco:Sfera del vento:Sfera Congelante:Sfera velenosa:Sfera cieca:Cancellare")) {
		case 1: callsub S_Old_Sphere_Exchange,13203,13228;
		case 2: callsub S_Old_Sphere_Exchange,13204,13229;
		case 3: callsub S_Old_Sphere_Exchange,13207,13230;
		case 4: callsub S_Old_Sphere_Exchange,13205,13231;
		case 5: callsub S_Old_Sphere_Exchange,13206,13232;
		case 6: callsub S_Close;
		}
	case 4:
		switch(select("Munizioni a pallini L:Munizioni a pallini M:Munizioni a pallini H:Munizioni a pallini SH:Munizioni a pallini XH:Cancellare")) {
		case 1: callsub S_Slug_Exchange,13210,24;
		case 2: callsub S_Slug_Exchange,13211,12;
		case 3: callsub S_Slug_Exchange,13212,8;
		case 4: callsub S_Slug_Exchange,13213,6;
		case 5: callsub S_Slug_Exchange,13214,5;
		case 6: callsub S_Close;
		}
	case 5:
		callsub S_Close;
	}

S_Casing:
	.@bullet_id = getarg(0);
	.@pack_id = getarg(1);
	.@bullet_str$ = getitemname(.@bullet_id);
	.@pack_str$ = getitemname(.@pack_id);
	.@bullet_count = countitem(.@bullet_id);
	.@pack_count = .@bullet_count / 500;

	if (.@bullet_count < 500 || getequipid(EQI_AMMO) != -1 || BaseClass != Job_Gunslinger) {
		mes "[Kenny]";
		mes "Ehi~ Hai bisogno della quantità esatta di oggetti da scambiare. Vai a prenderne un altro.";
		mes "Devi avere 500 o più di ^FF0000" + .@bullet_str$ + "^000000, ok? Inoltre, assicurati di non ^FF0000indossarli^000000 se vuoi scambiarli.";
		close;
	}

	mes "[Kenny]";
	mes "Hai scelto ^FF0000" + .@pack_str$ + "^000000.";
	mes "Te ne servono 500 ^3131FF" + .@bullet_str$ + "^000000 per fare 1 ^3131FF" + .@bullet_str$ + "^000000 articolo e costerà ^3131FF500^000000 zeny.";
	mes "Quanti ne vuoi scambiare?";
	next;
	switch(select("Scambia tutto:Realizzarne solo 1:Cancellare")) {
		case 1:
			mes "[Kenny]";
			mes "Conteggio proiettili: ^3131FF" + .@bullet_count;
			mes "Combinazione di sacchetti per proiettili: ^3131FF" + .@pack_count;
			mes "Zeny richiesto: ^3131FF" + .@pack_count * 500;
			next;
			mes "Vuoi scambiare?";
			next;
			switch(select("Scambio:Cancellare")) {
				case 1:
					if (Zeny < .@pack_count * 500) {
						mes "[Kenny]";
						mes "Ehi~ Hai bisogno dell'importo esatto di Zeny da scambiare. Vai a prenderne un altro.";
						mes "Devi avere ^FF0000" + .@pack_count * 500 + "^000000, ok?";
						close;
					}
					mes "[Kenny]";
					mes "Tutto fatto.";
					mes "La borsa è inutile quando il peso delle scorte supera il 70%. Per favore, fai attenzione.";
					Zeny -= .@pack_count * 500;
					delitem .@bullet_id, .@pack_count * 500;
					getitem .@pack_id, .@pack_count;
					close;
				case 2:
					close;
			}
		case 2:
			if (Zeny < 500) {
				mes "[Kenny]";
				mes "Ehi~ Hai bisogno dell'importo esatto di Zeny da scambiare. Vai a prenderne un altro.";
				mes "Devi avere ^FF0000500^000000, ok?";
				close;
			}
			mes "[Kenny]";
			mes "Tutto fatto.";
			mes "La borsa è inutile quando il peso delle scorte supera il 70%. Per favore, fai attenzione.";
			Zeny -= 500;
			delitem .@bullet_id, 500;
			getitem .@pack_id, 1;
			close;
		case 3:
			callsub S_Close;
	}

S_Sphere_Exchange:
	.@old_id = getarg(0);
	.@new_id = getarg(1);
	.@old_str$ = getitemname(.@old_id);
	.@new_str$ = getitemname(.@new_id);
	.@old_count = countitem(.@old_id);
	.@new_count = .@old_count / 10;

	if (.@old_count < 10 || getequipid(EQI_AMMO) != -1 || BaseClass != Job_Gunslinger) {
		mes "[Kenny]";
		mes "Ehi~ Hai bisogno della quantità esatta di oggetti da scambiare. Vai a prenderne un altro.";
		mes "Devi averne 10 o più di ^FF0000" + .@old_str$ + "^000000, ok? Inoltre, assicurati di non ^FF0000indossarli^000000 se vuoi scambiarli.";
		close;
	}

	mes "[Kenny]";
	mes "Hai scelto ^FF0000" + .@old_str$ + "^000000.";
	mes "Te ne servono 10 ^3131FF" + .@old_str$ + "^000000 per fare 1 ^3131FF" + .@new_str$ + "^000000 elemento.";
	mes "Quanti ne vuoi scambiare?";
	next;
	switch(select("Scambia tutto:Realizzarne solo 1:Cancellare")) {
		case 1:
			mes "[Kenny]";
			mes .@old_str$ + " count: ^3131FF" + .@old_count;
			mes .@new_str$ + " count: ^3131FF" + .@new_count;
			next;
			mes "Vuoi scambiare?";
			next;
			switch(select("Scambio:Cancellare")) {
				case 1:
					mes "[Kenny]";
					mes "Tutto fatto.";
					delitem .@old_id, .@new_count * 10;
					getitem .@new_id, .@new_count;
					close;
				case 2:
					close;
			}
		case 2:
			mes "[Kenny]";
			mes "Tutto fatto.";
			delitem .@old_id, 10;
			getitem .@new_id, 1;
			close;
		case 3:
			callsub S_Close;
	}

S_Old_Sphere_Exchange:
	.@old_id = getarg(0);
	.@new_id = getarg(1);
	.@old_str$ = getitemname(.@old_id);
	.@new_str$ = getitemname(.@new_id);
	.@old_count = countitem(.@old_id);

	if (.@old_count < 1 || getequipid(EQI_AMMO) != -1 || BaseClass != Job_Gunslinger) {
		mes "[Kenny]";
		mes "Ehi~ Hai bisogno della quantità esatta di oggetti da scambiare. Vai a prenderne un altro.";
		mes "Devi avere 1 o più di ^FF0000" + .@old_str$ + "^000000, ok? Inoltre, assicurati di non ^FF0000indossarli^000000 se vuoi scambiarli.";
		close;
	}

	mes "[Kenny]";
	mes "Hai scelto ^FF0000" + .@old_str$ + "^000000.";
	mes "Te ne serve 1 ^3131FF" + .@old_str$ + "^000000 per fare 1 ^3131FF" + .@new_str$ + "^000000 elemento.";
	mes "Quanti ne vuoi scambiare?";
	next;
	switch(select("Scambia tutto:Realizzarne solo 1:Cancellare")) {
		case 1:
			mes "[Kenny]";
			mes .@old_str$ + " count: ^3131FF" + .@old_count;
			mes .@new_str$ + " count: ^3131FF" + .@old_count;
			next;
			mes "Vuoi scambiare?";
			next;
			switch(select("Scambio:Cancellare")) {
				case 1:
					mes "[Kenny]";
					mes "Tutto fatto.";
					delitem .@old_id, .@old_count;
					getitem .@new_id, .@old_count;
					close;
				case 2:
					close;
			}
		case 2:
			mes "[Kenny]";
			mes "Tutto fatto.";
			delitem .@old_id, 1;
			getitem .@new_id, 1;
			close;
		case 3:
			callsub S_Close;
	}

S_Slug_Exchange:
	.@slug_id = getarg(0);
	.@req_count = getarg(1);
	.@slug_str$ = getitemname(.@slug_id);
	.@new_slug_str$ = getitemname(25187);
	.@count = countitem(.@slug_id);
	.@new_count = .@count / .@req_count;

	if (.@count < .@req_count || getequipid(EQI_AMMO) != -1 || BaseClass != Job_Gunslinger) {
		mes "[Kenny]";
		mes "Ehi~ Hai bisogno della quantità esatta di oggetti da scambiare. Vai a prenderne un altro.";
		mes "Devi averlo " + .@req_count + " o più di ^FF0000" + .@slug_str$ + "^000000, ok? Inoltre, assicurati di non ^FF0000indossarli^000000 se vuoi scambiarli.";
		close;
	}

	mes "[Kenny]";
	mes "Hai scelto ^FF0000" + .@slug_str$ + "^000000.";
	mes "Ne hai bisogno " + .@req_count + " ^3131FF" + .@slug_str$ + "^000000 per fare 1 ^3131FF" + .@new_slug_str$ + "^000000 elemento.";
	mes "Quanti ne vuoi scambiare?";
	next;
	switch(select("Scambia tutto:Realizzarne solo 1:Cancellare")) {
		case 1:
			mes "[Kenny]";
			mes .@slug_str$ + " count: ^3131FF" + .@count;
			mes .@new_slug_str$ + " count: ^3131FF" + .@new_count;
			next;
			mes "Vuoi scambiare?";
			next;
			switch(select("Scambio:Cancellare")) {
				case 1:
					mes "[Kenny]";
					mes "Tutto fatto.";
					delitem .@slug_id, .@new_count * .@req_count;
					getitem 25187, .@new_count;
					close;
				case 2:
					close;
			}
		case 2:
			mes "[Kenny]";
			mes "Tutto fatto.";
			delitem .@slug_id, .@req_count;
			getitem 25187, 1;
			close;
		case 3:
			callsub S_Close;
	}

S_Close:
	mes "[Kenny]";
	mes "Va bene.";
	mes "Se hai bisogno di qualcosa";
	mes "per favore torna indietro.";
	close;
}
