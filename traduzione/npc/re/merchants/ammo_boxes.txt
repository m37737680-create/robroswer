que_ng,187,149,3	script	Magazine Dealer Kenny::mdk	4_M_01,{
	mes "[Kenny]";
	mes "Le buste per proiettili ti aiuteranno a trasportare i proiettili molto più facilmente.";
	mes "Dai un'occhiata~";
	next;
	switch(select("Varie munizioni per bossoli:Scambio Sfera:Vecchia Borsa di Scambio:Scambio di slug:Annulla")) {
	case 1:
		switch(select("Cassa per munizioni:Caso di Proiettile Sangue:Custodia Proiettile d'Argento:Munizioni proiettili AP bossolo:Cartuccia proiettile Ammo Blazing:Cassa di proiettili congelanti senza munizioni:Custodia per munizioni Ammo Lightning Bullet:Custodia per proiettili Ammo Magic Stone Bullet:Cartuccione purificatrice per munizioni:Custodia per proiettili Flare Bullet:Custodia Proiettile Fulmineo:Custodia per proiettili di ghiaccio:Cassa di proiettili velenosi:Custodia per proiettili ciechi:Annulla")) {
			case 1: callsub S_Casing,13200,12149;
			case 2: callsub S_Casing,13222,22737;
			case 3: callsub S_Casing,13221,22738;
			case 4: callsub S_Casing,13215,22744;
			case 5: callsub S_Casing,13216,22745;
			case 6: callsub S_Casing,13217,22746;
			case 7: callsub S_Casing,13218,22747;
			case 8: callsub S_Casing,13219,22748;
			case 9: callsub S_Casing,13220,22749;
			case 10: callsub S_Casing,13228,23123;
			case 11: callsub S_Casing,13229,23124;
			case 12: callsub S_Casing,13230,23125;
			case 13: callsub S_Casing,13231,23126;
			case 14: callsub S_Casing,13232,23127;
			case 15: callsub S_Close;
		}
	case 2:
		switch(select("Colpo di fuoco:Proiettile di Vento:Proiettile Congelante:Palla avvelenata:Proiettile Cieco:Annulla")) {
		case 1: callsub S_Sphere_Exchange,13223,13228;
		case 2: callsub S_Sphere_Exchange,13224,13229;
		case 3: callsub S_Sphere_Exchange,13227,13230;
		case 4: callsub S_Sphere_Exchange,13225,13231;
		case 5: callsub S_Sphere_Exchange,13226,13232;
		case 6: callsub S_Close;
		}
	case 3:
		switch(select("Sfera di Fuoco:Sfera del Vento:Sfera ghiacciata:Sfera Veleno:Sfera Cieca:Annulla")) {
		case 1: callsub S_Old_Sphere_Exchange,13203,13228;
		case 2: callsub S_Old_Sphere_Exchange,13204,13229;
		case 3: callsub S_Old_Sphere_Exchange,13207,13230;
		case 4: callsub S_Old_Sphere_Exchange,13205,13231;
		case 5: callsub S_Old_Sphere_Exchange,13206,13232;
		case 6: callsub S_Close;
		}
	case 4:
		switch(select("Munizioni da proiettile Slug L:Munizioni da proiettile Slug Amm.:Munizioni da proiettile Slug Ammunition H:Munizioni da proiettile SH:Munizioni da cannone XH:Annulla")) {
		case 1: callsub S_Slug_Exchange,13210,24;
		case 2: callsub S_Slug_Exchange,13211,12;
		case 3: callsub S_Slug_Exchange,13212,8;
		case 4: callsub S_Slug_Exchange,13213,6;
		case 5: callsub S_Slug_Exchange,13214,5;
		case 6: callsub S_Close;
		}
	case 5:
		callsub S_Close;
	}

S_Casing:
	.@bullet_id = getarg(0);
	.@pack_id = getarg(1);
	.@bullet_str$ = getitemname(.@bullet_id);
	.@pack_str$ = getitemname(.@pack_id);
	.@bullet_count = countitem(.@bullet_id);
	.@pack_count = .@bullet_count / 500;

	if (.@bullet_count < 500 || getequipid(EQI_AMMO) != -1 || BaseClass != Job_Gunslinger) {
		mes "[Kenny]";
		mes "Ehi~ Hai bisogno della quantità esatta di articoli per lo scambio. Vai a prenderne altri.";
		mes "Devi avere 500 o più di ^FF0000" + .@bullet_str$ + "^000000, ok? Inoltre, assicurati di non ^FF0000indossare^000000li se vuoi cambiarli.";
		close;
	}

	mes "[Kenny]";
	mes "Hai scelto ^FF0000" + .@pack_str$ + "^000000.";
	mes "Hai bisogno di 500 ^3131FF" + .@bullet_str$ + "^000000 per fare 1 ^3131FF" + .@bullet_str$ + "l'oggetto ^000000, e costerà ^3131FF500^000000 zeni.";
	mes "Quanti vuoi scambiare?";
	next;
	switch(select("Scambiare tutti:Fai solo 1:Annulla")) {
		case 1:
			mes "[Kenny]";
			mes "Conteggio proiettili: ^3131FF" + .@bullet_count;
			mes "Combinazione di borse per proiettili: ^3131FF" + .@pack_count;
			mes "Zeny richiesto: ^3131FF" + .@pack_count * 500;
			next;
			mes "Vuoi scambiare?";
			next;
			switch(select("Scambio:Annulla")) {
				case 1:
					if (Zeny < .@pack_count * 500) {
						mes "[Kenny]";
						mes "Ehi~ Hai bisogno dell'esatto importo di Zeny per lo scambio. Vai a prenderne altri.";
						mes "Devi avere ^FF0000" + .@pack_count * 500 + "^000000, va bene?";
						close;
					}
					mes "[Kenny]";
					mes "Tutto fatto.";
					mes "La borsa è inutile quando il peso dell'inventario supera il 70%. Si prega di fare attenzione.";
					Zeny -= .@pack_count * 500;
					delitem .@bullet_id, .@pack_count * 500;
					getitem .@pack_id, .@pack_count;
					close;
				case 2:
					close;
			}
		case 2:
			if (Zeny < 500) {
				mes "[Kenny]";
				mes "Ehi~ Hai bisogno dell'esatto importo di Zeny per lo scambio. Vai a prenderne altri.";
				mes "Devi avere ^FF0000500^000000, ok?";
				close;
			}
			mes "[Kenny]";
			mes "Tutto fatto.";
			mes "La borsa è inutile quando il peso dell'inventario supera il 70%. Si prega di fare attenzione.";
			Zeny -= 500;
			delitem .@bullet_id, 500;
			getitem .@pack_id, 1;
			close;
		case 3:
			callsub S_Close;
	}

S_Sphere_Exchange:
	.@old_id = getarg(0);
	.@new_id = getarg(1);
	.@old_str$ = getitemname(.@old_id);
	.@new_str$ = getitemname(.@new_id);
	.@old_count = countitem(.@old_id);
	.@new_count = .@old_count / 10;

	if (.@old_count < 10 || getequipid(EQI_AMMO) != -1 || BaseClass != Job_Gunslinger) {
		mes "[Kenny]";
		mes "Ehi~ Hai bisogno della quantità esatta di articoli per lo scambio. Vai a prenderne altri.";
		mes "Devi avere 10 o più di ^FF0000" + .@old_str$ + "^000000, ok? Inoltre, assicurati di non ^FF0000indossare^000000li se vuoi cambiarli.";
		close;
	}

	mes "[Kenny]";
	mes "Hai scelto ^FF0000" + .@old_str$ + "^000000.";
	mes "Hai bisogno di 10 ^3131FF" + .@old_str$ + "^000000 per fare 1 ^3131FF" + .@new_str$ + "^000000 elemento.";
	mes "Quanti vuoi scambiare?";
	next;
	switch(select("Scambiare tutti:Fai solo 1:Annulla")) {
		case 1:
			mes "[Kenny]";
			mes .@old_str$ + " count: ^3131FF" + .@old_count;
			mes .@new_str$ + " count: ^3131FF" + .@new_count;
			next;
			mes "Vuoi scambiare?";
			next;
			switch(select("Scambio:Annulla")) {
				case 1:
					mes "[Kenny]";
					mes "Tutto fatto.";
					delitem .@old_id, .@new_count * 10;
					getitem .@new_id, .@new_count;
					close;
				case 2:
					close;
			}
		case 2:
			mes "[Kenny]";
			mes "Tutto fatto.";
			delitem .@old_id, 10;
			getitem .@new_id, 1;
			close;
		case 3:
			callsub S_Close;
	}

S_Old_Sphere_Exchange:
	.@old_id = getarg(0);
	.@new_id = getarg(1);
	.@old_str$ = getitemname(.@old_id);
	.@new_str$ = getitemname(.@new_id);
	.@old_count = countitem(.@old_id);

	if (.@old_count < 1 || getequipid(EQI_AMMO) != -1 || BaseClass != Job_Gunslinger) {
		mes "[Kenny]";
		mes "Ehi~ Hai bisogno della quantità esatta di articoli per lo scambio. Vai a prenderne altri.";
		mes "Devi avere almeno uno di ^FF0000" + .@old_str$ + "^000000, ok? Inoltre, assicurati di non ^FF0000indossare^000000li se vuoi cambiarli.";
		close;
	}

	mes "[Kenny]";
	mes "Hai scelto ^FF0000" + .@old_str$ + "^000000.";
	mes "Hai bisogno di 1 ^3131FF" + .@old_str$ + "^000000 per fare 1 ^3131FF" + .@new_str$ + "^000000 elemento.";
	mes "Quanti vuoi scambiare?";
	next;
	switch(select("Scambiare tutti:Fai solo 1:Annulla")) {
		case 1:
			mes "[Kenny]";
			mes .@old_str$ + " count: ^3131FF" + .@old_count;
			mes .@new_str$ + " count: ^3131FF" + .@old_count;
			next;
			mes "Vuoi scambiare?";
			next;
			switch(select("Scambio:Annulla")) {
				case 1:
					mes "[Kenny]";
					mes "Tutto fatto.";
					delitem .@old_id, .@old_count;
					getitem .@new_id, .@old_count;
					close;
				case 2:
					close;
			}
		case 2:
			mes "[Kenny]";
			mes "Tutto fatto.";
			delitem .@old_id, 1;
			getitem .@new_id, 1;
			close;
		case 3:
			callsub S_Close;
	}

S_Slug_Exchange:
	.@slug_id = getarg(0);
	.@req_count = getarg(1);
	.@slug_str$ = getitemname(.@slug_id);
	.@new_slug_str$ = getitemname(25187);
	.@count = countitem(.@slug_id);
	.@new_count = .@count / .@req_count;

	if (.@count < .@req_count || getequipid(EQI_AMMO) != -1 || BaseClass != Job_Gunslinger) {
		mes "[Kenny]";
		mes "Ehi~ Hai bisogno della quantità esatta di articoli per lo scambio. Vai a prenderne altri.";
		mes "Devi avere " + .@req_count + " o più di ^FF0000" + .@slug_str$ + "^000000, ok? Inoltre, assicurati di non ^FF0000indossare^000000li se vuoi cambiarli.";
		close;
	}

	mes "[Kenny]";
	mes "Hai scelto ^FF0000" + .@slug_str$ + "^000000.";
	mes "Hai bisogno di " + .@req_count + " ^3131FF" + .@slug_str$ + "^000000 per fare 1 ^3131FF" + .@new_slug_str$ + "^000000 elemento.";
	mes "Quanti vuoi scambiare?";
	next;
	switch(select("Scambiare tutti:Fai solo 1:Annulla")) {
		case 1:
			mes "[Kenny]";
			mes .@slug_str$ + " count: ^3131FF" + .@count;
			mes .@new_slug_str$ + " count: ^3131FF" + .@new_count;
			next;
			mes "Vuoi scambiare?";
			next;
			switch(select("Scambio:Annulla")) {
				case 1:
					mes "[Kenny]";
					mes "Tutto fatto.";
					delitem .@slug_id, .@new_count * .@req_count;
					getitem 25187, .@new_count;
					close;
				case 2:
					close;
			}
		case 2:
			mes "[Kenny]";
			mes "Tutto fatto.";
			delitem .@slug_id, .@req_count;
			getitem 25187, 1;
			close;
		case 3:
			callsub S_Close;
	}

S_Close:
	mes "[Kenny]";
	mes "Ok.";
	mes "Se hai bisogno di qualcosa";
	mes "per favore torna.";
	close;
}
