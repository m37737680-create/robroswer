//===== rAthena Script ======================================= 
//= Blessed Refiner
//===== Description: =========================================
//= [Official Conversion]
//= Refiners that use Blessed ores to refine equipment.
//= Upon failure, the equipment is not destroyed. The success
//= rate is identical to that for Enriched ores.
//= - "Blacksmith Dister" only refines from +6~12.
//= NOTE: This NPC is currently disabled on official servers.
//===== Changelog: ===========================================
//= 1.0 First version. [Euphy]
//= 1.1 Removed re-roll behavior. [Secret]
//============================================================

// Main NPC :: new_smelting612
//============================================================
-	script	::BlacksmithDister	-1,{
	disable_items;
	mes "[Fabbro Dister]";
	mes "In questa società altamente competitiva, dobbiamo essere diversi per sopravvivere!";
	mes "Raffino l'equipaggiamento solo ai livelli da +6 a +12.";
	next;
	mes "[Fabbro Dister]";
	mes "Avrò bisogno di ^ff9999Blessed Oridecon^000000 per le armi e di ^ff9999Blessed Elunium^000000 per gli armature.";
	mes "I raffinamenti falliti ^FF0000non romperanno né ridurranno il livello di raffinamento^000000.";
	next;
	mes "[Fabbro Dister]";
	mes "Che ne dici? Vuoi affinare?";
	next;
	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(set .@i,1; .@i<=10; set .@i,.@i+1)
		set .@menu$, .@menu$ + ( getequipisequiped(.@indices[.@i]) ? getequipname(.@indices[.@i]) : F_getpositionname(.@indices[.@i]) +"-[Unequipped]" ) +":";
	set .@part, .@indices[ select(.@menu$) ];
	if (!getequipisequiped(.@part)) {
		mes "[Fabbro Dister]";
		switch(.@part) {
		case EQI_HEAD_TOP:
			mes "Sono un fabbro, non uno stilista di capelli.";
			break;
		case EQI_ARMOR:
			mes "Con il mio martello, ti farò una stella del cielo.";
			break;
		case EQI_HAND_L:
		case EQI_HAND_R:
			mes "Fare mani artificiali non è la mia specialità.";
			break;
		case EQI_GARMENT:
			mes "Tira fuori l'articolo così posso rifinarlo!";
			break;
		case EQI_SHOES:
			mes "Da dove viene questo odore strano?";
			break;
		case EQI_ACC_L:
		case EQI_ACC_R:
			mes "Dov'è l'accessorio?";
			break;
		case EQI_HEAD_MID:
			mes "Cosa vuoi che io migliori?";
			break;
		case EQI_HEAD_LOW:
			mes "Eh? Cosa vuoi che faccia?";
			break;
		}
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "[Fabbro Dister]";
		mes "Questo oggetto non può essere raffinato.";
		close;
	}
	.@equip_id = getequipid(.@part);
	.@itemtype = getiteminfo( .@equip_id, ITEMINFO_TYPE );
	.@equip_refine = getequiprefinerycnt(.@part);
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);

	if (.@equip_refine < 6 || .@equip_refine > 12) {
		mes "[Fabbro Dister]";
		mes "Questo equipaggiamento si è raffinato per "+.@equip_refine+"Io gestisco solo oggetti con livelli di raffinazione da +6 a +12!";
		close;
	}
	if( .@itemtype == IT_ARMOR ){
		.@equip_lv = getequiparmorlv( .@part );
		.@type$ = "Armor";

		switch(.@equip_lv) {
			case 1:
				set .@price,20000;
				set .@material,6439; //Unbreakable_Def
				break;
			default:
				// TODO:
				close;
		}
	}else if( .@itemtype == IT_WEAPON ){
		.@equip_lv = getequipweaponlv( .@part );
		.@type$ = "Weapon";

		switch( .@equip_lv ){
			case 1:
				set .@price,1000;
				set .@material,6438; //Unbreakable_Weap
				break;
			case 2:
				set .@price,2000;
				set .@material,6438; //Unbreakable_Weap
				break;
			case 3:
				set .@price,20000;
				set .@material,6438; //Unbreakable_Weap
				break;
			case 4:
				set .@price,40000;
				set .@material,6438; //Unbreakable_Weap
				break;
			default:
				// TODO:
				close;
		}
	}else{
		// TODO:
		close;
	}
	set .@ore$,"^ff9999Blessed "+.@type$+" Ore^000000";
	mes "[Fabbro Dister]";
	mes "Questo "+.@type$+" è stato raffinato per "+.@equip_refine+"Per perfezionarlo, ho bisogno "+.@ore$+" e un "+callfunc("F_InserisciVirgola",.@price)+" commissione di raffinazione zeny.";
	mes "Vuoi procedere?";
	next;
	if(select("Sì:No") == 2) {
		mes "[Fabbro Dister]";
		mes "Sono impegnato/a, non scherzare con me!";
		close;
	}
	if (getequippercentrefinery(.@part) < 100) {
		mes "[Fabbro Dister]";
		mes "Questo "+.@type$+" è stato raffinato molte volte. Sebbene non scomparirà in caso di fallimento, il "+.@ore$+" scomparirà!";
		next;
		mes "[Fabbro Dister]";
		mes "Certo, la commissione di raffinazione non verrà restituita! Vuoi continuare?";
		next;
		if(select("Sì:No") == 2) {
			mes "[Fabbro Dister]";
			mes "Bene. Sarei triste se anche questo fallisse.";
			close;
		}
	}
	if (countitem(.@material) == 0 || Zeny < .@price) {
		mes "[Fabbro Dister]";
		mes "Materiali insufficienti.";
		mes "Rifinire un livello "+.@equip_lv+" "+.@type$+", ho bisogno di "+.@ore$+" e un "+callfunc("F_InserisciVirgola",.@price)+" commissione di raffinazione zeny.";
		close;
	}
	delitem .@material,1;
	set Zeny, Zeny-.@price;

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) || callfunc("F_IsEquipRefineHack", .@part, .@equip_refine))
		close;

	mes "[Fabbro Dister]";
	mes "Tac! Tac! Tac!";
	if (getequippercentrefinery(.@part, true) > rand(100)) {
		specialeffect EF_BLESSING;
		successrefitem .@part;
		next;
		mes "[Fabbro Dister]";
		mes "Ahahahahahaha!!!";
		next;
		mes "[Fabbro Dister]";
		mes "Refine è stato un grande successo~ Hahaha!";
		close;
	}
	specialeffect EF_CURSEATTACK;
	specialeffect2 EF_SUI_EXPLOSION;
	next;
	emotion (!rand(5))?ET_ANGER:ET_HUK;
	mes "[Fabbro Dister]";
	mes "Oh mio Dio!!!!";
	next;
	mes "[Fabbro Dister]";
	mes "Oh! Sebbene fosse benedetto da "+.@ore$+" per trattenere l'attrezzatura...";
	next;
	mes "[Fabbro Dister]";
	mes "Il mio umore è peggiore quando fallisco!!";
	close;
}
