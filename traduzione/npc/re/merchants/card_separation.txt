-	script	::CardSeparation_mal	-1,{
	disable_items;
	if (checkweight(1201,1) == 0) {
		mes "Hai troppi tipi di oggetti. Proviamo a continuare dopo aver ridotto quegli oggetti.";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "Non posso continuare perché hai troppi oggetti pesanti. Proviamo a riprendere dopo aver ridotto il peso.";
		close;
	}
	if (strnpcinfo(1) == "Jeremy") {
		set .@Jeremy,1;
		set .@n$, "[Jeremy]";
		setarray .@equip_name$[0],  "Armor",  "Shoes",  "Garment", "Upper Hat";
		setarray .@equip_slot[0], EQI_ARMOR,EQI_SHOES,EQI_GARMENT,EQI_HEAD_TOP;

		mes .@n$;
		mes "È passato tanto tempo!";
		mes "Ho imparato una nuova abilità che separa le carte da Armatura, Scarpe, Indumento e Copricapo. Vuoi provarla?";
	} else {
		set .@Jeremy,0;
		set .@n$, "[Richard]";
		setarray .@equip_name$[0], "Left hand", "Right hand";
		setarray .@equip_slot[0],   EQI_HAND_L,   EQI_HAND_R;

		mes .@n$;
		mes "Jeremy sciocco non vuole toccare armi e scudi perché prelevare diverse carte potrebbe danneggiarle. Ecco perché ho preparato un'abilità di separazione delle carte per armi e scudi...";
	}
	next;
	mes .@n$;
	mes "Generalmente la tariffa è di 1.000.000 Zeny. Durante la separazione delle carte, puoi usare ^990000oggetti speciali che riducono il tasso di distruzione dell'equipaggiamento o delle carte^000000. Non addebitiamo zeny aggiuntivi per questo.";
	next;
	mes .@n$;
	mes "There is a possibility of destroying them even using a special item. Also, ^ff0000the refine level might be lost^000000. Do you have any equipment to separate?";
	next;

	for(set .@i,0; .@i<getarraysize(.@equip_slot); set .@i,.@i+1) {
		if (getequipisequiped(.@equip_slot[.@i]))
			set .@menu$, .@menu$+getequipname(.@equip_slot[.@i])+":";
		else
			set .@menu$, .@menu$+"^999999"+.@equip_name$[.@i]+" (empty)^000000:";
	}

	set .@i, select("Ferma il lavoro:"+((.@Jeremy)?"Come è possibile?":"")+":"+.@menu$);
	switch(.@i) {
	case 1:
		mes .@n$;
		mes "Quando hai bisogno del lavoro, venimi qui.";
		close;
	case 2:
		mes .@n$;
		mes "Ti chiedi cosa sia così speciale. Beh, odio darti solo una guida, quindi lascia che ti racconti la storia...";
		next;
		mes "^000099Jeremy sta stirando le spalle e le mani. Potrebbe aspettare qualcuno con cui parlare.^000000";
		next;
		mes .@n$;
		mes "Sai che la specialità di Malangdo è il cibo in scatola?";
		next;
		select("Lo sapevo bene. È così limitato?");
		mes .@n$;
		mes "Ehi... A tutti piace. Ma prima c'erano dei problemi.";
		next;
		select("Problemi? C'è qualche pesce difettoso?");
		mes .@n$;
		mes "No, il pesce non ha alcun problema. Il problema è l'alto volume di olio di pesce prodotto dopo la lavorazione. Questo è un rifiuto industriale tale.";
		next;
		mes .@n$;
		mes "Tuttavia, dopo un processo di raffinazione rivitalizzante, questo olio di pesce divenne prezioso da usare per la cura di vecchi macchinari e come lubrificante industriale.";
		next;
		mes .@n$;
		mes "Inoltre, questo olio è così utile per separare i reperti dalle armi equipaggiate che non possiamo comprare Lubrificante Premium e Lubrificante Ordinario con Zeny.";
		next;
		mes .@n$;
		mes "Non preoccuparti dei soldi. Sicuramente il Lubrificante Premium è costoso. Se mi paghi un po' di zeny, posso darti un lubrificante più economico.";
		next;
		mes .@n$;
		mes "Non sono sicuro della qualità del successo. Comunque, questo è così economico, vero?";
		close;
	default:
		set .@equip_num, .@equip_slot[.@i-3];
		if (!getequipisequiped(.@equip_num)) {
			mes .@n$;
			if (.@Jeremy)
				mes "In questa parte, non c'è niente?";
			else
				mes "Non c'è niente su quella parte?";
			close;
		}
		break;
	}

	setarray .@check_equip_card[0], getequipcardid(.@equip_num,0),getequipcardid(.@equip_num,1),getequipcardid(.@equip_num,2),getequipcardid(.@equip_num,3);
	setarray .@equip_card[0], .@check_equip_card[0], .@check_equip_card[1], .@check_equip_card[2], .@check_equip_card[3];
	set .@mvp_list$,
		"|4408|4128|4456|4168|4142"+  //Gloom_Under_Night_Card, Golden_Bug_Card, Nidhogg_Shadow_Card, Dark_Lord_Card, Doppelganger_Card
		"|4134|4137|4386|4407|4357"+  //Dracula_Card, Drake_Card, Detale_Card, Randgris_Card, B_Seyren_Card
		"|4146|4132|4147|4372|4145"+  //Maya_Card, Mistress_Card, Baphomet_Card, Bacsojin_Card, Berzebub_Card
		"|4374|4352|4367|4236|4425"+  //Apocalips_H_Card, B_Ygnizem_Card, B_Shecil_Card, Amon_Ra_Card, Atroce_Card
		"|4359|4123|4144|4135|4143"+  //B_Eremes_Card, Eddga_Card, Osiris_Card, Orc_Load_Card, Orc_Hero_Card
		"|4263|4131|4430|4276|4419"+  //Incant_Samurai_Card, Moonlight_Flower_Card, Ifrit_Card, Lord_Of_Death_Card, Ktullanux_Card
		"|4403|4399|4376|4441|4302"+  //Kiel_Card, Thanatos_Card, Lady_Tanee_Card, Fallen_Bishop_Card, Tao_Gunka_Card
		"|4305|4148|4318|4121|4365"+  //Turtle_General_Card, Pharaoh_Card, Knight_Windstorm_Card, Phreeoni_Card, B_Katrinn_Card
		"|4363|4324|4361|4330|4342|"; //B_Magaleta_Card, Garm_Card, B_Harword_Card, Dark_Snake_Lord_Card, Rsx_0806_Card

	if (.@Jeremy) {
		for ( .@i = 0; .@i < MAX_SLOTS; .@i++ ) {
			if (getiteminfo(.@equip_card[.@i], ITEMINFO_SUBTYPE) == CARD_ENCHANT)
				.@equip_card[.@i] = 0;// Armor Enchant System
		}
		if (!getarraysize(.@equip_card)) {
			mes .@n$;
			mes "La carta non è equipaggiata. Vuoi controllare di nuovo?";
			close;
		}
		if ((.@equip_card[0] && compare(.@mvp_list$,"|"+.@equip_card[0]+"|")) ||
			(.@equip_card[1] && compare(.@mvp_list$,"|"+.@equip_card[1]+"|")) ||
			(.@equip_card[2] && compare(.@mvp_list$,"|"+.@equip_card[2]+"|")) ||
			(.@equip_card[3] && compare(.@mvp_list$,"|"+.@equip_card[3]+"|")))
			set .@boss_chk,1;
	} else {
		// Official "Richard" script uses a hardcoded list including every possible item.
		//if (!getequipisequiped(.@equip_num)) {
		//	mes "[Richard]";
		//	mes "I'm sorry. We don't provide that equipment yet.";
		//	close;
		//}

		mes "[Richard]";
		mes "Quale socket numerico vuoi per separare la carta? Dal socket di sinistra, sono ordinati 1,2,3,4.";
		next;
		set .@menu$,"";
		for ( .@i = 0; .@i < MAX_SLOTS; .@i++ ) {
			if (.@equip_card[.@i] && getiteminfo(.@equip_card[.@i], ITEMINFO_SUBTYPE) == CARD_NORMAL) // Armor Enchant System
				.@menu$ = .@menu$ + "{!tag:socket} " + (.@i+1) + " - " + getitemname(.@equip_card[.@i])+":";
			else {
				.@menu$ = .@menu$ + "^777777Socket " + (.@i+1) + " Nessuna carta^000000:";
			}
		}
		set .@i, select("Ferma il lavoro:"+.@menu$);
		switch(.@i) {
		case 1:
			mes .@n$;
			mes "Quando hai bisogno di lavorare, per favore vieni da me.";
			close;
		default:
			set .@slot, .@i-2;
			if (.@equip_card[.@slot] == 0 || getiteminfo(.@equip_card[.@slot], ITEMINFO_SUBTYPE) == CARD_ENCHANT) {
				mes .@n$;
				mes "Questo socket non è dotato di nessuna scheda. Perché non ricontrolli?";
				close;
			}
			break;
		}
		if (compare(.@mvp_list$,"|"+.@equip_card[.@slot]+"|"))
			set .@boss_chk,1;
	}
	if (.@boss_chk == 0) {
		mes .@n$;
		if (.@Jeremy)
			mes "Eccetto le carte, ^ff0000tutti gli effetti incantati scompariranno.^000000 Se acconsenti a questo, per favore scegli il tipo di lavoro:";
		else
			mes "Scegliere la tariffa di lavoro.";
		next;
		set .@menu$,
			"Next time...:"+
			((Zeny >= 1000000)?"Use 1,000,000z (Do not use special item):":"^999999Use 1,000,000z (Insufficient)^000000:")+
			((countitem(6441))?"Use Premium Lubricant:":"^999999Premium Lubricant (Insufficient)^000000:")+
			((countitem(6440))?"Use Ordinary Lubricant":"^999999Ordinary Lubricant (Insufficient)^000000");
		switch(select(.@menu$)) {
		case 1:
			mes .@n$;
			mes "Quando hai bisogno del lavoro, venimi qui.";
			close;
		case 2:
			if (Zeny < 1000000) {
				mes .@n$;
				mes "Non hai abbastanza zeny. Per favore torna con abbastanza tasse.";
				close;
			}
			mes .@n$;
			mes "Questo è un equipaggiamento piuttosto vecchio. C'è un alto tasso di distruzione delle carte o dell'equipaggiamento durante il lavoro. Sei sicuro di voler continuare?";
			next;
			if(select("La prossima volta...:Continua") == 1) {
				mes .@n$;
				mes "Quando hai bisogno del lavoro, venimi qui.";
				close;
			}
			set .@sf_c_num,150;
			set .@sf_r_num,150;
			set .@sf_w_num,150;
			set Zeny, Zeny - 1000000;
			break;
		case 3:
			if (countitem(6441) == 0) {
				mes .@n$;
				mes "Non hai Lubrificante Premium.";
				close;
			}
			mes .@n$;
			mes "Se usi il Lubrificante Premium, la velocità di distruzione sarà molto diminuita, ma non posso darti una garanzia del 100%. Sei sicuro di voler continuare?";
			next;
			if(select("La prossima volta...:Continua") == 1) {
				mes .@n$;
				mes "Quando hai bisogno del lavoro, venimi qui.";
				close;
			}
			set .@sf_c_num,75;
			set .@sf_r_num,75;
			set .@sf_w_num,75;
			delitem 6441,1; //High_RankLubricant
			break;
		case 4:
			if (countitem(6440) == 0) {
				mes .@n$;
				mes "Non hai Lubrificante Ordinario.";
				close;
			}
			mes .@n$;
			mes "Se usi l'Olio Lubrificante Ordinario, il tasso di distruzione sarà molto diminuito, ma non posso darti una garanzia del 100%. Sei sicuro di voler continuare?";
			next;
			if(select("La prossima volta...:Continua") == 1) {
				mes .@n$;
				mes "Quando hai bisogno del lavoro, venimi qui.";
				close;
			}
			set .@sf_c_num,75;
			set .@sf_r_num,150;
			set .@sf_w_num,150;
			delitem 6440,1; //General_Lubricant
			break;
		}
	} else if (.@boss_chk == 1) {
		mes .@n$;
		mes "Questo equipaggiamento contiene una preziosa carta MVP. Questa carta non può essere separata con lubrificante. Se porti il super tensioattivo ^0000ffSillit Pong^000000, sarò in grado di lavorare.";
		next;
		switch(select("La prossima volta...:Ho un Sillit Pong.")) {
		case 1:
			mes .@n$;
			mes "Quando hai bisogno del lavoro, venimi qui.";
			close;
		case 2:
			if (countitem(6443) == 0) {
				mes .@n$;
				mes "Non hai il Sillit Pong.";
				close;
			}
			break;
		}
		mes .@n$;
		if (.@Jeremy) {
			mes "Eccetto le carte, ^ff0000tutti gli effetti incantati scompariranno.^000000 Se acconsenti a questo, per favore scegli il tipo di lavoro:";
			set .@menu$,"Alright, let's do it!";
		} else {
			mes "Posso continuare?";
			set .@menu$,"I got it. Just do it quickly!";
		}
		next;
		switch(select("La prossima volta...:"+.@menu$)) {
		case 1:
			mes .@n$;
			mes "Quando hai bisogno del lavoro, venimi qui.";
			close;
		case 2:
			set .@sf_c_num,60;
			set .@sf_r_num,60;
			set .@sf_w_num,60;
			delitem 6443,1; //Sillit_Pong_Bottle
			break;
		}
	}

	set .@equip_id, getequipid(.@equip_num);
	set .@equip_refine, getequiprefinerycnt(.@equip_num);

	// anti-hack
	if (callfunc("F_IsEquipCardHack", .@equip_num, .@check_equip_card[0], .@check_equip_card[1], .@check_equip_card[2], .@check_equip_card[3]))
		close;

	delequip .@equip_num;

	// Chance of retaining refine level.
	if (rand(1,.@sf_r_num) >= 61)
		set .@equip_refine,0;

	if (.@Jeremy) {
		// Chance of retaining equipment.
		if (rand(1,.@sf_w_num) < 61) {
			set .@equip_safe,1;
			getitem2 .@equip_id,1,1,.@equip_refine,0,0,0,0,0;
		}

		// Chance of retaining cards.
		for(set .@i,0; .@i<4; set .@i,.@i+1) {
			if (.@equip_card[.@i]) {
				if (rand(1,.@sf_c_num) < 61)
					getitem .@equip_card[.@i],1;
				else
					set .@card_break,1;
			}
		}
	} else {
		set .@card, .@equip_card[.@slot];
		set .@check_equip_card[.@slot],0;

		// Chance of retaining equipment (all enchantments are preserved).
		if (rand(1,.@sf_w_num) < 61) {
			set .@equip_safe,1;
			getitem2 .@equip_id,1,1,.@equip_refine,0,.@check_equip_card[0],.@check_equip_card[1],.@check_equip_card[2],.@check_equip_card[3];
		}

		// Chance of retaining card.
		if (rand(1,.@sf_c_num) < 61)
			getitem .@card,1;
		else
			set .@card_break,1;
	}

	// Display corresponding effect.
	if (!.@equip_safe && .@card_break)
		specialeffect2 EF_LORD;
	else if (.@equip_safe && .@card_break)
		specialeffect2 EF_SUI_EXPLOSION;
	else if (!.@equip_safe && !.@card_break)
		specialeffect2 EF_FIREPILLAR;
	else
		specialeffect2 EF_MAXPOWER;

	// Output results.
	mes "-- Risultato della Separazione delle Carte --";
	if (.@equip_safe) {
		mes "Non si è verificata alcuna fessurazione durante il processo di separazione della carta.";
		mes "^0000FFLa separazione dell'attrezzatura è normale.^000000";
	} else {
		mes "Si è verificata una crepa durante il processo di separazione della carta.";
		mes "L'attrezzatura è stata danneggiata. ^ff0000Irrecuperabile.^000000";
	}
	mes "-------------------";
	if (!.@card_break) {
		mes "L'erosione della superficie non è avvenuta durante il processo di separazione delle carte.";
		mes "^0000ffLa separazione della carta è riuscita.^000000";
	} else {
		mes "L'erosione della superficie è avvenuta durante il processo di separazione delle carte.";
		mes "La carta è danneggiata. ^ff0000Irrecuperabile.^000000";
	}
	next;
	mes .@n$;
	mes "Questo è tutto per i risultati della separazione delle carte. Per favore, venite di nuovo.";
	close;
}
malangdo,215,166,4	duplicate(CardSeparation_mal)	Jeremy#pa0829	553	// Armors :: mal_yong
