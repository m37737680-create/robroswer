moc_para01,22,16,5	script	Catalog Magician#catal01	704,{
	if (!checkweight(1301,3)) {
		mes "Hai troppi elementi. Per favore fai spazio.";
		close;
	}
	set .@ticket_cost, 200;
	mes "[Mago del Catalogo]";
	mes "Guarda... l'Accademia Magica";
	mes "a Geffen è ora direttamente";
	mes "vendendo il nuovissimo";
	mes "^007777Catalogo Universale Silver^000000!";
	next;
	mes "Sta guardando le parole";
	mes "scritto sulla sua mano.";
	mes "Sembra troppo impegnato";
	mes "guardarti.";
	next;
	mes "[Mago del Catalogo]";
	mes "Un catalogo nuovissimo";
	mes "è appena uscito!";
	mes "Puoi controllare";
	mes "quali oggetti puoi scambiare";
	mes "e quali articoli sono in saldo";
	mes "dal fornitore.";
	next;
	mes "[Mago del Catalogo]";
	mes "Il catalogo universale ^007777Silver^000000";
	mes "costa solo 200z!";
	mes "Puoi comprare fino a 50 pezzi in una volta sola!";
	next;
	switch(select("Compra un ^007777Catalogo Universale Argento^000000:Non ne ho bisogno!:Non sembri un venditore!")) {
	case 1:
		mes "[Mago del Catalogo]";
		mes "Vuoi tutti i 50 pezzi?";
		mes "O quanti ne hai bisogno?";
		next;
		while (1) {
			input .@input;
			mes "[Mago del Catalogo]";
			if (.@input == 0) {
				mes "Il commercio è stato interrotto!";
				mes "Non so cosa fare dopo...";
				mes "Cosa devo fare?";
				emotion ET_PROFUSELY_SWEAT, playerattached();
				close;
			} else if (.@input < 1 || .@input > 50) {
				mes "Dovrebbe essere meno di 50 pezzi.";
				next;
			} else
				break;
		}
		set .@sell, .@ticket_cost * .@input;
		mes "Il numero totale di catalogo(i) che stai cercando di acquistare è " + .@input + " pezzi.";
		mes "Costa " + .@sell + " z.";
		if (Zeny < .@sell) {
			mes "Sembra che non abbia abbastanza soldi.";
			mes "Puoi vendere la tua attrezzatura";
			mes "per guadagnare più soldi... se vuoi.";
			close;
		}
		mes "Ok, eccolo/a.";
		set Zeny, Zeny - .@sell;
		getitem 12580,.@input; //Vending_Search_Scroll
		close;
	case 2:
		mes "[Mago del Catalogo]";
		mes "Argh... ne hai davvero bisogno";
		mes "articolo... Vuoi che io";
		mes "Spiegare di nuovo? Non lo so";
		mes "cosa fare dopo...";
		mes "Cosa devo fare?";
		emotion ET_PROFUSELY_SWEAT, playerattached();
		close;
	case 3:
		mes "[Mago del Catalogo]";
		mes "Il mio vero lavoro è quello di mago.";
		mes "Stavo per chiedere al mercante";
		mes "gilda per venderli, ma erano";
		mes "cercando di usarli come appendice";
		mes "per la vendita di spazzolini. Ecco perché";
		mes "Li sto vendendo da solo/a.";
		next;
		mes "[Mago del Catalogo]";
		mes "Ci sono molti avventurieri";
		mes "qui chi viene e va.";
		mes "Sono abbastanza sicuro di poter vendere a tutti loro presto.";
		mes "Non lo pensi tu?";
		mes "Allora... iniziamo!! Comprane di più, per favore! PER FAVORE...!";
		close;
	}
}
