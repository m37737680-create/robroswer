//===== rAthena Script =======================================
//= Clothing Effect Removal Service
//===== Description: =========================================
//= Service to remove clothing effects like Xmas, Wedding, ...
//===== Changelog: ===========================================
//= 1.0 Initial version. [Balferian, Everade, Lemongrass]
//============================================================

prontera,80,106,5	script	Clothing Effect Removal Service	4_EP16_COOK2,{
	mes "[Derek Tiro]";
	mes "Sei incapace di partecipare alla battaglia a causa dei potenziamenti dell'abbigliamento? Deve essere fastidioso.";
	next;
	mes "[Derek Tiro]";
	mes "Anche se ti sta bene, può anche diventare davvero fastidioso. Posso eliminarlo con un singolo gesto della mano!! Lo sistemerò in modo ordinato. Se sei pronto, scegli la tua opzione!";
	next;
	switch( select( "Conservalo", "Rilascialo immediatamente" ) ){
		case 1:
			mes "[Derek Tiro]";
			mes "Ritorna da me, se cambi idea.";
			close;
		case 2:
			sleep2 500;
			specialeffect2 EF_SPHERE;
			sleep2 500;
			specialeffect2 EF_SPHERE;
			sleep2 500;
			specialeffect2 EF_SPHERE;
			mes "[Derek Tiro]";
			mes "La rimozione è stata completata. Assicurati di poter muoverti liberamente ora.";
			atcommand "@changedress";
			close;
	}
}
