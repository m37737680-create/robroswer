malangdo,220,167,5	script	Coin Exchanger CX-1	564,{
	if (checkweight(1201,1) == 0) {
		mes "Hai troppi tipi di oggetti. Proviamo a continuare dopo aver ridotto quegli oggetti.";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "Impossibile continuare perché hai troppi oggetti pesanti. Proviamo a continuare dopo aver ridotto il peso.";
		close;
	}
	mes "[Scambiatore di monete CX-1]";
	mes "Benvenuti in CX-1. Se hai monete, puoi scambiarle con monete di alta o bassa qualità.";
	next;
	setarray .@coins[1],
		6417, //Silvervine
		6418, //Agrade_Coin
		6419, //Bgrade_Coin
		6420, //Cgrade_Coin
		6421, //Dgrade_Coin
		6422; //Egrade_Coin
	.@menu$ = "Fermare:";
	for(.@i = 1; .@i<=6; .@i++) {
		if (countitem(.@coins[.@i]))
			.@menu$ = .@menu$ + getitemname(.@coins[.@i]) + ":";
		else
			.@menu$ = .@menu$ + "^aaaaaa"+getitemname(.@coins[.@i])+" (Nessuno)^000000:";
	}
	.@i = select(.@menu$);
	switch(.@i) {
	case 1:
		mes "[Scambiatore di monete CX-1]";
		mes "Grazie per essere venuto.";
		close;
	default:
		.@coin = .@coins[.@i-1];
		.@coin_select = .@i-1;
		break;
	}
	if (countitem(.@coin) == 0) {
		mes "[Scambiatore di monete CX-1]";
		mes "Hai scelto una moneta che non hai.";
		mes "Grazie per essere venuto.";
		close;
	}
	mes "[Scambiatore di monete CX-1]";
	mes "----Tasso di cambio----";
	mes "Scambio più alto: da 30 a 10";
	mes "Scambio inferiore: da 10 a 30";
	mes "------------------------";
	mes "^ff3333Quando effettui lo scambio, addebitiamo una commissione in base al prezzo di mercato corrente.^000000";
	next;
	setarray .@exchange_name$[0],"higher","lower";
	setarray .@exchange_rate[0],30,10;
	setarray .@exchange_loss[0],rand(1,4),rand(1,2); // Amount deducted per exchange.
	setarray .@exchange_id[0],.@coins[.@coin_select-1],.@coins[.@coin_select+1];
	if (.@exchange_id[0] == .@coins[1]) .@exchange_id[0] = 0; // Cannot exchange for Silvervine.
	.@menu$ = "Fermare:";
	for(.@i = 0; .@i<2; .@i++) {
		if (.@exchange_id[.@i] == 0)
			.@menu$ = .@menu$ + "^ff3333Scambio non disponibile con "+.@exchange_name$[.@i]+" moneta^000000:";
		else if (countitem(.@coin) < .@exchange_rate[.@i])
			.@menu$ = .@menu$ + "^aaaaaaScambia con "+.@exchange_name$[.@i]+" moneta (nullo)^000000:";
		else
			.@menu$ = .@menu$ + "Scambia con "+.@exchange_name$[.@i]+" moneta - "+getitemname(.@coin)+" ("+.@exchange_rate[.@i]+" necessario):";
	}
	.@i = select(.@menu$)-2;
	if (.@i == -1) {
		mes "[Scambiatore di monete CX-1]";
		mes "Grazie per essere venuto.";
		close;
	}
	if (.@exchange_id[.@i] == 0) {
		mes "[Scambiatore di monete CX-1]";
		mes "Questa moneta non è disponibile per lo scambio "+.@exchange_name$[.@i]+" coniare.";
		mes "Grazie per essere venuto.";
		close;
	}
	if (countitem(.@coin) < .@exchange_rate[.@i]) {
		mes "[Scambiatore di monete CX-1]";
		mes "Non hai abbastanza monete con cui scambiare "+.@exchange_name$[.@i]+" coniare.";
		mes "Grazie per essere venuto.";
		close;
	}
	.@exchange_total = .@exchange_rate[(!.@i)] - .@exchange_loss[(!.@i)];
	delitem .@coin, .@exchange_rate[.@i];
	getitem .@exchange_id[.@i], .@exchange_total;
	mes "[Scambiatore di monete CX-1]";
	mes "^003399"+.@exchange_rate[.@i]+"^000000 unità di ^003399 "+getitemname(.@coin)+"^000000 preso.";
	mes "Li ho scambiati con ^003399"+.@exchange_name$[.@i]+" monete^000000, dopo aver detratto ^003399"+.@exchange_loss[(!.@i)]+"^000000 unità, ^003399"+.@exchange_total+"^000000 unità rimanenti.";
	close;
}
