malangdo,220,167,5	script	Coin Exchanger CX-1	564,{
	if (checkweight(1201,1) == 0) {
		mes "Hai troppi tipi di oggetti. Proviamo a continuare dopo aver ridotto quegli oggetti.";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "Non posso continuare perché hai troppi oggetti pesanti. Proviamo a riprendere dopo aver ridotto il peso.";
		close;
	}
	mes "[Cambiatore di Monete CX-1]";
	mes "Benvenuti a CX-1. Se avete monete, potete scambiarle per monete di grado alto o basso.";
	next;
	setarray .@coins[1],
		6417, //Silvervine
		6418, //Agrade_Coin
		6419, //Bgrade_Coin
		6420, //Cgrade_Coin
		6421, //Dgrade_Coin
		6422; //Egrade_Coin
	.@menu$ = "Fermati:";
	for(.@i = 1; .@i<=6; .@i++) {
		if (countitem(.@coins[.@i]))
			.@menu$ = .@menu$ + getitemname(.@coins[.@i]) + ":";
		else
			.@menu$ = .@menu$ + "^aaaaaa"+getitemname(.@coins[.@i])+" (None)^000000:";
	}
	.@i = select(.@menu$);
	switch(.@i) {
	case 1:
		mes "[Cambiatore di Monete CX-1]";
		mes "Grazie per essere venuto/a.";
		close;
	default:
		.@coin = .@coins[.@i-1];
		.@coin_select = .@i-1;
		break;
	}
	if (countitem(.@coin) == 0) {
		mes "[Cambiatore di Monete CX-1]";
		mes "Hai scelto una moneta che non hai.";
		mes "Grazie per essere venuto/a.";
		close;
	}
	mes "[Cambiatore di Monete CX-1]";
	mes "Tasso di cambio";
	mes "Cambio più alto - da 30 a 10";
	mes "Basso scambio - 10 a 30";
	mes "------------------------";
	mes "^ff3333When you exchange, we charge a fee according to the current market price.^000000";
	next;
	setarray .@exchange_name$[0],"higher","lower";
	setarray .@exchange_rate[0],30,10;
	setarray .@exchange_loss[0],rand(1,4),rand(1,2); // Amount deducted per exchange.
	setarray .@exchange_id[0],.@coins[.@coin_select-1],.@coins[.@coin_select+1];
	if (.@exchange_id[0] == .@coins[1]) .@exchange_id[0] = 0; // Cannot exchange for Silvervine.
	.@menu$ = "Fermati:";
	for(.@i = 0; .@i<2; .@i++) {
		if (.@exchange_id[.@i] == 0)
			.@menu$ = .@menu$ + "^ff3333Scambio non disponibile per "+.@exchange_name$[.@i]+" moneta^000000:";
		else if (countitem(.@coin) < .@exchange_rate[.@i])
			.@menu$ = .@menu$ + "^aaaaaaExchange to "+.@exchange_name$[.@i]+" moneta (null)^000000:";
		else
			.@menu$ = .@menu$ + "Scambiare con "+.@exchange_name$[.@i]+" moneta - "+getitemname(.@coin)+" ("+.@exchange_rate[.@i]+" (necessario):";
	}
	.@i = select(.@menu$)-2;
	if (.@i == -1) {
		mes "[Cambiatore di Monete CX-1]";
		mes "Grazie per essere venuto/a.";
		close;
	}
	if (.@exchange_id[.@i] == 0) {
		mes "[Cambiatore di Monete CX-1]";
		mes "Questa moneta non è disponibile per lo scambio con "+.@exchange_name$[.@i]+" moneta.";
		mes "Grazie per essere venuto/a.";
		close;
	}
	if (countitem(.@coin) < .@exchange_rate[.@i]) {
		mes "[Cambiatore di Monete CX-1]";
		mes "Non hai abbastanza monete per cambiare a "+.@exchange_name$[.@i]+" moneta.";
		mes "Grazie per essere venuto/a.";
		close;
	}
	.@exchange_total = .@exchange_rate[(!.@i)] - .@exchange_loss[(!.@i)];
	delitem .@coin, .@exchange_rate[.@i];
	getitem .@exchange_id[.@i], .@exchange_total;
	mes "[Cambiatore di Monete CX-1]";
	mes "^003399"+.@exchange_rate[.@i]+"^000000 unit(s) of ^003399 "+getitemname(.@coin)+"^000000 è stato preso.";
	mes "Li ho scambiati con ^003399"+.@exchange_name$[.@i]+" monete^000000, dopo aver dedotto ^003399"+.@exchange_loss[(!.@i)]+"^000000 unità, ^003399"+.@exchange_total+"Rimangono ^000000 unità.";
	close;
}
