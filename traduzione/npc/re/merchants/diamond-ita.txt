//============================================================
-	script	RareDiamondMerchant	58,{

	set .@npc$, "[Rare Diamond Merchant]";

	mes .@npc$;
	mes "Signore e signori! ! !";
	mes "Hai sentito delle voci ma non le hai mai viste con i tuoi occhi";
	mes "il diamante più raro del mondo!";
	mes "Il ^FF82FF'17 Carat Diamond'^000000 è attualmente in vendita a un prezzo conveniente!!";
	mes "Se i diamanti preziosi ti interessano, ascolta!";
	next;
	
	mes .@npc$;
	mes "Puoi regalare il diamante al tuo amante.";
	mes "Il destinatario del ^FF82FF'17 Carat Diamond'^000000 lo apprezzerà";
	mes "e ti custodiscono nel loro cuore.";
	emotion ET_BLABLA;
	next;
	
	mes .@npc$;
	mes "Che cosa? Non hai un amante?";
	mes "Puoi ancora tenertelo stretto per te stesso.";
	mes "L'aura incantevole del ^FF82FF'17 Carat Diamond'^000000 è probabile";
	mes "per portare l'amore nella tua vita.";
	next;
	
	mes .@npc$;
	mes "Conosciuto come l'affascinante diamante dei diamanti...";
	mes "È il ^FF82FF'17 carati di diamante'^000000!";
	next;

	set .@menu$, "^FF82FF'17 Carat Diamond'^000000?";
	if (countitem(6024))
		set .@menu$, .@menu$ + ":Exchange my 17 Carat Diamond for zeny...";

	switch(select(.@menu$)) {
	case 1:
		mes .@npc$;
		mes "SÌ!";
		mes "Hehe, il ^FF82FF'17 Carat Diamond'^000000 che vedi adesso";
		mes "è uno dei diamanti da 17 carati della più alta qualità.";
		mes "Confrontandolo con qualsiasi roccia incolore delle dimensioni di un pisello che potresti avere";
		mes "dissotterrato dal Marocco è indicibile!";
		emotion ET_HUK;
		next;
		
		mes .@npc$;
		mes "Alcuni sostengono che questo sia lo stesso diamante";
		mes "quello è stato visto da Kachua a Comodo!";
		next;
		
		if (select("Termina conversazione:Quanto costa?") == 1) {
			mes .@npc$;
			mes "È piuttosto economico!";
			mes "Se sei interessato, parla con me in qualsiasi momento!";
			close;
		}
		
		mes .@npc$;
		mes "Ovviamente! La cosa più importante è il prezzo!";
		mes "Come ho detto prima, il ^FF82FF'17 Carat Diamond'^000000 è stato";
		mes "valutato come un diamante da 17 carati.";
		mes "Non ce ne sono molti come questo al mondo.";
		next;
		
		mes .@npc$;
		mes "Il prezzo è di soli 500 milioni di zeny!";
		mes "C'è anche una commissione di servizio di 1.000.000 z.";
		mes "Rispetto alla qualità del diamante";
		mes "questo è un piccolo prezzo da pagare.";
		next;
		
		if (select("^828282Costa troppo caro.^000000:Lo comprerò.") == 1) {
			mes .@npc$;
			mes "Dire che un diamante come il";
			mes "^FF82FF'17 Carat Diamond'^000000 è troppo costoso, ";
			mes "mi lasci senza parole.";
			close;
		}
		
		mes .@npc$;
		mes "OOOH! Sapevo dal momento in cui ti ho visto che eri legittimo";
		mes "proprietario di questo meraviglioso diamante.";
		mes "Sono felice di non aver sbagliato!";
		mes "Credo che questo diamante ti starà molto bene.";
		mes "Tutto ciò che mi serve da te è completare questo contratto.";
		next;
		
		while(1) {
			if (select("^B9062FFirmerò il contratto.^000000:^828282Non importa, ho cambiato idea.^000000") == 2) {
				mes .@npc$;
				mes "Capisco, è un peccato..";
				close;
			}
			
			mes .@npc$;
			mes "Come ti chiami?";
			next;
			
			mes "[" + strcharinfo(0) + "]";
			mes "Mi chiamo " + strcharinfo(0) + ".";
			next;

			mes .@npc$;
			mes "Va bene allora, per favore firma qui.";
			next;
			
			input .@charname$;
			if (.@charname$ == strcharinfo(0)) {
				mes .@npc$;
				mes "Ok, bene.";
				mes "Ora non resta che il pagamento.";
				next;
				
				if (Zeny >= 501000000) {
					mes .@npc$;
					mes ". . . . . .";
					next;
					
					mes .@npc$;
					mes "Calcolo del prezzo e del costo del servizio..";
					mes "501.000.000 di zeni.";
					mes "L'importo è stato confermato.";
					emotion ET_BLABLA;
					set Zeny, Zeny - 501000000;
					getitem 6024, 1;
					next;

					mes .@npc$;
					mes "Puoi scambiare il diamante con Zeny in qualsiasi momento.";
					mes "Tuttavia, ci sarà una commissione di servizio di 1.000.000 z.";
					mes "Grazie per il tuo lavoro.";
					emotion ET_THANKS;
					close;
				}
				else {
					mes .@npc$;
					mes "Ehm?";
					mes "Mi dispiace, ma non hai fondi sufficienti.";
					mes "Inclusa la commissione di servizio, è richiesto un totale di 501.000.000 z.";
					mes "Controlla il tuo saldo Zeny e riprova.";
					emotion ET_SCRATCH;
					close;
				}
			}
			else {
				mes .@npc$;
				mes "È davvero questa la tua firma?";
				mes "Firmerà di nuovo per confermare?";
				next;
			}
		}
	case 2:
		mes .@npc$;
		mes "AAH. Vorresti scambiare il tuo ^FF82FF'17 Carat Diamond'^000000 con Zeny?";
		emotion ET_SURPRISE;
		next;
		if (select("SÌ:NO") == 2) {
			mes .@npc$;
			mes "Allora, cosa desideri?...";
			mes ". . . . . ....";
			emotion ET_THINK;
			close;
		}

		mes .@npc$;
		mes "Vedo. Dopo aver ricevuto la tua firma, ";
		mes "Devi offrire il tuo ^FF82FF'17 carati Diamond'^000000.";
		mes "Inoltre, non dimenticare che è previsto un costo di servizio di 1.000.000 z!";
		next;
		
		mes .@npc$;
		mes "Come ti chiami?";
		next;
		
		mes "[" + strcharinfo(0) + "]";
		mes "Mi chiamo " + strcharinfo(0) + ".";
		next;
		
		mes .@npc$;
		mes "Va bene allora, per favore firma qui.";
		next;
		
		while (1) {
			if (select("Cartello:Non firmare") == 2) {
				mes .@npc$;
				mes "Lo scambio non può essere completato senza la tua firma.";
				close;
			}
			
			input .@charname$;
			if (.@charname$ == strcharinfo(0)) {
				mes .@npc$;
				mes "La tua firma è stata ricevuta.";
				mes "Ora prenderò il tuo ^FF82FF'17 carati Diamond'^000000..";
				mes "Successivamente riceverai ^0000FF499.000.000z^000000.";
				next;
				
				mes .@npc$;
				mes "Grazie per il tuo lavoro.";
				mes "Ogni volta che hai bisogno di un diamante da ^FF82FF'17 carati'^000000,";
				mes "torna quando vuoi con 500 milioni di zeny.";
				emotion ET_THANKS;
				delitem 6024, 1;
				set Zeny, Zeny + 499000000;
				close;
			}
			else {
				mes .@npc$;
				mes "È davvero questa la tua firma?";
				mes "Firmerà di nuovo per confermare?";
				next;
			}
		}
	}
}
