-	script	RareDiamondMerchant	58,{

	set .@npc$, "[Rare Diamond Merchant]";

	mes .@npc$;
	mes "Signore e signori! ! !";
	mes "Hai sentito voci ma non hai mai visto con i tuoi occhi";
	mes "il diamante più raro del mondo!";
	mes "Il ^FF82FF'17 Carat Diamond'^000000 è attualmente in offerta a un prezzo economico!!";
	mes "Se i preziosi diamanti ti interessano allora ascolta!";
	next;
	
	mes .@npc$;
	mes "Puoi regalare il diamante al tuo amante.";
	mes "Il destinatario del ^FF82FF'17 Carat Diamond'^000000 lo custodirà con cura";
	mes "e tesoriarlo nel loro cuore.";
	emotion ET_BLABLA;
	next;
	
	mes .@npc$;
	mes "Cosa? Non hai un amante?";
	mes "Puoi ancora tenerlo per te.";
	mes "L'incantevole aura del ^FF82FF'17 Carat Diamond'^000000 è probabilmente";
	mes "portare l'amore nella tua vita.";
	next;
	
	mes .@npc$;
	mes "Conosciuto come il affascinante diamante dei diamanti...";
	mes "È il ^FF82FF'17 Carat Diamond'^000000!";
	next;

	set .@menu$, "^FF82FF'17 Carat Diamond'^000000?";
	if (countitem(6024))
		set .@menu$, .@menu$ + ":Exchange my 17 Carat Diamond for zeny...";

	switch(select(.@menu$)) {
	case 1:
		mes .@npc$;
		mes "Sì!";
		mes "Hehe, the ^FF82FF'17 Carat Diamond'^000000 you see right now";
		mes "è uno dei diamanti da 17 carati di più alta qualità.";
		mes "Confrontandolo con qualsiasi roccia traslucida e di dimensioni di un pisello che potresti avere";
		mes "Tirato fuori dal Marocco è indicibile!";
		emotion ET_HUK;
		next;
		
		mes .@npc$;
		mes "Alcuni sostengono che questo sia lo stesso diamante";
		mes "che fu visto da Kachua a Comodo!";
		next;
		
		if (select("Fine Conversazione:Quanto costa?") == 1) {
			mes .@npc$;
			mes "È piuttosto economico!";
			mes "Se sei interessato/a, parla con me in qualsiasi momento!";
			close;
		}
		
		mes .@npc$;
		mes "Certo! La cosa più importante è il prezzo!";
		mes "Come ho detto prima, il ^FF82FF'17 Carat Diamond'^000000 è stato";
		mes "stimato come un diamante da 17 carati.";
		mes "Non ce ne sono molti nel mondo.";
		next;
		
		mes .@npc$;
		mes "Il prezzo è un misero 500 milioni di zeni!";
		mes "C'è anche una commissione di servizio di 1.000.000z.";
		mes "Rispetto alla qualità del diamante";
		mes "è un piccolo prezzo da pagare.";
		next;
		
		if (select("^828282It's too expensive.^000000:Lo comprerò.") == 1) {
			mes .@npc$;
			mes "Dire che un diamante come il";
			mes "^FF82FF'17 Carat Diamond'^000000 è troppo costoso, ";
			mes "mi lasci senza parole";
			close;
		}
		
		mes .@npc$;
		mes "OOOH! Ho saputo dal momento in cui ti ho visto che eri il legittimo";
		mes "proprietario di questo meraviglioso diamante.";
		mes "Sono contento di non aver sbagliato!";
		mes "Credo che questo diamante ti stia molto bene.";
		mes "Tutto quello di cui ho bisogno da te è completare questo contratto.";
		next;
		
		while(1) {
			if (select("^B9062FFirmerò il contratto.^000000:^828282Non importa, ho cambiato idea.^000000") == 2) {
				mes .@npc$;
				mes "Capisco, è un peccato...";
				close;
			}
			
			mes .@npc$;
			mes "Come ti chiami?";
			next;
			
			mes "[" + strcharinfo(0) + "]";
			mes "Mi chiamo " + strcharinfo(0) + ".";
			next;

			mes .@npc$;
			mes "Allora, per favore firmi qui.";
			next;
			
			input .@charname$;
			if (.@charname$ == strcharinfo(0)) {
				mes .@npc$;
				mes "Ok, bene.";
				mes "Tutto ciò che resta ora è il pagamento.";
				next;
				
				if (Zeny >= 501000000) {
					mes .@npc$;
					mes ". . . . . .";
					next;
					
					mes .@npc$;
					mes "Calcolo del prezzo e della commissione di servizio..";
					mes "501.000.000 zeni.";
					mes "L'importo è stato confermato.";
					emotion ET_BLABLA;
					set Zeny, Zeny - 501000000;
					getitem 6024, 1;
					next;

					mes .@npc$;
					mes "Puoi scambiare il diamante per zeny in qualsiasi momento.";
					mes "Tuttavia, ci sarà una commissione di servizio di 1.000.000z.";
					mes "Grazie per la vostra fiducia.";
					emotion ET_THANKS;
					close;
				}
				else {
					mes .@npc$;
					mes "Hmm?";
					mes "Mi dispiace, ma non ha fondi sufficienti.";
					mes "Incluso il costo del servizio, è richiesto un totale di 501.000.000z.";
					mes "Controlla il tuo saldo di zeny e riprova.";
					emotion ET_SCRATCH;
					close;
				}
			}
			else {
				mes .@npc$;
				mes "È davvero la tua firma?";
				mes "Firmerà di nuovo per confermare?";
				next;
			}
		}
	case 2:
		mes .@npc$;
		mes "AAH. Vorresti scambiare il tuo ^FF82FF'17 Carat Diamond'^000000 con zeny?";
		emotion ET_SURPRISE;
		next;
		if (select("Sì:No") == 2) {
			mes .@npc$;
			mes "Allora, cosa desideri?";
			mes ". . . . . ....";
			emotion ET_THINK;
			close;
		}

		mes .@npc$;
		mes "Ho capito. Dopo che riceverò la tua firma, ";
		mes "Devi offrire il tuo ^FF82FF'17 Carat Diamond'^000000.";
		mes "E non dimenticare che c'è una commissione di servizio di 1.000.000z!";
		next;
		
		mes .@npc$;
		mes "Come ti chiami?";
		next;
		
		mes "[" + strcharinfo(0) + "]";
		mes "Mi chiamo " + strcharinfo(0) + ".";
		next;
		
		mes .@npc$;
		mes "Allora, per favore firmi qui.";
		next;
		
		while (1) {
			if (select("Segno:Non firmare") == 2) {
				mes .@npc$;
				mes "Il commercio non può essere completato senza la tua firma.";
				close;
			}
			
			input .@charname$;
			if (.@charname$ == strcharinfo(0)) {
				mes .@npc$;
				mes "La tua firma è stata ricevuta.";
				mes "Prenderò ora il tuo ^FF82FF'17 Carat Diamond'^000000.";
				mes "Riceverai ^0000FF499,000,000z^000000 dopo.";
				next;
				
				mes .@npc$;
				mes "Grazie per la vostra fiducia.";
				mes "Quando hai bisogno di un ^FF82FF'17 Carat Diamond'^000000,";
				mes "ritorna quando vuoi con 500 milioni di zeny.";
				emotion ET_THANKS;
				delitem 6024, 1;
				set Zeny, Zeny + 499000000;
				close;
			}
			else {
				mes .@npc$;
				mes "È davvero la tua firma?";
				mes "Firmerà di nuovo per confermare?";
				next;
			}
		}
	}
}
