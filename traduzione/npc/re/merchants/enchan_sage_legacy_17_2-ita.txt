		close;
	}
	disable_items;
	mes "[Lisa]";
	mes "Benvenuto benvenuto~";
	next;
	switch( select( "Esplora dispositivi", "Miglioramento automatico dell'ala del motore", "Potenziamento automatico dell'armatura", "Potenziamento automatico delle gambe", "Miglioramento degli accessori automatici(R).", "Miglioramento degli accessori automatici (L)." ) ) {

	case 3:

		.@part = EQI_SHOES;
		setarray .@equip_required[0],470022,470023;
		break;
	case 5:

		setarray .@equip_required[0],490024,490026;
		break;
	case 6:
		.@part = EQI_ACC_L;
		setarray .@equip_required[0],490025,490027;
		break;
	}
	.@equip_id = getequipid(.@part);
	.@equip_name$ = getequipname(.@part);
	.@refine = getequiprefinerycnt(.@part);
	for ( .@i = 0; .@i < 4; ++.@i )
		.@card[.@i] = getequipcardid(.@part,.@i);

	mes "[Lisa]";
	mes "Se porti equipaggiamento automatico e moduli di rinforzo, possiamo incantare il modulo appropriato sull'equipaggiamento.";
	next;
	mes "[Lisa]";
	mes strcharinfo(0) + ".";
	mes "Seleziona il livello del modulo che desideri utilizzare.";
	next;
	switch( select( "Cancellare", "Grado normale", "Grado raro", "Grado unico", "Grado leggendario", "Grado epico 1", "Grado epico 2", "Epico grado 3" ) ) {
	case 1:
		mes "[Lisa]";
		mes "Interrompere la conversazione.";
		close;
	case 2:
		setarray .@module_etc[0],1000105,1000106,1000107,1000108,1000109,1000110,1000111,1000112;

		break;
	}
	.@size = getarraysize(.@module_etc);
	.@menu$ = "Cancellare:";

	// Build the menu according to all the modules with the given grade in inventory
	for ( .@i = 0; .@i < .@size; ++.@i ) {
		if (countitem(.@module_etc[.@i]) < 1)
			.@menu$ += "^808080(Quantità modulo 0)^000000" + ":";
		else {

		mes "Interrompere la conversazione.";

	.@module_id = .@module_etc[.@s];

	if (countitem(.@module_id) < 1) {
		mes "[Lisa]";
		mes "È necessario possedere un modulo nel proprio inventario che corrisponda all'efficacia per procedere con il miglioramento.";
		close;
	}
	if (.@equip_required[0] != .@equip_id && .@equip_required[1] != .@equip_id) {
		mes "[Lisa]";
		mes "Assicurati di aver equipaggiato l'attrezzatura automatica adeguata che desideri migliorare.";
		close;

		mes "Questa attrezzatura automatica è già alla sua capacità massima.";
		mes "Provare un'altra apparecchiatura automatica.";
		close;
	}

		mes "[Lisa]";
		mes "Si è verificato un errore sconosciuto.";
		close;
	}

	// Each part has a list of allowed module and a maximum allowed number of the same module
	switch( .@part ) {
	case EQI_GARMENT:
		setarray .@module_etc_data[0],1000105,3,1000106,3,1000125,2,1000126,2,1000127,2,1000135,1,1000142,1,1000208,1;

		// Epic enchants. Max 2 epic enchants of the same enchant allowed
		for (.@i = 1000152; .@i < 1000203; .@i++) {
			setarray .@module_etc_data[.@size], .@i, 2;
			.@size += 2;
		}
		break;
	case EQI_SHOES:
		setarray .@module_etc_data[0],

		break;

	case EQI_ACC_R:
		setarray .@module_etc_data[0],1000107,3,1000108,3,1000109,3,1000110,3,1000111,3,1000112,3,1000113,2,1000114,2,1000115,1,1000116,1,1000117,1,1000118,1,1000136,1,1000137,1,1000138,1,1000139,1,1000143,1;
		break;
	}

	.@data_index = inarray(.@module_etc_data, .@module_id);

	if (.@data_index == -1) {
		mes "[Lisa]";
		mes "Non puoi incantare questo modulo su questa attrezzatura automatica.";
		close;
	}
	// Max number of the selected module allowed
	.@max_module = .@module_etc_data[.@data_index+1];
	
	// Bind the module etc to the specific enchant
	for ( .@i = 1000105; .@i < 1000203; ++.@i ) {
		if (.@i == 1000150 || .@i == 1000151)
			continue;
		.@auto_module[.@i] = 310082 + .@size_auto_module;
		.@size_auto_module++;
	}
	.@auto_module[1000207] = 310178;
	.@auto_module[1000208] = 310179;

	// Check the count of the corresponding enchant in card slots
	.@enchant_id = .@auto_module[.@module_id];
	if (.@enchant_id == 0) {
		mes "[Lisa]";
		mes "Si è verificato un errore sconosciuto.";
		close;
	}
	if (countinarray(.@card[0], .@enchant_id) >= .@max_module) {
		mes "[Lisa]";
		mes "Questa apparecchiatura automatica ha già massimizzato il miglioramento di questo modulo.";
		mes "Prova un altro modulo.";
		close;
	}
	mes "[Lisa]";
	mes "Sei sicuro di voler migliorare il tuo ^FF0000" + .@equip_name$ + "^000000 con ^0000CD" + getitemname(.@module_id) + "^000000? Questo processo non può essere invertito.";
	next;
	if (select( "Aspetta... Non farlo", "Sono sicuro che. Fallo." ) == 1) {
		mes "[Lisa]";
		mes "Interrompere la conversazione.";
		close;
	}
	delitem .@module_id,1;

	if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipRefineHack", .@part, .@refine) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]))
		end;

	for ( .@i = 3; .@i > 0; .@i--) {
		if (.@card[.@i] == 0) {
			.@card[.@i] = .@enchant_id;
			break;
		}
	}
	delequip .@part;
	getitem2 .@equip_id,1,1,.@refine,0,.@card[0],.@card[1],.@card[2],.@card[3];
	mes "[Lisa]";
	mes "Woaaaah~ Sento già la potenza della tua attrezzatura automatica~!";
	close;
}
