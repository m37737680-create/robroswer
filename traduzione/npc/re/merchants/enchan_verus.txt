		mes "Si prega di prestare particolare attenzione se l'attrezzatura verrà rimossa.";
		close;
	}

	switch( .@equip_id ) {
	case 20773:	// Excelion_Wing
	case 15128:	// Excelion_Suit
		break;
	default:
		mes "[MARS_01]";
		mes "Il prodotto non ha un numero di serie, non soddisfa le specifiche del prodotto e non può essere rafforzato.";
		close;
	}
	mes "[MARS_01]^0000ff";
	mes "Scegli " + getequipname(.@part) + " + " + getitemname(.@item_req) + ",";
	mes "^000000------------------";
	mes "Il tuo design selezionato può essere aggiornato al limite di attrezzatura ^0000ff" + .@max_num_enchant + " lo stesso design, ^000000finora, potenziato ^0000ff" + .@slot_sum + " volte^000000, vuoi continuare?";
	next;
	if (select( "Ci penserò...", "Vai avanti." ) == 1) {
		mes "[MARS_01]";
		mes "Torna la prossima volta che mi serve~";
		close;
	}

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) || callfunc("F_IsEquipRefineHack", .@part, .@equip_refine))
		close;

	if (.@card[3] == 0) {
		.@enchant_count = 1;
		.@card[3] = .@item_enchant_id;
	}
	else if (.@card[2] == 0) {
		.@enchant_count = 2;
		.@card[2] = .@item_enchant_id;
	}
	else if (.@card[1] == 0) {
		.@enchant_count = 3;
		.@card[1] = .@item_enchant_id;
	}
	else {
		mes "[MARS_01]";
		mes "L'attrezzatura ha raggiunto il limite superiore di rafforzamento Ciao!";
		close;
	}
	specialeffect2 EF_REPAIRWEAPON;
	mes "[MARS_01]";
	if (.@enchant_count < 5)
		mes "Il primo equipaggiamento è stato aggiornato ^990000" + .@enchant_count + " Tempi^000000.";
	else
		mes "Per un aggiornamento dell'attrezzatura di prestazioni aggiuntive^990000 ^000000.";	// never displayed

	delitem .@item_req, 1;
	delequip .@part;
	getitem2 .@equip_id,1,1,.@equip_refine,0,.@card[0],.@card[1],.@card[2],.@card[3];
	close;
}
