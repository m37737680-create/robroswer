grademk,34,184,4	script	Sratos#sratos	4_JP_GARM_H,{
	mes "[Sratos]";
	mes "Ciao caro cliente entrato, speriamo in un miracolo oggi.";
	mes "Come posso aiutarla?";
	next;
	switch( select( "Migliora il grado dell'attrezzatura.", "Scambia oggetti Etel", "Cos'è il miglioramento del voto?" ) ){
		case 1:
			mes "[Sratos]";
			mes "Il nostro cliente. Vuoi migliorare il livello dell'attrezzatura.";
			mes "Non è facile trasferire il potere magico dei gioielli.";
			next;
			mes "[Sratos]";
			mes "Il potere magico può esplodere sul posto.";
			mes "Poi esplode anche l'arma che erediterà il potere magico!";
			next;
			switch( select( "Lo farò ancora!", "Ci penserò." ) ){
				case 1:
					mes "[Sratos]";
					mes "Auguro buona fortuna ai nostri clienti coraggiosi!";
					close2;
					enchantgradeui();
					end;
				case 2:
					mes "[Sratos]";
					mes "Ogni volta che hai il coraggio di sfidare, per favore torna indietro.";
					close;
			}
		case 2:
			mes "[Sratos]";
			mes "Si tratta di gioielli utilizzati per migliorare i gradi.";
			mes "Etel Dust, i gioielli che saranno la base, e se mi dai un piccolo compenso lo scambio con i gioielli Etel.";
			close2;
			callshop "EnchantGradeExchange";
			end;

		case 3:
			mes "[Sratos]";
			mes "Occasionalmente vengono trovati gioielli instabili con molto potere magico puro.";
			mes "Le chiamiamo gemme eteriche, giusto?";
			next;
			mes "[Sratos]";
			mes "Puoi trasferire il potere magico della gemma eterica ad altre armi, ma ovviamente c'è qualche rischio.";
			mes "È delicato, quindi se non stai attento, pop! Esploderà.";
			next;
			mes "[Sratos]";
			mes "Ad ogni modo, se usi questa gemma eterica, puoi potenziare l'arma che stai utilizzando.";
			mes "Le prestazioni migliorano. Chiamiamo questo miglioramento del grado.";
			next;
			mes "[Sratos]";
			mes "Anche tu, non saresti felice se le armi che ami crescessero ancora di più?";
			mes "Noi siamo le persone che lo assistono. È un po' rischioso, però. ahahahah.";
			close;
	}

OnInit:
	setunittitle(getnpcid(0), "<Grade Enhancer>");
	end;
}
