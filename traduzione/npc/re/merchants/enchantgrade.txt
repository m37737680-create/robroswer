grademk,34,184,4	script	Sratos#sratos	4_JP_GARM_H,{
	mes "[Sratos]";
	mes "Ciao, caro cliente che è entrato, speriamo per un miracolo oggi.";
	mes "Come posso aiutarti?";
	next;
	switch( select( "Migliorare il grado dell'attrezzatura.", "Scambiare oggetti Etel", "Cos'è il potenziamento del voto?" ) ){
		case 1:
			mes "[Sratos]";
			mes "Il nostro cliente. Volete migliorare il grado di attrezzature.";
			mes "Non è facile spostare il potere magico dei gioielli.";
			next;
			mes "[Sratos]";
			mes "Il potere magico può esplodere sul posto.";
			mes "Poi anche l'arma che erediterà il potere magico esplode!";
			next;
			switch( select( "Ci lo farò comunque!", "Ci penserò." ) ){
				case 1:
					mes "[Sratos]";
					mes "Auguro buona fortuna ai nostri coraggiosi clienti!";
					close2;
					enchantgradeui();
					end;
				case 2:
					mes "[Sratos]";
					mes "Quando avrai il coraggio di sfidare, per favore torna.";
					close;
			}
		case 2:
			mes "[Sratos]";
			mes "Questi sono gioielli usati per migliorare i voti.";
			mes "Polvere Etel, i gioielli che saranno la base, e se mi dai un piccolo compenso li scambierò con gioielli Etel.";
			close2;
			callshop "EnchantGradeExchange";
			end;

		case 3:
			mes "[Sratos]";
			mes "Occasionalmente si trovano gemme instabili con un sacco di potere magico puro.";
			mes "Le chiamiamo gemme eteree, giusto?";
			next;
			mes "[Sratos]";
			mes "Puoi trasferire il potere magico della gemma eterica ad altre armi, ma ovviamente c'è qualche rischio.";
			mes "È delicato, quindi se non fai attenzione, pop! Esploderà.";
			next;
			mes "[Sratos]";
			mes "Comunque, se usi questo gioiello etereo, puoi potenziare l'arma che stai usando.";
			mes "Le prestazioni migliorano. Chiamiamo questo potenziamento del grado.";
			next;
			mes "[Sratos]";
			mes "Anche tu, non saresti felice se le armi che ami crescessero un altro passo?";
			mes "Siamo le persone che lo assistono. È un po' rischioso, però. ahahaha.";
			close;
	}

OnInit:
	setunittitle(getnpcid(0), "<Grade Enhancer>");
	end;
}
