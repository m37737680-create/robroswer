		mes "Con un flauto magico puoi chiamare il tuo Falcon ovunque!";
		next;
		mes "[Commerciante di flauto Falcon]";
		mes "È meraviglioso, non è vero? Ah ah ah!";
		next;
		mes "[Commerciante di flauto Falcon]";
		mes "Perché vendo questi flauti Falcon?";
		mes "È perché non ho idea di come funzionino.";
		mes "Ah ah ah!";
		next;
		mes "[Commerciante di flauto Falcon]";
		mes "Ovviamente avrai bisogno dell'abilità ^ff0000Falcon Mastery^000000.";
		mes "Non ho l'abilità, quindi non so come gestire i Falcons.";
		mes "Per ora, forse non avrò mai un Falcon.";
		next;
		mes "[Commerciante di flauto Falcon]";
		mes "Ma non posso dartene uno gratis!";
		mes "Devo guadagnare soldi perché ho anche bisogno di mangiare!";
		next;
		mes "[Commerciante di flauto Falcon]";
		mes "Se ti serve un flauto Falcon, lo venderò per 12.500 zeny. O è troppo costoso?";
		next;
		if (eaclass()&EAJL_THIRD) {
			mes "[Commerciante di flauto Falcon]";
			mes "Per i Rangers, invece, ho un prezzo speciale di 10.000 zeny!";
			set .@price,10000;
			next;
		} else
			set .@price,12500;
		if(select("Non comprarne uno:Acquista un flauto Falcon") == 1) {
			mes "[Commerciante di flauto Falcon]";
			mes "E' un peccato. Se ne hai bisogno, torna!";
			close;
		}
		if (Zeny >= .@price) {
			if (countitem(12848) >= 1) {
				mes "[Commerciante di flauto Falcon]";
				mes "Come posso vedere, hai già un Flauto Falcon!";
				mes "Non preoccuparti, mio ​​giovane amico, una volta acquistato il Flauto Falcon non avrai più bisogno di rifarlo";
				mes "dal momento che il Flauto Falcon non scomparirà con l'uso, non è fantastico?";
				close;
			}
			mes "[Commerciante di flauto Falcon]";
			mes "Ecco, prendi questo flauto.";
			mes "Con esso puoi fischiare per il tuo Falcon da qualsiasi luogo.";
		  	set Zeny, Zeny - .@price;
			getitem 12848,1; //Falcon_Flute
			next;
			mes "[Commerciante di flauto Falcon]";
			mes "Solo un promemoria.";
			mes "Per poterla utilizzare è necessaria l'abilità ^ff0000Falcon Mastery^000000!";
			close;
		}
		mes "[Commerciante di flauto Falcon]";
		mes "Mi dispiace ma non hai abbastanza soldi per comprare un Flauto Falcon, torna quando avrai più soldi!";
		close;
	}
	if (BaseJob == Job_Archer) {
		mes "Ciao, giovane!";
		mes "Al momento non posso aiutarti,";
		mes "ma perché non torni da me quando sarai diventato un grande cacciatore e avrai imparato a maneggiare adeguatamente i falchi?";
		close;
	}
	mes "Non è una bella giornata oggi?";
	close;
}
