-	script	Falcon Flute Trader::fflute	-1,{
	mes "[Trader del Flauto Falco]";
	if (BaseJob == Job_Hunter) {
		mes "Hai bisogno di un Falco?";
		mes "Con un flauto magico, puoi chiamare il tuo falco da qualsiasi parte!";
		next;
		mes "[Trader del Flauto Falco]";
		mes "È meraviglioso, non è vero? Ah ah ah!";
		next;
		mes "[Trader del Flauto Falco]";
		mes "Perché sto vendendo questi flauti Falcon?";
		mes "È perché non ho idea di come funzionino.";
		mes "Ah ah ah!";
		next;
		mes "[Trader del Flauto Falco]";
		mes "Certo, avrai bisogno della abilità ^ff0000Padronanza del Falco^000000.";
		mes "Non ho la Skill, quindi non so come gestire i Falchi.";
		mes "Per ora, potrei non avere mai un Falcon.";
		next;
		mes "[Trader del Flauto Falco]";
		mes "Ma non posso darti uno gratis!";
		mes "Devo guadagnare soldi perché devo anche mangiare!";
		next;
		mes "[Trader del Flauto Falco]";
		mes "Se hai bisogno di un Flauto Falco te li venderò per 12.500 zeny. O è troppo caro?";
		next;
		if (eaclass()&EAJL_THIRD) {
			mes "[Trader del Flauto Falco]";
			mes "Per i Rangers, tuttavia, ho un prezzo speciale di 10.000 zeny!";
			set .@price,10000;
			next;
		} else
			set .@price,12500;
		if(select("Non comprarne uno:Compra un flauto Falco") == 1) {
			mes "[Trader del Flauto Falco]";
			mes "Peccato. Se ne hai bisogno, torna!";
			close;
		}
		if (Zeny >= .@price) {
			if (countitem(12848) >= 1) {
				mes "[Trader del Flauto Falco]";
				mes "Come vedo che hai già un Falcon Flute!";
				mes "Non preoccuparti, mio giovane amico, una volta che avrai acquistato il Flauto del Falco non dovrai farlo più";
				mes "dato che il flauto Falco non scomparirà dopo l'uso, non è fantastico?";
				close;
			}
			mes "[Trader del Flauto Falco]";
			mes "Qui, prendi questo flauto.";
			mes "Con questo puoi richiamare il tuo Falco da qualsiasi posto.";
		  	set Zeny, Zeny - .@price;
			getitem 12848,1; //Falcon_Flute
			next;
			mes "[Trader del Flauto Falco]";
			mes "Solo un promemoria.";
			mes "Hai bisogno della abilità ^ff0000Padronanza del Falco^000000 per usarla!";
			close;
		}
		mes "[Trader del Flauto Falco]";
		mes "Mi dispiace ma non hai abbastanza soldi per comprare un Falcon Flute, per favore torna quando avrai più soldi!";
		close;
	}
	if (BaseJob == Job_Archer) {
		mes "Ciao, giovane!";
		mes "Attualmente non posso aiutarti,";
		mes "ma perché non torni da me quando sarai un grande Cacciatore e saprai gestire i Falchi correttamente?";
		close;
	}
	mes "Non è una giornata bellissima oggi?";
	close;
}
