dali02,51,130,4	script	Dimension Traveler	4_F_SHABBY,{
	if (checkweight(28100,1) == 0) {// Thanos Axe
		mes "- Non puoi procedere perché hai troppi oggetti con te. -";
		close;
	}
	.@item_name_req$ = getitemname(6672);// Gray_Shard
	mes "[Viaggiatore dimensionale]";
	mes "Amico avventuriero, " + mesitemlink( 6672 ) + " è ciò di cui ho bisogno. Che ne dici di scambiarlo con qualcosa che ho?";
	next;
	.@type = select( "Mostrami la tua arma.", "Mostrami la tua armatura.", "Qualcosa di speciale?", "Non ne ho bisogno." ) - 1;
	mes "[Viaggiatore dimensionale]";
	switch(.@type) {
	case 0:
		mes "OH! Ti interessano le armi? Abbiamo tutte le armi che possiamo ottenere dal Palazzo Fantasma. Perché non ne scegli uno?";
		break;
	case 1:
		mes "Armatura? Abbiamo tutte le armature che possiamo ottenere dal Palazzo Fantasma. Perché non ne scegli uno?";
		break;
	case 2:
		mes "Eh! Sembra che molti di loro abbiano già sentito che ho le cose migliori. Sentiti libero di dare un'occhiata alle mie collezioni.";
		break;
	case 3:
		mes "Addio amico~ Vieni di nuovo!";
		close;
	}
	next;
	mes "[Viaggiatore dimensionale]";
	mes "Ah! Il numero scritto accanto " + mesitemlink( 6672 ) + " sarà necessario.";
	mes "Per favore, tienilo a mente.";

	switch(.@type) {
	case 0:
		setarray .@items[0],
			13441, 200,		// Thanos Sword
			21009, 200, 	// Thanos Great Sword
			1438,  200,		// Thanos Spear
			1496,  200,		// Thanos Long Spear
			1669,  200,		// Thanos Staff
			2023,  200,		// Thanos Rod
			18119, 200,		// Thanos Bow
			13093, 200,		// Thanos Dagger
			28000, 200,		// Thanos Katar
			1836,  200,		// Thanos Knuckle
			16028, 200,		// Thanos Hammer
			28100, 200,		// Thanos Axe
			1933,  200,		// Thanos Violin
			1988,  200;		// Thanos Whip
		break;
	case 1:
		setarray .@items[0],
			2187,  100,		// Shield Of Gray
			15090, 100, 	// Armor Of Gray
			15091, 100,		// Gray Robe
			20721, 100,		// Cloak Of Gray
			22033, 100,		// Boots Of Gray
			18820, 100;		// Gray Helmet
		break;
	case 2:
		setarray .@items[0],
			27184, 10000,		// Knight Sakray Card
			23676, 50, 			// Gray Charcoal Melee (Melee)
			23677, 50,			// Gray Charcoal Magic (Magic)
			23678, 50;			// Gray Charcoal Range (Range)
		break;
	}
	.@size = getarraysize(.@items);
	for ( .@i = 0; .@i < .@size; .@i += 2 )
		.@menu$ += getitemname(.@items[.@i]) + " - " + .@items[.@i+1] + ":";

	while(true) {
		next;
		.@s = (select(.@menu$) - 1) * 2;
		mes "[Viaggiatore dimensionale]";
		if (.@type != 1)	// armor type doesn't display item info
			mes "Vuoi dire " + mesitemlink( .@items[.@s] ) + "?";
		mes "" + .@items[.@s+1] + " ^006400" + .@item_name_req$ + "^000000 sono richiesti in cambio.";
		next;
		switch( select( "Scambiamoci.", "Fammi vedere altre cose.", "Torno più tardi." ) ) {
		case 1:
			break;
		case 2:
			mes "[Viaggiatore dimensionale]";
			switch(.@type) {
			case 0:
				mes "Ho armi più che sufficienti.";
				break;
			case 1:
				mes "Abbiamo molte armature per te. Perché non ne scegli uno?";
				break;
			case 2:
				mes "Se mi porti " + .@item_name_req$ + ", Lo scambierò con qualsiasi cosa tu voglia.";
				break;
			}
			continue;
		case 3:
			mes "[Viaggiatore dimensionale]";
			mes "Abbi cura di te amico ~ Alla prossima volta!";
			close;
		}
		if (countitem(6672) < .@items[.@s+1]) {
			mes "[Viaggiatore dimensionale]";
			mes "Mi hai messo in una situazione difficile, amico.";
			mes "Ne ho bisogno " + .@item_name_req$ + ". Lo scambierò con ^006400" + .@items[.@s+1] + " " + .@item_name_req$ + "^000000.";
			close;
		}
		mes "[Viaggiatore dimensionale]";
		mes "Grazie. Torna se ti serve altro.";
		delitem 6672, .@items[.@s+1];
		getitem .@items[.@s], 1;
		close;
	}
}
