-	script	GuildWarehouse	-1,{
	mes "[Responsabile del magazzino]";
	mes "Come stai? Siamo specializzati nei magazzini delle corporazioni. Questa è la nostra ^0000cdstoria^000000:";
	mes "Perché i membri della gilda non possono condividere un deposito? Abbiamo iniziato con questa semplice domanda.";
	next;
	mes "[Responsabile del magazzino]";
	mes "Puoi aprire il magazzino per 1000 Zeny o consultare la cronologia di utilizzo della gilda.";
	mes "Come posso aiutarla?";
	next;
	if( select( "Aprire il magazzino della gilda:Visualizza la cronologia di utilizzo del magazzino" ) == 1 ){
		.@guildid = getcharid( 2 );

		if( .@guildid == 0 ){
			mes "[Responsabile del magazzino]";
			mes "Il deposito della gilda è disponibile solo per i membri della gilda.";
			close;
		}

		if( getgdskilllv( .@guildid, "GD_GUILD_STORAGE" ) == 0 || !guild_has_permission( GUILD_PERM_STORAGE ) ){
			mes "[Responsabile del magazzino]";
			mes "Sembra che non sia ancora possibile per la tua gilda utilizzare il magazzino della gilda oppure non disponi dei permessi di accesso al magazzino.";
			mes "Per favore torna dopo aver controllato l'abilità della gilda e il permesso di accedere al magazzino.";
			close;
		}

		if( Zeny < 1000 ){
			mes "[Responsabile del magazzino]";
			mes "Temo che tu non abbia abbastanza soldi per saldare la tariffa.";
			mes "La tariffa è di 1000 Zeny.";
			close;
		}

		mes "[Responsabile del magazzino]";
		mes "Allora aprirò per te il deposito della gilda. Trascorri dei momenti indimenticabili!";
		close2;

		if( Zeny < 1000 ){
			// Cheat prevention
			end;
		}

		if( guildopenstorage() == GSTORAGE_OPEN ){
			Zeny -= 1000;
			end;
		}else{
			mes "[Responsabile del magazzino]";
			mes "Temo che qualcun altro stia utilizzando il magazzino in questo momento.";
			mes "Per favore riprova più tardi.";
			close;
		}
	}else{
		.@guildid = getcharid( 2 );

		if( .@guildid == 0 ){
			mes "[Responsabile del magazzino]";
			mes "Il deposito della gilda è disponibile solo per i membri della gilda.";
			close;
		}

		if( getgdskilllv( .@guildid, "GD_GUILD_STORAGE" ) == 0 || !guild_has_permission( GUILD_PERM_STORAGE ) ){
			mes "[Responsabile del magazzino]";
			mes "Sembra che non sia ancora possibile per la tua gilda utilizzare il magazzino della gilda oppure non disponi dei permessi di accesso al magazzino.";
			mes "Per favore torna dopo aver controllato l'abilità della gilda e il permesso di accedere al magazzino.";
			close;
		}

		mes "[Responsabile del magazzino]";
		mes "Ti mostrerò la cronologia di utilizzo del magazzino. La cronologia di utilizzo verrà conservata per un massimo di 3 mesi.";
		mes "Trascorri dei momenti indimenticabili!";
		close2;
		guildopenstorage_log();
		end;
	}
}
