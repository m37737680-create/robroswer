-	script	GuildWarehouse	-1,{
	mes "Responsabile del Magazzino";
	mes "Come state? Siamo specializzati in magazzini di gilda. Questa è la nostra ^0000cdstoria^000000:";
	mes "Perché i membri della gilda non possono condividere un deposito? Abbiamo iniziato con quella semplice domanda.";
	next;
	mes "Responsabile del Magazzino";
	mes "Puoi aprire il magazzino per 1000 Zeny o consultare la cronologia di utilizzo del gilda.";
	mes "Come posso aiutarti?";
	next;
	if( select( "Apri magazzino gilda:Visualizza la cronologia di utilizzo del magazzino" ) == 1 ){
		.@guildid = getcharid( 2 );

		if( .@guildid == 0 ){
			mes "Responsabile del Magazzino";
			mes "Lo stoccaggio della gilda è disponibile solo per i membri della gilda.";
			close;
		}

		if( getgdskilllv( .@guildid, "GD_GUILD_STORAGE" ) == 0 || !guild_has_permission( GUILD_PERM_STORAGE ) ){
			mes "Responsabile del Magazzino";
			mes "Sembra che non sia ancora possibile per la tua gilda utilizzare il magazzino della gilda o che tu non abbia i permessi di accesso per il magazzino.";
			mes "Ritorna dopo aver controllato la abilità della gilda e il permesso di accesso al magazzino.";
			close;
		}

		if( Zeny < 1000 ){
			mes "Responsabile del Magazzino";
			mes "Temo che non abbia abbastanza soldi per saldare la tariffa.";
			mes "La tariffa è di 1000 Zeny.";
			close;
		}

		mes "Responsabile del Magazzino";
		mes "Ti aprirò il deposito della gilda allora. Divertiti!";
		close2;

		if( Zeny < 1000 ){
			// Cheat prevention
			end;
		}

		if( guildopenstorage() == GSTORAGE_OPEN ){
			Zeny -= 1000;
			end;
		}else{
			mes "Responsabile del Magazzino";
			mes "Temo che qualcun altro stia usando il magazzino in questo momento.";
			mes "Riprova tra un po'.";
			close;
		}
	}else{
		.@guildid = getcharid( 2 );

		if( .@guildid == 0 ){
			mes "Responsabile del Magazzino";
			mes "Lo stoccaggio della gilda è disponibile solo per i membri della gilda.";
			close;
		}

		if( getgdskilllv( .@guildid, "GD_GUILD_STORAGE" ) == 0 || !guild_has_permission( GUILD_PERM_STORAGE ) ){
			mes "Responsabile del Magazzino";
			mes "Sembra che non sia ancora possibile per la tua gilda utilizzare il magazzino della gilda o che tu non abbia i permessi di accesso per il magazzino.";
			mes "Ritorna dopo aver controllato la abilità della gilda e il permesso di accesso al magazzino.";
			close;
		}

		mes "Responsabile del Magazzino";
		mes "Ti mostrerò la cronologia di utilizzo del magazzino. La cronologia di utilizzo sarà conservata per un massimo di 3 mesi.";
		mes "Buon divertimento!";
		close2;
		guildopenstorage_log();
		end;
	}
}
