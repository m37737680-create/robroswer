//===== rAthena Script =======================================
//= HD Refiners
//===== Description: =========================================
//= [Official Conversion]
//= Refiners that use HD ores to refine equipment. Upon
//= failure, the equipment is not destroyed; rather, its
//= refine level decreases by 1. The success rate is identical
//= to that for Enriched ores.
//= - "Blacksmith Mighty Hammer" only refines from +7~9.
//= - "Basta" only refines from +10 and up.
//===== Changelog: ===========================================
//= 1.0 First version. [Euphy]
//= 1.1 Removed re-roll behavior. [Secret]
//============================================================

// Blacksmith Mighty Hammer (+7~9) :: cash_smelting79
//============================================================
-	script	::MightyHammer	-1,{
	disable_items;
	mes "[Il potente martello del fabbro]";
	mes "A differenza di altri, sono un fabbro che affina un numero molto limitato di oggetti.";
	mes "Perfeziono solo gli elementi da ^CC0000+7 a +9^000000.";
	next;
	mes "[Il potente martello del fabbro]";
	mes "La mia specialità è che, anche se la raffinazione fallisce, il livello di raffinazione diminuisce di 1 senza perdere l'equipaggiamento. Non è fantastico?";
	next;
	mes "[Il potente martello del fabbro]";
	mes "Quindi diamo una marcia in più a questa cosa, che ne dici? Quale articolo vuoi perfezionare?";
	next;
	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for ( .@i = 1; .@i <= 10; ++.@i )
		.@menu$ = .@menu$ + ( getequipisequiped(.@indices[.@i]) ? getequipname(.@indices[.@i]) : F_getpositionname(.@indices[.@i]) + "-[Non equipaggiato]" ) + ":";
	.@part = .@indices[ select(.@menu$) ];
	if (!getequipisequiped(.@part)) {
		mes "[Il potente martello del fabbro]";
		switch(.@part) {
		case EQI_HEAD_TOP:
			mes "Sono un fabbro, non un parrucchiere.";
			break;
		case EQI_ARMOR:
			mes "Con il mio martello farò di te una stella del cielo.";
			break;
		case EQI_HAND_L:
		case EQI_HAND_R:
			mes "Fare mani artificiali non è la mia specialità.";
			break;
		case EQI_GARMENT:
			mes "Tira fuori l'oggetto così posso perfezionarlo!";
			break;
		case EQI_SHOES:
			mes "Da dove viene questo odore di piedi?";
			break;
		case EQI_ACC_L:
		case EQI_ACC_R:
			mes "Dov'è l'accessorio?";
			break;
		case EQI_HEAD_MID:
			mes "Cosa vuoi che perfezioni?";
			break;
		case EQI_HEAD_LOW:
			mes "Eh? Cosa vuoi che faccia?";
			break;
		}
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "[Il potente martello del fabbro]";
		mes "Questo oggetto non può essere perfezionato.";
		close;
	}
	switch( getequiprefinerycnt(.@part) ) {
	case 7:
		.@blacksmith_blessing_count = 1;
		break;
	case 8:
		.@blacksmith_blessing_count = 2;
		break;
	case 9:
		.@blacksmith_blessing_count = 4;
		break;
	default:
		mes "[Il potente martello del fabbro]";
		mes "Gestisco solo oggetti con livelli di perfezionamento da +7 a +9.";
		close;
	}

	.@refineitemid = getequipid(.@part); // save id of the item
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_HD, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_HD, REFINE_MATERIAL_ID);

	mes "[Il potente martello del fabbro]";
	mes "Per perfezionare l'attrezzatura selezionata hai bisogno di ^ff9999" + getitemname(.@material) + "^000000 e 20.000 zeny come compenso.";
	mes "Li hai pronti?";
	next;
	if (select("SÌ:NO") == 2) {
		mes "[Il potente martello del fabbro]";
		mes "Aspetterò finché non sarai pronto.";
		close;
	}
	if (getequippercentrefinery(.@part) < 100) {
		mes "[Il potente martello del fabbro]";
		mes "Sembra che questo oggetto probabilmente non verrà perfezionato.";
		mes "Ebbene, anche se fallisce, diminuisce solo di 1 livello di raffinazione.";
		mes "Desideri continuare a perfezionare?";
		next;
		if (countitem(6635) < .@blacksmith_blessing_count)
			setarray .@menu$[0], "", "Yes", "Not yet";
		else {
			mes "[Il potente martello del fabbro]";
			mes "Ah! È la ^0000ffBenedizione del fabbro^000000?";
			mes "Con la Benedizione del Fabbro, l'equipaggiamento non svanirà se la raffinazione fallisce!";
			next;
			mes "[Il potente martello del fabbro]";
			if ( .@itemtype != IT_WEAPON )
				mes "Per +" + getequiprefinerycnt(.@part) + " attrezzatura, perfeziona con^316AC5 " + .@blacksmith_blessing_count + " le unità Blacksmith Blessing^000000 possono impedire la scomparsa dell'equipaggiamento. Vuoi usarlo per affinare?";
			else
				mes "Per +" + getequiprefinerycnt(.@part) + " arma, perfeziona con^316AC5 " + .@blacksmith_blessing_count + " le unità Blacksmith Blessing^000000 possono impedire la scomparsa dell'equipaggiamento. Vuoi usarlo per affinare?";
			next;
			setarray .@menu$[0], "Use it to refine", "Refine directly without it", "Don't refine yet";
		}
		switch( select(.@menu$[0], .@menu$[1], .@menu$[2]) ) {
		case 1:
			.@bless_who = 1;
			break;
		case 2:
			break;
		case 3:
			mes "[Il potente martello del fabbro]";
			mes "Solo chi supera la paura di fallire otterrà un capolavoro.";
			close;
		}
	}
	if ((.@bless_who && countitem(6635) < .@blacksmith_blessing_count) || countitem(.@material) == 0 || Zeny < .@price) {
		mes "[Il potente martello del fabbro]";
		mes "Non hai appena detto che avevi tutto pronto?";
		close;
	}
	if (.@bless_who)
		delitem 6635, .@blacksmith_blessing_count;
	delitem .@material,1;
	Zeny = Zeny - .@price;

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
		mes "[Il potente martello del fabbro]";
		emotion ET_FRET;
		mes "Aspetta un secondo...";
		mes "Pensi che io sia stupido?!";
		mes "Hai cambiato oggetto mentre non guardavo! Fuori di qui!";
		close;
	}

	mes "[Il potente martello del fabbro]";
	mes "Tac! Tac! Tac!";
	if (getequippercentrefinery(.@part, true) > rand(100)) {
		successrefitem .@part;
		next;
		emotion ET_BEST;
		mes "[Il potente martello del fabbro]";
		mes "Il suono mi rinfresca la mente ogni volta che lo ascolto.";
		mes "Ecco, prendilo. Il perfezionamento è riuscito in modo impeccabile!";
		close;
	}
	if (.@bless_who == 1) {
		specialeffect EF_HOLYHIT;
		next;
		emotion ET_HUK;
		mes "[Il potente martello del fabbro]";
		mes "Che cosa?!!";
		next;
		mes "[Il potente martello del fabbro]";
		mes "Aiya! Sono senza volto adesso. HmHm.";
		close;
	}
	downrefitem .@part;
	next;
	emotion ET_HUK;
	mes "[Il potente martello del fabbro]";
	mes "Ops!!";
	next;
	mes "[Il potente martello del fabbro]";
	mes "Sono sicuro che una persona come te non mi incolperebbe mai per una diminuzione del livello di raffinamento di 1. Hmm.";
	close;
}

// Basta (+10 and up) :: cash_smelting
//============================================================
-	script	::Basta	-1,{
	disable_items;
	mes "[Basta]";
	mes "Sono il miglior fabbro del mondo, Basta.";
	mes "Ma non fornisco un normale servizio di rifinitura.";
	mes "Perfeziono l'attrezzatura solo ^CC0000oltre +10^000000.";
	next;
	mes "[Basta]";
	mes "Quale attrezzatura vuoi perfezionare?";
	next;
	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for ( .@i = 1; .@i <= 10; ++.@i )
		.@menu$ = .@menu$ + ( getequipisequiped(.@indices[.@i]) ? getequipname(.@indices[.@i]) : F_getpositionname(.@indices[.@i]) + "-[Non equipaggiato]" ) + ":";
	.@part = .@indices[ select(.@menu$) ];
	if (!getequipisequiped(.@part)) {
		mes "[Basta]";
		switch(.@part) {
		case EQI_HEAD_TOP:
			mes "La tua testa è un'attrezzatura?";
			break;
		case EQI_ARMOR:
			mes "Cosa vuoi che faccia?";
			break;
		case EQI_HAND_L:
		case EQI_HAND_R:
			mes "Fare mani artificiali non è la mia specialità.";
			break;
		case EQI_GARMENT:
			mes "Sai almeno cos'è una veste?";
			break;
		case EQI_SHOES:
			mes "Se vuoi affinare i tuoi piedi, non venire da me, prova a correre una maratona.";
			break;
		case EQI_ACC_L:
		case EQI_ACC_R:
			mes "Dov'è l'accessorio?";
			break;
		case EQI_HEAD_MID:
			mes "Beh... non vedo nessuna attrezzatura che valga la pena perfezionare.";
			break;
		case EQI_HEAD_LOW:
			mes "Non posso renderti intelligente. Vai a vedere un insegnante di scuola per quello.";
			break;
		}
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "[Basta]";
		mes "Nemmeno io riesco a perfezionare questo articolo. Non c'è modo.";
		close;
	}
	.@refine_count = getequiprefinerycnt(.@part);
	if (.@refine_count < 10) {
		mes "[Basta]";
		mes "Non te l'ho detto? Perfeziono solo l'equipaggiamento +10 e superiore.";
		close;
	}
	else if (.@refine_count == 10)
		.@blacksmith_blessing_count = 7;
	else if (.@refine_count == 11)
		.@blacksmith_blessing_count = 11;
	else if (.@refine_count == 20) {
		mes "[Basta]";
		mes "Quest'arma è perfetta, non c'è più bisogno di perfezionarla~";
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_HD, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_HD, REFINE_MATERIAL_ID);

	if ( .@itemtype != IT_WEAPON )
		.@type$ = "armor";
	else
		.@type$ = "weapon";
	mes "[Basta]";
	mes "Hmm... è questo quello che vuoi perfezionare?";
	mes "Per perfezionare questa attrezzatura, mi serve 1 ^ff9999" + getitemname(.@material) + "^000000 e " + callfunc("F_InserisciComma",.@price) + " zeny a pagamento.";
	mes "Vuoi davvero perfezionarlo?";
	next;
	if (select("SÌ:NO") == 2) {
		mes "[Basta]";
		mes "Va bene. Se è quello che vuoi...";
		close;
	}
	if (getequippercentrefinery(.@part, true) < 100) {
		mes "[Basta]";
		mes "Questo " + .@type$ + " è già stato raffinato piuttosto in alto.";
		mes "Se provi a perfezionarlo ulteriormente, il livello di perfezionamento potrebbe diminuire.";
		next;
		mes "[Basta]";
		mes "Sono diverso dai fabbri degli altri posti.";
		mes "È impossibile che il livello di raffinazione scenda, diciamo, di 3 o 4... sembra spaventoso.";
		mes "Qui può diminuire solo di 1 livello.";
		next;
		if (.@blacksmith_blessing_count == 0 || countitem(6635) < .@blacksmith_blessing_count) {
			mes "[Basta]";
			mes "Rispetto ad altri fabbri, il rischio è minore.";
			mes "Ho dato tutte le precauzioni. Vuoi provarlo?";
			next;
			setarray .@menu$[0], "", "Yes", "No";
		}
		else {
			mes "[Basta]";
			mes "Woah~ È la ^316AC5Benedizione del fabbro^000000? È difficile da ottenere~ Ma ce l'hai!";
			next;
			mes "[Basta]";
			mes "Per +" + .@refine_count + " " + .@type$ + " bisogno di ^316AC5" + .@blacksmith_blessing_count + " le unità Blacksmith Blessing^000000 possono impedire la scomparsa dell'equipaggiamento. Vuoi perfezionarti con la Benedizione del Fabbro?";
			next;
			setarray .@menu$[0], "Refine with Blacksmith Blessing", "Refine without Blacksmith Blessing", "Don't refine yet";
		}
		switch( select(.@menu$[0], .@menu$[1], .@menu$[2]) ) {
		case 1:
			.@bless_who = 1;
			break;
		case 2:
			break;
		case 3:
			mes "[Basta]";
			mes "Bene~";
			mes "Non essere affatto impegnativo potrebbe anche essere una sorta di saggezza nella vita.";
			close;
		}
	}
	if ((.@bless_who && countitem(6635) < .@blacksmith_blessing_count) || countitem(.@material) == 0 || Zeny < .@price) {
		mes "[Basta]";
		mes "Hmm... Non hai portato tutto il materiale necessario.";
		mes "Torna quando li avrai tutti.";
		close;
	}
	if (.@bless_who)
		delitem 6635, .@blacksmith_blessing_count;
	delitem .@material,1;
	Zeny = Zeny - .@price;

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
		mes "[Basta]";
		emotion ET_FRET;
		mes "Aspetta un secondo...";
		mes "Pensi che io sia stupido?!";
		mes "Hai cambiato oggetto mentre non guardavo! Fuori di qui!";
		close;
	}

	mes "Pow! Pow! Pow! Pow!";
	if (getequippercentrefinery(.@part, true) > rand(100)) {
		successrefitem .@part;
		next;
		emotion ET_BEST;
		mes "[Basta]";
		mes "Grande! Ben fatto!!";
		mes "Sono davvero il miglior fabbro del mondo intero!";
		close;
	}
	if (.@bless_who == 1) {
		specialeffect EF_HOLYHIT;
		next;
		emotion (!rand(5))?ET_MONEY:ET_HUK;
		mes "[Basta]";
		mes "Aaaaaaaaaa!!!";
		next;
		mes "[Basta]";
		mes "La raffinazione è fallita!";
		mes "Il miglior fabbro del mondo come me...";
		mes "non garantisce il successo al 100%~";
	}
	else {
		downrefitem .@part;
		next;
		emotion (!rand(5))?ET_MONEY:ET_HUK;
		mes "[Basta]";
		mes "Aaaaaaaaaa!!!";
		next;
		mes "[Basta]";
		mes "Accidenti!";
		mes "La raffinazione è fallita e il livello di raffinazione è diminuito!";
		mes "Anche il miglior fabbro del mondo non garantisce il successo al 100%!";
	}
	mes "Peccato.";
	next;
	mes "[Basta]";
	mes "Farò meglio la prossima volta! Non preoccuparti!";
	close;
}
