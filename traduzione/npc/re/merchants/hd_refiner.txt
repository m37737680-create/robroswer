//===== rAthena Script =======================================
//= HD Refiners
//===== Description: =========================================
//= [Official Conversion]
//= Refiners that use HD ores to refine equipment. Upon
//= failure, the equipment is not destroyed; rather, its
//= refine level decreases by 1. The success rate is identical
//= to that for Enriched ores.
//= - "Blacksmith Mighty Hammer" only refines from +7~9.
//= - "Basta" only refines from +10 and up.
//===== Changelog: ===========================================
//= 1.0 First version. [Euphy]
//= 1.1 Removed re-roll behavior. [Secret]
//============================================================

// Blacksmith Mighty Hammer (+7~9) :: cash_smelting79
//============================================================
-	script	::MightyHammer	-1,{
	disable_items;
	mes "[Martello Potente del Fabbro]";
	mes "A differenza degli altri, sono un fabbro che raffina un numero molto limitato di oggetti.";
	mes "Raffino solo gli elementi che sono da ^CC0000+7 a +9^000000.";
	next;
	mes "[Martello Potente del Fabbro]";
	mes "La mia specialità è che anche se il mio raffinamento fallisce, il livello di raffinazione diminuisce di 1 senza perdere l'equipaggiamento. Non è fantastico?";
	next;
	mes "[Martello Potente del Fabbro]";
	mes "Allora facciamo partire questo a tutta velocità, che ne dici? Quale oggetto vuoi raffinare?";
	next;
	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for ( .@i = 1; .@i <= 10; ++.@i )
		.@menu$ = .@menu$ + ( getequipisequiped(.@indices[.@i]) ? getequipname(.@indices[.@i]) : F_getpositionname(.@indices[.@i]) + "Non equipaggiato" ) + ":";
	.@part = .@indices[ select(.@menu$) ];
	if (!getequipisequiped(.@part)) {
		mes "[Martello Potente del Fabbro]";
		switch(.@part) {
		case EQI_HEAD_TOP:
			mes "Sono un fabbro, non uno stilista di capelli.";
			break;
		case EQI_ARMOR:
			mes "Con il mio martello, ti farò una stella del cielo.";
			break;
		case EQI_HAND_L:
		case EQI_HAND_R:
			mes "Fare mani artificiali non è la mia specialità.";
			break;
		case EQI_GARMENT:
			mes "Tira fuori l'articolo così posso rifinarlo!";
			break;
		case EQI_SHOES:
			mes "Da dove viene questo odore di piedi?";
			break;
		case EQI_ACC_L:
		case EQI_ACC_R:
			mes "Dov'è l'accessorio?";
			break;
		case EQI_HEAD_MID:
			mes "Cosa vuoi che io migliori?";
			break;
		case EQI_HEAD_LOW:
			mes "Eh? Cosa vuoi che faccia?";
			break;
		}
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "[Martello Potente del Fabbro]";
		mes "Questo oggetto non può essere raffinato.";
		close;
	}
	switch( getequiprefinerycnt(.@part) ) {
	case 7:
		.@blacksmith_blessing_count = 1;
		break;
	case 8:
		.@blacksmith_blessing_count = 2;
		break;
	case 9:
		.@blacksmith_blessing_count = 4;
		break;
	default:
		mes "[Martello Potente del Fabbro]";
		mes "Gestisco solo oggetti con livelli di raffinazione da +7 a +9.";
		close;
	}

	.@refineitemid = getequipid(.@part); // save id of the item
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_HD, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_HD, REFINE_MATERIAL_ID);

	mes "[Martello Potente del Fabbro]";
	mes "Per affinare l'ingranaggio che hai selezionato ti serve ^ff9999" + getitemname(.@material) + "^000000 e 20.000 zeny come commissione.";
	mes "Li hai pronti?";
	next;
	if (select("Sì:No") == 2) {
		mes "[Martello Potente del Fabbro]";
		mes "Aspetterò finché non sarai pronto.";
		close;
	}
	if (getequippercentrefinery(.@part) < 100) {
		mes "[Martello Potente del Fabbro]";
		mes "Sembra che questo oggetto probabilmente non riuscirà a essere raffinato.";
		mes "Beh, anche se fallisce, diminuisce solo di un livello di raffinazione.";
		mes "Vuoi continuare a perfezionare?";
		next;
		if (countitem(6635) < .@blacksmith_blessing_count)
			setarray .@menu$[0], "", "Yes", "Not yet";
		else {
			mes "[Martello Potente del Fabbro]";
			mes "Ah! È la ^0000ffBenedizione del Fabbro^000000?";
			mes "Con la Benedizione del Fabbro, l'equipaggiamento non svanirà se il raffinare fallisce!";
			next;
			mes "[Martello Potente del Fabbro]";
			if ( .@itemtype != IT_WEAPON )
				mes "Per +" + getequiprefinerycnt(.@part) + " attrezzatura, affinare con^316AC5 " + .@blacksmith_blessing_count + " unità di Benedizione del Fabbro^000000 possono impedire che l'equipaggiamento svanisca. Vuoi usarlo per raffinare?";
			else
				mes "Per +" + getequiprefinerycnt(.@part) + " arma, raffinare con^316AC5 " + .@blacksmith_blessing_count + " unità di Benedizione del Fabbro^000000 possono impedire che l'equipaggiamento svanisca. Vuoi usarlo per raffinare?";
			next;
			setarray .@menu$[0], "Use it to refine", "Refine directly without it", "Don't refine yet";
		}
		switch( select(.@menu$[0], .@menu$[1], .@menu$[2]) ) {
		case 1:
			.@bless_who = 1;
			break;
		case 2:
			break;
		case 3:
			mes "[Martello Potente del Fabbro]";
			mes "Solo coloro che superano la paura del fallimento otterranno un capolavoro.";
			close;
		}
	}
	if ((.@bless_who && countitem(6635) < .@blacksmith_blessing_count) || countitem(.@material) == 0 || Zeny < .@price) {
		mes "[Martello Potente del Fabbro]";
		mes "Non hai appena detto che avevi tutto pronto?";
		close;
	}
	if (.@bless_who)
		delitem 6635, .@blacksmith_blessing_count;
	delitem .@material,1;
	Zeny = Zeny - .@price;

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
		mes "[Martello Potente del Fabbro]";
		emotion ET_FRET;
		mes "Aspetta un secondo...";
		mes "Pensi che sono stupido?!";
		mes "Hai cambiato l'oggetto mentre non guardavo! Via da qui!";
		close;
	}

	mes "[Martello Potente del Fabbro]";
	mes "Tac! Tac! Tac!";
	if (getequippercentrefinery(.@part, true) > rand(100)) {
		successrefitem .@part;
		next;
		emotion ET_BEST;
		mes "[Martello Potente del Fabbro]";
		mes "Il suono rinfresca la mia mente ogni volta che lo sento.";
		mes "Ecco, tieni. Rifinito con successo senza intoppi!";
		close;
	}
	if (.@bless_who == 1) {
		specialeffect EF_HOLYHIT;
		next;
		emotion ET_HUK;
		mes "[Martello Potente del Fabbro]";
		mes "Cosa?!!";
		next;
		mes "[Martello Potente del Fabbro]";
		mes "Aiya! Sono senza volto ora. HmHm.";
		close;
	}
	downrefitem .@part;
	next;
	emotion ET_HUK;
	mes "[Martello Potente del Fabbro]";
	mes "Ops!!";
	next;
	mes "[Martello Potente del Fabbro]";
	mes "Sono sicuro che una persona come te non mi incolperà per un calo di livello di raffinatezza di 1. Mmh.";
	close;
}

// Basta (+10 and up) :: cash_smelting
//============================================================
-	script	::Basta	-1,{
	disable_items;
	mes "[Basta]";
	mes "Sono il miglior fabbro di tutto il mondo, basta.";
	mes "Ma io non fornisco un normale servizio di raffinamento.";
	mes "Raffino solo l'equipaggiamento ^CC0000sopra +10^000000.";
	next;
	mes "[Basta]";
	mes "Quale attrezzatura vuoi affinare?";
	next;
	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for ( .@i = 1; .@i <= 10; ++.@i )
		.@menu$ = .@menu$ + ( getequipisequiped(.@indices[.@i]) ? getequipname(.@indices[.@i]) : F_getpositionname(.@indices[.@i]) + "-[Scollegato]" ) + ":";
	.@part = .@indices[ select(.@menu$) ];
	if (!getequipisequiped(.@part)) {
		mes "[Basta]";
		switch(.@part) {
		case EQI_HEAD_TOP:
			mes "È la tua testa un apparecchiatura?";
			break;
		case EQI_ARMOR:
			mes "Cosa vuoi che faccia?";
			break;
		case EQI_HAND_L:
		case EQI_HAND_R:
			mes "Fare mani artificiali non è la mia specialità.";
			break;
		case EQI_GARMENT:
			mes "Sai anche cosa sia un accappatoio?";
			break;
		case EQI_SHOES:
			mes "Se vuoi affinare i tuoi piedi, non venire da me, prova a correre una maratona.";
			break;
		case EQI_ACC_L:
		case EQI_ACC_R:
			mes "Dov'è l'accessorio?";
			break;
		case EQI_HEAD_MID:
			mes "Beh... non vedo nessun equipaggiamento degno di essere raffinato.";
			break;
		case EQI_HEAD_LOW:
			mes "Non posso renderti intelligente. Vai a vedere un insegnante per questo.";
			break;
		}
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "[Basta]";
		mes "Nemanche io posso raffinare questo articolo. Non c'è modo.";
		close;
	}
	.@refine_count = getequiprefinerycnt(.@part);
	if (.@refine_count < 10) {
		mes "[Basta]";
		mes "Non ti ho detto? Rinfresco solo attrezzature che sono +10 e superiori.";
		close;
	}
	else if (.@refine_count == 10)
		.@blacksmith_blessing_count = 7;
	else if (.@refine_count == 11)
		.@blacksmith_blessing_count = 11;
	else if (.@refine_count == 20) {
		mes "[Basta]";
		mes "Questa arma è perfetta, non c'è bisogno di rifinarla più~";
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_HD, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_HD, REFINE_MATERIAL_ID);

	if ( .@itemtype != IT_WEAPON )
		.@type$ = "armor";
	else
		.@type$ = "weapon";
	mes "[Basta]";
	mes "Mmh... È questo che vuoi migliorare?";
	mes "Per perfezionare questa attrezzatura, ho bisogno di 1 ^ff9999" + getitemname(.@material) + "^000000 e " + callfunc("F_InserisciVirgola",.@price) + " zeny come tariffa.";
	mes "Vuoi davvero raffinare questo?";
	next;
	if (select("Sì:No") == 2) {
		mes "[Basta]";
		mes "Va bene. Se è quello che vuoi...";
		close;
	}
	if (getequippercentrefinery(.@part, true) < 100) {
		mes "[Basta]";
		mes "Questo " + .@type$ + " già stato raffinato piuttosto bene.";
		mes "Se provi a raffinare di più, il livello di raffinatezza potrebbe diminuire.";
		next;
		mes "[Basta]";
		mes "Sono diverso dagli fabbri in altri posti.";
		mes "È impossibile che il livello di raffinazione scenda di, diciamo, 3 o 4... suona spaventoso.";
		mes "Qui può diminuire solo di 1 livello.";
		next;
		if (.@blacksmith_blessing_count == 0 || countitem(6635) < .@blacksmith_blessing_count) {
			mes "[Basta]";
			mes "Rispetto ad altri fabbri, il rischio è minore.";
			mes "Ho dato tutte le precauzioni. Vuoi provarci?";
			next;
			setarray .@menu$[0], "", "Yes", "No";
		}
		else {
			mes "[Basta]";
			mes "Wow~ È la Benedizione del Fabbro ^316AC5Blacksmith Blessing^000000? È difficile da ottenere~ Ma ce l'hai!";
			next;
			mes "[Basta]";
			mes "Per +" + .@refine_count + " " + .@type$ + " ho bisogno di ^316AC5" + .@blacksmith_blessing_count + " unità di Benedizione del Fabbro^000000 possono impedire che l'equipaggiamento svanisca. Vuoi raffinare con la Benedizione del Fabbro?";
			next;
			setarray .@menu$[0], "Refine with Blacksmith Blessing", "Refine without Blacksmith Blessing", "Don't refine yet";
		}
		switch( select(.@menu$[0], .@menu$[1], .@menu$[2]) ) {
		case 1:
			.@bless_who = 1;
			break;
		case 2:
			break;
		case 3:
			mes "[Basta]";
			mes "Beh~";
			mes "Non affatto impegnativo potrebbe anche essere una specie di saggezza nella vita.";
			close;
		}
	}
	if ((.@bless_who && countitem(6635) < .@blacksmith_blessing_count) || countitem(.@material) == 0 || Zeny < .@price) {
		mes "[Basta]";
		mes "Mmh... Non hai portato tutti i materiali necessari.";
		mes "Torna quando li hai tutti.";
		close;
	}
	if (.@bless_who)
		delitem 6635, .@blacksmith_blessing_count;
	delitem .@material,1;
	Zeny = Zeny - .@price;

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
		mes "[Basta]";
		emotion ET_FRET;
		mes "Aspetta un secondo...";
		mes "Pensi che sono stupido?!";
		mes "Hai cambiato l'oggetto mentre non guardavo! Via da qui!";
		close;
	}

	mes "Pow! Pow! Pow! Pow!";
	if (getequippercentrefinery(.@part, true) > rand(100)) {
		successrefitem .@part;
		next;
		emotion ET_BEST;
		mes "[Basta]";
		mes "Ottimo! Ben fatto!!";
		mes "Sono davvero il miglior fabbro di tutto il mondo!";
		close;
	}
	if (.@bless_who == 1) {
		specialeffect EF_HOLYHIT;
		next;
		emotion (!rand(5))?ET_MONEY:ET_HUK;
		mes "[Basta]";
		mes "Aaaaaaaaaaak!!!";
		next;
		mes "[Basta]";
		mes "Il raffinamento è fallito!";
		mes "Il miglior fabbro del mondo come me...";
		mes "non garantisce il successo al 100%~";
	}
	else {
		downrefitem .@part;
		next;
		emotion (!rand(5))?ET_MONEY:ET_HUK;
		mes "[Basta]";
		mes "Aaaaaaaaaaak!!!";
		next;
		mes "[Basta]";
		mes "Accidenti!";
		mes "Il raffinamento è fallito e il livello di raffinazione è diminuito!";
		mes "Nemanche il miglior fabbro del mondo garantisce il 100% di successo!";
	}
	mes "Peccato.";
	next;
	mes "[Basta]";
	mes "Farò meglio la prossima volta! Non preoccuparti!";
	close;
}
