moro_cav,32,73,5	script	Hero Token Exchanger#1	4_F_SHADOWCHASER,{
	disable_items;
	mes "[Hero Token Exchanger]";
	mes "Se avessi combattuto Morocchino";
	mes "e raccolto";
	mes "Token Eroe,";
	mes "Posso scambiarli con un anello speciale per te.";
	mes "Vuoi dare un'occhiata al mio anello?";
	while(true) {
		next;
		if (checkweight(2163,2) == 0) {
			mes "[Hero Token Exchanger]";
			mes "Il tuo zaino ha molti oggetti,";
			mes "Per favore torna dopo averlo sistemato!";
			close;
		}
		switch( select( "Descrizione dell'anello", "Scambio con l'anello.", "Termina conversazione." ) ) {
		case 1:
			mes "[Hero Token Exchanger]";
			mes "Questo Anello Eroe";
			mes "non può";
			mes "guarda";
			mes "impressionante,";
			next;
			mes "[Hero Token Exchanger]";
			mes "ma portalo a";
			mes "Incantatore Esperto Byrnes,";
			mes "e può Incantare";
			mes "tutte e 4 le prese in esso";
			mes "e portare fuori";
			mes "sono abilità latenti.";
			next;
			mes "[Hero Token Exchanger]";
			mes "Anello Eroe";
			mes "può aiutare";
			mes "nel sottomettere";
			mes "Marocco";
			mes "Altre domande?";
			continue;
		case 2:
			mes "Anello Eroe";
			mes "^006400A ring made for warriors^000000";
			mes "^006400per sconfiggere Morocc..^000000";
			mes "^006400When Enchanted,^000000";
			mes "^006400it lends its owner^000000";
			mes "^006400the power to confront Morocc.^000000";
			mes "^006400Classe: Accessory/DEF: 0^000000";
			mes "^006400Peso: 20/Livello Richiesto: 160^000000";
			mes "^006400Slot: 0/Equip To: Tutti i Lavori^000000";
			next;
			mes "[Hero Token Exchanger]";
			mes "Anello Eroe richiede";
			mes "30 Token Eroe";
			mes "per lo scambio.";
			mes "Vuoi scambiare?";
			next;
			if (select( "Menu precedente.", "Scambiare 30 Token Eroe." ) == 1) {
				mes "[Hero Token Exchanger]";
				mes "Pensare";
				mes "di nuovo.";
				continue;
			}
			if (countitem(6684) < 30) {
				mes "[Hero Token Exchanger]";
				mes "^006400Eroe Token^000000";
				mes "è dato a";
				mes "coloro che combatterono coraggiosamente i Marocchi.";
				mes "Non hai";
				mes "30 ^006400Gettoni Eroe^000000,";
				mes "e quindi non posso assisterti.";
				continue;
			}
			mes "[Hero Token Exchanger]";
			mes "Scambierò";
			mes "30 ^006400Hero Tokens^000000";
			mes "con 1 Anello Eroe ^006400Anello Eroe^000000.";
			mes "Qualcos'altro?";
			delitem 6684,30;	// TokenOfHero
			getitem 2981,1;		// RingOfHero
			continue;
		case 3:
			mes "[Hero Token Exchanger]";
			mes "Un giorno";
			mes "ti ritroverai";
			mes "ho bisogno dell'anello degli eroi.";
			close;
		}
	}
}

moro_cav,34,65,5	script	Expert Enchanter Byrnes	1_M_03,{
	disable_items;
	mes "[Byrnes]";
	mes "Hai sentito parlare dell'Anello Eroe? Solo i veri eroi meritano di possedere quell'anello magnifico, e io ne aspettavo uno.";
	mes "Cosa posso fare per te?";
	next;
	switch( select( "Incantare Anello Eroe.", "Resetta gli incantesimi sull'anello eroe.", "Annulla." ) ) {
	case 1:
		mes "[Byrnes]";
		mes "Posso Incantare solo gli anelli equipaggiati. Quale degli anelli equipaggiati vuoi che io incanti?";
		next;
		setarray .@eqi_slot[0], EQI_ACC_L, EQI_ACC_R;
		.@s = select( "Sinistra", "Giusto" ) - 1;
		switch( getequipid(.@eqi_slot[.@s]) ) {
		case -1:
			mes "[Byrnes]";
			mes "Equipare l'Anello dell'Eroe che vuoi incantare.";
			close;
		case 2981:
			break;
		default:
			mes "[Byrnes]";
			mes "Equipare il tuo Anello Eroe. Non posso Incantare nient'altro.";
			close;
		}
		// custom at this point
		setarray .@card[0],
			getequipcardid(.@eqi_slot[.@s],0),
			getequipcardid(.@eqi_slot[.@s],1),
			getequipcardid(.@eqi_slot[.@s],2),
			getequipcardid(.@eqi_slot[.@s],3);

		if (.@card[0] > 0) {
			mes "[Byrnes]";
			mes "Un Anello Eroe completamente incantato non può essere più incantato.";
			close;
		}
		if (.@card[3] == 0) {
			setarray .@enchant_list$[0], "DEF", "MDEF";
			.@slot = 4;
		}
		else if (.@card[2] == 0) {
			setarray .@enchant_list$[0], "Max HP", "Max SP";
			.@slot = 3;
		}
		else if (.@card[1] == 0) {
			setarray .@enchant_list$[0], "Atk", "MAtk";
			.@slot = 2;
		}
		else {
			setarray .@enchant_list$[0], "STR", "INT", "AGI", "VIT", "DEX", "LUK";
			.@slot = 1;
		}
		mes "[Byrnes]";
		mes "Quale abilità vorresti come incantesimo su slot " + .@slot + "?";
		next;
		.@enchant_list_size = getarraysize(.@enchant_list$);
		for ( .@i = 0; .@i < .@enchant_list_size; ++.@i )
			.@menu$ += .@enchant_list$[.@i] + ":";
		.@type = select(.@menu$) - 1;

		switch(.@slot) {
		case 4:
			switch(.@type) {// enchant list index
			case 0:	// DEF
				setarray .@bonus[0], 4791, 4792, 4793, 4794, 4893;
				break;
			default:	// MDEF
				setarray .@bonus[0], 4890, 4786, 4891, 4787, 4892;
				break;
			}
			break;
		case 3:
			switch(.@type) {
			case 0:	// Max HP
				setarray .@bonus[0], 4861, 4862, 4867, 4868, 4900;
				break;
			default:	// Max SP
				setarray .@bonus[0], 4800, 4870, 4871, 4801, 4802;
				break;
			}
			break;
		case 2:
			switch(.@type) {
			case 0:	// Atk
				setarray .@bonus[0], 4882, 4766, 4767, 4894, 4895;
				break;
			default:	// Matk
				setarray .@bonus[0], 4883, 4896, 4897, 4898, 4899;
				break;
			}
			break;
		case 1:
			switch(.@type) {
			case 0:	// STR
				setarray .@bonus[0], 4700, 4701, 4702, 4703, 4704;
				break;
			case 1:	// INT
				setarray .@bonus[0], 4710, 4711, 4712, 4713, 4714; 
				break;
			case 2:	// AGI
				setarray .@bonus[0], 4730, 4731, 4732, 4733, 4734;
				break;
			case 3:	// VIT
				setarray .@bonus[0], 4740, 4741, 4742, 4743, 4744;
				break;
			case 4:	// DEX
				setarray .@bonus[0], 4720, 4721, 4722, 4723, 4724;
				break;
			default:	// LUK
				setarray .@bonus[0], 4750, 4751, 4752, 4753, 4754;
				break;
			}
			break;
		}

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@eqi_slot[.@s], 2981) || callfunc("F_IsEquipCardHack", .@eqi_slot[.@s], .@card[0], .@card[1], .@card[2], .@card[3]))
			close;

		.@bonus_size = getarraysize(.@bonus);
		.@card[.@slot-1] = .@bonus[ rand(.@bonus_size) ];	// unknown rates
		specialeffect2 EF_ANGEL2;
		mes "[Byrnes]";
		mes "Incantesimo completato!";
		delequip .@eqi_slot[.@s];
		getitem2 2981,1,1,0,0,.@card[0],.@card[1],.@card[2],.@card[3];
		close;
	case 2:
		mes "[Byrnes]";
		mes "Per resettare gli Incantesimi sul tuo Anello Eroe, ho bisogno di 3 Gettoni Eroe.";
		next;
		if (countitem(6684) < 3) {
			mes "[Byrnes]";
			mes "Non hai abbastanza Token Eroe. Non posso resettare il tuo Anello Eroe.";
			close;
		}
		mes "[Byrnes]";
		mes "Posso resettare solo gli anelli equipaggiati. Quale degli anelli equipaggiati vuoi resettare?";
		next;
		setarray .@eqi_slot[0], EQI_ACC_L, EQI_ACC_R;
		.@s = select( "Sinistra", "Giusto" ) - 1;
		switch( getequipid(.@eqi_slot[.@s]) ) {
		case -1:
			mes "[Byrnes]";
			mes "Devi prima equipaggiare l'oggetto per resettare il suo stato di Incantesimo.";
			close;
		case 2981:
			break;
		default:
			mes "[Byrnes]";
			mes "Equipare il tuo Anello Eroe. Non posso resettare nient'altro.";
			close;
		}
		// custom at this point
		if (countitem(6684) < 3) {
			mes "[Byrnes]";
			mes "Non hai abbastanza Token Eroe. Non posso resettare il tuo Anello Eroe.";
			close;
		}
		setarray .@card[0],
			getequipcardid(.@eqi_slot[.@s],0),
			getequipcardid(.@eqi_slot[.@s],1),
			getequipcardid(.@eqi_slot[.@s],2),
			getequipcardid(.@eqi_slot[.@s],3);

		if (.@card[3] == 0) {
			mes "[Byrnes]";
			mes "Il tuo Anello Eroe è attualmente in uno stato senza incantesimo.";
			close;
		}

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@eqi_slot[.@s], 2981) || callfunc("F_IsEquipCardHack", .@eqi_slot[.@s], .@card[0], .@card[1], .@card[2], .@card[3]))
			close;

		specialeffect2 EF_ANGEL2;
		mes "[Byrnes]";
		mes "Inizializzazione completata.";
		delitem 6684,3;
		delequip .@eqi_slot[.@s];
		getitem 2981,1;
		close;
	case 3:
		close;
	}
}
