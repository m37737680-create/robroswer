prontera,82,108,5	script	Heart Merchant#life01	4_M_NFDEADMAN2,5,5,{
	.@old_item = 12209;
	.@new_item = 6413;
	.@old_item_name$ = getitemname(.@old_item);
	.@new_item_name$ = getitemname(.@new_item);

	if (!checkweight(1201,1) || (MaxWeight - Weight) < 500) {
		mes "Non puoi procedere con la conversazione perché hai una grande quantità di articoli.";
		mes "Organizza i tuoi elementi e riprova.";
		close;
	}
	mes "[Mercante di cuori]";
	mes "Ciao! Vedo che sei un avventuriero che mette il cuore nel viaggiare, usa una barca per navigare attraverso il continente e ama il brivido del pericolo!";
	next;
	if(!countitem(.@old_item)){
		mes "[Mercante di cuori]";
		mes "Ne hai qualcuno? " + mesitemlink( .@old_item ) + " nel tuo magazzino?";
		next;
		mes "[Mercante di cuori]";
		mes "Se mi porti un " + .@old_item_name$ + ", lo scambierò con " + mesitemlink( .@new_item ) + ". Cosa ne pensi? Hai idea di cosa sia questo oggetto?";
	} else {
		mes "[Mercante di cuori]";
		mes "Hai mai pensato di sostituire il tuo " + mesitemlink( .@old_item ) + " a qualcosa di meglio? Per esempio... " + mesitemlink( .@new_item ) + ". Cosa ne pensi?";
	}
	next;
	switch(select("Che cos'è?",(countitem(.@old_item)?"Scambiatelo per favore!":""),"Non mi interessa")) {
	case 1:
		mes "[Mercante di cuori]";
		mes mesitemlink( .@old_item ) + " is an item that prevents you from losing experience on death once within the 30 minutes duration after use.";
		next;
		mes "[Mercante di cuori]";
		mes "Ma molte persone sono tristi per il limite di tempo, giusto? Quindi l'ho esaminato e ho preparato il " + mesitemlink( .@new_item ) + ".";
		next;
		mes "[Mercante di cuori]";
		mes mesitemlink( .@new_item ) + " is an item that will prevent the loss of experience on death once by just having it ^EE0000regardless of the duration^000000. Of course, after death, the "+.@new_item_name$+" is consumed.";
		next;
		select("Perché lo stai scambiando?");
		mes "[Mercante di cuori]";
		mes "Sei curioso a riguardo? Sono un commerciante. Comprare e vendere cose è il mio destino, sia che sia legato alla propria vita o alla propria morte. Pensi che perderò questa opportunità di fare soldi? Guadagno soldi e i vivi hanno un margine di manovra dal dolore della morte. Non è un buon affare?";
		next;
		mes "[Mercante di cuori]";
		mes "Il tasso di cambio è 5 " + mesitemlink( .@old_item ) + " per 1 " + mesitemlink( .@new_item ) + ". Hai qualche domanda?";
		next;
		mes "[Mercante di cuori]";
		mes "Hm~ Cosa ne pensi? Non è un commercio equo? Vuoi preoccuparti sempre del tempo? È un must per gli avventurieri che amano le avventure mozzafiato.";
		close;
		
	case 2:	
		mes "[Mercante di cuori]";
		mes "Yeah Yeah. Buona decisione. È meglio cambiarlo con uno migliore, non è vero?";
		next;
		mes "[Mercante di cuori]";
		mes "Posso scambiare il tuo ^0000cd5 "+.@old_item_name$+" a 1 "+.@new_item_name$+".";
		next;
		mes "[Mercante di cuori]";
		mes "Tuttavia, ^EE0000[Non in vendita] " + .@old_item_name$ + " non è idoneo allo scambio^000000, quindi è inutile portarlo qui e chiedere che venga scambiato.";
		next;
		mes "[Mercante di cuori]";
		mes "Lo scambiamo adesso?";
		next;
		if (select("Scambiatelo per favore!:Non voglio scambiarlo.") == 2)
			break;
		mes "[Mercante di cuori]";
		mes "Quanti ne vuoi scambiare? ^0000cdPuoi scambiarne fino a 10 "+.@new_item_name$+" alla volta^000000.";
		next;
		input .@amount;
		if (.@amount == 0) {
			mes "[Mercante di cuori]";
			mes "Ehm? Non sei interessato? Questo è un ottimo prodotto ed è davvero ottimo per qualcuno come te. Ma se non vuoi proteggere la tua vita, non ti costringerò neanche io.";
			close;
		}
		if (.@amount > 10) {
			mes "[Mercante di cuori]";
			mes "Come ho detto, ^0000cdpuoi scambiare solo fino a 10^000000 alla volta. Controlla di nuovo la quantità.";
			close;
		}
		if (countitem(.@old_item) < (.@amount*5)) {
			mes "[Mercante di cuori]";
			mes "Uhm, scusami ma... non ne hai abbastanza " +.@old_item_name$+ ", NO? Sono le 5 " + mesitemlink( .@old_item ) + " per 1 " + mesitemlink( .@new_item ) + ". Si prega di verificare la quantità.";
			close;
		}
		mes "[Mercante di cuori]";
		mes "Ora, con questo, ho scambiato il tuo ^0000cd" + .@old_item_name$ + "^000000 per il ^EE0000" + .@new_item_name$ + "^000000!! È stato un buon affare, vero? La prossima volta che metti le mani su a "+.@old_item_name$+", vieni a trovarmi di nuovo!";
		delitem .@old_item,.@amount*5;
		getitem .@new_item,.@amount;
		close;
	case 3:
		break;
	}
	mes "[Mercante di cuori]";
	mes "Questo è un ottimo affare! Avventuriero, è un oggetto indispensabile per chi come te si avventura in luoghi pericolosi!";
	next;
	mes "[Mercante di cuori]";
	mes "Se ti interessa, assicurati di tornare per lo scambio. Va bene?";
	close;
	
OnTouch:
	if(countitem(12209))
		npctalk "Hai qualcosa di buono. Vorresti sostituire il tuo " + getitemname(12209) + " with a better one?", "", bc_self;
	end;
}
