prontera,82,108,5	script	Heart Merchant#life01	4_M_NFDEADMAN2,5,5,{
	.@old_item = 12209;
	.@new_item = 6413;
	.@old_item_name$ = getitemname(.@old_item);
	.@new_item_name$ = getitemname(.@new_item);

	if (!checkweight(1201,1) || (MaxWeight - Weight) < 500) {
		mes "Non puoi procedere con la conversazione perché hai una grande quantità di elementi.";
		mes "Per favore, organizza i tuoi oggetti e riprova.";
		close;
	}
	mes "Mercante di Cuori";
	mes "Ciao! Vedo che sei un avventuriero che dona il suo cuore ai viaggi, usa una barca per navigare attraverso il continente e ama il brivido del pericolo!";
	next;
	if(!countitem(.@old_item)){
		mes "Mercante di Cuori";
		mes "Hai qualche " + mesitemlink( .@old_item ) + " nel tuo archivio?";
		next;
		mes "Mercante di Cuori";
		mes "Se mi porti un/una " + .@old_item_name$ + ", lo scambierò con " + mesitemlink( .@new_item ) + "Cosa ne pensi? Hai qualche idea di cosa sia questo oggetto?";
	} else {
		mes "Mercante di Cuori";
		mes "Hai mai pensato di sostituire il tuo " + mesitemlink( .@old_item ) + " a qualcosa di meglio? Per esempio... " + mesitemlink( .@new_item ) + "Cosa ne pensi?";
	}
	next;
	switch(select("Cos'è?",(countitem(.@old_item)?"Scambialo per favore!":""),"Non sono interessato")) {
	case 1:
		mes "Mercante di Cuori";
		mes mesitemlink( .@old_item ) + " is an item that prevents you from losing experience on death once within the 30 minutes duration after use.";
		next;
		mes "Mercante di Cuori";
		mes "Ma molte persone sono tristi per il limite di tempo, giusto? Quindi ho approfondito e preparato la " + mesitemlink( .@new_item ) + ".";
		next;
		mes "Mercante di Cuori";
		mes mesitemlink( .@new_item ) + " is an item that will prevent the loss of experience on death once by just having it ^EE0000regardless of the duration^000000. Of course, after death, the "+.@new_item_name$+" is consumed.";
		next;
		select("Perché lo stai scambiando?");
		mes "Mercante di Cuori";
		mes "Sei curioso di questo? Sono un mercante. Comprare e vendere cose è il mio destino, che sia legato alla vita o alla morte. Pensi che mi perderò questa opportunità di fare soldi? Io faccio soldi, e i vivi ottengono una tregua dal dolore della morte. Non è un buon affare?";
		next;
		mes "Mercante di Cuori";
		mes "Il tasso di cambio è 5 " + mesitemlink( .@old_item ) + " per 1 " + mesitemlink( .@new_item ) + "Hai domande?";
		next;
		mes "Mercante di Cuori";
		mes "Hmm~ Cosa ne pensi? Non è un buon affare? Vuoi preoccuparti sempre del tempo? È un must-have per gli avventurieri che amano avventure mozzafiato.";
		close;
		
	case 2:	
		mes "Mercante di Cuori";
		mes "Sì, sì. Buona decisione. È meglio scambiarlo con uno migliore, non è vero?";
		next;
		mes "Mercante di Cuori";
		mes "Posso scambiare il tuo ^0000cd5 "+.@old_item_name$+" a 1 "+.@new_item_name$+".";
		next;
		mes "Mercante di Cuori";
		mes "Tuttavia, ^EE0000[Non in vendita] " + .@old_item_name$ + " non è eleggibile per lo scambio^000000, quindi è inutile portarlo qui e chiedere che venga scambiato.";
		next;
		mes "Mercante di Cuori";
		mes "Scambiamolo ora?";
		next;
		if (select("Scambialo per favore!:Non voglio cambiarlo.") == 2)
			break;
		mes "Mercante di Cuori";
		mes "Quanti vuoi scambiare? ^0000cdPuoi scambiarne fino a 10 "+.@new_item_name$+" alla volta^000000.";
		next;
		input .@amount;
		if (.@amount == 0) {
			mes "Mercante di Cuori";
			mes "Hm? Non sei interessato? Questo è un ottimo prodotto, ed è davvero buono per qualcuno come te. Ma se non vuoi proteggere la tua vita, non ti costringerò nemmeno.";
			close;
		}
		if (.@amount > 10) {
			mes "Mercante di Cuori";
			mes "Come ho detto, ^0000cdpuoi scambiare al massimo 10^000000 alla volta. Controlla di nuovo la quantità.";
			close;
		}
		if (countitem(.@old_item) < (.@amount*5)) {
			mes "Mercante di Cuori";
			mes "Um, scusi ma... non ha abbastanza " +.@old_item_name$+ ", no? È 5 " + mesitemlink( .@old_item ) + " per 1 " + mesitemlink( .@new_item ) + "Per favore controlla la quantità.";
			close;
		}
		mes "Mercante di Cuori";
		mes "Ora, con questo, ho scambiato il tuo ^0000cd" + .@old_item_name$ + "^000000 per il ^EE0000" + .@new_item_name$ + "^000000!! Era un buon affare, non è vero? La prossima volta che avrai tra le mani un "+.@old_item_name$+", vieni a vedermi di nuovo!";
		delitem .@old_item,.@amount*5;
		getitem .@new_item,.@amount;
		close;
	case 3:
		break;
	}
	mes "Mercante di Cuori";
	mes "È un affare fantastico! Avventuriero, è un oggetto indispensabile per persone come te che ti avventuri in posti pericolosi!";
	next;
	mes "Mercante di Cuori";
	mes "Se diventi interessato, assicurati di tornare per lo scambio. Ok?";
	close;
	
OnTouch:
	if(countitem(12209))
		npctalk "Hai qualcosa di buono. Vorresti sostituire il tuo " + getitemname(12209) + " with a better one?", "", bc_self;
	end;
}
