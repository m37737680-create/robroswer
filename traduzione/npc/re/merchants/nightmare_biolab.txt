	}
	else if (.@r == 7) {
		showscript "Non avresti dovuto venire qui...";

	else if (.@r == 8) {
		showscript "Ciao... C'è qualcuno?";
		sc_start SC_BLIND,5000,0;
	}
	else if (.@r == 9) {
		showscript "Ho così paura....paura....";

		showscript "Caro Odino...";
