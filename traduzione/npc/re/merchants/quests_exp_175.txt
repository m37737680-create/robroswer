einbech,149,257,4	script	Site manager Elmen#extdq0	4_TOWER_16,{
	if (checkweight(25814,100) == 0 || checkweight(1201,1) == 0 || (MaxWeight - Weight) < 10000) {
		mes "Non puoi procedere perché hai troppi elementi.";
		close;
	}
	mes "[Elmen]";
	mes "Ciao. Avventuriero.";
	mes "Se sei qui per acquistare attrezzature minerarie Dynite o raffinare minerali in Dynite, sei nel posto giusto.";
	next;
	switch( select( "Raffinazione Dynite", "Negozio Dynite", "Scambio di Attrezzature Dynite" ) ) {
	case 1:
		npctalk "Elmen: Puoi raffinare fino a 100 Dynite alla volta. La macchina è piccola.", "", bc_self;
		mes "[Elmen]";
		mes "" + mesitemlink(25814) + " Minerale raffinato.";
		mes "" + mesitemlink(25809) + " ^777777(currently " + countitem(25809) + ")^000000";
		mes "" + mesitemlink(25810) + " ^777777(currently " + countitem(25810) + ")^000000";
		mes "" + mesitemlink(25811) + " ^777777(currently " + countitem(25811) + ")^000000";
		mes "" + mesitemlink(25812) + " ^777777(currently " + countitem(25812) + ")^000000";
		mes "" + mesitemlink(25813) + " ^777777(currently " + countitem(25813) + ")^000000";
		mes "Ti servono cinque tipi di minerale.";
		next;
		input .@val;
		if (.@val > 100) {
			mes "[Elmen]";
			mes "Mi dispiace. Avventuriero.";
			mes "Si può raffinare la Dynite solo fino a 100 alla volta? La macchina è piccola e non aiuta. Ti prego di capire.";
			close;
		}
		if (.@val == 0) {
			mes "[Elmen]";
			mes "Annullare?";
			mes "Se hai bisogno di affinare il Dynite, puoi trovarlo in qualsiasi momento. Non c'è nessuno che possa farlo senza di me.";
			close;
		}
		if (countitem(25809) < .@val || countitem(25810) < .@val || countitem(25811) < .@val || countitem(25812) < .@val || countitem(25813) < .@val) {
			mes "[Elmen]";
			mes "Ehi! Questo rende difficile per ognuno di noi, non ti esaurisci dei materiali necessari per il raffinamento? I materiali!";
			close;
		}
		specialeffect EF_REPAIRWEAPON;
		progressbar_npc "FFFF00",1;
		delitem 25809,.@val;	// Ein_REDIUM
		delitem 25810,.@val;	// Ein_RINDIUM
		delitem 25811,.@val;	// Ein_ODIUM
		delitem 25812,.@val;	// Ein_PURDIUM
		delitem 25813,.@val;	// Ein_WHIDIUM
		getitem 25814,.@val;	// Ein_DYNITE
		mes "[Elmen]";
		mes "" + .@val + " Dynite è stato raffinato. È perfetto oggi!";
		close;
	case 2:
		mes "[Elmen]";
		mes "Ci sono vari accessori e permessi di modifica. Guarda attentamente quello che vuoi prima di comprare.";
		close2;
		callshop "barter_ein_dun03";
		end;
	case 3:
		mes "[Elmen]";
		mes "Posso scambiare l'attrezzatura con il Dynite, ma è già in una scatola ben sigillata.";
		mes "Dammi 50 ^4d4dffDynites^000000 e ti aprirò una ^4d4dffscatola di armi^000000 per te.";
		next;
		mes "[Elmen]";
		mes "Non so cosa uscirà, ma la qualità è buona. Vorresti aprirlo?";
		next;
		if (select( "Chiudi gli occhi e scambia.", "Non scambiare." ) == 2) {
			mes "[Elmen]";
			mes "Kee, questo è uno spreco, era un buon affare.";
			close;
		}
		if (countitem(25814) < 50) {
			mes "[Elmen]";
			mes "Non hai abbastanza amici Dynite.";
			mes "Non so cosa uscirà, quindi è difficile andare da qualche parte e trovare questo tipo di attrezzatura a questo prezzo.";
			close;
		}
		delitem 25814,50;
		// unknown text
		getgroupitem(IG_EIN_1HITEMBOX);
		end;
	}
}
