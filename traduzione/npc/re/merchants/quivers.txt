//= arg(1): How many of each 'getarg(0)' arrow per quiver.
//= arg(2): The cost of making a 'getarg(0)' quiver.

// Mora :: ep14_1_tre
//============================================================
mora,106,117,3	script	Quiver Maker#mora	4_M_RAFLE_GR,{
	if (checkweight(1201,1) == 0) {
		mes "Costruttore di faretra";
		mes "Hai troppe cose con te.";
		mes "Fai un po' di spazio nel tuo inventario e torna indietro. Ti racconterò qualcosa di interessante.";
		close;
	}
	if (MaxWeight - Weight < 2000) {
		mes "Costruttore di faretra";
		mes "Sembri stanco con tutte quelle cose.";
		mes "Fai un po' di spazio nel tuo inventario e torna indietro. Ti racconterò qualcosa di interessante.";
		close;
	}
	mes "Costruttore di faretra";
	mes "Gli abitanti di Mora chiedono quali sono i buoni sacche. Semplicemente non sanno come funziona il mondo.";
	mes "Non è affatto sorprendente che non sappiano nulla di quiver—spalmare marmellata sulle foglie tutto il giorno.";
	next;
	if (getbattleflag("feature.barter")) {
		.@menu$ = "Voglio frecce";
	}
	switch( select("Fammi un sacchetto di frecce.", .@menu$, "Cos'è un carcaj?") ) {
	case 1:
		if (getbattleflag("feature.barter")) {
			mes "Costruttore di faretra";
			mes "Finalmente è apparso qualcuno che conosce il vero valore di una frusta. Lo farò per coloro che ne conoscono il merito.";
			close2;
			callshop "Quiver_Archer", 1;
			end;
		}
		mes "Costruttore di faretra";
		mes "Finalmente qualcuno apprezza un carcaj!";
		mes "Posso fare Frecoviali Elfici e Frecoviali da Caccia.";
		mes "Quale ti serve?";
		next;
		switch(select("Un sacchemino elfico:Una frecciera da caccia:Non ho bisogno di un sacchetto frecce.")) {
		case 1: callsub S_BuyQuiver,1773,500,500,12575; //Arrow_Of_Elf_Cntr
		case 2: callsub S_BuyQuiver,1774,500,500,12576; //Hunting_Arrow_Cntr
		case 3:
			mes "Costruttore di faretra";
			mes "Puoi comprare frecce dal mercante di strumenti accanto a me.";
			close;
		}
		end;
	case 2:
		if (getbattleflag("feature.barter")) {
			mes "Costruttore di faretra";
			mes "Non vendo frecce ordinarie. Queste sono frecce artigianali che possono essere acquistate solo qui. Riconosceresti davvero il loro valore, giusto?";
			close2;
			callshop "Catch_Arrows", 1;
		}
		end;
	case 3:
		mes "Costruttore di faretra";
		mes "Una freccia può essere sottile e leggera, ma portare centinaia e migliaia di frecce è come portare un intero tronco.";
		next;
		mes "Costruttore di faretra";
		mes "Ma quando hai i quiver, puoi metterci le frecce e risparmiare peso e spazio.";
		mes "Se sei interessato a uno, te ne cucirò uno.";
		close;
	}
	end;

S_BuyQuiver:
	.@item_req = getarg(0);
	.@req_amount = getarg(1);
	if (countitem(getarg(0)) < .@req_amount) {
		mes "Costruttore di faretra";
		mes "Portare più di "+ .@req_amount +" "+ getitemname(.@item_req) +" e ti farò un sacchetto di frecce.";
		close;
	}
	mes "Costruttore di faretra";
	mes "Oh, vedo che hai "+ getitemname(.@item_req) +" nelle tue mani!";
	mes "Sei interessato ad usare un manubrio? È davvero comodo!";
	mes "Se sei interessato, posso scambiare "+ .@req_amount +" di quelle frecce per uno di questi quiver per ^FF3131"+getarg(2)+" zeny^000000.";
	next;
	switch(select("Scambia tutte le frecce che hai:Prendi solo un carcaj:Non scambiare")) {
	case 1:
		set .@arrows, countitem(.@item_req);
		set .@quiver, .@arrows / .@req_amount;
		set .@arrows_used, .@quiver * .@req_amount;
		set .@arrow_zeny01, .@quiver * getarg(2);
		mes "Il numero di frecce che hai: ^3131FF"+.@arrows+"^000000";
		mes "Il numero di quiver disponibili: ^3131FF"+.@quiver+"^000000";
		mes "Zeny necessari per lo scambio: ^3131FF"+.@arrow_zeny01+" zeny^000000";
		next;
		mes "Commercio?";
		next;
		if(select("Commercio:Non scambiare") == 2) {
			mes "Costruttore di faretra";
			mes "Non dubiti delle mie capacità, vero?";
			close;
		}
		break;
	case 2:
		set .@quiver, 1;
		set .@arrows_used, .@req_amount;
		set .@arrow_zeny01, getarg(2);
		set .@zeny_mes,1;
		break;
	case 3:
		mes "Costruttore di faretra";
		mes "Non dubiti delle mie capacità, vero?";
		close;
	}
	if (Zeny < .@arrow_zeny01) {
		mes "Costruttore di faretra";
		if (.@zeny_mes == 1)
			mes "Ho detto che avrei accettato monete umane solo per te, e tu ancora non vuoi spendere i soldi?";
		else
			mes "Non ti aspetti di ottenere un capolavoro gratis, vero?";
		close;
	}
	mes "Costruttore di faretra";
	mes "Ehi, eccoti.";
	mes "C'è qualcosa che devi sapere^3131FF qualcosa che devi sapere^000000 - prova a ricordarlo.";
	mes "^FF0000You can't use quivers when your encumbrance is over 70%.^000000";
	mes "Meglio tenerlo a mente, o potresti avere problemi più tardi.";
	set Zeny, Zeny-.@arrow_zeny01;
	delitem .@item_req,.@arrows_used;
	getitem getarg(3),.@quiver;
	close;
}
