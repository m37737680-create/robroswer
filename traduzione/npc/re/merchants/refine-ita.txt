//============================================================
// +11 and above Refiner Function
//============================================================
//= To allow auto safe refining/multiple refining set the
//= second argument to '1' in the function call.
//= If you enable this function, be sure to edit the value of
//= .@safe to the max safe refine in refine.yml as well.
//=
//= On official servers, if an item is unsuccessfully refined
//= it will break at a 20% rate and downgrade at an 80% rate.
//============================================================
function	script	refinenew	{
	.@npc_name$ = getarg(0);
	disable_items;
	mes "["+ .@npc_name$ +"]";
	mes "Sono il miglior armaiolo di sempre!";
	mes "Non perfeziono oggetti normali e noiosi.";
	mes "Perfeziono solo gli oggetti di livello 10 o superiore.";
	next;
	mes "["+ .@npc_name$ +"]";
	mes "Ad ogni modo, puoi utilizzare i miei servizi se il tuo oggetto è di livello 10 o superiore.";
	mes "Cosa vuoi che perfezioni?";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(.@i = 1; .@i<=10; ++.@i) {
		if (getequipisequiped(.@indices[.@i])) {
			.@menu$ = .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			.@equipped = 1;
		}
		.@menu$ = .@menu$ + ":";
	}
	if (.@equipped == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "Non credo di poter perfezionare gli elementi che hai...";
		close;
	}
	.@part = .@indices[ select(.@menu$) ];

	if (!getequipisequiped(.@part)) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "Non indossi";
		mes "qualsiasi cosa lì";
		mes "Posso perfezionare.";
		emotion ET_FRET;
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "["+ .@npc_name$ +"]";
		mes "Non penso di poterlo fare";
		mes "perfezionare completamente questo articolo...";
		close;
	}
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	if (.@refinerycnt < 10) {
		mes "["+ .@npc_name$ +"]";
		mes "Ho detto che non lavoro con oggetti inferiori al livello 10.";
		close;
	}
	if (.@refinerycnt >= 20) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "Non posso perfezionarlo";
		mes "più. Questo è come";
		mes "raffinato come arriva!";
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);
	.@safe = 10;
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	
	if( .@itemtype == IT_WEAPON ){
		.@article$ = "a";
		.@type$ = "weapon";
	}else if( .@itemtype == IT_ARMOR ){
		.@article$ = "an";
		.@type$ = "armor";
	}else{
		// TODO:
		close;
	}

	mes "["+ .@npc_name$ +"]";
	mes "Hmm " + .@article$ + " " + .@type$ + ", va bene?";
	mes "Se vuoi perfezionarlo " + .@type$ + ",";
	mes "Me ne servirà 1 ^003366" + getitemname(.@material) + "^000000 e " + callfunc("F_InserisciComma",.@price) + " zenico.";
	mes "Sei sicuro di voler continuare?";
	next;
	if (select("SÌ:NO") == 2) {
		mes "["+ .@npc_name$ +"]";
		mes "Hm... se ti dispiace... non importa...";
		close;
	}
	if (getarg(1) != 1) {
		if (getequippercentrefinery(.@part) < 100) {
			mes "["+ .@npc_name$ +"]";
			mes "Questo "+.@type$+" è già stato perfezionato più volte.";
			mes "Potrebbe essere distrutto se provo di nuovo.";
			mes "Non si romperà di sicuro, ma c'è una piccola possibilità.";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "Potresti ^FF0000abbassare il livello di aggiornamento^000000 di "+.@type$+",";
			mes "oppure, se si rompe, perderai ^FF0000tutte le carte^000000 o le proprietà speciali aggiunte ad essa.";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "Vuoi ancora che lo rifinisca?";
			mes "Penso di averti dato abbastanza avvertimento.";
			next;
			if(select("SÌ.:NO.") == 2) {
				mes "["+ .@npc_name$ +"]";
				mes "Beh, nessuna sfida è una strada da percorrere...";
				mes "Nessun rischio... potrebbe essere saggio.";
				close;
			}
		}
		if (countitem(.@material) < 1 || Zeny < .@price) {
			mes "["+ .@npc_name$ +"]";
			mes "Uhm. Sembra che tu non abbia abbastanza soldi o "+getitemname(.@material)+".";
			mes "Per favore, torna quando li avrai.";
			close;
		}
		Zeny = Zeny - .@price;
		delitem .@material,1;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
			mes "["+ .@npc_name$ +"]";
			emotion ET_FRET;
			mes "Aspetta un secondo...";
			mes "Pensi che io sia stupido?!";
			mes "Hai cambiato oggetto mentre non guardavo! Fuori di qui!";
			close;
		}

		if (getequippercentrefinery(.@part) > rand(100)) {
			mes "Clang! Clang! Clang! Clang!";
			successrefitem .@part;
			next;
			emotion ET_BEST;
			mes "["+ .@npc_name$ +"]";
			mes "Bene! Successo!!!";
			mes "Sono il miglior armaiolo.";
			close;
		} else {
			if (rand(100) < 80) {
				mes "["+ .@npc_name$ +"]";
				mes "Clang! Clang! Clang! Clang!";
				downrefitem .@part,3; // Failed refine attempts decrease the item's refine level by 3
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "Ahhh!!!";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "Dio mio!";
				mes "Il livello di aggiornamento è sceso...";
			} else {
				mes "["+ .@npc_name$ +"]";
				mes "Clang! Clang! Clang!";
				failedrefitem .@part;
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "Hmm!";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "Oh mio Dio! Non sono riuscito a perfezionare le cose...";
				mes "Non lo intendevo!";
			}
			mes "Avrei potuto commettere un errore anche se sono il miglior armaiolo di sempre.";
			mes "Semplicemente non doveva succedere.";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "La prossima volta farò un lavoro migliore! Non preoccuparti!";
			close;
		}
	}
// New +11 and above Refining Functions ========================
	if (.@refinerycnt < .@safe) {
		mes "["+ .@npc_name$ +"]";
		mes "Posso perfezionarlo fino al limite di sicurezza o al numero di volte desiderato. È una tua scelta.";
		next;
		.@menu2 = select("Al limite di sicurezza, per favore.","Deciderò quante volte.","Ho cambiato idea...");
	} else
		.@menu2 = 2;
	switch(.@menu2){
	case 1: 
		.@refinecnt = .@safe - .@refinerycnt;
		break;
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "Quante volte vorresti che perfezionassi il tuo articolo?";
		next;
		input .@refinecnt;
		.@refinecheck = .@refinecnt + .@refinerycnt;
		if (.@refinecnt < 1 || .@refinecheck > 20) {
			mes "["+ .@npc_name$ +"]";
			mes "Non riesco a perfezionare questo elemento così tante volte.";
			close;
		}
		if (.@refinecheck > .@safe) {
			.@refinecheck = .@refinecheck - .@safe;
			mes "["+ .@npc_name$ +"]";
			mes "Questo proverà a perfezionare l'attrezzatura " + .@refinecheck + " volte oltre il limite di sicurezza. La tua attrezzatura potrebbe essere distrutta... va bene?";
			next;
			if(select("SÌ...","NO...") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "L'avevi detto tu... Così sia.";
				close;
			}
		}
		break;
	case 3:
		mes "["+ .@npc_name$ +"]";
		mes "L'avevi detto tu... Così sia.";
		close;
	}
	.@fullprice = .@price * .@refinecnt;
	mes "["+ .@npc_name$ +"]";
	mes "Questo ti costerà " + .@refinecnt + " " + getitemname(.@material) + " E " + .@fullprice + " Zeny. Va bene?";
	next;
	if(select("SÌ:NO...") == 2){
		mes "["+ .@npc_name$ +"]";
		mes "L'avevi detto tu... Così sia.";
		close;
	}
	if (countitem(.@material) < .@refinecnt || Zeny < .@fullprice) {
		mes "["+ .@npc_name$ +"]";
		mes "Uhm. Sembra che tu non abbia abbastanza soldi o "+getitemname(.@material)+".";
		mes "Per favore, torna quando li avrai.";
		close;
	}
	Zeny = Zeny - .@fullprice;
	delitem .@material,.@refinecnt;
	while(.@refinecnt){
		if (getequipisequiped(.@part) == 0) {
			mes "["+ .@npc_name$ +"]";
			mes "Guarda qui... non hai nessun articolo su...";
			close;
		}
		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt))
			close;

		if (getequipid(.@part) != .@refineitemid || (.@menu2 == 1 && getequippercentrefinery(.@part) < 100)) {
			mes "["+ .@npc_name$ +"]";
			mes "Clang... No, ma immaginavi che potessi essere così stupido?!";
			mes "L'hai cambiato...";
			mes "Esci prima che ti stordisca con il mio martello!!";
			close;
		}
		if (getequippercentrefinery(.@part) > rand(100)) {
			mes "Clang! Clang! Clang! Clang!";
			successrefitem .@part;
			.@refinecnt = .@refinecnt - 1;
			next;
		} else {
			if (rand(100) < 80) {
				mes "["+ .@npc_name$ +"]";
				mes "Clang! Clang! Clang! Clang!";
				downrefitem .@part,3; // Failed refine attempts decrease the item's refine level by 3
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "Ahhh!!!";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "Dio mio!";
				mes "Il livello di aggiornamento è sceso...";
			} else {
				mes "["+ .@npc_name$ +"]";
				mes "Clang! Clang! Clang!";
				failedrefitem .@part;
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "Hmm!";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "Oh mio Dio! Non sono riuscito a perfezionare le cose...";
				mes "Non lo intendevo!";
			}
			mes "Avrei potuto commettere un errore anche se sono il miglior armaiolo di sempre.";
			mes "Semplicemente non doveva succedere.";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "La prossima volta farò un lavoro migliore! Non preoccuparti!";
			close;
		}
		.@refinerycnt = getequiprefinerycnt(.@part);
	}
	emotion ET_BEST;
	mes "["+ .@npc_name$ +"]";
	mes "Bene! Successo!!!";
	mes "Sono il miglior fabbro.";
	close;
}
