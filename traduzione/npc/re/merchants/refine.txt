//============================================================
// +11 and above Refiner Function
//============================================================
//= To allow auto safe refining/multiple refining set the
//= second argument to '1' in the function call.
//= If you enable this function, be sure to edit the value of
//= .@safe to the max safe refine in refine.yml as well.
//=
//= On official servers, if an item is unsuccessfully refined
//= it will break at a 20% rate and downgrade at an 80% rate.
//============================================================
function	script	refinenew	{
	.@npc_name$ = getarg(0);
	disable_items;
	mes "["+ .@npc_name$ +"]";
	mes "Sono il miglior fabbro di braccia che ci sia!";
	mes "Non affino oggetti normali e noiosi.";
	mes "Rifino solo gli elementi che sono Livello 10 o superiore.";
	next;
	mes "["+ .@npc_name$ +"]";
	mes "Comunque, puoi usare i miei servizi se il tuo articolo è di Livello 10 o superiore.";
	mes "Cosa vuoi che io migliori?";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(.@i = 1; .@i<=10; ++.@i) {
		if (getequipisequiped(.@indices[.@i])) {
			.@menu$ = .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			.@equipped = 1;
		}
		.@menu$ = .@menu$ + ":";
	}
	if (.@equipped == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "Non credo di poter raffinare nessun articolo che hai...";
		close;
	}
	.@part = .@indices[ select(.@menu$) ];

	if (!getequipisequiped(.@part)) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "Non indossi";
		mes "qualcosa lì che";
		mes "Posso affinare.";
		emotion ET_FRET;
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "["+ .@npc_name$ +"]";
		mes "Non credo di potercela fare";
		mes "affinare questo elemento in qualsiasi modo...";
		close;
	}
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	if (.@refinerycnt < 10) {
		mes "["+ .@npc_name$ +"]";
		mes "Ho detto che non lavoro con articoli inferiori al Livello 10.";
		close;
	}
	if (.@refinerycnt >= 20) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "Non posso affinare questo";
		mes "ancora. Questo è come";
		mes "sempre più raffinato!";
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);
	.@safe = 10;
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	
	if( .@itemtype == IT_WEAPON ){
		.@article$ = "a";
		.@type$ = "weapon";
	}else if( .@itemtype == IT_ARMOR ){
		.@article$ = "an";
		.@type$ = "armor";
	}else{
		// TODO:
		close;
	}

	mes "["+ .@npc_name$ +"]";
	mes "Mmm " + .@article$ + " " + .@type$ + "va bene?";
	mes "Se vuoi affinare questo " + .@type$ + ",";
	mes "Avrò bisogno di 1 ^003366" + getitemname(.@material) + "^000000 e " + callfunc("F_InserisciVirgola",.@price) + " zeny.";
	mes "Sei sicuro di voler continuare?";
	next;
	if (select("Sì:No") == 2) {
		mes "["+ .@npc_name$ +"]";
		mes "Mh... se non ti dispiace... lascia stare...";
		close;
	}
	if (getarg(1) != 1) {
		if (getequippercentrefinery(.@part) < 100) {
			mes "["+ .@npc_name$ +"]";
			mes "Questo "+.@type$+" già è stato raffinato diverse volte.";
			mes "Potrebbe essere distrutto se ci riprovo.";
			mes "Non si romperà di sicuro, ma c'è una piccola possibilità.";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "You could be ^FF0000lowering the upgrade level^000000 of the "+.@type$+",";
			mes "o se si rompe, perderai tutte le carte ^FF0000 o proprietà speciali aggiunte ad esso.^000000";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "Vuoi ancora che lo migliori?";
			mes "Penso di averti dato abbastanza avvertimenti.";
			next;
			if(select("Sì.:No.") == 2) {
				mes "["+ .@npc_name$ +"]";
				mes "Beh, nessuna sfida è un modo per andare...";
				mes "Nessun rischio... quello potrebbe essere saggio.";
				close;
			}
		}
		if (countitem(.@material) < 1 || Zeny < .@price) {
			mes "["+ .@npc_name$ +"]";
			mes "Hm. Non sembri avere abbastanza soldi o "+getitemname(.@material)+".";
			mes "Torna quando li hai.";
			close;
		}
		Zeny = Zeny - .@price;
		delitem .@material,1;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
			mes "["+ .@npc_name$ +"]";
			emotion ET_FRET;
			mes "Aspetta un secondo...";
			mes "Pensi che sono stupido?!";
			mes "Hai cambiato l'oggetto mentre non guardavo! Via da qui!";
			close;
		}

		if (getequippercentrefinery(.@part) > rand(100)) {
			mes "Clang! Clang! Clang! Clang!";
			successrefitem .@part;
			next;
			emotion ET_BEST;
			mes "["+ .@npc_name$ +"]";
			mes "Bene! Successo!!!";
			mes "Sono il miglior fabbro di braccia.";
			close;
		} else {
			if (rand(100) < 80) {
				mes "["+ .@npc_name$ +"]";
				mes "Clang! Clang! Clang! Clang!";
				downrefitem .@part,3; // Failed refine attempts decrease the item's refine level by 3
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "Ahhh!!!";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "Oh mio dio!";
				mes "Il livello di aggiornamento è sceso...";
			} else {
				mes "["+ .@npc_name$ +"]";
				mes "Clang! Clang! Clang!";
				failedrefitem .@part;
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "Mmh!";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "Oh mio Dio! Ho fallito nel raffinare le cose...";
				mes "Non l'ho voluto dire!";
			}
			mes "Avrei potuto fare un errore anche se sono l'Armsmith migliore di sempre.";
			mes "Non era destinato ad essere.";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "Farò un lavoro migliore la prossima volta! Non preoccuparti!";
			close;
		}
	}
// New +11 and above Refining Functions ========================
	if (.@refinerycnt < .@safe) {
		mes "["+ .@npc_name$ +"]";
		mes "Posso affinare questo al limite di sicurezza o a un numero desiderato di volte. È la tua scelta.";
		next;
		.@menu2 = select("Al limite sicuro, per favore.","Deciderò quante volte.","Ho cambiato idea...");
	} else
		.@menu2 = 2;
	switch(.@menu2){
	case 1: 
		.@refinecnt = .@safe - .@refinerycnt;
		break;
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "Quante volte vorresti che io perfezionassi il tuo articolo?";
		next;
		input .@refinecnt;
		.@refinecheck = .@refinecnt + .@refinerycnt;
		if (.@refinecnt < 1 || .@refinecheck > 20) {
			mes "["+ .@npc_name$ +"]";
			mes "Non posso raffinare questo articolo così tante volte.";
			close;
		}
		if (.@refinecheck > .@safe) {
			.@refinecheck = .@refinecheck - .@safe;
			mes "["+ .@npc_name$ +"]";
			mes "Questo cercherà di raffinare l'attrezzatura " + .@refinecheck + " oltre il limite di sicurezza. Il tuo equipaggiamento potrebbe essere distrutto... va bene così?";
			next;
			if(select("Sì...","No...") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "Hai detto così... Sia fatto.";
				close;
			}
		}
		break;
	case 3:
		mes "["+ .@npc_name$ +"]";
		mes "Hai detto così... Sia fatto.";
		close;
	}
	.@fullprice = .@price * .@refinecnt;
	mes "["+ .@npc_name$ +"]";
	mes "Ti costerà " + .@refinecnt + " " + getitemname(.@material) + " e " + .@fullprice + " Va bene Zeny.";
	next;
	if(select("Sì:No...") == 2){
		mes "["+ .@npc_name$ +"]";
		mes "Hai detto così... Sia fatto.";
		close;
	}
	if (countitem(.@material) < .@refinecnt || Zeny < .@fullprice) {
		mes "["+ .@npc_name$ +"]";
		mes "Hm. Non sembri avere abbastanza soldi o "+getitemname(.@material)+".";
		mes "Torna quando li hai.";
		close;
	}
	Zeny = Zeny - .@fullprice;
	delitem .@material,.@refinecnt;
	while(.@refinecnt){
		if (getequipisequiped(.@part) == 0) {
			mes "["+ .@npc_name$ +"]";
			mes "Guarda qui... non hai nessun articolo su...";
			close;
		}
		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt))
			close;

		if (getequipid(.@part) != .@refineitemid || (.@menu2 == 1 && getequippercentrefinery(.@part) < 100)) {
			mes "["+ .@npc_name$ +"]";
			mes "Clang... No, ma ti sei immaginato che potessi essere così stupido?!";
			mes "L'hai cambiato...";
			mes "Vattene prima che ti stordisca con il mio Martello!!";
			close;
		}
		if (getequippercentrefinery(.@part) > rand(100)) {
			mes "Clang! Clang! Clang! Clang!";
			successrefitem .@part;
			.@refinecnt = .@refinecnt - 1;
			next;
		} else {
			if (rand(100) < 80) {
				mes "["+ .@npc_name$ +"]";
				mes "Clang! Clang! Clang! Clang!";
				downrefitem .@part,3; // Failed refine attempts decrease the item's refine level by 3
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "Ahhh!!!";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "Oh mio dio!";
				mes "Il livello di aggiornamento è sceso...";
			} else {
				mes "["+ .@npc_name$ +"]";
				mes "Clang! Clang! Clang!";
				failedrefitem .@part;
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "Mmh!";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "Oh mio Dio! Ho fallito nel raffinare le cose...";
				mes "Non l'ho voluto dire!";
			}
			mes "Avrei potuto fare un errore anche se sono l'Armsmith migliore di sempre.";
			mes "Non era destinato ad essere.";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "Farò un lavoro migliore la prossima volta! Non preoccuparti!";
			close;
		}
		.@refinerycnt = getequiprefinerycnt(.@part);
	}
	emotion ET_BEST;
	mes "["+ .@npc_name$ +"]";
	mes "Bene! Successo!!!";
	mes "Sono il miglior fabbro.";
	close;
}

	Zeny = Zeny-.@price;

		emotion ET_FRET;
