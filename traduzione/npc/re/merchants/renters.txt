job3_rune01,88,62,5	script	Dragon Breeder	105,{
	mes "Allevatore di Draghi";
	if ((eaclass()&EAJ_THIRDMASK)==EAJ_RUNE_KNIGHT) {
		mes "Benvenuto/a. Vorresti affittare un Drago?";
		next;
		switch(select("Sì:No")) {
		case 1:
			//if (!getskilllv("RK_DRAGONTRAINING")) {
			if (!getskilllv("KN_RIDING")) {
				mes "Allevatore di Draghi";
				mes "Impara prima a cavalcare un Drago.";
				close;
			}
			else if (checkdragon()) {
				mes "Allevatore di Draghi";
				mes "Hai già un Drago.";
				close;
			}
			else if (ismounting()) {
				mes "Allevatore di Draghi";
				mes "Rimuovi il tuo deposito contante.";
				close;
			}
			setdragon;
			close;
		case 2:
			mes "Allevatore di Draghi";
			mes "Ho capito. Allora ti auguro una buona giornata.";
			close;
		}
	}
	mes "Cosa fai qui?";
	mes "Solo i Cavalieri Runici possono affittare un Drago.";
	close;
}

// Mado Gear Renter :: madogear
//============================================================
-	script	::mgm	-1,{
	mes "[Armaiolo Meccanico di Mado]";
	if ((eaclass()&EAJ_THIRDMASK)==EAJ_MECHANIC) {
		mes "Benvenuto, Meccanico.";
		mes "Vorresti noleggiare un carretto o";
		mes "andare su un Mado Gear?";
		next;
		switch(select("Noleggiare un carretto:Guidare un Mado Gear:Acquista attrezzatura di emergenza Mado:Aggiornare il dispositivo di raffreddamento:Annulla")) {
		case 1:
			if (checkcart()) {
				mes "[Armaiolo Meccanico di Mado]";
				mes "Mi dispiace, ma tu già";
				mes "ho un carretto.";
				close;
			}
			setcart;
			mes "[Armaiolo Meccanico di Mado]";
			mes "Eccolo/a!";
			close;
		case 2:
			if (checkmadogear()) {
				mes "[Armaiolo Meccanico di Mado]";
				mes "Mi dispiace, ma sei già";
				mes "guidando un Mado Gear.";
				close;
			}
			else if (!getskilllv("NC_MADOLICENCE")) {
				mes "[Armaiolo Meccanico di Mado]";
				mes "Impara prima l'abilità di ottenere la Licenza Mado Gear.";
				close;
			}
			else if(ismounting()) {
				mes "[Armaiolo Meccanico di Mado]";
				mes "Rimuovi il tuo deposito contante.";
				close;
			}
			setmadogear;
			mes "[Armaiolo Meccanico di Mado]";
			mes "Divertiti e per favore torna!";
			close;
		case 3:
			mes "[Armaiolo Meccanico di Mado]";
			mes "L'Attrezzatura Mado di Emergenza è davvero utile per situazioni di emergenza ed è venduta a 1.000.000 Zeny.";
			next;
			if (select("Acquisto:Annulla") == 2) {
				mes "[Armaiolo Meccanico di Mado]";
				mes "Capisco. Sentiti libero/a di chiedermi qualsiasi cosa.";
				mes "se cambi idea.";
				close;
			}
			if (countitem(23277) > 0) {
				mes "[Armaiolo Meccanico di Mado]";
				mes "Mi dispiace, ma hai già un Emergency Mado Gear.";
				close;
			}
			if (Zeny < 1000000) {
				mes "[Armaiolo Meccanico di Mado]";
				mes "Mi dispiace, ma non hai abbastanza Zeny per acquistare l'Equipaggiamento Mado di Emergenza.";
				close;
			}
			Zeny -= 1000000;
			getitem 23277,1; //Mado_Box
			mes "[Armaiolo Meccanico di Mado]";
			mes "Eccolo/a!";
			close;
		case 4:
			mes "[Armaiolo Meccanico di Mado]";
			mes "Quale dispositivo vuoi aggiornare?";
			next;
			if (select("Dispositivo di raffreddamento:Raffreddatore di alta qualità") == 1) {
				mes "[Armaiolo Meccanico di Mado]";
				mes "Aggiornare il dispositivo di raffreddamento a un dissipatore di alta qualità richiede 1 Dispositivo di Raffreddamento e 2.000.000 Zeny.";
				next;
				.@itemid = 2804;
				.@cost = 2000000;
			} else {
				mes "[Armaiolo Meccanico di Mado]";
				mes "Aggiornare un Dissipatore di Alta Qualità a un Dissipatore Speciale richiede 1 Dissipatore di Alta Qualità e 4.000.000 Zeny.";
				next;
				.@itemid = 2809;
				.@cost = 4000000;
			}
			if (select("Aggiornare:Annulla") == 2) {
				mes "[Armaiolo Meccanico di Mado]";
				mes "Capisco. Sentiti libero/a di chiedermi qualsiasi cosa.";
				mes "se cambi idea.";
				close;
			}
			if (!countitem(.@itemid)) {
				mes "[Armaiolo Meccanico di Mado]";
				mes "Mi dispiace, ma non hai il " + getitemname(.@itemid) + ".";
				close;
			}
			if (Zeny < .@cost) {
				mes "[Armaiolo Meccanico di Mado]";
				mes "Mi dispiace, ma non hai abbastanza Zeny per aggiornare il dispositivo.";
				close;
			}
			Zeny -= .@cost;
			delitem .@itemid,1;
			getitem (.@itemid == 2804 ? 2809 : 2810),1; //High_Quality_Cooler,Special_Cooler
			mes "[Armaiolo Meccanico di Mado]";
			mes "Eccolo qui! Il tuo proprio " + getitemname(.@itemid) + ".";
			close;
		case 5:
			close;
		}
	}
	mes "Come posso aiutarla?";
	mes "I marchi Gears sono disponibili solo per i Meccanici.";
	close;
}
