//===== rAthena Script =======================================
//= Shadow Blacksmith
//===== Description: =========================================
//= [Official Conversion]
//= Shadow equipments refining NPC.
//===== Changelog: ===========================================
//= 1.0 First version [Aleos]
//= 1.1 Removed re-roll behavior and fetch materials from db
//=     [Secret]
//= 1.2 Added Refine UI [Atemo, Lemongrass]
//= 1.3 Moved to Cash Mall [Lemongrass]
//============================================================
-	script	::ShadowBlacksmith	-1,{
	if( getbattleflag( "feature.refineui" ) ){
		mes "[Fabbro dell'Ombra]";
		mes "Ora Basta è responsabile del perfezionamento degli oggetti ombra.";
		mes "Se desideri perfezionare un oggetto ombra, visitalo.";
		close;
	}

	.@zeny_cost = 200000; // Zeny cost is 200,000 according to official script [Secret]
	disable_items;
	mes "[Fabbro dell'Ombra]";
	mes "Vuoi perfezionare un oggetto ombra? Scegli il tuo veleno!";
	next;
	setarray .@indices[1], EQI_SHADOW_ARMOR, EQI_SHADOW_WEAPON, EQI_SHADOW_SHIELD, EQI_SHADOW_SHOES, EQI_SHADOW_ACC_R, EQI_SHADOW_ACC_L;
	.@indlen = getarraysize(.@indices) - 1;
	for(.@i = 1; .@i <= .@indlen; .@i++)
		.@menu$ = .@menu$ + (getequipisequiped(.@indices[.@i]) ? getequipname(.@indices[.@i]) : F_getpositionname(.@indices[.@i]) +"-[Non equipaggiato]") +":";
	.@menu$ = .@menu$ + "Perfeziona le informazioni";
	.@choice = select(.@menu$);
	if (.@choice == .@indlen + 1) { // Refine info
		mes "[Fabbro dell'Ombra]";
		mes "Quando un oggetto ombra viene raffinato, ottiene bonus extra proprio come gli oggetti normali.";
		next;
		mes "[Fabbro dell'Ombra]";
		mes "Arma: ATK, MATK + 1 incremento per ogni +1 successo di raffinamento.";
		mes "Ecc: aumento di HP + 10 per ogni successo di raffinazione +1.";
		next;
		mes "[Fabbro dell'Ombra]";
		mes "Tutti i tipi di Oridecon ed Elunium possono essere utilizzati per rifinire gli oggetti ombra. Ogni tentativo costerà inoltre 20.000 zeny.";
		next;
		mes "[Fabbro dell'Ombra]";
		mes "I minerali HD possono essere usati per equipaggiamenti che siano almeno di livello +7 e prevengono la rottura finché rimani a parlare con me.";
		close;
	}

	.@part = .@indices[.@choice];

	if (!getequipisequiped(.@part)) {
		mes "[Fabbro dell'Ombra]";
		mes "Non c'è niente qui!";
		close;
	}
	.@equip_id = getequipid(.@part);
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);

	while(1) {
		.@equip_refine = getequiprefinerycnt(.@part);
		mes "[Fabbro dell'Ombra]";
		mes "Ho bisogno " + callfunc("F_InserisciComma", .@zeny_cost) + " zeny come tariffa per OGNI tentativo di raffinazione.";
		mes "Scegli il tuo minerale e inizia a raffinare.";
		next;

		.@material[0] = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);
		.@material[1] = getequiprefinecost(.@part, REFINE_COST_ENRICHED, REFINE_MATERIAL_ID);
		.@material[2] = getequiprefinecost(.@part, REFINE_COST_HD, REFINE_MATERIAL_ID);
		.@is_enriched = false;

		if (countitem(.@material[0]))
			.@mate$[0] = getitemname(.@material[0]);
		else
			.@mate$[0] = "^8C8C8C"+ getitemname(.@material[0]) +"^000000";
		if (countitem(.@material[1]))
			.@mate$[1] = getitemname(.@material[1]);
		else
			.@mate$[1] = "^8C8C8C"+ getitemname(.@material[1]) +"^000000";
		if (.@equip_refine > 6 && countitem(.@material[2]))
			.@mate$[2] = getitemname(.@material[2]);
		else
			.@mate$[2] = "^8C8C8C"+ getitemname(.@material[2]) +"^000000";

		switch( select( .@mate$[0], .@mate$[1], .@mate$[2], "Cancellare" ) ) {
		case 1:// NORMAL
			.@choose = .@material[0];
			break;
		case 2:// ENRICHED
			.@is_enriched = true;
			.@choose = .@material[1];
			break;
		case 3:// HD
			if (.@equip_refine < 7) {
				mes "[Fabbro dell'Ombra]";
				mes "Il minerale HD può essere utilizzato solo per oggetti di livello di raffinazione +7 o superiore.";
				close;
			}
			.@hoihoi = 1;
			.@choose = .@material[2];
			break;
		case 4:
			mes "[Fabbro dell'Ombra]";
			mes "Hai annullato la raffinazione.";
			close;
		}
		if (!countitem(.@choose)) {
			mes "[Fabbro dell'Ombra]";
			mes "Non ne hai abbastanza "+ getitemname(.@choose) +".";
			close;
		}
		if (Zeny < .@zeny_cost) {
			mes "[Fabbro dell'Ombra]";
			mes "Non hai abbastanza Zeny.";
			close;
		}
		if (.@equip_refine > 9) {
			mes "[Fabbro dell'Ombra]";
			mes "L'equipaggiamento dell'ombra può essere perfezionato fino a un massimo di 10...";
			close;
		}
		if (!getequipisenableref(.@part)) {
			mes "[Fabbro dell'Ombra]";
			mes "Questo oggetto non può essere raffinato.";
			close;
		}
		if (getequippercentrefinery(.@part) < 100) {
			mes "[Fabbro dell'Ombra]";
			mes "Il livello di miglioramento della sicurezza per l'Equipaggiamento Ombra è +4.";
			if (!.@hoihoi)
				mes "L'equipaggiamento ombra può essere distrutto nei successivi perfezionamenti. Vuoi iniziare?";
			else
				mes "Il successivo perfezionamento, se fallisce, degraderà il perfezionamento. Vuoi perfezionare?";
			next;
			if (select("Procedere","Cancellare") == 2) {
				mes "[Fabbro dell'Ombra]";
				mes "Eh, lo sapevo!";
				close;
			}
		}
		mes "[Fabbro dell'Ombra]";
		mes "Ebbene... qui non va niente!";
		next;
		delitem .@choose,1;
		Zeny -= .@zeny_cost;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) || callfunc("F_IsEquipRefineHack", .@part, .@equip_refine))
			close;

		if (getequippercentrefinery(.@part, .@is_enriched) > rand(100)) {
			successrefitem .@part;
			mes "[Fabbro dell'Ombra]";
			mes "Ha funzionato! Ha funzionato!";
			next;
		} else {
			if (.@hoihoi)
				downrefitem .@part;
			else
				failedrefitem .@part;
			mes "[Fabbro dell'Ombra]";
			mes "Oh Odino No!";
			close;
		}
	}
}
