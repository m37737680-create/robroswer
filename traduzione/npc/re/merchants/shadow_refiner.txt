//===== rAthena Script =======================================
//= Shadow Blacksmith
//===== Description: =========================================
//= [Official Conversion]
//= Shadow equipments refining NPC.
//===== Changelog: ===========================================
//= 1.0 First version [Aleos]
//= 1.1 Removed re-roll behavior and fetch materials from db
//=     [Secret]
//= 1.2 Added Refine UI [Atemo, Lemongrass]
//= 1.3 Moved to Cash Mall [Lemongrass]
//============================================================
-	script	::ShadowBlacksmith	-1,{
	if( getbattleflag( "feature.refineui" ) ){
		mes "[Fabbro Ombra]";
		mes "Ora Basta è responsabile del perfezionamento per gli oggetti ombra.";
		mes "Se desideri perfezionare un oggetto ombra, per favore visita lui.";
		close;
	}

	.@zeny_cost = 200000; // Zeny cost is 200,000 according to official script [Secret]
	disable_items;
	mes "[Fabbro Ombra]";
	mes "Vuoi affinare un oggetto ombra? Scegli il tuo veleno!";
	next;
	setarray .@indices[1], EQI_SHADOW_ARMOR, EQI_SHADOW_WEAPON, EQI_SHADOW_SHIELD, EQI_SHADOW_SHOES, EQI_SHADOW_ACC_R, EQI_SHADOW_ACC_L;
	.@indlen = getarraysize(.@indices) - 1;
	for(.@i = 1; .@i <= .@indlen; .@i++)
		.@menu$ = .@menu$ + (getequipisequiped(.@indices[.@i]) ? getequipname(.@indices[.@i]) : F_getpositionname(.@indices[.@i]) +"Non equipaggiato") +":";
	.@menu$ = .@menu$ + "Rifinire informazioni";
	.@choice = select(.@menu$);
	if (.@choice == .@indlen + 1) { // Refine info
		mes "[Fabbro Ombra]";
		mes "Quando un oggetto ombra viene raffinato, acquisisce bonus extra molto simili agli oggetti normali.";
		next;
		mes "[Fabbro Ombra]";
		mes "Arma: Aumento di ATK e MATK di 1 per ogni successo di rifinitura +1.";
		mes "Etc: HP + 10 aumento per ogni successo di raffinazione +1.";
		next;
		mes "[Fabbro Ombra]";
		mes "Tutti i tipi di Oridecon e Elunium possono essere usati per raffinare oggetti d'ombra. Ogni tentativo costerà anche 20.000 zeny.";
		next;
		mes "[Fabbro Ombra]";
		mes "Le vene di HD possono essere utilizzate per equipaggiamento che ha almeno livello di raffinazione +7 e impediranno di rompersi finché continuerai a parlare con me.";
		close;
	}

	.@part = .@indices[.@choice];

	if (!getequipisequiped(.@part)) {
		mes "[Fabbro Ombra]";
		mes "Non c'è niente qui!";
		close;
	}
	.@equip_id = getequipid(.@part);
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);

	while(1) {
		.@equip_refine = getequiprefinerycnt(.@part);
		mes "[Fabbro Ombra]";
		mes "Richiedo " + callfunc("F_InserisciVirgola", .@zeny_cost) + " zeny come tariffa per OGNI tentativo di affinamento.";
		mes "Scegli la tua Mena e inizia a raffinare.";
		next;

		.@material[0] = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);
		.@material[1] = getequiprefinecost(.@part, REFINE_COST_ENRICHED, REFINE_MATERIAL_ID);
		.@material[2] = getequiprefinecost(.@part, REFINE_COST_HD, REFINE_MATERIAL_ID);
		.@is_enriched = false;

		if (countitem(.@material[0]))
			.@mate$[0] = getitemname(.@material[0]);
		else
			.@mate$[0] = "^8C8C8C"+ getitemname(.@material[0]) +"^000000";
		if (countitem(.@material[1]))
			.@mate$[1] = getitemname(.@material[1]);
		else
			.@mate$[1] = "^8C8C8C"+ getitemname(.@material[1]) +"^000000";
		if (.@equip_refine > 6 && countitem(.@material[2]))
			.@mate$[2] = getitemname(.@material[2]);
		else
			.@mate$[2] = "^8C8C8C"+ getitemname(.@material[2]) +"^000000";

		switch( select( .@mate$[0], .@mate$[1], .@mate$[2], "Annulla" ) ) {
		case 1:// NORMAL
			.@choose = .@material[0];
			break;
		case 2:// ENRICHED
			.@is_enriched = true;
			.@choose = .@material[1];
			break;
		case 3:// HD
			if (.@equip_refine < 7) {
				mes "[Fabbro Ombra]";
				mes "Ore HD può essere usato solo per oggetti di livello raffinato +7 o superiore.";
				close;
			}
			.@hoihoi = 1;
			.@choose = .@material[2];
			break;
		case 4:
			mes "[Fabbro Ombra]";
			mes "Hai annullato il raffinamento.";
			close;
		}
		if (!countitem(.@choose)) {
			mes "[Fabbro Ombra]";
			mes "Non hai abbastanza "+ getitemname(.@choose) +".";
			close;
		}
		if (Zeny < .@zeny_cost) {
			mes "[Fabbro Ombra]";
			mes "Non hai abbastanza zeny.";
			close;
		}
		if (.@equip_refine > 9) {
			mes "[Fabbro Ombra]";
			mes "L'Equipaggiamento Ombra può essere raffinato fino al massimo di 10...";
			close;
		}
		if (!getequipisenableref(.@part)) {
			mes "[Fabbro Ombra]";
			mes "Questo oggetto non può essere raffinato.";
			close;
		}
		if (getequippercentrefinery(.@part) < 100) {
			mes "[Fabbro Ombra]";
			mes "Il livello di rifinitura della sicurezza per l'Equipaggiamento Ombra è +4.";
			if (!.@hoihoi)
				mes "L'equipaggiamento ombra può essere distrutto nei raffinamenti successivi. Vuoi iniziare?";
			else
				mes "Il prossimo affinamento, se fallisce, degraderà l'affinamento. Vuoi affinare?";
			next;
			if (select("Procedi","Annulla") == 2) {
				mes "[Fabbro Ombra]";
				mes "Eh, lo sapevo!";
				close;
			}
		}
		mes "[Fabbro Ombra]";
		mes "Beh allora... qui va niente!";
		next;
		delitem .@choose,1;
		Zeny -= .@zeny_cost;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) || callfunc("F_IsEquipRefineHack", .@part, .@equip_refine))
			close;

		if (getequippercentrefinery(.@part, .@is_enriched) > rand(100)) {
			successrefitem .@part;
			mes "[Fabbro Ombra]";
			mes "Ha funzionato! Ha funzionato!";
			next;
		} else {
			if (.@hoihoi)
				downrefitem .@part;
			else
				failedrefitem .@part;
			mes "[Fabbro Ombra]";
			mes "Oh Odino No!";
			close;
		}
	}
}
