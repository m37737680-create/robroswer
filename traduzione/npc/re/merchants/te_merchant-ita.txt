	mes "[Gestione articoli a noleggio]";

		mes "Hai già troppi articoli per ricevere una nuova attrezzatura a noleggio.";

	mes "Seleziona l'articolo desiderato.";
	next;
	for ( .@i = 1; .@i < getargcount(); .@i += 2 ) {
		if (rentalcountitem(getarg(.@i)) > isequippedcnt(getarg(.@i)))// don't count item equipped
			.@menu$ = .@menu$ + "^4d4dff"+ getarg(.@i+1) +" - Controllato^000000:";
		else
			.@menu$ = .@menu$ + getarg(.@i+1) + ":";
	}
	.@total_item = (getargcount() -1) /2;
	.@s = select( .@menu$ + ":Visualizza altri elementi" ) -1;
	if (.@s <= .@total_item) {
		.@index = .@s *2 +1;
		mes "[Gestione articoli a noleggio]";
		if (rentalcountitem(getarg(.@index)) > isequippedcnt(getarg(.@index)))
			mes "L'hai già fatto "+ getarg(.@index+1) +".";
		else {
			mes "Ecco "+ getarg(.@index+1) +".";
			mes "^4d4dffQuesto articolo è valido per le prossime 24 ore^000000 e successivamente scade.";
			rentitem getarg(.@index),86400;
		}
		mes "Hai bisogno di qualcos'altro?";
		next;
	}
	return;
