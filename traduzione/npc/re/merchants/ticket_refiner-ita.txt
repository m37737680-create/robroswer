// Main NPC :: safety_Ref_NPC
//============================================================
prontera,184,177,6	script	Refine Master	851,{
	disable_items;
	setarray .@cert_weapon[0],
		6864,	// Guarantee_Weapon_19Up
		6875,	// Guarantee_Weapon_18Up
		6874,	// Guarantee_Weapon_17Up
		6873,	// Guarantee_Weapon_16Up
		6872,	// Guarantee_Weapon_15Up
		6871,	// Guarantee_Weapon_14Up
		6870,	// Guarantee_Weapon_13Up
		6584,	// Guarantee_Weapon_12Up
		6238,	// Guarantee_Weapon_11Up
		6228,	// Guarantee_Weapon_9Up
		6229,	// Guarantee_Weapon_8Up
		6230,	// Guarantee_Weapon_7Up
		6231,	// Guarantee_Weapon_6Up
		6456;	// Guarantee_Weapon_5Up
	setarray .@cert_armor[0],
		6865,	// Guarantee_Armor_19Up
		6881,	// Guarantee_Armor_18Up
		6880,	// Guarantee_Armor_17Up
		6879,	// Guarantee_Armor_16Up
		6878,	// Guarantee_Armor_15Up
		6877,	// Guarantee_Armor_14Up
		6876,	// Guarantee_Armor_13Up
		6585,	// Guarantee_Armor_12Up
		6239,	// Guarantee_Armor_11Up
		6232,	// Guarantee_Armor_9Up
		6233,	// Guarantee_Armor_8Up
		6234,	// Guarantee_Armor_7Up
		6235,	// Guarantee_Armor_6Up
		6457;	// Guarantee_Armor_5Up
	setarray .@cert_level[0],19,18,17,16,15,14,13,12,11,9,8,7,6,5;
	.@size_cert = getarraysize(.@cert_weapon);

	for ( .@i = 0; .@i < .@size_cert; ++.@i ) {
		if (countitem(.@cert_weapon[.@i]) > 0 || countitem(.@cert_armor[.@i]) > 0) {
			.@check = 1;
			break;
		}
	}

	if (.@check == 0) {
		mes "[Perfeziona maestro]";
		mes "Ciao!";
		mes "Che cosa succede?";
		mes "Sono uno specialista";
		mes "per perfezionare gli oggetti,";
		mes "ma non lavoro più.";
		next;
		switch(select("andrò per la mia strada.:Hmm... questo mi incuriosisce.")) {
		case 1:
			mes "[Perfeziona maestro]";
			mes "Abbi cura di te, avventuriero.";
			close;
		case 2:
			mes "[Perfeziona maestro]";
			mes "In realtà, a volte fornisco servizi di perfezionamento agli avventurieri con un ^006400Refine Ticket^000000...";
			mes "Ciao ciao~!";
			close;
		}
	}
	emotion ET_SURPRISE;
	mes "[Perfeziona maestro]";
	mes "Saluti!";
	mes "Posso perfezionare un articolo fino allo ^006400stesso livello del tuo biglietto^000000.";
	mes "Non devi preoccuparti! Non c'è alcuna possibilità di rompere il tuo oggetto.";
	next;
	if(select("Tornerò più tardi.:Perfeziona l'articolo con il biglietto.") == 1) {
		mes "[Perfeziona maestro]";
		mes "Va bene.";
		mes "Puoi tornare più tardi.";
		close;
	}
	mes "[Perfeziona maestro]";
	mes "Quale attrezzatura vorresti perfezionare?";
	next;
	setarray .@position$[1],"Head upper","Armor","Left hand","Right hand","Robe","Shoes","Accessory 1","Accessory 2","Head middle","Head lower";
	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for ( .@i = 1; .@i <= 10; ++.@i )
		.@menu$ += (getequipisequiped(.@indices[.@i]) ? getequipname(.@indices[.@i]) : .@position$[.@i] + "- [Vuoto]") + ":";

	.@part = .@indices[ select(.@menu$) ];
	if (!getequipisequiped(.@part)) {
		mes "[Perfeziona maestro]";
		mes "Devi equipaggiare l'oggetto che vuoi perfezionare.";
		close;
	}
	if (!getequipisenableref(.@part)) {
		emotion ET_OTL;
		mes "[Perfeziona maestro]";
		mes "Oh, mi dispiace.";
		mes "Questo articolo è impossibile da perfezionare.";
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	if( .@itemtype == IT_WEAPON ){
		switch( getequipweaponlv( .@part ) ){
			case 1:
			case 2:
			case 3:
			case 4:
				copyarray .@tickets[0], .@cert_weapon[0], .@size_cert;
				.@type$ = "Weapon";
				break;
			default:
				// TODO:
				close;
		}
	}else if( .@itemtype == IT_ARMOR ){
		switch( getequiparmorlv( .@part ) ){
			case 1:
				copyarray .@tickets[0], .@cert_armor[0], .@size_cert;
				.@type$ = "Armor";
				break;
			default:
				// TODO:
				close;
		}
	}else{
		// TODO:
		close;
	}
	.@check = 0;
	for ( .@i = 0; .@i < .@size_cert; ++.@i ) {
		if (countitem(.@tickets[.@i]) > 0) {
			.@check = 1;
			break;
		}
	}
	if (.@check == 0) {
		emotion ET_THINK;
		mes "[Perfeziona maestro]";
		mes "Se vuoi perfezionare questo ^006400"+.@type$+"^000000, per favore vieni con ^006400"+.@type$+" Perfeziona biglietto^000000.";
		mes "Arrivederci!";
		close;
	}
	mes "[Perfeziona maestro]";
	mes "Per favore scegli quale ^006400"+.@type$+" Perfeziona il Ticket^000000 che desideri utilizzare.";
	next;
	.@menu$ = "";
	for ( .@i = 0; .@i < .@size_cert; ++.@i )
		.@menu$ += getitemname(.@tickets[.@i]) + ":";
	.@select = select(.@menu$)-1;
	.@ticket_lv = .@cert_level[.@select];
	.@ticket_id = .@tickets[.@select];

	if (countitem(.@ticket_id) == 0) {
		emotion ET_QUESTION;
		mes "[Perfeziona maestro]";
		mes getitemname(.@ticket_id)+" is not in your inventory. Did you put it in your storage?";
		mes "Per favore controlla di nuovo.";
		mes "Arrivederci!";
		close;
	}
	if (getequiprefinerycnt(.@part) >= .@ticket_lv) {
		emotion ET_PROFUSELY_SWEAT;
		mes "[Perfeziona maestro]";
		mes "^8B4513Questo oggetto è già raffinato tanto quanto la tua azione.^000000";
		mes "Per favore, porta con te un oggetto perfezionato meno del tuo biglietto.";
		close;
	}
	mes "[Perfeziona maestro]";
	mes "Vado a perfezionare ^006400"+getequipname(.@part)+"^8B4513 fino al +"+.@ticket_lv+" livello^000000 con ^006400"+getitemname(.@ticket_id)+"^000000.";
	mes "Posso procedere?";
	next;
	if(select("NO.:SÌ.") == 1) {
		emotion ET_THINK;
		mes "[Perfeziona maestro]";
		mes "Oh, hai cambiato idea.";
		mes "OK.";
		mes "Puoi tornare più tardi.";
		close;
	}
	mes "[Perfeziona maestro]";
	mes "Grande.";
	mes "Come si desidera!";
	mes "Ho il mio modo speciale per perfezionare...";
	mes ".......ka boom!";
	specialeffect EF_SUI_EXPLOSION;
	if (countitem(.@ticket_id) < 1) {
		next;
		mes "Errore!";
		mes "Per favore segnalalo.";
		close;
	}
	delitem .@ticket_id,1;

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) ||
		callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3])) {
		mes "[Perfeziona maestro]";
		emotion ET_FRET;
		mes "Aspetta un secondo...";
		mes "Pensi che io sia stupido?!";
		mes "Hai cambiato oggetto mentre non guardavo! Fuori di qui!";
		close;
	}
	successrefitem .@part, .@ticket_lv - getequiprefinerycnt(.@part);
	next;
	emotion ET_DELIGHT;
	mes "[Perfeziona maestro]";
	mes "Va bene, eccolo qui~";
	mes "Bene, ^0000FF"+strcharinfo(0)+"^000000!";
	mes "Congratulazioni per la tua brillantezza "+.@type$+".";
	mes "Sei FANTASTICO!";
	mes "Addio~!";
	close;
}
