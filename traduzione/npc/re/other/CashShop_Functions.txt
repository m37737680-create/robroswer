function	script	VIP_Third_Class	{
	if (!vip_status(VIP_STATUS_ACTIVE)) {
		mes "[Voce da nessuna parte]";
		mes "L'accesso ai contenuti Premium è disponibile solo per coloro che possiedono una certa quantità di Pietre di Resetta.";
		next;
		if (countitem(6320) < 1) {
			mes "[Voce da nessuna parte]";
			mes "Per passare alla terza classe, devi portare 1 ^0000CCPietra di Resetta^000000 o avere un account Premium.";
			//mes "Please visit our iShop to purchase the ^0000CCReset Stone^000000.";
			close;
		} else {
			mes "[Voce da nessuna parte]";
			mes "Vorresti partecipare a questa ricerca per 1 ^0000CCPietra di Reset^000000?";
			next;
			switch(select("Sì:No")) {
			case 1:
				mes "[Voce da nessuna parte]";
				mes "Buona fortuna, avventuriero.";
				delitem 6320,1; //Premium_Reset_Stone
				next;
				break;
			case 2:
				mes "[Voce da nessuna parte]";
				mes "Molto bene.";
				close;
			}
		}
	}
	return;
}
