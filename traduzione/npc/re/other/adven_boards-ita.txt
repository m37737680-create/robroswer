function	script	F_AvenBoard	{
	.@mo_board = select( "Look up monsters by floor",
					"Monster Information Retrieval System",
					( countitem(909) == 30000 ? "Request" : "" ),
					"Cancel" );
	if (.@mo_board == 1)
		return 1;
	else if (.@mo_board == 2) {
		while(1) {
			mes "[Sistema di recupero delle informazioni sui mostri]";
			mes "Questo è il sistema di recupero delle informazioni sui mostri. È possibile cercare informazioni sui mostri che appaiono in questo dungeon. Inserisci il nome del mostro, spazi inclusi, durante la ricerca.";
			next;
			input .@inputstr$;
			setarray .@element$[0],"Neutral","Water","Earth","Fire","Wind","Poison","Holy","Shadow","Ghost","Undead";
			setarray .@size$[0],"Medium","Small","Big";
			setarray .@race$[0],"Formless","Undead","Animal","Plant","Insect","Fish","Demon","Demihuman","Angel","Dragon";
			.@size = getargcount();
			.@i = 0;
			while( .@i < .@size && getarg(.@i+1,"") != .@inputstr$ ) .@i += 3;
			mes "[Sistema di recupero delle informazioni sui mostri]";
			mes "Ora sto cercando";
			next;
			mes "[Sistema di recupero delle informazioni sui mostri]";
			mes "La ricerca è stata completata.";
			mes "Ora vengono visualizzati i risultati della ricerca.";
			next;
			mes "[Sistema di recupero delle informazioni sui mostri]";
			if (.@i >= .@size) {
				mes "Non ci sono informazioni disponibili per quel mostro.";
				mes "Per favore controlla il nome del tuo mostro.";
			}
			else {
				.@mob_id = getarg(.@i);
				mes "- Nome : "+ getarg(.@i+1);
				mes "- Elemento: "+ .@element$[ getmonsterinfo(.@mob_id,MOB_ELEMENT) ];
				mes "- Misurare : "+ .@size$[ getmonsterinfo(.@mob_id,MOB_SIZE) ];
				mes "- Gara : "+ .@race$[ getmonsterinfo(.@mob_id,MOB_RACE) ];
				mes "------------------^FD8D02";
				mes getarg(.@i+2);
			}
			next;
			mes "[Sistema di recupero delle informazioni sui mostri]";
			mes "Desideri cercare altre informazioni sui mostri?";
			next;
			if (select( "Ricerca","Cancellare" ) == 1)
				continue;
			mes "[Sistema di recupero delle informazioni sui mostri]";
			mes "Il sistema di ricerca dei mostri ora si spegnerà.";
			next;
			return 2;
		}
	}
	else if (.@mo_board == 3) {
		mes "Verranno visualizzate le informazioni richieste: se vengono visualizzate, non ci sono problemi";
		next;
		return 3;
	}
	else if (.@mo_board == 4)
		close;
	else {
		mes "Errore di operazione.";
		close;
	}
}

function	script	F_InfoBoard	{
	next;
	mes "[Gruppo avventurieri]";
	mes "Per ulteriori informazioni sui mostri";
	mes "Sistema di recupero delle informazioni sui mostri";
	mes "Deve essere usato.";
	next;
	return;
}
