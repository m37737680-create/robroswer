function	script	F_AvenBoard	{
	.@mo_board = select( "Look up monsters by floor",
					"Monster Information Retrieval System",
					( countitem(909) == 30000 ? "Request" : "" ),
					"Cancel" );
	if (.@mo_board == 1)
		return 1;
	else if (.@mo_board == 2) {
		while(1) {
			mes "Sistema di Recupero Informazioni sui Mostri";
			mes "Questo è il Sistema di Recupero Informazioni sui Mostri. È possibile cercare informazioni sui mostri che appaiono in questa dungeon. Inserire il Nome del Mostro, inclusi gli spazi, quando si cerca.";
			next;
			input .@inputstr$;
			setarray .@element$[0],"Neutral","Water","Earth","Fire","Wind","Poison","Holy","Shadow","Ghost","Undead";
			setarray .@size$[0],"Medium","Small","Big";
			setarray .@race$[0],"Formless","Undead","Animal","Plant","Insect","Fish","Demon","Demihuman","Angel","Dragon";
			.@size = getargcount();
			.@i = 0;
			while( .@i < .@size && getarg(.@i+1,"") != .@inputstr$ ) .@i += 3;
			mes "Sistema di Recupero Informazioni sui Mostri";
			mes "Ora sto cercando";
			next;
			mes "Sistema di Recupero Informazioni sui Mostri";
			mes "La ricerca è stata completata.";
			mes "Ora vengono mostrati i risultati di ricerca.";
			next;
			mes "Sistema di Recupero Informazioni sui Mostri";
			if (.@i >= .@size) {
				mes "Non ci sono informazioni disponibili su quel mostro.";
				mes "Controlla il nome del tuo mostro.";
			}
			else {
				.@mob_id = getarg(.@i);
				mes "Nome : "+ getarg(.@i+1);
				mes "Elemento: "+ .@element$[ getmonsterinfo(.@mob_id,MOB_ELEMENT) ];
				mes "Dimensione: "+ .@size$[ getmonsterinfo(.@mob_id,MOB_SIZE) ];
				mes "Razza : "+ .@race$[ getmonsterinfo(.@mob_id,MOB_RACE) ];
				mes "--------------------------------^FD8D02";
				mes getarg(.@i+2);
			}
			next;
			mes "Sistema di Recupero Informazioni sui Mostri";
			mes "Vuoi cercare altre informazioni sui mostri?";
			next;
			if (select( "Cerca","Annulla" ) == 1)
				continue;
			mes "Sistema di Recupero Informazioni sui Mostri";
			mes "Il sistema di ricerca mostri si spegnerà ora.";
			next;
			return 2;
		}
	}
	else if (.@mo_board == 3) {
		mes "Le informazioni richieste appariranno - Non c'è problema se appare";
		next;
		return 3;
	}
	else if (.@mo_board == 4)
		close;
	else {
		mes "Errore operativo.";
		close;
	}
}

function	script	F_InfoBoard	{
	next;
	mes "Gruppo di Avventurieri";
	mes "Per maggiori informazioni sui mostri";
	mes "Sistema di Recupero Informazioni sui Mostri";
	mes "Deve essere usato.";
	next;
	return;
}
