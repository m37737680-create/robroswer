	next;

	case 1:
		if (Zeny > 199) {
			if(!callfunc("F_CanOpenStorage")){
				mes "[Responsabile della logistica]";
				mes "Non sei ancora qualificato per utilizzare lo spazio di archiviazione.";
			}
			else {
				Zeny -= 200;
				openstorage;
			}
		}
		else {
			mes "[Responsabile della logistica]";
			mes "Sono necessari almeno 200 zeny per utilizzare lo spazio di archiviazione.";
		}
		break;
	case 2:
		mes "[Responsabile della logistica]";
		mes "Fammi sapere se cambi idea.";
	}
	close;
}
