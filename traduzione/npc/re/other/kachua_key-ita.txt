//===== rAthena Script =======================================
//= Dynamic NPC: Kachua's Secret Box
//===== Description: =========================================
//- [Official conversion]
//= Kachua's Secret Box is a Gachapon NPC.
//= It lets the player exchange a Kachua's Secret Key for a random item.
//= It also gives a Kachua's Mileage Coupon for each pull.
//===== Changelogs: ==========================================
//= 1.0 First version. [secretdataz]
//============================================================

sec_in02,126,178,3	script	Kachua's Secret Box#bm	4_TREASURE_BOX,{
	.@key$ = "K_Secret_Key";
	.@keyname$ = getitemname( .@key$ );
	.@box$ = "Main_Lucky_Box";
	.@boxname$ = getitemname( .@box$ );
	mes "Una scatola segreta dove nessuno sa cosa c'è dentro.";
	mes "^4d4dffPuoi aprire questa scatola consumando " + mesitemlink( .@key$ ) + ".^000000";
	next;
	switch(select("^4d4dffApri la scatola 1 volta (1 " + .@keyname$ + ")^000000","^4d4dffApri la scatola 10 volte (10 " + .@keyname$ + "s)^000000")) {
		case 1:
			mes "^FF0000[Avviso]^000000";
			mes "^FF0000Scambia un oggetto casuale con uno " + .@keyname$ + ".^000000";
			mes "^FF0000L'articolo scambiato sopra non può essere ritirato, né può essere scambiato con " + .@keyname$ + ".^000000";
			next;
			if(select("Continuare.","Interrompi la conversazione.") == 2) {
				mes "Hai deciso di non aprire la Scatola.";
				close;
			} else if (countitem(.@key$) < 1) {
				mes "Non abbastanza " + mesitemlink( .@key$ ) + ".";
				close;
			} else {
				if (checkweight(1201,1) == 0 || ((MaxWeight - Weight) * 100 / MaxWeight) < 10) {
					mes "^4d4dffAssicurati di avere spazio sufficiente nel tuo inventario.^000000";
					close;
				}
				delitem(.@key$, 1);
				consumeitem(.@box$);
				mes .@boxname$ + " was opened!";
				mes "Sei stato fortunato?";
				specialeffect2 EF_VALLENTINE;
				close;
			}
		case 2:
			mes "^FF0000[Avviso]^000000";
			mes "^FF0000Scambia 10 oggetti casuali con 10 " + .@keyname$ + "s.^000000";
			mes "^FF0000Gli articoli scambiati sopra non possono essere ritirati, né possono essere scambiati " + .@keyname$ + ".^000000";
			next;
			if(select("Continuare.","Interrompi la conversazione.") == 2) {
				mes "Hai deciso di non aprire la Scatola.";
				close;
			} else if (countitem(.@key$) < 10) {
				mes "Non ne hai abbastanza " + mesitemlink( .@key$ ) + ".";
				close;
			} else {
				for (.@i = 1; .@i <= 10; ++.@i) {
					progressbar "4d4dff",2;
					if (checkweight(1201,1) == 0 || ((MaxWeight - Weight) * 100 / MaxWeight) < 10) {
						mes "^4d4dffAssicurati di avere spazio sufficiente nel tuo inventario.^000000";
						close;
					}
					if (countitem(.@key$) < 1) { // Custom check, just in case
						close;
					}
					delitem(.@key$, 1);
					consumeitem(.@box$);
					dispbottom .@boxname$ + " è stato aperto " + .@i + " volte. Un altro è in fase di apertura.",0xFFFFFF;
					specialeffect2 EF_VALLENTINE;
				}
				mes .@boxname$ + " was opened 10 times!";
				mes "Sei stato fortunato?";
				close;
			}
	}
}
