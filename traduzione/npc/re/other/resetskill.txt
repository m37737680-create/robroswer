izlude,127,175,3	script	Hypnotist#novice	124,{
	mes "Ipnotista";
	mes "Saluti, avventuriero.";
	mes "Sono un membro di Hypnotist";
	mes "Accademia inviata qui a Izlude";
	mes "fornire servizi di reset delle competenze";
	mes "a certi personaggi di prima classe a un prezzo davvero buono: gratis!";
	next;
	mes "Ipnotista";
	mes "Sebbene io offra un numero illimitato di";
	mes "reset di stat/abilità per ora, io ho";
	mes "due condizioni che devono essere";
	mes "adempito. Innanzitutto, devi essere";
	mes "inferiore a ^FF0000Livello Base 50^000000.";
	next;
	mes "Ipnotista";
	mes "Secondo, devi essere un";
	mes "^FF0000Spadaccino, Acolita, Mago,";
	mes "Ladro, Arciere, Mercante,";
	mes "Ragazzo di Taekwon, o ragazza di Taekwon";
	mes "Coltivatore e Ninja^000000";
	mes "Carattere lavorativo per qualificarsi.";
	mes "Adesso, hai qualche domanda?";
	next;
	switch(select("^FF0000Resettaggio Statistiche/Abilità?:Resetta le statistiche:Resetta abilità^000000:Annulla")) {
	case 1:
		mes "Ipnotista";
		mes "Reset di Stat/Abilità permettono agli avventurieri";
		mes "ridistribuire le loro abilità";
		mes "Punti se sono infelici";
		mes "con le loro attuali competenze.";
		next;
		mes "Ipnotista";
		mes "Prima di procedere con";
		mes "un reset di Stat/Abilità, devi";
		mes "ridurre tutto il peso";
		mes "di tutti gli oggetti trasportati sul tuo";
		mes "carattere a 0. Puoi mettere";
		mes "articoli extra in Kafra Storage.";
		next;
		mes "Ipnotista";
		mes "Ah, è anche importante";
		mes "per rimuovere il tuo ^FF0000Pushcart^000000";
		mes "se ne hai uno equipaggiato.";
		mes "Altrimenti, l'ipnosi non";
		mes "lavorare, o farà effetto contrario...";
		close;
	case 2:
		set .@str$,"Stat";
	case 3:
		if (.@str$ == "")
			set .@str$,"Skill";
		mes "Ipnotista";
		mes "Sei sicuro che tu";
		mes "voglio procedere con";
		mes "mio ^FF0000"+.@str$+" Resettare il servizio ^000000?";
		next;
		if(select("Sì:Annulla") == 1)
			break;
	case 4:
		mes "Ipnotista";
		mes "Grazie e buono";
		mes "buona fortuna per le tue avventure.";
		mes "Viaggiate in sicurezza~";
		close;
	}
	if ((Class >= Job_Swordman && Class <= Job_Thief) || Class == Job_Taekwon || Class == Job_Gunslinger || Class == Job_Ninja) {
		if (checkcart()) {
			mes "Ipnotista";
			mes "Oh! Per favore rimuovi il tuo";
			mes "Carretto prima di procedere";
			mes "con il "+.@str$+" Ripristina servizio.";
			mes "Grazie per la collaborazione~";
			close;
		}
		if (Weight != 0) {
			mes "Ipnotista";
			mes "Se sei qui per il mio "+.@str$;
			mes "Ripristinare il servizio, per favore";
			mes "ricordare che non puoi";
			mes "resettare il tuo "+.@str$+"fino a il";
			mes "^FF0000weight of your carried items in";
			mes "il tuo inventario è ridotto a 0^000000.";
			next;
			mes "Ipnotista";
			mes "Perché non metti il tuo";
			mes "cose nel magazzino di Kafra";
			mes "per ora? In questo modo, puoi";
			mes "conserva tutti i tuoi beni in sicurezza.";
			close;
		}
		if (BaseLevel >= 50 && (Class == Job_Taekwon || Class == Job_Gunslinger || Class == Job_Ninja)) {
			mes "Ipnotista";
			mes "Mi dispiace, ma caratteri";
			mes "con Livelli Base più alti";
			mes "meno 50 sono ineleggibili per il";
			mes .@str$+" Reset service I provide.";
			close;
		}
		mes "Ipnotista";
		mes "Grazie per aver usato";
		mes "mio "+.@str$+" Ridistribuzione";
		mes "servizi. Oh, e meglio";
		mes "di buona fortuna a te per il tuo";
		mes "viaggiatore, avventuriero.";
		if (.@str$ == "Stat")
			resetstatus;
		else
			resetskill;
		close;
	} else {
		mes "Ipnotista";
		mes "Mi dispiace, ma il tuo";
		mes "Classe lavorativa non qualifica";
		mes "per il "+.@str$+" Ripristina servizio";
		mes "che io fornisco. Posso solo";
		mes "offrire "+.@str$+" Si reimposta a";
		mes "seguendo i lavori...";
		next;
		mes "Ipnotista";
		mes "^FF0000Spadaccino, Acolita,";
		mes "Mago, Ladro, Arciere";
		mes "Mercante, Taekwon";
		mes "Ragazzo, Ragazza di Taekwon-Do,";
		mes "Gunslinder e Ninja^000000.";
		close;
	}
}
