geffen_in,99,174,5	script	Ore Tester#sorty	8_F_GIRL,{
	mes "[Ordinamento del tester del minerale]";
	mes "Hai ^3131FFqualche tipo di minerale nei tuoi inventari^000000 che occupa molti spazi?";
	mes "Posso distinguere i tuoi minerali e organizzarli subito~!";
	next;
	switch(select("Raffina il mio minerale.:Cancellare.")) {
	case 1:
		mes "[Ordinamento del tester del minerale]";
		mes "Al momento riesco solo a distinguere ^FF0000Steel^000000, ^FF0000Iron^000000, ^FF0000Iron Ore^000000 e ^FF0000Rough Oridecon^000000 e organizzarli.";
		mes "Tieni presente che ^3131FFaltri elementi non sono^000000!";
		next;
		mes "[Ordinamento del tester del minerale]";
		mes "Allora, vuoi che organizzi i tuoi minerali?";
		next;
		.@i = select("Acciaio, per favore.:Ferro, per favore.:Minerale di ferro, per favore.:Oridecon grezzo, per favore.:No grazie.");
		if (.@i == 5) {
			mes "[Ordinamento del tester del minerale]";
			mes "Eh, odio davvero le persone disordinate!";
			close;
		}
		setarray .@ores[0],999,998,1002,756;
		.@item = .@ores[.@i-1];
		if (countitem(.@item) == 0) {
			mes "[Ordinamento del tester del minerale]";
			mes "Non ne hai "+getitemname(.@item)+"...";
			close;
		}
		// while(1) {
			// if (countitem(.@item)) {
				// set .@count, countitem(.@item);
				// set .@total, .@total + .@count;
				// delitem .@item, .@count;
			// } else {
				// getitem .@item, .@total;
				// break;
			// }
		// }
		.@count = countitem(.@item);
		delitem .@item, .@count;
		getitem .@item, .@count;
		mes "[Ordinamento del tester del minerale]";
		mes "È fatto. Sembra fantastico! Non la pensi così?";
		close;
	case 2:
		mes "[Ordinamento del tester del minerale]";
		mes "Sembra che la tua borsa sia pulita? Uh! Potresti essere il maestro della pulizia!";
		close;
	}
}
