	mes "[Ore Tester Sorty]";
	mes "Hai ^3131FFqualche altro tipo di minerali simili nei tuoi inventari^000000 che occupano molti spazi?";
	mes "Posso distinguere i tuoi minerali e organizzarli subito~!";
	next;
	switch(select("Raffina il mio minerale.:Annulla.")) {
	case 1:
		mes "[Ore Tester Sorty]";
		mes "Al momento posso distinguere solo ^FF0000Acciaio^000000, ^FF0000Ferro^000000, ^FF0000Minerario di Ferro^000000, e ^FF0000Oridecon Grezzo^000000 e organizzarli.";

		next;
		mes "[Ore Tester Sorty]";
		mes "Quindi, vuoi che ti organizzi i tuoi minerali?";
		next;
		.@i = select("Acciaio, per favore.:Ferro, per favore.:Minerario di ferro, per favore.:Oridecon grezzo, per favore.:No, grazie.");
		if (.@i == 5) {
			mes "[Ore Tester Sorty]";
			mes "Uffa, odio i tipi disordinati!";
			close;
		}
		setarray .@ores[0],999,998,1002,756;
		.@item = .@ores[.@i-1];
		if (countitem(.@item) == 0) {
			mes "[Ore Tester Sorty]";
			mes "Non hai nessun/a "+getitemname(.@item)+"...";
			close;
		}
		// while(1) {
			// if (countitem(.@item)) {
				// set .@count, countitem(.@item);
				// set .@total, .@total + .@count;
				// delitem .@item, .@count;
			// } else {
				// getitem .@item, .@total;
				// break;
			// }
		// }
		.@count = countitem(.@item);
		delitem .@item, .@count;
		getitem .@item, .@count;
		mes "[Ore Tester Sorty]";
		mes "È fatto. Sembra ottimo! Non credi?";
		close;
	case 2:
		mes "[Ore Tester Sorty]";
		mes "Sembra che la tua borsa sia pulita? Huhu! Potresti essere il maestro della pulizia!";
		close;
	}
}
