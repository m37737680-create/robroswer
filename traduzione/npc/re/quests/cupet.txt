function	script	cute_pet_manager	{
	// getarg(0) -> .@tame_id Array
	// getarg(1) -> .@tame_amount Array
	// getarg(2) -> .@hunt_id Array
	// getarg(3) -> .@hunt_amount Array
	// getarg(4) -> .@mob_id Array
	// getarg(5) ->  Attempt Variable
	// getarg(6) -> .@tame_gets Variable
	// getarg(7) -> .@hunt_id2 Array
	// getarg(8) -> .@hunt_amount2 Array

	mes "[Gestore di Animali Domestici Carini]";
	mes "Ciao~! Sono un Cute Pet Manager";
	mes "chi è responsabile delle pubbliche relazioni";
	mes "per il nuovo sistema di animali domestici carini e potenziati.";
	next;
	
	mes "[Gestore di Animali Domestici Carini]";
	mes "Se hai qualsiasi preoccupazione riguardo a";
	mes "il sistema di animali domestici carini sono qui per guidarti.";
	mes "Fammi sapere quale addomesticamento";
	mes "articolo che vuoi,";
	mes "poi ti racconterò il mostro";
	mes "che puoi addomesticare con l'oggetto.";
	next;
	
	mes "[Gestore di Animali Domestici Carini]";
	mes "Anche spiegherò quali materiali";
	mes "hai bisogno di portare per";
	mes "prendi l'oggetto di addomesticamento.";
	next;
	
	mes "[Gestore di Animali Domestici Carini]";
	mes "Se hai già portato i materiali";
	mes "per scambiare con l'oggetto di addomesticamento,";
	mes "Posso scambiarlo con oggetti per addomesticare immediatamente.";
	next;
	
	mes "[Gestore di Animali Domestici Carini]";
	mes "Tuttavia, c'è un limite";
	mes "per scambiare oggetti di addomesticamento.";
	mes "Puoi ottenere solo oggetti di addomesticamento";
	mes getarg(6) + " times. So please make sure";
	mes "quante volte hai scambiato.";
	next;
	
	mes "[Gestore di Animali Domestici Carini]";
	mes "Allora, quale oggetto per addomesticare vuoi?";
	mes "Scegliere uno dall'elenco.";
	next;
	
	// Create Menu System
	set .@size, getarraysize(getarg(0));
	for (set .@a, 0; .@a < .@size; set .@a, .@a + 1)
		set .@menu$, .@menu$ + (.@menu$ == "" ? "" : ":") + getitemname(getelementofarray(getarg(0), .@a));
	
	// Query Player Choice
	set .@choice, select(.@menu$) - 1;

	// Store Variables (Less Lookup)
	set .@tame_id, getelementofarray(getarg(0), .@choice);
	set .@tame_amount, getelementofarray(getarg(1), .@choice);
	set .@hunt_id, getelementofarray(getarg(2), .@choice);
	set .@hunt_amount, getelementofarray(getarg(3), .@choice);
	set .@mob_id, getelementofarray(getarg(4), .@choice);
	set .@hunt_id2, getelementofarray(getarg(7), .@choice);
	set .@hunt_amount2, getelementofarray(getarg(8), .@choice);

	// dispbottom "Tame ID: " + .@tame_id;
	// dispbottom "Tame Amount: " + .@tame_amount;
	// dispbottom "Hunt ID: " + .@hunt_id;
	// dispbottom "Hunt Name: " + getitemname(.@hunt_id);
	// dispbottom "Hunt Amount: " + .@hunt_amount;
	// dispbottom "Mob ID: " + .@mob_id;
	if (.@hunt_id2) { 
		// dispbottom "Hunt ID2: " + .@hunt_id2;
		// dispbottom "Hunt Amount2: " + .@hunt_amount2;
		set .@hunt2_count, countitem(.@hunt_id2);
	}
	if (countitem(.@hunt_id) >= .@hunt_amount && countitem(6083) > 0 && .@hunt2_count >= .@hunt_amount2) {
		mes "[Gestore di Animali Domestici Carini]";
		mes "Ah, hai raccolto tutti gli oggetti";
		mes "con i tuoi amici. Ora puoi";
		mes "scambio per un " + getitemname(.@tame_id) + ".";
		mes "Vuoi scambiare con me?";
		next;
		if  (select("Scambio:Non Scambiare") == 2) { 
			mes "[Gestore di Animali Domestici Carini]";
			mes "Non lo fai? Tornerai di nuovo.";
			close;
		}
		delitem .@hunt_id, .@hunt_amount;
		delitem 6083, 1;
		if (.@hunt_id2)
			delitem .@hunt_id2, .@hunt_amount2;
		set getarg(5), getarg(5) + 1;
		getitem .@tame_id, .@tame_amount;
		
		mes "[Gestore di Animali Domestici Carini]";
		mes "Scelta saggia.";
		mes "Spero che tu e il tuo animale domestico vi troviate bene.";
		close;
	} else {
		mes "[Gestore di Animali Domestici Carini]";
		mes "Hai scelto " + getitemname(.@tame_id) + ".";
		mes "Puoi usarlo per addomesticare";
		mes "^FF0000" + getmonsterinfo(.@mob_id, MOB_NAME) + "mostri ^000000.";
		next;
		mes "[Gestore di Animali Domestici Carini]";
		mes "Porta ^FF0000" + .@hunt_amount + " " + getitemname(.@hunt_id) + "^000000" + (.@hunt_id2 ? ", ^FF0000" + .@hunt_amount2 + " " + getitemname(.@hunt_id2) : "") + " ^000000 e ^FF00001 Dolly Capsule^000000";
		mes "poi puoi scambiarli";
		mes "per un " + getitemname(.@tame_id) + ".";
		close;
	}
}
