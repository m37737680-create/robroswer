		if (countitem(getarg(5)) < getarg(6))
			set .@items,1; //incomplete
		else
			set .@items,2; //complete
	}

		mes "Hai una missione in corso. Vorresti controllare i dettagli?";
		next;
		if(select("Controlla i dettagli.:Annulla.") == 1)
			callsub L_Details, .@quest_id;
		close;
	} else {
		mes "Ho fatto abbastanza bene per la missione. Devo segnalarlo ora?";
		next;

			delitem getarg(3),getarg(4);
			if (.@items)

			getexp getarg(1),getarg(2);
			erasequest .@quest_id;
			mes "Hai completato la missione. Ottieni i premi.";
