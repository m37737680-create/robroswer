L_Quest:
	set .@quest1, getarg(0);
	set .@quest2, .@quest1+31;
	.@cutin$ = getarg(3);
	set .@playtime, checkquest(.@quest2,PLAYTIME);
	if (.@playtime == 0 || .@playtime == 1) {
		mes "[86-90 Bacheca della missione]";
		mes "Devi aspettare 3 ore prima di poter accettare questa missione di nuovo.";
		close;
	}
	set .@hunting, checkquest(.@quest1,HUNTING);
	if (.@hunting == 0 || .@hunting == 1) {
		mes "[86-90 Bacheca della missione]";
		mes "Hai finito di cacciare tutti i 30? "+getarg(1)+getarg(5,"")+" ancora?";
		close;
	} else if (.@hunting == 2) {
		mes "[86-90 Bacheca della missione]";
		mes "Hai completato la caccia.";
		mes "Accetta questo premio come un risarcimento.";
		erasequest .@quest1;
		if (.@playtime > -1) erasequest .@quest2;
		setquest .@quest2;
		getexp getarg(2),0;
		close;
	}/* else if (.@hunting == 3) {
//		recall_completequest .@quest1

		mes "[86-90 Bacheca della missione]";
		mes "Hai riaccettato il "+getarg(1)+" Missione di caccia.";
		close;
	}*/ else {

		mes getarg(4);
		next;
		if (.@cutin$ != "") cutin .@cutin$,3;
		mes "[86-90 Bacheca della missione]";
		mes "Caccia 30 "+getarg(1)+" mostri lì e pubblica il tuo successo su questo bacheca.";
		next;
		if (.@cutin$ != "") cutin .@cutin$,255;
		if(select("Li caccierò.:No grazie.") == 1) {
			mes "[86-90 Bacheca della missione]";
			mes "Quando avrai completato la missione, pubblicala su questo bacheca e ritira il tuo premio.";
			if (.@playtime > -1) erasequest .@quest2;
