S_Quest:
	if (isbegin_quest(getarg(0)))
		erasequest getarg(0);
	setquest getarg(1);
	if (getarg(2,0))
		getexp getarg(2),getarg(3);
	return;
S_Erasequest:
	.@total = getargcount();
	for ( .@i = 0; .@i < .@total; .@i += 3 ) {
		.@quest_id = getarg(.@i);
		switch( checkquest(.@quest_id,HUNTING) ) {
		case -1:
			if (checkquest(.@quest_id+1) > -1)
				erasequest (.@quest_id+1);
			continue;
		case 0:
		case 1:
			mes "[Ragi]";
			switch( getarg(.@i+1) ) {
			case 1:
				mes "Il resto della storia di "+ getarg(.@i+2) +" Va bene, chiederò a qualcun altro.";
				break;
			case 2:
			case 3:
				mes "Chiederò ad altri avventurieri la storia di "+ getarg(.@i+2) +".";
				break;
			}
			break;
		case 2:
			mes "[Ragi]";
			switch( getarg(.@i+1) ) {
			case 1:
				mes "Hai sentito la storia di "+ getarg(.@i+2) +"Grazie. Prometto di fare una bella canzone.";
				getexp 1100000,300000;
				break;
			case 2:
				mes "La storia di Morocc è abbastanza buona per canzoni? Hai fatto un buon lavoro.";
				getexp 1100000,200000;
				break;
			case 3:
				mes "Com'è stata la tua avventura? È stato divertente? Per favore, raccontami la storia della tua avventura un giorno.";
				getexp 1100000,200000;
				break;
			}
		}
		erasequest .@quest_id;
		next;
	}
	return;
