function	script	VIP_iRO_Acolyte	{
	if (!vip_status(VIP_STATUS_ACTIVE)) {
		mes getarg(0);
		mes "Ma i miei poteri completi sono disponibili solo per coloro che sono VIP.";
		mes "Posso ancora offrirti un varco verso "+getarg(1)+" tuttavia.";
		next;
		switch(select(getarg(0)+" -> 1800z:Annulla")) {
		case 1:
			if (Zeny < 1800) {
				mes getarg(0);
				mes "Mi dispiace, ma non hai";
				mes "abbastanza zeny per il Teletrasporto";
				mes "Servizio. La tariffa per teletrasportare";
				mes "a "+getarg(1)+" sono 1.800 zeny.";
				close;
			}
			specialeffect2 EF_READYPORTAL;
			specialeffect2 EF_TELEPORTATION;
			specialeffect2 EF_PORTAL;
			next;
			set Zeny, Zeny - 1800;
			if (getarg(1) == "Juno")
				warp "yuno",158,125;
			else
				warp "prontera",116,72;
			close;
		case 2:
			close;
		}
	}
	return;
}
