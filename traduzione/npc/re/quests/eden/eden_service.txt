-	script	Goods Cabinet#00::pggc	-1,{
	mes "È un armadio per il deposito di beni.";
	mes "Un messaggio è scritto su un pezzo di carta.";
	next;
	mes "+ Armadio è esclusivamente +";
	mes "+ per il gruppo Eden +";
	mes "Se vuoi usare questo";
	mes "armadio per favore controlla";
	mes "il seguente:";
	next;
	mes "Sei un membro";
	mes "   del gruppo di Edenn?";
	mes "Hai imparato";
	mes "   abbastanza competenze di base?";
	mes "La commissione di cabina è";
	mes "  ^4d4dff500 zeny^000000!";
	next;
	if (callfunc("F_HasEdenGroupMark")) {
		mes "Devi inserire zeny per usare l'armadio.";
		mes "Costo: 500 Zeny ";
		mes "Vorresti usarlo?";
		next;
		switch(select("Usare il Gabinetto.:Annulla.")) {
			case 1:
				if (Zeny > 499) {
					if(!callfunc("F_CanOpenStorage")){
						mes "L'armadio non funziona per me.";
						mes "Forse non sono ancora qualificato per usare Cabinet.";
						close;
					}
					set Zeny, Zeny - 500;
					close2;
					openstorage;
					end;
				}
				mes "Non ho abbastanza zeny.";
				mes "Ho bisogno di almeno 500 zeny per usare l'Armadio.";
				close;
			case 2:
				mes "Lo userò la prossima volta.";
				close;
		}
	}
	mes "Ho bisogno di un marchio Eden Group per usare questo mobile.";
	close;
}
