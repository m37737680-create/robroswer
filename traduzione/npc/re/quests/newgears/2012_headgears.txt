L_CheckHunting:
	for (set .@quest,getarg(0)+1; .@quest<getarg(1)+getarg(2); set .@quest,.@quest+1) {
		if (checkquest(.@quest,HUNTING) == 2) {
			mes "[Reno]";
			mes "Finito "+getarg(3)+" caccia a collezioni missione";
			next;
			setquest getarg(0);
			erasequest .@quest;
			getitem getarg(4),getarg(5);
			mes "Ricevi "+getarg(5)+" "+getitemname(getarg(4))+"come il "+getarg(3)+" collezione ricompensa.";
			close;
		}
	}
	return;
