que_ng,21,72,6	script	Master Armor Craftsman	4_M_RASWORD,{
	mes "[Gyo]";
	mes "Come definiresti la vittoria?";
	mes "Per sconfiggere il nemico...";
	mes "Tuttavia, la cosa più importante è...";
	mes "Mai cadere tu stesso!";
	next;
	mes "[Gyo]";
	mes "Per questo, devi proteggerti meglio.";
	mes "Non puoi proteggere gli altri mentre ostacoli la tua protezione.";
	next;
	if(select("Combina armature.:Termina conversazione.") == 2) {
		mes "[Gyo]";
		mes "La prudenza è anche una virtù importante in battaglia.";
		close;
	}
	switch(select("Bracciale da lupo[1]:Bracciale a mezzaluna[1]:Armatura Squamata Ninja[1]:Tenebri Latitanti[1]:Basta.")) {
	case 1:
		mes "[Gyo]";
		mes "Armatura per braccio di lupo... Un pezzo d'armatura bestiale.";
		next;
		callsub L_Select,2172;
		if (countitem(2171) > 0 && countitem(6252) > 9) {  //custom translation
			delitem 2171,1; //Fox_Armguard
			delitem 6252,10; //Wolf_Blood
			getitem 2172,1; //Wolf_Armguard
			mes "[Gyo]";
			mes "Senti la natura del lupo?";
			mes "Sia selvaggi che affascinanti...";
			close;
		}
		break;
	case 2:
		mes "[Gyo]";
		mes "Bracciale a mezzaluna... Una cosa di bellezza e forza... Con nessuno il minore.";
		next;
		callsub L_Select,2173;
		if (countitem(2171) > 0 && countitem(7321) > 99) {  //custom translation
			delitem 2171,1; //Fox_Armguard
			delitem 7321,100; //Fragment_Of_Crystal
			getitem 2173,1; //Crescent_Armguard
			mes "[Gyo]";
			mes "Che bellezza...";
			mes "Come se tutto il corpo stesse traboccando di magia.";
			close;
		}
		break;
	case 3:
		mes "[Gyo]";
		mes "Armatura Squamata da Ninja... Si dice che un leggendario Ninja l'abbia indossata.";
		next;
		callsub L_Select,15054;
		if (countitem(7451) > 29 && countitem(7562) > 29 && countitem(6091) > 29) {  //custom translation
			delitem 7451,30; //Scale_Of_Red_Dragon
			delitem 7562,30; //Ice_Scale
			delitem 6091,30; //Dark_Red_Scale
			getitem 15054,1; //Ninja_Scale_Armor
			mes "[Gyo]";
			mes "Questa è veramente un'armatura leggendaria...";
			mes "Anche se non credo nelle leggende...";
			close;
		}
		break;
	case 4:
		mes "[Gyo]";
		mes "L'armatura del Re Ombra... Era fatta per proteggere gli altri.";
		next;
		callsub L_Select,15055;
		if (countitem(15056) > 0 && countitem(6089) > 9) {  //custom translation
			delitem 15056,1; //Special_Ninja_Suit_
			delitem 6089,10; //Piece_Of_Darkness
			getitem 15055,1; //Tenebris_Latitantes
			mes "[Gyo]";
			mes "Come è?";
			mes "L'armatura non sembra di avere due?";
			close;
		}
		break;
	case 5:
		close;
	}
	mes "[Gyo]";
	mes "Non hai abbastanza ingredienti.";
	mes "Guarda ancora...";
	close;

L_Select:
	.@string1$ = "Check Ingredients.";
	.@string2$ = "Check Stats.";
	while(1) {
		switch(select(.@string1$,.@string2$,"Combinare.","Basta.")) {
		case 1:
			.@string1$ = "";
			callsub L_Ingredients, getarg(0);
			break;
		case 2:
			.@string2$ = "";
			callsub L_Stats, getarg(0);
			break;
		case 3:
			callsub L_Combine;
			return;
		case 4:
			mes "[Gyo]";
			mes "Sembra che tu abbia bisogno di più tempo per decidere.";
			close;
		}
	}

L_Ingredients:
	mes "[Gyo]";
	switch (getarg(0)) {
	case 2172:	//Wolf_Armguard
		mes "Per creare un Bracciale da Lupo ^ff0000, hai bisogno di 1 Bracciale per Polso di Volpe e 10 Sangue di Lupo^000000.";
		mes "Senti? Il battito del sangue di lupo...";
		break;
	case 2173:	//Crescent_Armguard
		mes "Per fare un Bracciale per braccio Mezzaluna ^ff0000Crescent Armguard, hai bisogno di 1 Bracciale per braccio della Volpe e 100 Frammenti di Cristallo^000000.";
		mes "Una mezzaluna bella e chiara come un cristallo sarà incisa su di esso.";
		break;
	case 15054:	//Ninja_Scale_Armor
		mes "Per creare un'Armatura Scala Ninja ^ff0000, hai bisogno di 30 Scaglie di Drago Fuoco, 30 Scaglie di Ghiaccio e 30 Pezzi di Scaglia Rosso Scuro^000000.";
		mes "Puoi sentire i suoi poteri leggendari.";
		break;
	case 15055:	//Tenebris_Latitantes
		mes "Per creare l'Armatura del Re Ombra ^ff0000, avrai bisogno di un Tuta Ninja Speciale con uno slot e 10 Pezzi Oscuri^000000.";
		mes "Con l'oscurità cancella l'esistenza stessa...";
		break;
	}
	next;
	return;

L_Stats:
	mes "[" + getitemname(getarg(0)) + "]";
	switch (getarg(0)) {
	case 2172:	//Wolf_Armguard
		mes "Possibilità di avere 5 sec. di ATK+100, FLEE -50 al colpo corpo a corpo.";
		mes "Tipo di scudo, Difesa 45, Slot 1";
		mes "Livello Richiesto 100, Solo Tipo Ninja";
		break;
	case 2173:	//Crescent_Armguard
		mes "Diminuisce il ritardo delle abilità dopo l'acquisizione del 2% per ogni potenziamento.";
		mes "Tipo di scudo, Difesa 70, Slot 1";
		mes "Livello Richiesto 100, Solo Tipo Ninja";
		break;
	case 15054:	//Ninja_Scale_Armor
		mes "MHP+15%, MSP-30%";
		mes "Tipo di Armatura, Difesa 90, Slot 1";
		mes "Livello Richiesto 100, Solo Tipo Ninja";
		break;
	case 15055:	//Tenebris_Latitantes
		mes "Opportunità di attivare Illusione Livello 1 - Ombra quando attaccato da attacchi corpo a corpo.";
		mes "Tipo di Armatura, Difesa 60, Slot 1";
		mes "Livello Richiesto 100, Solo Tipo Ninja";
		break;
	}
	next;
	return;

L_Combine:
	mes "[Gyo]";
	mes "Make sure that ^ff0000you have the correct ingredients and equipment at hand^000000.";
	mes "Non piangere sopra di esso più tardi.";
	next;
	switch(select("Combinare.:Basta.")) {
	case 1:
		if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2000) {
			mes "Aspetta!";
			mes "Non puoi ricevere oggetti";
			mes "perché porti troppo.";
			mes "Per favore riprova";
			mes "dopo aver alleggerito il tuo peso.";
			close;
		}
		return;
	case 2:
		mes "[Gyo]";
		mes "Sembra che tu abbia bisogno di più tempo per decidere.";
		close;
	}
	end;
}
