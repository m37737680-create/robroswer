yuno,179,174,4	script	Gomer	851,{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 1000) {
		mes "[Gomer]";
		mes "Stai portando così tanto. Lascia un po' dei tuoi bagagli e torna da me.";
		close;
	}
	if (countitem(1549)) {
		mes "[Gomer]";
		mes "Allora, come sta andando per te il Pilebuncker?";
		close;
	}
	if (Class != Job_Mechanic && Class != Job_Mechanic_T && Class != Job_Baby_Mechanic) {
		mes "[Gomer]";
		mes "Sono un meccanico in pensione e non credo di avere nessun rapporto con voi.";
		close;
	}
	mes "[Gomer]";
	mes "Allora? Com'è essere su un Madogear?";
	mes "È qualcosa che non è permesso a coloro che hanno fallito nel cercare il proprio percorso.";
	next;
	mes "[Gomer]";
	mes "Io? Mi sono ritirato da molto tempo fa. Perché ho avuto un incidente... mi sono fatto male gravemente e non potrò più andare in bici. Sono triste per questo.";
	next;
	mes "[Gomer]";
	mes "Quindi ho deciso di restare qui a forgiare ingranaggi per Madogear.";
	mes "Eh? Non dico di avere nessun reclamo su questo lavoro.";
	next;
	mes "[Gomer]";
	mes "Essere costretti a fare qualcosa che non si vuole è la cosa più misera del mondo.";
	next;
	mes "[Gomer]";
	mes "Ci sono persone che sono felici di essere solo oggetti di scena, sai, come essere il muro su cui qualcuno può appoggiarsi.";
	next;
	mes "[Gomer]";
	mes "Mi fa sentire bene e sto per creare un nuovo ingranaggio. Voglio che tu mi aiuti. Che ne dici?";
	next;
	switch(select("Non sono così interessato.:Cosa farai?:Da dove prendi i materiali?:Ho portato i materiali.")) {
	case 1:
		mes "[Gomer]";
		mes "Niente che posso fare";
		mes "riguardo questo allora.";
		mes "A presto!";
		close;
	case 2:
		mes "[Gomer]";
		mes "Un bastone metallico enorme che potrebbe schiacciare un avversario con un colpo!";
		next;
		mes "[Gomer]";
		mes "Non sembra elegante, ma ho progettato un dispositivo a molla per attaccare con grande forza.";
		next;
		mes "[Gomer]";
		mes "È nascosto, quindi quando trovi un'occasione, un colpo potrebbe mettere un avversario in condizioni critiche!";
		mes "Non vedo l'ora di vederlo in azione!";
		next;
		mes "[Gomer]";
		mes "Se dovessi immaginare quanto possa essere buono questo tipo di arma, lo farò diventare realtà.";
		next;
		mes "[Gomer]";
		mes "Portami 200 Acciaio, 30 Tubi Flessibili e 1 Brocca. Potrei fare quello che ho appena descritto con quei materiali.";
		close;
	case 3:
		mes "[Gomer]";
		mes "Dovresti essere in grado di ottenere acciaio da molti mostri come i Koboldi.";
		next;
		mes "[Gomer]";
		mes "I Tubi Flessibili sono più rari, lasciati cadere dai Metalings intorno a Einbroch e Lighthalzen.";
		next;
		mes "[Gomer]";
		mes "Infine, trovare una Brocca è molto difficile, ma dovresti essere in grado di fare qualcosa al riguardo. Haha, vai e prendila!";
		close;
	case 4:
		if (countitem(999) < 200 || countitem(7325) < 30 || countitem(1415) < 1) {
			mes "[Gomer]";
			mes "Non hai portato tutti i materiali che ho chiesto.";
			next;
			mes "[Gomer]";
			mes "Ho l'abitudine di abbandonare quello che sto creando e provare a fare cose nuove quando arriva l'ispirazione.";
			mes "Meglio che tu abbia fretta!";
			close;
		} else {
			mes "[Gomer]";
			mes "Ho aspettato. Non sei venuto in tempo, quindi mi chiedevo se te ne fossi dimenticato. Per favore aspetta lì un po'.";
			specialeffect EF_REPAIRWEAPON;
			next;
			mes "[Gomer]";
			mes "È fatto. Cosa ne pensi? Non male, eh? Anche se l'ho fatto io stesso, devo dire che è piuttosto decentemente fatto.";
			next;
			mes "[Gomer]";
			mes "Combinazione di peso, volume e guarda il suo colore e la sua lucentezza. Mi conquista!";
			next;
			mes "[Gomer]";
			mes "...........";
			next;
			mes "[Gomer]";
			mes "Ehm, con un piccolo miglioramento, potrebbe essere prodotto in serie e consegnato. Sono sicuro di poterne fare altri, quindi te ne darò questo.";
			next;
			mes "[Gomer]";
			mes "E poi si scopre che non avevo bisogno di tutto quell'acciaio dopo. Quindi, prendi la maggior parte indietro qui.";
			delitem 999,50; //Steel
			delitem 7325,30; //Tube
			delitem 1415,1; //Skewer
			getitem 1549,1; //Pilebuncker
			getitem 1360,1; //Two_Handed_Axe
			close;
		}
	}
}
