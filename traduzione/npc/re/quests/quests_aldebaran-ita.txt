c_tower1,232,222,5	script	Belljamin Button#ct	968,{
	mes "[Pulsante campanello]";
	mes "I mostri qui sono in grado di controllare il tempo.";
	if( BaseLevel < 150 ){
		mes "Torna quando avrai raggiunto il ^0000FFLivello 150^000000.";
		close;
	}
	mes "Tempo, ricordi, ambienti... tutto può deformarsi qui.";
	next;
	mes "[Pulsante campanello]";
	mes "Ti prego di non chiedere altro a riguardo. Potrebbe mettere in pericolo la tua vita.";
	next;
	.@menu = select("Non chiedere:Vai al 2° piano della torre deformata:Vai al 3° piano della torre deformata");
	switch(.@menu){
		case 1:
			break;
		case 2:
		case 3:
			if( .@menu == 2 ){
				.@map$ = "c_tower2_";
				.@x = 268;
				.@y = 26;
			}else{
				.@map$ = "c_tower3_";
				.@x = 65;
				.@y = 147;
			}
			mes "[Pulsante campanello]";
			mes "Fai attenzione alla tua scelta.";
			next;
			mes "[Pulsante campanello]";
			mes "Mi servono ^0000FF10.000z^000000 per farti entrare. È un posto piuttosto pericoloso dove stare.";
			next;
			switch( select( "Esentato:Paga 10.000 z:Perché hai bisogno di soldi?" ) ){
				case 1:
					break;
				case 2:
					if( Zeny < 10000 ){
						mes "[Pulsante campanello]";
						mes "Veramente? Non hai i soldi e mi hai chiesto di portarti lì?";
						close;
					}
					Zeny -= 10000;
					mes "[Pulsante campanello]";
					mes "Non sono sicuro di come farai a tenere traccia del tempo lì dentro, ma stai più attento che puoi.";
					close2;
					warp .@map$, .@x, .@y;
					end;
				case 3:
					mes "[Pulsante campanello]";
					mes "Sono qui tutto il giorno per la sicurezza degli avventurieri. Li avverto sempre del pericolo. Ecco perché ho bisogno di una qualche forma di pagamento.";
					emotion ET_SWEAT;
					close;
			}
			break;
	}
	mes "[Pulsante campanello]";
	mes "Uff... hai fatto una buona scelta. Questo posto è brutale, quindi probabilmente dovresti tornare in una zona sicura.";
	close;
}
