c_tower1,232,222,5	script	Belljamin Button#ct	968,{
	mes "[Pulsante Belljamin]";
	mes "I mostri qui sono in grado di controllare il tempo.";
	if( BaseLevel < 150 ){
		mes "Torna quando avrai raggiunto il livello ^0000FFLevel 150^000000.";
		close;
	}
	mes "Tempo, ricordi, ambienti... tutto può diventare distorto qui.";
	next;
	mes "[Pulsante Belljamin]";
	mes "Ti prego di non chiedere più di questo. Potrebbe mettere la tua vita in pericolo.";
	next;
	.@menu = select("Non chiedere:Vai al secondo piano della torre deforme:Vai al terzo piano della torre deforme");
	switch(.@menu){
		case 1:
			break;
		case 2:
		case 3:
			if( .@menu == 2 ){
				.@map$ = "c_tower2_";
				.@x = 268;
				.@y = 26;
			}else{
				.@map$ = "c_tower3_";
				.@x = 65;
				.@y = 147;
			}
			mes "[Pulsante Belljamin]";
			mes "Attento alla tua scelta.";
			next;
			mes "[Pulsante Belljamin]";
			mes "Ho bisogno di ^0000FF10,000z^000000 per farti entrare. È un posto piuttosto pericoloso.";
			next;
			switch( select( "Smetti:Paga 10,000z:Perché hai bisogno di soldi?" ) ){
				case 1:
					break;
				case 2:
					if( Zeny < 10000 ){
						mes "[Pulsante Belljamin]";
						mes "Davvero? Non hai i soldi e mi hai chiesto di portarti lì?";
						close;
					}
					Zeny -= 10000;
					mes "[Pulsante Belljamin]";
					mes "Non sono sicuro di come ti terrai traccia del tempo lì dentro, ma fai molta attenzione.";
					close2;
					warp .@map$, .@x, .@y;
					end;
				case 3:
					mes "[Pulsante Belljamin]";
					mes "Sono qui tutto il giorno per la sicurezza degli avventurieri. Li allerto sempre sul pericolo. Ecco perché ho bisogno di una forma di pagamento.";
					emotion ET_SWEAT;
					close;
			}
			break;
	}
	mes "[Pulsante Belljamin]";
	mes "Uffa...Hai fatto una buona scelta. Questo posto è brutale quindi dovresti probabilmente tornare in un'area sicura.";
	close;
}
