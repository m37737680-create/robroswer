		close;

	mes "Mi hai preso! Dannazione... non mi aspettavo che la mia reputazione di Tomb Raider venisse danneggiata in questo modo...";
	next;
	mes "[Gatto sospettoso]";
	mes "Ma andiamo, sono così strani! Sono mummie! Solo mummie, sai? Non dovrebbero essere così forti!";
	next;
	if(select("Mummie forti? Dove sono?:Termina la conversazione.") == 2) {
		mes "[Gatto sospettoso]";
		mes "Cos'è~ non vuoi sentire la mia storia? Bene! Faresti meglio a non menzionare il mio nome da nessun'altra parte!";
		close;
	}
	mes "[Gatto sospettoso]";
	mes "Ahah, è un segreto.";
	mes "L'ho scoperto mentre rubavo... No, voglio dire, semplicemente camminando attraverso la piramide.";
	next;
	mes "[Gatto sospettoso]";
	mes "Quando ho visto una mummia venire verso di me, ho pensato 'non importa' e non mi importava. Ma poi mi ha buttato a terra con uno schiaffo in faccia! Come ha potuto fare una cosa del genere a un gatto carino come me! Tu, mostro bendato... aspetta e vedrai!";
	next;
	if(select("Vorrei andare anch'io.:Termina la conversazione.") == 2)
		close;
	mes "[Gatto sospettoso]";
	mes "Non c'è modo! Ho passato così tanto tempo a trovare quel percorso!!";
	next;
	mes "[Gatto sospettoso]";
	mes "Hmm... ma se insisti, dammi solo 5000 Zeny. Ti guiderò.";
	mes "Cosa~? Non lanciarmi quello sguardo! Hai idea di quanto sia difficile per un gatto sopravvivere a Midgard!";
	next;
	mes "[Gatto sospettoso]";
	mes "^FF0000Ma ora che ci penso, un debole come te non sopravviverà lì. Forse per un secondo se sei fortunato?^000000";
	mes "Beh, non sono affari miei. Ma ti avevo comunque avvisato!";
	next;
	switch(select("Paga 5.000 zeny e parti.:Lasci perdere.")) {
	case 1:
		if (Zeny < 5000) {
			mes "[Gatto sospettoso]";
			mes "Dove pensi di andare! Sei così al verde!";
			close;
		}
		set Zeny, Zeny-5000;
		mes "[Gatto sospettoso]";
		mes "Bah, che succede con questi vecchi soldi... non hai niente di scintillante o luccicante?!";
		mes "Ahh, seguimi!";
		close2;
		warp "moc_prydn1",93,96;
		end;
	case 2:
		mes "[Gatto sospettoso]";
		mes "Cavolo, non hai nemmeno 5000 Zeny, vero?";
		close;
	}
}
