		next;
		mes "[Jason]";
		mes "Il carrello grigio solido.";
		next;
		mes "[Jason]";
		mes "Posso rimodellarlo in nuovi e carini design.";
		next;
		switch( select( "Cos'è la decorazione di carretti?", "Imparare a decorare i carretti", "Fine dialogo" ) ) {
		case 1:
			mes "[Jason]";
			mes "Vorrei ristrutturare completamente il povero carretto.";

			mes "Un piccolo rifacimento non basta, non sei d'accordo?";
			next;
			mes "[Jason]";
			mes "Quindi ho inventato alcuni nuovi design.";
			next;
			mes "[Jason]";
			mes "È carino e rinfrescante!";
			close;
		case 2:
			if (BaseClass != Job_Merchant) {
				mes "[Jason]";
				mes "Perché vuoi imparare la decorazione di carretti? Non disturbarmi!";
				close;
			}

				mes "Ti insegnerò se mi porti il ^3131FFCute Cart Remodel Coupon^000000.";

			mes "[Jason]";
			mes "Hai portato il buono per la ristrutturazione del Cute Cart. Sei sicuro di voler imparare a decorare un carrello spingibile?";
			next;
			if (select("SÌ", "NO") == 2) {
				mes "[Jason]";
				mes "Come ancora dopo che hai cambiato idea!";
				close;
			}
			mes "[Jason]";
			mes "Va bene! Seguimi.";
			next;
			mes "[Jason]";
			mes "Questo design";
			mes "deve afferrare il carretto...";
			next;
			mes "[Jason]";
			mes "Questo design";

			mes "[Jason]";
			mes "Sei un genio! L'hai imparato!";
			next;
			mes "[Jason]";
			mes "Ora è il tuo turno di farlo.";
			mes "Premi Alt+S per cambiare il tuo carretto!";
			next;
			mes "[Jason]";
			mes "Questo è il nuovo carretto carino. Ora fai parte della nostra famiglia...!";
			delitem 6606,1;	// P_Cart_C
			cart_remodel = 1;
			skill "MC_CARTDECORATE",1,SKILL_PERM;
			close;
		case 3:
			close;
		}
	}
	mes "[Jason]";
	mes "Hai decorato il carretto?";
	next;
	switch( select( "C'è altro da imparare?", "Ho dimenticato come decorare il mio carretto" ) ) {
	case 1:
		if (cart_remodel == .@max_skill_lvl) {
			mes "[Jason]";
			mes "No. Sto ancora esplorando nuovi design. Ti insegnerò quando l'avrò imparato.";
			close;
		}
		mes "[Jason]";
		mes "Ho qualcosa da insegnare, vuoi imparare?";

		mes "Ho bisogno di un coupon per il rimodellamento del carrello 'Cute Cart' ^3131FFCute Cart Remodel Coupon^000000.";

		if (select("Voglio imparare", "No") == 2) {
			mes "[Jason]";
			mes "Puoi tornare quando vuoi.";
			close;
		}
		if (countitem(6606) < 0) {
			mes "[Jason]";
			mes "Non hai nessun coupon per il rimodellamento di Cute Cart?";
			close;
		}

		mes "Ok, cominciamo!";

		mes "[Jason]";
		mes "....";
		mes "Facciamolo!";
		next;
		mes "[Jason]";
		mes "Ti ho insegnato la capacità, puoi provarla ora.";

		skill "MC_CARTDECORATE", cart_remodel ,SKILL_PERM;
		close;

		mes "[Jason]";
		mes "Oh? Lasciami vedere.";
		next;
		if (BaseClass != Job_Merchant) {
			mes "[Jason]";
			mes "Torna quando potrai usare il carrello!";
			close;
		}
		mes "[Jason]";
		mes "Non ti ricordi quello che ti ho insegnato?";

		mes "[Jason]";
		mes "Hai dimenticato?";
		mes "Ricordiamo ora!";
		next;
		mes "[Jason]";
		mes "Pensa attentamente, te lo ricorderai di nuovo.";
		skill "MC_CARTDECORATE",1,SKILL_PERM;
		close;
	}
}
