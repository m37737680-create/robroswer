	end;

	.@map$ = strnpcinfo(4);
	if (compare( .@map$,"te_alde" ))
		callsub S_Event, "alde0"+ charat( .@map$,getstrlen(.@map$)-1 ) + "::OnCommandOn";// #popa_alde0[1-5]::OnCommandOn
	else
		callsub S_Event, "prt0"+ charat( .@map$,getstrlen(.@map$)-1 ) + "::OnCommandOn";
	end;

OnReset:
	.@map$ = strnpcinfo(4);
	if (compare( .@map$,"te_alde" ))
		callsub S_Event, "alde0"+ charat( .@map$,getstrlen(.@map$)-1 ) + "::OnReset";

	.@string$ = getarg(0);
	donpcevent "#popa_"+ .@string$;
	donpcevent "#popb_"+ .@string$;
	donpcevent "#popc_"+ .@string$;
	donpcevent "#popd_"+ .@string$;
	end;
}

			mes "È una richiesta di trovare ^4d4dff"+ .@num_items +" o più di "+ .@string$[2] +"^000000.";
			callsub( S_Acceptquest, .@main_quest, .@npc_name$ );
		}
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "Poiché la regione TE ha un'atmosfera più informale,";
		mes "non è divertente fare solo battaglie d'assedio, vero?";
		mes "Allora, ho portato ^4d4dffalcuni lavori che puoi fare durante le ore del sacco^000000~";
		next;
		mes "["+ .@npc_name$ +"]";
		mes "Puoi accettare missioni ^4d4dffa partire da 1 ora prima dell'inizio del assedio e fino alla fine della battaglia assedio TE^000000,";
		mes "and every mission has ^4d4dffa 4-hour time limit^000000 for completion.";
		next;
		mes "["+ .@npc_name$ +"]";
		mes "Semplificando, accetti una missione da me durante le ore del assedio e";
		mes "^4d4dfftorna a darmi un rapporto entro il termine stabilito^000000.";
		next;
		mes "["+ .@npc_name$ +"]";
		mes "Provalo per divertimento.";
		close;
	case 3:
		mes "["+ .@npc_name$ +"]";
		mes "Capisco";
		mes "Torna pure quando cambi idea.";
		mes "Forse non sarò disponibile, però. Hehehe.";
		close;
	}

S_Cancelquest:
	.@npc_name$ = getarg(1);
	if (WoeTETimeStart(3600) == 1 || agitcheck3()) {
		mes "Sembra che tu non abbia ancora compiuto la missione?";
		mes getarg(2);
		next;
		mes "["+ .@npc_name$ +"]";
		mes "Vuoi annullare per caso?";
		next;
		switch( select( "No.","Sì." ) ) {
		case 1:
			mes "["+ .@npc_name$ +"]";
			mes "Buona fortuna!";
			close;
		case 2:
			mes "["+ .@npc_name$ +"]";
			mes "Come desideravi, questa missione è annullata.";
			erasequest getarg(0);
			close;
		}
	}
	else {
		mes "Ancora hai un po' di tempo, ma non è ancora l'ora del assedio";
		mes "Non credo che tu debba andare avanti.";
		mes "Dovresti annullare questa missione. Direi che è fallita.";
		next;
		switch( select( "Vorrei andare avanti.","Sì, lo annullerò." ) ) {
		case 1:
			mes "["+ .@npc_name$ +"]";
			mes "Non credo che ne valga la pena. Beh, buona fortuna!";
			close;
		case 2:
			mes "["+ .@npc_name$ +"]";
			mes "Come desideravi, questa missione è annullata.";
			erasequest getarg(0);
			close;
		}
	}

S_Acceptquest:
	.@npc_name$ = getarg(1);
	next;
	switch( select( "Accettalo.","Non accettarlo." ) ) {
	case 1:
		mes "["+ .@npc_name$ +"]";
		mes "^4d4dffIl tempo limite è di 4 ore.^000000";
		mes "Compi la missione entro quel tempo e riportami.";
		setquest getarg(0);
		close;
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "Prova un'altra missione allora.";
		close;
	}

S_Rentbox:
	setarray .@item_id[0], 2950,15067,20709,22012;// Rune_Ring, Rune_Suit, Mana_Manteau, Mana_Boots
	for ( .@count = 0; .@count < getarg(0); .@count++ ) {
		.@miss_item01 = rand(100);
		if (.@miss_item01 > 3)
			mes "Non hai trovato nulla.";
		else {
			rentitem .@item_id[ .@miss_item01 ],86400;
			mes "Hai trovato '"+ getitemname(.@item_id[ .@miss_item01 ]) +" !'";
		}
	}
	close;

S_Menu:
	for ( .@i = 1; .@i < getargcount(); .@i += 2 ) {
		if ( isbegin_quest(getarg(.@i)) )
			.@menu$ = .@menu$ + getarg(.@i+1) + " ^4d4dffCheck the result^000000:";
		else
			.@menu$ = .@menu$ + getarg(.@i+1) + ":";
		set getelementofarray( getarg(0),.@index ), getarg(.@i);
		.@index++;
	}
	return .@menu$;
}
