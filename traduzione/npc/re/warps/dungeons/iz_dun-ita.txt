iz_dun04,130,234,5	script	Gatekeeper#iz_dun	413,{
	mes "[Portiere]";
	mes "Fermati!";
	mes "Cosa ti porta qui, avventuriero?";
	mes "Abbiamo scoperto un nuovo piano di questa città sottomarina.";
	next;
	mes "[Portiere]";
	mes "Pensi di essere all'altezza della sfida?";
	mes "I mostri qui sono tra i livelli 110 e 124.";
	mes "E devi stare attento al Kraken!";
	next;
	switch(select("Libera il Kraken!:Starò qui.")) {
	case 1:
		mes "[Portiere]";
		mes "Sii cauto e buona fortuna.";
		mes "Se vedi il Kraken assicurati di avere i tuoi amici per aiutarti o fai quello che facciamo noi e scappa.";
		close2;
		warp "iz_dun05",141,187;
		end;
	case 2:
		mes "[Portiere]";
		mes "È sempre meglio essere preparati prima di esplorare una nuova area.";
		mes "Torna quando sei pronto per partire.";
		close;
	}
}
