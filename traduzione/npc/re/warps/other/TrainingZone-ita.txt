prontera,246,62,3	script	Training Instructor#sol1	8W_SOLDIER,{
	mes "[Istruttore di formazione]";
	mes "Se vuoi allenarti in un'area più ampia, ti faccio entrare.";
	next;
	mes "[Istruttore di formazione]";
	mes "Il campo di allenamento è diviso in due aree principali.";
	next;
	mes "[Istruttore di formazione]";
	mes "Area superiore dove puoi controllare la forza applicata in base alle dimensioni del mostro.";
	next;
	mes "[Istruttore di formazione]";
	mes "L'area inferiore dove puoi controllare la forza applicata a seconda dell'attributo e della forma del mostro.";
	next;
	mes "[Istruttore di formazione]";
	mes "Dove vorresti entrare?";
	next;
	switch( select( "Zona superiore", "Zona inferiore", "Accesso Vietato" ) ) {
	case 1:
		mes "[Istruttore di formazione]";
		mes "Allora ti guiderò.";
		close2;
		warp "tra_fild",112,167;
		end;
	case 2:
		mes "[Istruttore di formazione]";
		mes "Allora ti guiderò.";
		close2;
		warp "tra_fild",111,74;
		end;
	case 3:
		mes "[Istruttore di formazione]";
		mes "Se hai bisogno di una guida, vieni da me.";
		close;
	}
}
