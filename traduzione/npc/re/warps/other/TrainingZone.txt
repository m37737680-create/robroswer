prontera,246,62,3	script	Training Instructor#sol1	8W_SOLDIER,{
	mes "I formatore";
	mes "Se vuoi allenarti in un'area più ampia, ti farò entrare.";
	next;
	mes "I formatore";
	mes "Il campo di addestramento è diviso in due aree principali.";
	next;
	mes "I formatore";
	mes "Area superiore dove è possibile controllare la forza applicata in base alla dimensione del mostro.";
	next;
	mes "I formatore";
	mes "L'area inferiore dove è possibile controllare la forza applicata a seconda dell'attributo e della forma del mostro.";
	next;
	mes "I formatore";
	mes "Dove vorresti entrare?";
	next;
	switch( select( "Area superiore", "Area inferiore", "Non entrare" ) ) {
	case 1:
		mes "I formatore";
		mes "Allora ti guiderò.";
		close2;
		warp "tra_fild",112,167;
		end;
	case 2:
		mes "I formatore";
		mes "Allora ti guiderò.";
		close2;
		warp "tra_fild",111,74;
		end;
	case 3:
		mes "I formatore";
		mes "Se hai bisogno di guida, vieni da me.";
		close;
	}
}
