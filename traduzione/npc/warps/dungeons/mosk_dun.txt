mosk_dun02,53,220,4	script	#babayagain	WARPNPC,1,1,{
	end;
OnTouch:
	if (mos_nowinter == 11) {
		donpcevent "Soldier1#mos::OnEnable";
		donpcevent "Soldier2#mos::OnEnable";
		donpcevent "Soldier3#mos::OnEnable";
		mes "Soldato";
		mes "Fermati lì!";
		next;
		mes "Soldato";
		mes "Abbiamo informazioni";
		mes "che sei stato segretamente";
		mes "comunicare con";
		mes "Baba Yaga, l'Orribile.";
		next;
		mes "Soldato";
		mes "Se hai qualcosa da dire,";
		mes "parla con il Czar.";
		mes "Vieni con noi.";
		close2;
		donpcevent "Soldier1#mos::OnInit";
		donpcevent "Soldier2#mos::OnInit";
		donpcevent "Soldier3#mos::OnInit";
		set mos_nowinter,12;
		warp "mosk_in",125,92;
		end;
	}
	else if (mos_nowinter == 12 || mos_nowinter == 13) {
		donpcevent "Soldier1#mos::OnEnable";
		donpcevent "Soldier2#mos::OnEnable";
		donpcevent "Soldier3#mos::OnEnable";
		mes "Soldato";
		mes "Ci aspettavamo";
		mes "che tu riapparissi.";
		mes "Qual è il tuo affare con lo Zar?";
		next;
		if(select("Scappa.:Spiega la situazione.") == 1) {
			mes "["+strcharinfo(0)+"]";
			mes "Dovrei uscire da qui...";
			close2;
			warp "mosk_dun02",100,100;
		}
		else {
			mes "Spiego cosa è successo finora.";
			next;
			mes "Soldato";
			mes "Se hai qualcosa da dire, parla";
			mes "al Czar.";
			mes "Se vai a questo posto,";
			mes "sarai considerato il nostro nemico.";
			close2;
		}
		donpcevent "Soldier1#mos::OnInit";
		donpcevent "Soldier2#mos::OnInit";
		donpcevent "Soldier3#mos::OnInit";
		end;
	}
	else {
		warp "mosk_in",215,41;
		end;
	}
}
