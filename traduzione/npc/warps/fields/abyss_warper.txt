				mes "tremore dalla colonna.^000000";
				specialeffect EF_WINDHIT;
				next;
				mes "^3355FFInserisci con cautela";
				mes "una coda di drago nel";
				mes "finale groove. Le luci brillano";
				mes "forse da crepe nel";
